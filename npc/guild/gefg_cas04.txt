// Guild Flag (Outside Castle)
//============================================================
-	script	Bergel#1::OutsideFlagsG4	722,{

	set .@GID, GetCastleData("gefg_cas04",1);

	if (.@GID == 0) {
		mes "[神聖盧恩-米德加茲王國的敕令]";
		mes " ";
		mes "1. 遵循神聖盧恩-米德加茲王國的條例，";
		mes "我們宣告";
		mes "此城堡沒有正式的主人。";
		mes " ";
		mes "2. 對於能夠";
		mes "克服所有試煉";
		mes "並摧毀 Emperium 的人，";
		mes "國王將賦予其";
		mes "此城堡的所有權。";
		close;
	}
	if (getcharid(2) == .@GID) {
		mes "[迴盪的聲音]";
		mes "勇敢的人們...";
		mes "你們想要回到你們光榮的地方嗎？";
		next;
		if(select("返回公會城堡.:離開。") == 1) {
			close2;
			if (getcharid(2) == GetCastleData("gefg_cas04",1)) {
				warp "gefg_cas04",168,43;
				end;
			}
			end;
		}
		close;
	}
	mes "[神聖盧恩-米德加茲王國的敕令]";
	mes " ";
	mes "1. 遵循神聖盧恩-米德加茲王國的條例，";
	mes "我們批准此地為";
	mes "^ff0000"+GetGuildName(.@GID)+"^000000 公會的私人財產。";
	mes " ";
	mes "2. ^ff0000"+GetGuildName(.@GID)+"^000000 公會的公會長是";
	mes "^ff0000"+GetGuildMaster(.@GID)+"^000000";
	mes "如果有人反對，";
	mes "就用你手中的鋼刃來證明你的力量和榮譽吧。";
	close;

OnRecvCastleG04:
	FlagEmblem GetCastleData("gefg_cas04",1);
	end;
}
