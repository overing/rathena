// Guild Flags (Outside Castle)
//============================================================
-	script	Bamboo Grove Hill#1::OutsideFlagsPA5	722,{

	set .@GID, GetCastleData("payg_cas05",1);

	if (.@GID == 0) {
		mes "[神聖盧恩-米德加茲王國的敕令]";
		mes " ";
		mes "1. 遵守神聖盧恩-米德加茲王國的條例，";
		mes "我們宣告";
		mes "此城堡沒有正式的主人。";
		mes " ";
		mes "2. 對於能夠";
		mes "克服所有試煉";
		mes "並摧毀神之金屬的人，";
		mes "國王將賦予其";
		mes "此城堡的所有權。";
		close;
	}
	if (getcharid(2) == .@GID) {
		mes "[迴盪的聲音]";
		mes "勇敢的人們...";
		mes "你們希望回到你們光榮的地方嗎？";
		next;
		if(select("返回公會城堡.:離開。") == 1) {
			close2;
			if (getcharid(2) == GetCastleData("payg_cas05",1)) {
					warp "payg_cas05",243,27;
				end;
			}
			end;
		}
		close;
	}
	mes "[神聖盧恩-米德加茲王國的敕令]";
	mes " ";
	mes "1. 遵守神聖盧恩-米德加茲王國的條例，";
	mes "我們批准此地為";
	mes "^ff0000"+GetGuildName(.@GID)+"^000000 公會的私人財產。";
	mes " ";
	mes "^ff0000"+GetGuildName(.@GID)+"^000000 公會的";
	mes "公會會長為 ^ff0000"+GetGuildMaster(.@GID)+"^000000";
	mes "若有任何人對此有異議，";
	mes "用你手中的鋼刃證明你的力量與榮譽。";
	close;

OnRecvCastlePy05:
	FlagEmblem GetCastleData("payg_cas05",1);
	end;
}
