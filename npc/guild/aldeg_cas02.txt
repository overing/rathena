// Guild Flags (Outside Castle)

-	script	Hohenschwangau#1::OutsideFlagsA2	722,{

	set .@GID, GetCastleData("aldeg_cas02",1);

	if (.@GID == 0) {
		mes "[神聖盧恩-米德加爾特王國敕令]";
		mes " ";
		mes "1. 遵守神聖盧恩-米德加爾特王國的法令，";
		mes "我們宣告";
		mes "此城堡沒有正式的主人。";
		mes " ";
		mes "2. 賜予能夠";
		mes "克服所有試煉";
		mes "並摧毀 Emperium 的人，";
		mes "國王將授予此人";
		mes "此城堡的所有權。";
		close;
	}
	if (getcharid(2) == .@GID) {
		mes "[迴盪的聲音]";
		mes "勇敢的人們...";
		mes "你們是否想回到你們光榮的地方？";
		next;
		if(select("返回公會城堡。:離開。") == 1) {
			close2;
			if (getcharid(2) == GetCastleData("aldeg_cas02",1)) {
				warp "aldeg_cas02",220,190;
				end;
			}
			end;
		}
		close;
	}
	mes "[神聖盧恩-米德加爾特王國敕令]";
	mes " ";
	mes "1. 遵守神聖盧恩-米德加爾特王國的法令，";
	mes "我們批准此地為";
	mes "^ff0000"+GetGuildName(.@GID)+"^000000 公會的私人財產。";
	mes " ";
	mes "2. ^ff0000"+GetGuildName(.@GID)+"^000000 公會的公會長是";
	mes "^ff0000"+GetGuildMaster(.@GID)+"^000000";
	mes "如果有人對此有異議，";
	mes "用你手中的鋼刃證明你的力量與榮譽。";
	close;

OnRecvCastleA02:
	FlagEmblem GetCastleData("aldeg_cas02",1);
	end;
}
