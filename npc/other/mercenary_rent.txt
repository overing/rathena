-	script	傭兵商品商人#dummy::MercMerchant	-1,{
	mes "[傭兵商品商人]";
	mes "您好，我販賣";
	mes "傭兵可以使用的";
	mes "商品。有什麼";
	mes "特別的東西";
	mes "您在尋找嗎？";
	next;
	setarray .@item, 12184,	12185,	12241,	12242,	12243;
	setarray .@cost, 2500,	5000,	800,	1500,	3000;
	for (.@size = getarraysize(.@item); .@i < .@size; .@i++)
		.@menu$ = .@menu$ + getitemname(.@item[.@i]) + ":";
	.@m = select(.@menu$)-1;
	.@str$ = .@cost[.@m] + "";
	.@len = getstrlen(.@str$);
	.@str_cost$ = (.@len <= 3)? .@str$ : insertchar(.@str$,",",.@len-3);
	mes "[傭兵商品商人]";
	mes getitemname(.@item[.@m]);
	mes "每個花費 " + .@str_cost$ + " zeny。";
	mes "您想要多少個？";
	next;
	input .@input;
	if (!.@input) {
		mes "[傭兵商品商人]";
		mes "您改變主意了嗎？";
		mes "好吧，當您想要";
		mes "購買任何傭兵藥水時，請隨時回來";
		mes "找我。";
		close;
	}
	else if (.@input < 0 || .@input > 10000) {
		mes "[傭兵商品商人]";
		mes "很抱歉，您";
		mes "一次只能購買";
		mes "最多 10,000 瓶這些藥水";
		mes "請輸入";
		mes "1 到 10,000 之間的數字。";
		close;
	}
	else if (set(.@total_po, .@input * .@cost[.@m]) > Zeny) {
		mes "[傭兵商品商人]";
		mes "很抱歉，您沒有";
		mes "足夠的zeny購買";
		mes "這麼多藥水。好吧，我會";
		mes "在這裡等您準備好";
		mes "為您的傭兵";
		mes "購買東西。";
		close;
	}
	else if (!checkweight(.@item[.@m], .@input)) {
		mes "[傭兵商品商人]";
		mes "如果我給您那麼多";
		mes "藥水，您將無法";
		mes "隨身攜帶它們。";
		mes "請在";
		mes "清理一些空間後";
		mes "回來您的背包中。";
		close;
	}
	mes "[傭兵商品商人]";
	if (.@input == 1) 
		mes "這是您的 " + getitemname(.@item[.@m]) + "。";
	else {
		mes "這是您要的，總共";
		mes callfunc("F_InsertPlural",.@input,getitemname(.@item[.@m])) + "。";
	}
	mes "謝謝您，當您需要更多";
	mes "藥水給您的傭兵時，請再來";
	mes "。";
	Zeny = Zeny - .@total_po;
	getitem .@item[.@m], .@input;
	close;
}

// Mercenary Switch NPCs
//============================================================
-	script	Mercenary_Switch	-1,{

	setarray .@name$, "Spear","Sword","Bow";
	.@npc$ = strnpcinfo(2);
	for (.@size = getarraysize(.@name$); .@type < .@size; .@type++)
		if (.@npc$ == .@name$[.@type]) break;

	mes "[檢查員]";
	mes "請輸入密碼。";
	next;
	.@i = callfunc("F_GM_NPC",1854,0,0,10000);
	mes "[檢查員]";
	if (.@i == -2)
		mes "錯誤。";
	else if (.@i == 0)
		mes "錯誤號碼";
	else if (.@i == 1) {
		mes "請選擇。";
		next;
		switch( select( "關閉傭兵NPC","開啟傭兵NPC" ) ) {
		case 1:
			mes "NPC已關閉。";
			disablenpc "傭兵管理人#" + .@name$[.@type];
			disablenpc "傭兵商品商人#" + .@name$[.@type];
			break;
		case 2:
			mes "NPC已開啟。";
			enablenpc "傭兵管理人#" + .@name$[.@type];
			enablenpc "傭兵商品商人#" + .@name$[.@type];
			break;
		}
	}
	close;
}
