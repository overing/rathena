pvp_y_room,77,85,4	script	Guild Battle Guide	105,{

	setarray .@map1X[1],7,9,16,8,20;
	setarray .@map2X[1],91,90,83,91,79;
	setarray .@mapY[1],49,49,50,49,50;

	mes "[公會戰鬥指南]";
	mes "你好嗎？";
	mes "我是公會戰鬥指南，";
	mes "專為新的 PvP 地圖服務。";
	mes "告訴我你想去哪張地圖，";
	mes "我會將你移動過去。";
	next;
	set .@i, select("地圖 1:地圖 2:地圖 3:地圖 4:地圖 5");
	mes "[公會戰鬥指南]";
	mes "你選擇了地圖 "+.@i+"。";
	mes "那麼，你屬於哪個隊伍？";
	mes "你可以選擇隊伍 1 或隊伍 2。";
	next;
	set .@j, select("隊伍 1:隊伍 2");
	mes "[公會戰鬥指南]";
	mes "好的，我會將你移動";
	mes "到地圖 "+.@i+"，身為";
	mes "隊伍 "+.@j+" 的一員。你準備好";
	mes "現在被傳送過去了嗎？";
	next;
	if(select("是:否") == 2) {
		mes "[公會戰鬥指南]";
		mes "好的，我希望";
		mes "能在 PvP 戰場上";
		mes "再次見到你！";
		close;
	}
	mes "[公會戰鬥指南]";
	mes "太棒了！玩得開心～";
	close2;
	warp "guild_vs"+.@i,getd(".@map"+.@j+"X["+.@i+"]"),.@mapY[.@i];
	end;
}
