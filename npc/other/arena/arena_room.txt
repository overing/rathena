	mes "您可以用 ^3131FF"+ .@arena_point +" 競技場點數兌換 "+ .@amount +" "+ getitemname(.@item_id) +"^000000。";
	mes "如果您想要取消，請輸入 0。如果您不取消，請輸入您想花費競技場點數的 ^3131FF次數^000000。";
	next;
	input .@reward;
	if (.@reward <= 0) {
		mes "[基夫]";
		mes "您已取消您的請求。";
		close;
	} else if (.@reward > 1500) {
		mes "[基夫]";
		mes "您已超過最大容量。";
		close;
	}
	if (arena_point < (.@reward * .@arena_point)) {
		mes "[基夫]";
		mes "您沒有足夠的競技場點數。";
		mes "請檢查您擁有的競技場點數總額。";
		close;
	}
	arena_point = arena_point - (.@reward * .@arena_point);
	getitem .@item_id,.@reward * .@amount;
	mes "[基夫]";
	mes "謝謝惠顧，歡迎再次光臨。";
	close;
}
