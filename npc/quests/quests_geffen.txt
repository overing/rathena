// Number Match Game :: gef_yagu
//============================================================
geffen,122,77,4	script	妮雅#yagu	700,{
	mes "[妮雅]";
	mes "你好，我是妮雅，是遊戲";
	mes "鬼才馬王第五位";
	mes "弟子。那麼，";
	mes "你想要和我玩";
	mes "數字配對遊戲嗎？";
	next;
	switch(select("數字配對遊戲?:練習數字配對:玩數字配對:紀錄保持人:取消")) {
	case 1:
		mes "[妮雅]";
		mes "嗯，數字配對遊戲的";
		mes "目標是猜出我正在想";
		mes "的數字。當然，有一些";
		mes "規則限制了我可以";
		mes "選擇的數字。";
		next;
		mes "[妮雅]";
		mes "我所想的所有數字都";
		mes "必須是 3 位數，我不能";
		mes "在任何位數上使用 0，";
		mes "而且我不能在任何位數上";
		mes "多次使用相同的";
		mes "數字。";
		next;
		mes "[妮雅]";
		mes "所以，舉例來說，我不能";
		mes "使用數字 103、112、";
		mes "252 或 701。這些數字";
		mes "要么有 0，要么它們在";
		mes "各個位數上多次使用";
		mes "相同的數字。明白了嗎？";
		next;
		mes "[妮雅]";
		mes "現在，你只有 5 次機會，";
		mes "而且每次猜完之後，我會告訴";
		mes "你猜對了幾個數字，";
		mes "然後告訴你猜對的數字中";
		mes "有幾個數字在正確的";
		mes "位置上。";
		next;
		mes "[妮雅]";
		mes "例如，假設";
		mes "我選的數字是 168。";
		mes "如果你的第一次猜測是 678，那麼你猜對了兩個數字，";
		mes "6 和 8。然而，";
		mes "只有 8 在正確的位置上。";
		next;
		mes "[妮雅]";
		mes "你知道嗎，最好";
		mes "先試試練習模式。";
		mes "然後，當你更";
		mes "熟悉的時候，玩實際的";
		mes "數字配對遊戲。別忘了速度會影響你的分數喔！";
		next;
		mes "[妮雅]";
		mes "啊，如果你碰巧";
		mes "在數字配對遊戲中";
		mes "創造了新的紀錄，你的";
		mes "名字就會被記錄下來，";
		mes "所以不要錯過這個成名的";
		mes "機會，甚至可能帶來財富～";
		close;
	case 2:
		mes "[妮雅]";
		mes "太好了，我們來玩一個練習";
		mes "數字配對遊戲吧。請";
		mes "試著猜猜我的數字，好嗎？";
		mes "如果你想取消，請輸入一個大於 1,000 的數字。";
		set .@practice,1;
		break;
	case 3:
		mes "[妮雅]";
		mes "太好了，我們來玩數字配對吧！";
		mes "你將有 5 次機會猜出我正在想的數字。";
		mes "如果你想取消，請輸入一個大於 1,000 的數字。";
		set .@practice,0;
		break;
	case 4:
		set .@name_record$, $050908_minus1_yagu$;
		if (.@name_record$ == "") //custom
			set .@name_record$,"(null)";
		set .@score_record, $050908_minus1_yagu;
		set .@score_min, .@score_record / 60;
		set .@score_sec, .@score_record % 60;
		mes "[妮雅]";
		mes "^ff0000" + .@name_record$ + "^000000 是數字";
		mes "配對遊戲的紀錄";
		mes "保持人，時間是";
		mes "^ff0000" + .@score_min + " 分鐘, " + .@score_sec + " 秒^000000。";
		close;
	case 5:
		mes "[妮雅]";
		mes "照顧好你自己，";
		mes "冒險者。如果你";
		mes "想玩數字配對遊戲，";
		mes "就來和我說說話，";
		mes "好嗎？再見～";
		close;
	}
	next;

	// Generate 3 digits of a number.
	while(1) {
		set .@yagu100, rand(1,9);
		set .@yagu10, rand(1,9);
		set .@yagu1, rand(1,9);
		if ((.@yagu100 != .@yagu10) && (.@yagu100 != .@yagu1) && (.@yagu10 != .@yagu1)) {
			// Store the actual number (unofficial).
			set .@yagu, (100*.@yagu100) + (10*.@yagu10) + .@yagu1;
			break;
		}
	}

	// The official time calculation is based on ServerTime,
	// an int HHMMSS equivalent to atoi(gettimestr("%H%M%S",7)).
	// UNIX time is really a better way to do this.
	set .@start_time, gettimetick(2);

	while(1) {

		// Guess a number that meets all conditions.
		while(1) {
			if (input(.@input,100,999) != 0) {
				mes "[妮雅]";
				mes "你已經取消了";
				mes "數字配對遊戲。";
				close;
			}
			set .@input100, .@input / 100;
			set .@input10, (.@input % 100) / 10;
			set .@input1, .@input % 10;
			if ((.@input100 > 0) && (.@input10 > 0) && (.@input1 > 0)) {
				if ((.@input100 != .@input10) && (.@input100 != .@input1) && (.@input10 != .@input1))
					break;
			}
			mes "[妮雅]";
			mes "你在其中一個";
			mes "位數輸入了 0，或者在";
			mes "各個位數中多次輸入";
			mes "相同的數值。";
			mes "請輸入另一個猜測。";
			next;
		}

		set .@retry, .@retry+1;
		mes "[妮雅]";
		mes "^ff0000"+strcharinfo(0)+"^000000，";
		mes "你的猜測是 ^0000ff"+.@input+"^000000。";
		mes "給我一點時間來";
		mes "計算出你的結果。";
		next;

		// Correct digits in correct locations.
		set .@strike,0;
		if (.@yagu100 == .@input100)
			set .@strike, .@strike+1;
		if (.@yagu10 == .@input10)
			set .@strike, .@strike+1;
		if (.@yagu1 == .@input1)
			set .@strike, .@strike+1;

		// Correct digits in incorrect locations.
		set .@ball,0;
		if ((.@yagu100 == .@input10) || (.@yagu100 == .@input1))
			set .@ball, .@ball+1;
		if ((.@yagu10 == .@input100) || (.@yagu10 == .@input1))
			set .@ball, .@ball+1;
		if ((.@yagu1 == .@input100) || (.@yagu1 == .@input10))
			set .@ball, .@ball+1;

		if (.@strike == 3) {
			if (.@practice) {
				mes "[妮雅]";
				mes "是的，我猜的數字是";
				mes "^ff0000"+.@yagu+"^000000!";
				mes "恭喜你，你";
				mes "贏得了這個練習";
				mes "遊戲！現在你應該";
				mes "準備好迎接真正的挑戰了～";
				close;
			} else {

				// Calculate final time and check against the previous record.
				set .@end_time, gettimetick(2);
				if (.@end_time < .@start_time)
					set .@end_time, .@end_time + 3600;
				set .@total_time, .@end_time - .@start_time;
				set .@total_min, .@total_time / 60;
				set .@total_sec, .@total_time % 60;
				set .@name_record$, $050908_minus1_yagu$;
				if (.@name_record$ == "") //custom
					set .@name_record$,"(null)";
				set .@score_record, $050908_minus1_yagu;
				set .@score_min, .@score_record / 60;
				set .@score_sec, .@score_record % 60;

				if (.@score_record >= .@total_time || !.@score_record) { //custom (second condition)
					set $050908_minus1_yagu$, strcharinfo(0);
					set $050908_minus1_yagu, .@total_time;
					mes "[妮雅]";
					mes "太棒了！你猜中了";
					mes "我的數字，是";
					mes "^ff0000" + .@yagu + "^000000。你花了";
					mes "^ff0000" + .@total_min + " 分鐘, " + .@total_sec + " 秒^000000 完成，";
					mes "打破了之前的紀錄";
					mes "^ff0000" + .@score_min + " 分鐘, " + .@score_sec + " 秒^000000。";
					next;
					mes "[妮雅]";
					mes "^ff0000" + .@name_record$ + "^000000";
					mes "創下了之前的紀錄，";
					mes "但現在你是在配對";
					mes "遊戲中需要被擊敗的";
					mes "新對象。恭喜";
					mes "你創下新紀錄！";
					close;
				}
				mes "[妮雅]";
				mes "做得好，你猜中了";
				mes "我的數字！當然，答案是";
				mes "^ff0000" + .@yagu + "^000000。你設法";
				mes "在 ^ff0000" + .@total_min + " 分鐘, " + .@total_sec + " 秒^000000 內正確地";
				mes "猜出這個數字。";
				next;
				mes "[妮雅]";
				mes "^ff0000" + .@name_record$ + "^000000 是目前";
				mes "數字配對";
				mes "紀錄保持人，時間是";
				mes "^ff0000" + .@score_min + " 分鐘, " + .@score_sec + " 秒^000000。";
				mes "祝你好運，希望有一天你";
				mes "可以打破這個紀錄～";
				close;
			}
		} else {
			if (.@practice) {
				mes "[妮雅]";
				mes "你的猜測有 ^ff0000" + .@ball + "^000000 個正確";
				mes "的數字。你猜測的";
				mes "^ff0000" + .@strike + "^000000 個位數";
				mes "在正確的數字位置上";
				mes "有正確的數字。";
				mes "好吧，再試一次～";
				next;
			} else {
				mes "[妮雅]";
				mes "你上次的猜測有 ^ff0000" + .@ball + "^000000";
				mes "個正確的數字，";
				mes "而且你猜測的 ^ff0000" + .@strike + "^000000 個位數";
				mes "在正確的位置上";
				mes "有正確的數字。";
				switch(.@retry) {
					case 1: mes "你還剩下四次機會。"; break;
					case 2: mes "你還剩下三次機會。"; break;
					case 3: mes "你還剩下兩次機會。"; break;
					case 4: mes "你只剩下一次機會了..."; break;
					case 5: mes "答案是 ^ff0000" + .@yagu + "^000000。"; close;
				}
				next;
			}
		}
	}
}
