pay_arche,122,100,0	script	Dollshoi	88,{
	mes "[魔法師公會人員]";
	mes "你想要溶液？ 嗯，給我 50 Zeny 並且至少給我一個空的試管。";
	next;
	if (select("好吧，成交。:不，算了。") == 1) {
		mes "[魔法師公會人員]";
		if (Zeny < 50) {
			mes "嘿！你沒有足夠的錢支付我的 50 Zeny 費用。";
			close;
		}
		if (countitem(1092) == 0) {
			mes "沒有瓶子你無法攜帶溶液！帶給我一個空的試管。";
			close;
		}
		delitem 1092,1; //Empty_Cylinder
		set Zeny, Zeny-50;
		getitem 1089,1; //Payon_Potion
	}
	close;
}
