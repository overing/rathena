		mes "噗，反問句。";
		next;
		mes "[古德曼]";
		mes "好消息。我負責";
		mes "教授由鐵匠和商人";
		mes "公會開發的技能，叫做^FF0000貪婪^000000。";
		mes "是否要學習取決於你，";
		mes "但我建議你學。";
		next;
		mes "[古德曼]";
		mes "如果你願意，我可以";
		mes "現在開始教你貪婪";
		mes "技能。你覺得";
		mes "如何，孩子？";
		next;
		if (select("聽起來不錯！:晚點再說～") == 1) {
			mes "[古德曼]";
			mes "嘿！好選擇。現在，";
			mes "貪婪技能會自動";
			mes "撿起施法者周圍 5*5";
			mes "格範圍內的所有物品。";
			mes "它對於撿拾怪物掉落的一堆物品非常方便。";
			next;
			mes "[古德曼]";
			mes "首先要做的是。我需要";
			mes "測試你，看看你是否有資格";
			mes "學習這個技能。去用任何物品填滿你的背包";
			mes "直到它滿到無法放入";
			mes "一個重量為 500 的物品為止。";
			next;
			mes "[古德曼]";
			mes "當你完成那個";
			mes "任務時，回來";
			mes "和我談談。好";
			mes "吧，祝你好運～";
			BLACK_SK2 = 1;
			close;
		}
		mes "[古德曼]";
		mes "我明白了。忙於其他事情";
		mes "是件好事，";
		mes "但要記住時間是";
		mes "一種商品，你應該";
		mes "始終將其花在做最";
		mes "有價值的事情上。明白了嗎？";
		close;
	}
	if (BLACK_SK2 == 1) {
		if (MaxWeight - Weight >= 5000) {
			mes "[古德曼]";
			mes "嗯。你回來得";
			mes "太早了。而且你沒有";
			mes "完成我為你設定的";
			mes "目標。仔細聽好。";
			next;
			mes "[古德曼]";
			mes "去用任何物品填滿你的背包";
			mes "直到它太滿，無法放入";
			mes "一個重量為 500 的物品為止。";
			mes "當你完成這個任務時，";
			mes "回來向我報告。";
			close;
		}
		mes "[古德曼]";
		mes "已經回來了，嗯？";
		mes "好吧，讓我檢查";
		mes "你攜帶的東西。";
		mes "讓我想想，背包，背包...";
		next;
		mes "[古德曼]";
		mes "做得好：你通過了。";
		mes "你現在是否意識到";
		mes "收集這些物品所花費的時間和經驗比";
		mes "它們的錢幣價格更有價值？";
		next;
		mes "[古德曼]";
		if (Upper == 1)
			mes "我相信神工匠";
		else
			mes "我相信鐵匠";
		mes "首先應該是";
		mes "工匠。變得富有和出名";
		mes "只是它的副作用而已。";
		if (Upper == 1)
			mes "光榮的神工匠知道價值在於工藝...";
		else
			mes "光榮的鐵匠知道價值在於工藝...";
		next;
		if (Upper == 1) {
			mes "[古德曼]";
			mes "現在，如果你不太過";
			mes "重視物品的";
			mes "錢幣價值，你";
			mes "就能夠更容易地放開它們。";
			mes "看著我做這個。";
			mes "看到了嗎？現在你也可以做到了。";
			next;
		}
		mes "[古德曼]";
		mes "你學到的技能是";
		mes "叫做貪婪，但我希望";
		mes "你以一種與它的名字";
		mes "不同的方式使用它。";
		mes "別忘了這一點，我希望";
		mes "你創造出真正的傑作。";
		skill "BS_GREED",1,SKILL_PERM;
		if (Upper == 1)
			BLACK_SK2 = 4;	// note: official script is using another variable when a transcendent character is doing the quest for the first time
		else
			BLACK_SK2 = 2;
		close;
	}
	mes "[古德曼]";
	mes "你如何使用";
	mes "貪婪技能？希望";
	mes "它在你的";
	mes "努力中為你帶來了便利，為你所有的";
	mes "同志們打造更高品質的";
	mes "商品。";
	if (getskilllv("BS_GREED") == 0)	// debug
		skill "BS_GREED",1,SKILL_PERM;
	close;
}
