function	script	F_BardSkillYhelle	{
	if (checkweight(1201,1) == 0 || MaxWeight - Weight < 2000) {
		mes "^3355FF等等！";
		mes "你攜帶太多物品，沒有足夠的物品欄空間來接收任何獎勵。請空出更多物品欄空間，然後再來接受這個挑戰。";
		close;
	}
	emotion ET_HUK, playerattached();
	emotion ET_HUK;
	mes "[伊耶魯母雞]";
	mes "咕咕咕？";
	mes "咕咕咕咕！";
	next;
	if (rand(1,3) == 2) {
		disablenpc "Yhelle#bard_chick"+getarg(0);
		donpcevent "Yhelle#bard_chick"+getarg(1)+"::OnEnable";
		mes "^3355FF察覺到你的";
		mes "存在，母雞";
		mes "迅速地跑開了。^000000";
		if (qskill_bard > 1 && qskill_bard < 7) {
			mes "^3355FF您在";
			mes "伊耶魯母雞棲息的";
			mes "地方發現了 ^3333331 個雞蛋^3355FF。^000000";
			set qskill_bard,qskill_bard+1;
			getitem 574,1; //Egg
		}
	}
	else {
		disablenpc "Yhelle#bard_chick"+getarg(0);
		donpcevent "Yhelle#bard_chick"+getarg(1)+"::OnEnable";
		mes "^3355FF察覺到你的";
		mes "存在，母雞";
		mes "迅速地跑開了。^000000";
	}
	return;
}
