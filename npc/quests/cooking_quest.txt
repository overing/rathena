S_SellSets:
	.@item_id = getarg(0);
	.@item_cost = getiteminfo(.@item_id, ITEMINFO_BUY);
	.@item_weight = getiteminfo(.@item_id, ITEMINFO_WEIGHT);
	mes "[瑪德琳·朱]";
	mes "您想要購買多少個 "+ (.@item_id == 12125 ? "戶外":"室內");
	mes "烹飪工具組呢？如果您";
	mes "想要取消，請";
	mes "輸入數字 0。";
	next;
	while(1) {
		input .@sell;
		if (.@sell == 0) {
			mes "[瑪德琳·朱]";
			mes "您改變主意了？";
			mes "好吧，如果您需要";
			mes "之後購買 "+ getitemname(.@item_id);
			mes "工具組，隨時回來";
			mes "找我就行了，好嗎？";
			close;
		}
		else if (.@sell > 100) {
			mes "[瑪德琳·朱]";
			mes "哦，我很抱歉，但是";
			mes "我為了安全起見，";
			mes "一次最多只賣";
			mes "100 個 "+ getitemname(.@item_id);
			next;
		}
		else break;
	}
	set .@total_cost,.@sell * .@item_cost;
	set .@total_weight,.@sell * .@item_weight;
	if (Zeny < .@total_cost) {
		mes "[瑪德琳·朱]";
		mes "哦，我很抱歉，但是您";
		mes "買不起這麼多";
		mes getitemname(.@item_id);
		mes "請在購買我的工具組之前";
		mes "檢查您的錢幣喔~";
		close;
	}
	if (!checkweight(.@item_id,.@sell)) {
		mes "[瑪德琳·朱]";
		mes "我很抱歉，但是您的";
		mes "物品欄沒有足夠的空間";
		mes "容納這麼多";
		mes getitemname(.@item_id) +"...";
		close;
	}
	set Zeny, Zeny-.@total_cost;
	getitem .@item_id,.@sell;
	mes "[瑪德琳·朱]";
	mes "給您~";
	mes "祝您";
	mes "烹飪訓練順利！";
	return;
}
