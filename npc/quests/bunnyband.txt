alberta,26,229,0	script	卡普拉職員#bunny	83,{
	if (BUNYBND == 1) {
		mes "[卡普拉職員]";
		mes "您好～！";
		mes "您想要";
		mes "參與卡普拉";
		mes "公司的特別";
		mes "^529DFF兔耳髮圈活動^000000 嗎？";
		next;
		switch(select("當然，我帶了物品。:活動資訊:取消")) {;
		case 1:
			mes "[卡普拉職員]";
			mes "好的～";
			mes "讓我檢查";
			mes "看看您是否帶了";
			mes "所有的物品...";
			next;
			setarray .@Items[0],949,100,706,1,722,1,2213,1;
			for(set .@i,0; .@i<8; set .@i,.@i+2)
				if (countitem(.@Items[.@i]) < .@Items[.@i+1]) {
					mes "[卡普拉職員]";
					mes "喔，我很抱歉";
					mes "但是您需要";
					mes "至少帶";
					mes callfunc("F_InsertPlural",.@Items[.@i+1],getitemname(.@Items[.@i]))+"。";
					close;
				}
			mes "[卡普拉職員]";
			mes "太好了，我看到";
			mes "您收集了";
			mes "我需要製作";
			mes "兔耳髮圈的所有東西。";
			mes "請稍等一下";
			mes "我把它們組裝起來...";
			next;
			for(set .@i,0; .@i<8; set .@i,.@i+2)
				if (countitem(.@Items[.@i]) < .@Items[.@i+1]) {
					mes "[卡普拉職員]";
					mes "嗯？我很抱歉，";
					mes "但我現在實際上";
					mes "無法製作這個。您";
					mes "需要 "+callfunc("F_InsertPlural",.@Items[.@i+1],getitemname(.@Items[.@i]));
					mes "才能將這個";
					mes "兔耳髮圈組裝起來...";
					close;
				}
			delitem 949,100; //Feather
			delitem 706,1; //Four_Leaf_Clover
			delitem 722,1; //Scarlet_Jewel
			delitem 2213,1; //Cat_Hairband
			mes "[卡普拉職員]";
			mes "啊，給您～";
			mes "完美的兔耳髮圈！";
			mes "希望您喜歡。";
			getitem 2214,1; //Bunny_Band
			set BUNYBND,0;
			next;
			mes "[卡普拉職員]";
			mes "感謝您";
			mes "參與這次";
			mes "特別活動以及您";
			mes "對卡普拉服務的";
			mes "持續惠顧～";
			close;
		case 2:
			goto L_Info;
		case 3:
			close;
		}
	}
	mes "[卡普拉職員]";
	mes "您好～！您想要加入我們由卡普拉";
	mes "公司主辦，艾爾貝塔商人工會贊助的";
	mes "特別 ^529DFF兔耳髮圈活動^000000 嗎？";
	next;
	switch(select("加入活動:活動資訊:取消")) {;
	case 1:
		set BUNYBND,1;
		mes "[卡普拉職員]";
		mes "太好了！感謝您的";
		mes "參與！如果您";
		mes "還沒聽說過，";
		mes "如果您想要我幫您";
		mes "組裝一個兔耳髮圈，";
		mes "您需要收集這些物品...";
		next;
		mes "[卡普拉職員]";
		mes "100 個羽毛、";
		mes "1 個四葉幸運草、";
		mes "1 個貓耳髮圈，以及";
		mes "1 個珍珠。就是這樣！";
		mes "祝您好運，我會在";
		mes "這裡等您～";
		close;
	case 2:
		break;
	case 3:
		close;
	}
L_Info:
	mes "[卡普拉職員]";
	mes "為了感謝我們尊貴的";
	mes "客戶，卡普拉公司";
	mes "準備了一項特別活動，";
	mes "卡普拉員工將為";
	mes "攜帶所需物品的冒險者";
	mes "組裝兔耳髮圈。";
	next;
	mes "[卡普拉職員]";
	mes "為了這個特別的，";
	mes "獨一無二的物品，";
	mes "請攜帶 100 個羽毛、";
	mes "1 個四葉幸運草、";
	mes "1 個貓耳髮圈，以及";
	mes "1 個珍珠。";
	next;
	mes "[卡普拉職員]";
	mes "當您準備好時，";
	mes "回來並把";
	mes "那些物品帶給我。";
	mes "（由";
	mes "艾爾貝塔商人工會贊助）。";
	close;
}
