payon_in03,188,146,5	script	馬克斯·漢森#juice	86,{
	if (checkweight(1201,1) == 0) {
		mes "- 請稍等一下！ -";
		mes "- 目前您身上攜帶了 -";
		mes "- 太多物品了。 -";
		mes "- 請稍後再回來 -";
		mes "- 在您把一些物品放入克拉夫倉庫之後。 -";
		close;
	}
	if (MISC_QUEST&1) {
		mes "[商人馬克斯·漢森]";
		mes "歡迎光臨。";
		mes "您也是來";
		mes "處理水果的嗎？";
		next;
		switch(select("製作果汁。:交談並獲取有關水果加工的訊息。:取消。")) {
		case 1:
			mes "[商人馬克斯·漢森]";
			mes "您想要製作哪種果汁呢？";
			next;
			switch(select("蘋果汁:香蕉汁:紅蘿蔔汁:葡萄汁:取消")) {
			case 1:
				set .@fruit,512;
				set .@juice,531;
				break;
			case 2:
				set .@fruit,513;
				set .@juice,532;
				break;
			case 3:
				set .@fruit,515;
				set .@juice,534;
				break;
			case 4:
				set .@fruit,514;
				set .@juice,533;
				break;
			case 5:
				mes "[商人馬克斯·漢森]";
				mes "那麼...";
				mes "下次見。";
				close;
			}
			if (countitem(.@fruit) == 0 || countitem(713) == 0 || Zeny < 3) {
				mes "[商人馬克斯·漢森]";
				mes "糟糕...";
				mes "您沒有所有必要的材料。要製作 "+getitemname(.@juice)+"，我需要 1 個 "+getitemname(.@fruit)+" 和 1 個空瓶。我還需要 3 Zeny的手續費。";
				next;
				mes "[商人馬克斯·漢森]";
				mes "當您準備好一切時，我會將水果混合，為您提供美味的果汁。";
				close;
			}

			mes "[商人馬克斯·漢森]";
			mes "您想要多少呢？";
			next;
			switch(select("盡可能多。:我想要特定數量。:取消。")) {
			case 1:
				set .@make,countitem(.@fruit);
				if (countitem(713) < .@make) set .@make, countitem(713);
				if (Zeny/3 < .@make) set .@make, Zeny/3;
				break;
			case 2:
				mes "[商人馬克斯·漢森]";
				mes "選擇一個小於 100 的數字。如果您不想，請輸入 '0'。您最多可以製作 " + countitem(.@fruit) + " 瓶果汁。";
				next;
				while(1) {
					input .@input;
					if (.@input == 0) {
						mes "[商人馬克斯·漢森]";
						mes "那麼...";
						mes "下次見。";
						close;
					}
					else if (.@input > 100) {
						mes "[商人馬克斯·漢森]";
						mes "超過 100 瓶是不可能的。請選擇不同的數量。";
						next;
					}
					else {
						break;
					}
				}
				set .@make,.@input;
				break;
			case 3:
				mes "[商人馬克斯·漢森]";
				mes "那麼...";
				mes "下次見。";
				close;
			}

			set .@total_zeny,3 * .@make;

			if (countitem(.@fruit) < .@make || countitem(713) < .@make || Zeny < .@total_zeny) {
				mes "[商人馬克斯·漢森]";
				mes "糟糕...";
				mes "您沒有所有必要的材料。我無法幫助這種情況。我想您需要自己收集所需的東西。";
				close;
			}
			delitem .@fruit,.@make;
			delitem 713,.@make;
			set Zeny, Zeny-.@total_zeny;
			getitem .@juice,.@make;

			mes "[商人馬克斯·漢森]";
			mes "給您！新鮮又美味的果汁，正如承諾的那樣。它應該非常清爽可口。";
			next;
			mes "[商人馬克斯·漢森]";
			mes "那麼...";
			mes "下次見。";
			close;
		case 2:
			mes "[商人馬克斯·漢森]";
			mes "在人類能夠發展出如此廣闊的社會之前，他們會從樹上採集水果以求生存。水果可能是大自然的祝福，讓我們得以存在於這個世界上。";
			next;
			mes "[商人馬克斯·漢森]";
			mes "由於生活變得如此繁榮，年輕一代似乎不再吃水果了。所以，我開始思考如何讓水果更容易食用。";
			next;
			mes "[商人馬克斯·漢森]";
			mes "我意識到，當你製作果汁時，它更容易食用，而且味道更好。";
			next;
			mes "# 果汁資訊 #";
			mes "^CC4E5C- 蘋果汁 -^000000";
			mes "蘋果 x 1 個, 空瓶 x 1 個, 3 Zeny。";
			mes "^E3CF57- 香蕉汁 -^000000";
			mes "香蕉 x 1 個, 空瓶 x 1 個, 3 Zeny。";
			mes "^ED9121- 紅蘿蔔汁 -^000000";
			mes "紅蘿蔔 x 1 個, 空瓶 x 1 個, 3 Zeny。";
			mes "^CC00FF- 葡萄汁 -^000000";
			mes "葡萄 x 1 個, 空瓶 x 1 個, 3 Zeny。";
			close;
		case 3:
			mes "[商人馬克斯·漢森]";
			mes "嘿！";
			mes "如果您拜訪了";
			mes "某人，請與他們交談！";
			close;
		}
	}
	else {
		mes "[商人馬克斯·漢森]";
		mes "歡迎光臨。";
		mes "我可以為您做些什麼？";
		next;
		if (select("交談:取消") == 1) {
			mes "[商人馬克斯·漢森]";
			mes "在人類能夠發展出如此廣闊的社會之前，他們會從樹上採集水果以求生存。水果可能是大自然的祝福，讓我們得以存在於這個世界上。";
			next;
			mes "[商人馬克斯·漢森]";
			mes "由於生活變得如此繁榮，年輕一代似乎不再吃水果了。所以，我開始思考如何讓水果更容易食用。";
			next;
			mes "[商人馬克斯·漢森]";
			mes "我意識到，當你製作果汁時，它更容易食用，而且味道更好。";
			close;
		}
		mes "[商人馬克斯·漢森]";
		mes "嘿！";
		mes "如果您拜訪了";
		mes "某人，請與他們交談！";
		close;
	}
}
