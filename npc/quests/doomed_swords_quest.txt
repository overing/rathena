		mes "是我經歷過的最糟糕的！";
		mes "這裡太潮濕";
		mes "而且令人不舒服...";
		close;
	}

S_GetSword:
	if (getarg(2) == 1) {
		mes "^3355FF你伸入";
		mes "刺眼的光芒中並抓住";
		mes "劍的把手。當";
		mes "光芒消散時，你可以";
		mes "看到你已經";
		mes "獲得了一把 "+getitemname(getarg(0))+"。^000000";
		set magum,getarg(3);
		next;
	}
	if (checkweight(getarg(0),1) == 0) {
		mes "[休伊]";
		mes "等一下，你攜帶";
		mes "太多的東西了。";
		mes "你最好存放一些東西";
		mes "以便騰出更多的物品欄";
		mes "空間。然後，我們終於可以取得你的注定之劍了。";
		close;
	}
	.@num = getarg(1);
	mes "[休伊]";
	if (.@num == 0) {
		mes "這-這是格林圖斯！";
		mes "自從我從";
		mes "那些矮人那裡學到五芒星技術以來，我從未";
		mes "想像過我會看到那把劍。請明智地使用那件武器。";
	}
	else if (.@num == 1) {
		mes "我不敢相信！";
		mes "你... 你被認為";
		mes "配得上密斯特汀？";
		mes "小心點，不要讓";
		mes "它的力量壓倒你。";
		mes "只為了正義的目的揮舞它...";
	}
	else if (.@num == 2) {
		mes "這-這是...！";
		mes "行刑者已經";
		mes "選擇你來揮舞它？";
		mes "你一定擁有非常";
		mes "堅強的意志，或者其他什麼...";
		mes "不要屈服於嗜血。";
	}
	next;
	mes "[休伊]";
	mes "嗯，看來我的";
	mes "工作已經完成了。我很高興";
	mes "我有機會使用這些";
	mes "古老的技能，而且很榮幸能遇到一位有價值的冒險家。";
	mes "我的兄弟們應該會感到自豪。";
	next;
	mes "[休伊]";
	mes "說到這個，";
	mes "請向莫夫和";
	mes "沃克致以我的問候，如果你";
	mes "碰巧看到他們。那麼";
	mes "祝你保重，並小心使用那把劍。";
	set dmdswrd_Q,45;
	return;
}
