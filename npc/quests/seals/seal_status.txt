	mes "======== 神之封印狀態 ========";
	setarray .@seal$[1], "史雷普尼爾之靴","雷神腰帶","女神頸鍊","雷神之錘";
	for(set .@i,1; .@i<=4; set .@i,.@i+1) {
		set .@val, getd("$God"+.@i);
		if (.@val == 0)
			set .@status$,"未設置";
		else if (.@val < $@god_check1)
			set .@status$,"啟動";
		else if (.@val < $@god_check2)
			set .@status$,"封印";
		else
			set .@status$,"Released";
		mes .@seal$[.@i]+" 封印: "+.@status$;
	}
	mes " ";
	mes "======= 你的封印狀態 ========";
	if (god_sl_1 < 51)
		mes "^ff0000史雷普尼爾之靴 封印: 未完成^000000";
	else
		mes "^00ff00史雷普尼爾之靴 封印: 已完成^000000";
	if (god_eremes < 28)
		mes "^ff0000雷神腰帶 封印: 未完成^000000";
	else
		mes "^00ff00雷神腰帶 封印: 已完成^000000";
	if (god_brising < 50)
		mes "^ff0000女神頸鍊 封印: 未完成^000000";
	else
		mes "^00ff00女神頸鍊 封印: 已完成^000000";
	if (god_mjo_0 < 11)
		mes "^ff0000雷神之錘 封印: 未完成^000000";
	else
		mes "^00ff00雷神之錘 封印: 已完成^000000";
	close;
}
