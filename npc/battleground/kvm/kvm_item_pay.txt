	else if (.@item_type == 2) {

		mes "等級超過 80 級的人的盔甲是：^ff0000"+ getitemname(.@item_id) +"^000000。";
	}
	else if (.@item_type == 3) {
		mes "您已選擇 '量產型 "+ callfunc("F_GetArmorType",.@item_id) +"'。";
		mes "只有一種量產型 "+ callfunc("F_GetArmorType",.@item_id) +"：^ff0000"+ getitemname(.@item_id) +"^000000。";

		mes "只有一種普及型 "+ callfunc("F_GetArmorType",.@item_id) +"：^ff0000"+ getitemname(.@item_id) +"^000000。";
	}

	if (!.@req_setting)
		mes "需要 ^0000ff2,000^000000 KVM 點數，"+ .@name$ +"，您的點數是：^580080"+ kvm_point +"^000000。";
	else if (.@req_setting == 4)
		mes "需要 ^0000ff1,200^000000 KVM 點數，"+ .@name$ +"，您的點數是：^580080"+ kvm_point +"^000000。";
	else

		mes "購買物品時請小心，恕不退款。";
		break;
	case 2:
		if (kvm_point >= .@prices[.@req_setting]) {
			kvm_point = kvm_point -.@prices[.@req_setting];
			getitem .@item_id,1;

			mes "您已購買了一件 "+ getitemname(.@item_id) +"。";
