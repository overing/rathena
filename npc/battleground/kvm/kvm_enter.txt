// KvM Officer - Guillaume
//============================================================
bat_room,164,178,5	script	KVM Mercenary Officer#1	418,{
	if (getbattleflag("feature.bgqueue"))
		end;
	/*
	set .@kvm,checkquest(6026,PLAYTIME);
	if ((.@kvm == 0) || (.@kvm == 1)) {
		mes "[克羅伊傭兵隊軍官]";
		mes "我知道你已經和克羅伊簽約了。回去加入他們的隊伍。我們不會接受叛徒為我們而戰。";
		close;
	}
	else if (.@kvm == 2) erasequest 6026;
	*/
	set .@permit,checkquest(6025,PLAYTIME);
	if ((.@permit == 0) || (.@permit == 1)) {
		mes "[紀堯姆傭兵隊軍官]";
		mes "看來你最近參加了一場 KVM 戰鬥。你必須等待 5 分鐘才能再次報名。";
		close;
	}
	else if (.@permit == 2) erasequest 6025;
	mes "[紀堯姆傭兵隊軍官]";
	mes "讓他們知道紀堯姆真正的力量！";
	next;
	switch(select("我將與你並肩作戰。:結束對話。")) {
	case 1:
		mes "[紀堯姆傭兵隊軍官]";
		mes "讓他們看看我們有多強大。";
		mes "今天，每個人都會聽到來自紀堯姆的勝利歡呼！";
		close2;
		//setquest 6025;
		if (BaseLevel > 79)  warp "bat_room",169,223;
		else if (BaseLevel < 60) warp "bat_room",197,223;
		else warp "bat_room",225,223;
		end;
	case 2:
		mes "[紀堯姆傭兵隊軍官]";
		mes "我們一定會贏！";
		close;
	}
}

// Guillaume Knight - KvM
//============================================================
bat_room,167,178,5	script	Guillaume Knight#kvm	417,{
	mes "[紀堯姆騎士]";
	mes "你好。";
	mes "你想知道什麼？";
	next;
	switch(select("申請 KVM。:什麼是 KVM？:我該如何參與 KVM？:我想知道我的克雷格點數。")) {
	case 1:
		mes "[紀堯姆騎士]";
		if (getbattleflag("feature.bgqueue"))
			mes "如果你準備好戰鬥了，點擊介面上的 ^0000FF戰鬥按鈕^000000 來申請 KVM 戰鬥！";
		else {
			mes "目前無法申請。";
			mes "要申請，你需要去找 KVM 傭兵隊軍官。";
		}
		close;
	case 2:
		mes "[紀堯姆騎士]";
		mes "KVM 是 Kreiger Von Midgard 的縮寫。";
		mes "冒險者，你是否知道通往新世界的道路已經開啟？";
		next;
		mes "[紀堯姆騎士]";
		mes "我們，紀堯姆政府，有好幾個進軍新世界的計畫。";
		mes "而 KVM 只是其中之一。";
		next;
		mes "[紀堯姆騎士]";
		mes "這意味著我們，米德加爾特大陸上的眾多國家之一，";
		mes "決定雇用許多冒險者來立即進軍新世界。";
		next;
		mes "[紀堯姆騎士]";
		mes "因此，為了選出最優秀的冒險者，我們正在舉辦 KVM。";
		mes "我們期待許多冒險者的回應。";
		next;
		mes "[紀堯姆騎士]";
		mes "我們將給予他們參與獎勵！";
		mes "至於獎勵，請聯絡 KVM 後勤官。";
		close;
	case 3:
		mes "[紀堯姆騎士]";
		if (getbattleflag("feature.bgqueue")) {
			mes "基本上，KVM 是一個 3 對 3 的戰鬥。";
			mes "透過點擊介面上的 ^0000FF戰鬥按鈕^000000 申請，可以組隊或單獨申請。";
		} else {
			mes "基本上，KVM 是一個 5 對 5 的戰鬥。";
			mes "首先，你要在 KVM 接待員那裡申請，可以組隊或單獨申請。";
		}
		next;
		mes "[紀堯姆騎士]";
		if (getbattleflag("feature.bgqueue"))
			mes "組隊申請適用於當你打算與你的隊伍或公會成員一起進入 KVM 的情況，";
		else
			mes "組隊申請適用於當你打算與你的隊伍成員一起進入 KVM 的情況，";
		mes "而個人申請適用於當你打算單獨進入 KVM 的情況。";
		next;
		mes "[紀堯姆騎士]";
		if (getbattleflag("feature.bgqueue"))
			mes "請注意，一旦你取消，你在 ^0000FF1 分鐘內無法再次申請^000000。";
		else {
			mes "請向 KVM 軍官申請，當你在普隆德拉時，他稍後會與你聯絡。";
			next;
			mes "[紀堯姆騎士]";
			mes "然後你進入並遵循戰場上的指示。";
			next;
			mes "[紀堯姆騎士]";
			mes "但是，請注意，除非你在 KVM 辦公室，否則他無法與你聯絡。";
		}
		close;
	case 4:
		mes "[紀堯姆騎士]";
		mes "你的克雷格點數是：";
		mes ""+kvm_point+"。";
		close;
	}
}
