bat_room,124,178,5	script	Tierra Gorge Officer#01a	418,{
	if (getbattleflag("feature.bgqueue"))
		end;
	if (checkweight(1201,1) == 0) {
		mes "- 等一下！！ -";
		mes "- 目前你攜帶的 -";
		mes "- 物品太多了。 -";
		mes "- 請稍後再試 -";
		mes "- 等你減輕一些重量之後。 -";
		close;
	}
	mes "[紀堯姆軍隊軍官]";
	mes "讓我們向那些臭氣熏天的克羅伊軍隊展示紀堯姆軍隊的力量！";
	next;
	switch(select("我想加入你的軍隊！:結束對話")) {
	case 1:
		if ((BaseJob == Job_Novice) || (BaseJob == Job_Super_Novice)) {
			mes "[紀堯姆軍隊軍官]";
			mes "我很高興你想加入我們並與克羅伊軍隊作戰，但很抱歉：我不能派像你這樣的新手去殘酷的戰場上送死。";
		}
		else {
			if (BaseLevel < 80) {
				mes "[紀堯姆軍隊軍官]";
				mes "我很高興你想加入我們並與克羅伊軍隊作戰，但很抱歉：我不能派像你這樣的新手去殘酷的戰場上送死。";
			}
			else {
				set .@chk_urtime,checkquest(2069,PLAYTIME);
				if ((.@chk_urtime == 0)|| (.@chk_urtime == 1)) {
					mes "[紀堯姆軍隊軍官]";
					mes "你似乎剛從戰場回來。";
					mes "現在回去還太早了。去休息吧，把敵人交給我們！";
				}
				else {
					if (.@chk_urtime == 2)
						erasequest 2069;
					set .@mapcount,getmapusers("bat_a01");
					if (.@mapcount > 0) {
						mes "[紀堯姆軍隊軍官]";
						mes "我收到一份報告，通知我一支精銳部隊已經被派往戰場與克羅伊軍隊作戰。";
						mes "你何不去等一會兒？我建議你在那之前磨礪你的武器並準備你的補給。";
					}
					else {
						if ($@TierraBG1) {
							mes "[紀堯姆軍隊軍官]";
							mes "一支精銳部隊已經待命，準備被派往戰場。";
							mes "你何不去等一會兒？我建議你在那之前磨礪你的武器並準備你的補給。";
						}
						else {
							mes "[紀堯姆軍隊軍官]";
							mes "你肯定看起來已經準備好戰鬥了！";
							mes "去向克羅伊軍隊展示恐懼的真正含義吧！";
							mes "今天，我們的勝利吶喊將響徹整個戰場！";
							close2;
							warp "bat_room",57,223;
							end;
						}
					}
				}
			}
		}
		break;
	case 2:
		mes "[紀堯姆軍隊軍官]";
		mes "今天，我們必將勝利！";
		break;
	}
	close;
}
