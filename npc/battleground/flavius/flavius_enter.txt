bat_room,133,178,5	script	Flavius Officer#01a	418,{
	if (getbattleflag("feature.bgqueue"))
		end;
	if (checkweight(1201,1) == 0) {
		mes "- 等等 !! -";
		mes "- 目前你攜帶了 -";
		mes "- 太多的物品. -";
		mes "- 請稍後再試 -";
		mes "- 在你減輕一些重量之後. -";
		close;
	}
	mes "[紀堯姆軍隊軍官]";
	mes "讓我們向那些臭氣熏天的克洛伊展示紀堯姆軍隊的力量！";
	next;
	switch(select("我想加入你們的軍隊！:結束對話")) {
	case 1:
		if ((BaseJob == Job_Novice) || (BaseJob == Job_Super_Novice)) {
			mes "[紀堯姆軍隊軍官]";
			mes "我很高興你願意加入我們並與克洛伊軍隊作戰，但很抱歉：我不能派像你這樣的新手去殘酷的戰場送死。";
		}
		else {
			if (BaseLevel < 80) {
				mes "[紀堯姆軍隊軍官]";
				mes "我很高興你願意加入我們並與克洛伊軍隊作戰，但很抱歉：我不能派像你這樣的新手去殘酷的戰場送死。";
			}
			else {
				set .@chk_urtime,checkquest(2070,PLAYTIME);
				if ((.@chk_urtime == 0)|| (.@chk_urtime == 1)) {
					mes "[紀堯姆軍隊軍官]";
					mes "你似乎剛從戰場上回來。";
					mes "現在回去還太早了。去休息吧，把敵人留給我們！";
				}
				else {
					if (.@chk_urtime == 2)
						erasequest 2070;
					set .@mapcount2,getmapusers("bat_b01");
					if (.@mapcount2 > 0) {
						mes "[紀堯姆軍隊軍官]";
						mes "我收到一份報告，通知我一支精銳部隊已經被派遣到戰場與克洛伊作戰了。";
						mes "你為什麼不等等呢？我建議你磨利你的武器，準備好你的補給品，直到那時。";
					}
					else {
						if ($@FlaviusBG1 > 0) {
							mes "[紀堯姆軍隊軍官]";
							mes "一支精銳部隊已經待命，準備被派遣到戰場。";
							mes "你為什麼不等等呢？我建議你磨利你的武器，準備好你的補給品，直到那時。";
						}
						else {
							mes "[紀堯姆軍隊軍官]";
							mes "你絕對看起來已經準備好戰鬥了！";
							mes "去向克洛伊展示恐懼的真正含義！";
							mes "今天，我們的勝利吶喊將響徹整個戰場！";
							close2;
							warp "bat_room",85,223;
							end;
						}
					}
				}
			}
		}
		break;
	case 2:
		mes "[紀堯姆軍隊軍官]";
		mes "今天，我們必將勝利！";
		break;
	}
	close;
}
