// Badge Reward Function
//============================================================
function	script	F_BG_Badge	{
	.@team$ = getarg(1);
	.@bg_type$ = getarg(2);

	if (.@bg_type$ == "Tierra") {
		set .@badge,7828; //BF_Badge1
		set .@amount_win,3;
		set .@amount_lose,1;
	} else if (.@bg_type$ == "Flavius") {
		set .@badge,7829; //BF_Badge2
		set .@amount_win,9;
		set .@amount_lose,3;
	} else
		return;

	if (getarg(0) == 1) {
		set .@amount, .@amount_win;
		if (.@team$ == "Guillaume") {
			mes "[艾克索·羅斯]";
			mes "蒙受吉永的祝福！";
		} else if (.@team$ == "Croix") {
			mes "[斯旺德里]";
			mes "蒙受克羅伊克斯的祝福！";
		}
		mes "讓我們享受我們光榮的勝利！";
		mes strcharinfo(0)+", 這是反映勝利的徵兆。";
	} else {
		set .@amount, .@amount_lose;
		if (.@team$ == "Guillaume") {
			mes "[艾克索·羅斯]";
			mes "你輸了，但你全心投入了這場戰鬥。";
			mes "這是吉永·馬洛洛對你偉大奉獻的獎勵！";
			mes "只要把這次失敗當作一次教訓，下次你一定會贏。";
		} else if (.@team$ == "Croix") {
			mes "[斯旺德里]";
			mes "喔，"+strcharinfo(0)+" 別難過。";
			mes "即使我們沒有贏，我們也盡了最大的努力。";
			mes "這是來自克羅伊克斯的皇家禮物，請不要忘記這場戰鬥。我們下次一定會贏。";
		}
	}
	close2;

	// If the VIP system is enabled, VIP players receive an extra 2 badges.
	if (VIP_SCRIPT && vip_status(VIP_STATUS_ACTIVE))
		set .@amount, .@amount+2;

	set .@medal_gap, 500 - countitem(.@badge);
	if (.@medal_gap >= .@amount)
		getitem .@badge,.@amount;
	else
		getitem .@badge,.@medal_gap;
	return;
}
