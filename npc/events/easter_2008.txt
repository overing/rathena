	mes "可以用來恢復";
	mes "你朋友的生命。";
	mes "你想要我為你";
	mes "製作一些嗎？";
	next;
	switch(select("製作聖蛋需要什麼？:我想製作聖蛋。:下次再說吧。")) {
	case 1:
		mes "[奈倫]";
		mes "對於每個你想要";
		mes "我製作的聖蛋";
		mes "，你需要";
		mes "帶給我...";
		next;
		mes "[奈倫]";
		mes "^4D4DFF1 白色藥草^000000，";
		mes "^4D4DFF1 綠色藥草^000000，";
		mes "^4D4DFF1 黃色藥草^000000，";
		mes "^4D4DFF1 聖水^000000，以及";
		mes "^4D4DFF1 雞蛋^000000。";
		close;
	case 2:
		mes "[奈倫]";
		mes "好的，你想要";
		mes "我為你製作";
		mes "多少個聖蛋？";
		mes "我一次最多只能製作";
		mes "10 個聖蛋。";
		mes "輸入 ''0'' 以取消。";
		next;
		while(1) {
			input .@input;
			if (.@input == 0) {
				mes "[奈倫]";
				mes "喔，你改變";
				mes "主意了嗎？隨時都可以來找我";
				mes "只要你想要我為你";
				mes "製作一些聖蛋～";
				close;
			}
			else if (.@input < 0 || .@input > 10) {
				mes "[奈倫]";
				mes "我很抱歉，但我一次";
				mes "最多只能製作 10 個聖蛋";
				mes "。如果你想";
				mes "取消，那就輸入 ''0''。";
				next;
			}
			else {
				if ((countitem(509) < .@input) || (countitem(511) < .@input) || (countitem(508) < .@input) || (countitem(523) < .@input) || (countitem(7605) < .@input)) {
					mes "[奈倫]";
					mes "我很抱歉，但你沒有";
					mes "足夠的物品來製作";
					mes "聖蛋。你需要以下";
					mes "的物品我才能為你";
					mes "製作每個聖蛋。";
					next;
					mes "[奈倫]";
					mes "^4D4DFF1 白色藥草^000000，";
					mes "^4D4DFF1 綠色藥草^000000，";
					mes "^4D4DFF1 黃色藥草^000000，";
					mes "^4D4DFF1 聖水^000000，以及";
					mes "^4D4DFF1 雞蛋^000000。";
					close;
				}
				else {
					break;
				}
			}
		}
		if (checkweight(12019,.@input) == 0) {
			mes "[奈倫]";
			mes "我很抱歉，但你";
			mes "現在無法攜帶";
			mes "那麼多的聖蛋...";
			mes "你最好先空出更多";
			mes "物品欄空間。";
			close;
		}
		else {
			delitem 509,.@input; //White_Herb
			delitem 511,.@input; //Green_Herb
			delitem 508,.@input; //Yellow_Herb
			delitem 523,.@input; //Holy_Water
			delitem 7605,.@input; //Egg
			getitem 12019,.@input; //Holy_Egg
			mes "[奈倫]";
			mes "謝謝！";
			mes "希望你喜歡";
			mes "這些聖蛋～";
			mes "希望你與上帝的愛同在。";
			close;
		}
	case 3:
		mes "[奈倫]";
		mes "好的，我明白了。";
		mes "願上帝保佑你。";
		close;
	}
}
