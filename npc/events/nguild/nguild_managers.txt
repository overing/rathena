//=   arg(3): y1 coordinate for warp to treasure room

function	script	F_GldManager	{
	.@npc_name$ = getarg(0);
	.@map_name$ = getarg(1);
	.@gid = getcastledata(.@map_name$,1);
	.@gm_name$ = getguildmaster(.@gid);

	mes "[ "+ .@npc_name$ +" ]";
	if (.@gid == 0){
		mes "我正在等待我的主人。勇敢的冒險者，追隨你的命運吧！";
		return;
	}
	if (getcharid(2) != .@gid){
		mes "我是來遵從 ^5533FF" + getguildmaster(.@gid) + "^000000 的命令！嘿！你甚至不是公會的一份子！！";
		mes "守護者在哪裡？消滅這些入侵者！";
		return;
	}
	if (is_guild_leader() == false){
		mes "你不是 ^5533FF" + getguildmaster(.@gid) + "^000000！我只會遵從 ^5533FF" + getguildmaster(.@gid) + "^000000 的命令";
		return 0;
	}

	mes "歡迎主人 ^5533FF" + getguildmaster(.@gid) + "^000000！我會盡我所能幫助您！";
	next;

	menu "Kafra Staff Employment / Dismissal",M_Kaf, "Enter Treasure Room",M_Treas, "Cancel",M_End;

	//===========================
	M_Kaf:
	//======
		mes "[ "+ .@npc_name$ +" ]";
		if (GetCastleData(.@map_name$,9) == 1) goto L_Dismiss;
		if (getgdskilllv(.@gid,10001) == 0){
			mes "主人，您沒有與卡普拉員工公司的合約。";
			mes "為了雇用卡普拉，您必須先學習公會技能 ^5533FF與卡普拉簽訂合約^000000。";
			return;
		}

		L_Hire:
			mes "您想要雇用卡普拉嗎？您需要 ^5533FF10,000 Zeny^000000 才能這麼做... ";
			next;
			menu "Employ Kafra.",-,"Cancel",sM_KafEnd;

				mes "[ "+ .@npc_name$ +" ]";
				if (Zeny < 10000){
					mes "主人，您沒有足夠的錢雇用卡普拉。雇用已取消。";
					return;
				}
				set Zeny,Zeny-10000;
				enablenpc "卡普拉工作人員#"+getarg(4);
				SetCastleData .@map_name$,9,1;
				mes "您已經與卡普拉員工公司建立了合約。";
				next;
				cutin "kafra_01",2;
				mes "[ 卡普拉員工 ]";
				mes "您好！我來為您提供有用的服務！我會盡我所能為您服務。";
				next;
				cutin "kafra_01",255;
				mes "[ "+ .@npc_name$ +" ]";
				//mes "您的雇用合約為期 ^5533FF1 month^000000。在這段期限結束後，您將需要建立新的合約。";
				mes "我認為卡普拉員工將使我們的公會成員受益。";
				return;

			sM_KafEnd:
				mes "[ "+ .@npc_name$ +" ]";
				mes "如您所願，主人。但我建議我們盡快找一位卡普拉！";
				return;

		L_Dismiss:
			mes "您想要解僱目前的卡普拉嗎？";
			next;
			menu "Dismissal",-,"Cancel",sM_KafEnd2;

				cutin "kafra_01",2;
				mes "[ 卡普拉員工 ]";
				mes "我做錯了什麼嗎？如果有的話，請您原諒我好嗎？";
				next;
				menu "Dismiss",-,"Cancel",ssM_KafEnd2;

					mes "[ 卡普拉員工 ]";
					mes "很遺憾我無法再為您的公會服務了....";
					next;
					disablenpc "卡普拉工作人員#"+getarg(4);
					SetCastleData .@map_name$,9,0;
					cutin "kafra_01",255;
					mes "[ "+ .@npc_name$ +" ]";
					mes "卡普拉已被解僱。但是...我們真的應該盡快找一位卡普拉！";
					return;
				ssM_KafEnd2:
					mes "[ 卡普拉員工 ]";
					mes "謝謝您，主人，我會盡我所能！^^。";
					cutin "kafra_01",255;
					return;
			sM_KafEnd2:
				mes "[ "+ .@npc_name$ +" ]";
				mes "主人，我認為您應該保留目前的卡普拉員工，因為她已經盡力為我們服務了";
				return;

	//=========================
	M_Treas:
	//========
		mes "[ "+ .@npc_name$ +" ]";
		mes "您想要前往我們的寶物室嗎？只有您，公會會長，才被允許進入這個房間。";
		next;
		menu "Enter Treasure room.",-,"Cancel",sM_TresEnd;

			mes "[ "+ .@npc_name$ +" ]";
			mes "請跟隨我穿過秘密通道。";
			mes "您必須拉下秘密開關才能出去。";
			next;
			warp .@map_name$,getarg(2),getarg(3);
			return;
		sM_TresEnd:
			mes "[ "+ .@npc_name$ +" ]";
			mes "貨物每天都會生產。";
			mes "您應該盡可能地取得它們，因為如果您在錯誤的時間拿取它們，它們可能會消失。";
			return;


	//==========================
	M_End:
	//=======
		mes "[ "+ .@npc_name$ +" ]";
		mes "如您所願，主人。";
		return;
}
