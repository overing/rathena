mosk_in,21,246,5	script	Pinkamenia	71,{
	set .@n$,"^0000FF[Pinkamenia]^000000";
	if (#V_QUE12==3) {
		set .@LFlowerID,7864;
		setarray .@SweetsID[0],538,558,539,573,559,560,12062,596,597,12414,12319;
		setarray .@SweetsCost[0],1,2,5,10,10,10,15,15,15,20,20;
		mes .@n$;
		mes "哈囉 "+strcharinfo(0)+", 你";
		mes "想要買些東西嗎？";
		mes "你必須給我愛心";
		mes "花朵來交換，當然！";
		next;
		if(select("是的，當然！","不，沒關係。") == 2) close;
		next;
		mes .@n$;
		mes "選擇一個物品。";
		mes "你需要的愛心花朵數量在括號裡。";
		for(set .@i,0; .@i < getarraysize(.@SweetsID); set .@i,.@i+1)
			set .@menu$, .@menu$+"^00AA00["+.@SweetsCost[.@i]+"]^000000 "+getitemname(.@SweetsID[.@i])+":";
		next;
		set .@choice, select(.@menu$)-1;
		mes .@n$;
		if (countitem(.@LFlowerID) < .@SweetsCost[.@choice]) {
			mes "如果你想要那個，你需要更多的愛心花朵！";
			close;
		}
		mes "你確定要用 ^00aa00"+.@SweetsCost[.@choice]+"個愛心花朵 ^000000 交換 ^0055FF"+getitemname(.@SweetsID[.@choice])+"^000000 嗎？";
		if(select("不，我改變主意了。:是的，交換！") == 1) close;
		set .@new, .@LFlowerID - .@SweetsCost[.@choice];
		if (.@new < 0) set .@new, 0;
		delitem .@LFlowerID, .@SweetsCost[.@choice];
		getitem .@SweetsID[.@choice],1;
		mes "祝你使用物品愉快！";
		close;
	} else if (#V_QUE12==2) {
		mes .@n$;
		if (countitem(7864) < 15) {
			mes "你必須帶給我 15 朵愛心花朵！";
			close;
		}
		mes "非常感謝你";
		mes "幫我們找回花朵！";
		delitem 7864,15;
		next;
		mes .@n$;
		mes "拿去，這是一點點";
		mes "『謝禮』。";
		getexp 500000,400000;
		getitem 617,1;
		getitem 12319,2;
		next;
		mes .@n$;
		mes "如果你得到更多的愛心";
		mes "花朵，你可以來這裡";
		mes "換取一些甜點。";
		mes "再見！";
		set #V_QUE12,3;
		close;
	} else if (#V_QUE12==1) {
		mes .@n$;
		mes "請將 +8 蛋糕帽";
		mes "拿給站在我旁邊的";
		mes "蛋糕帽製作大師！";
		close;
	} else if (BaseLevel >= 45) {
		mes .@n$;
		mes "哈囉 "+strcharinfo(0)+", 現在";
		mes "是情人節，我們";
		mes "喜歡傳遞甜蜜的";
		mes "巧克力。";
		next;
		mes .@n$;
		mes "我們的問題是我們";
		mes "需要像你一樣強壯的";
		mes "人來幫助";
		mes "我們，但首先你必須";
		mes "將 +8 蛋糕帽";
		mes "拿給蛋糕帽製作大師，他";
		mes "就站在我旁邊！";
		set #V_QUE12,1;
		close;
	}
	mes .@n$;
	mes "哈囉 "+strcharinfo(0)+"!";
	close;

OnInit:
	// Unknown item drop rate (currently set at 5%)
	setitemscript 5105,"{ bonus bDex,1; bonus bMaxSP,80; bonus3 bAddMonsterDropItem,7864,RC_DemiHuman,500; }",0;
	end;
}

			mes "帽給我。現在我要";
			mes "將你的 +8 蛋糕";

		}
	} else {

		close;
	}
}
