prontera,147,61,3	script	鑽孔師#pron	87,{
	disable_items;
	if (checkweight(1201,1) == 0) {
		mes "你攜帶太多物品在你的背包裡了。去卡普拉倉庫整理一下再試一次。";
		close;
	}
	if (MaxWeight - Weight < 10000) {
		mes "你因為超重而無法繼續。";
		close;
	}
	set .@part, EQI_ACC_L;
	if (!getequipisequiped(.@part)) {
		mes "[鑽孔師]";
		mes "我的工作是在 RWC 紀念飾品上鑽一個卡片插槽。";
		next;
		mes "[鑽孔師]";
		mes "很抱歉，但你沒有在右邊飾品欄位裝備任何物品。";
		close;
	}
	mes "[鑽孔師]";
	mes "我的工作是在 RWC 紀念飾品上鑽一個卡片插槽。此外，我只處理 ^ff0000純淨的物品^000000，或是那些沒有被附魔過的。";
	next;
	set .@equip_id, getequipid(.@part);
	if (.@equip_id != 2966 && .@equip_id != 2968) {
		mes "[鑽孔師]";
		mes "然而，我發現你現在戴在右側的飾品無法被處理。請裝備一個 RWC 紀念飾品。";
		close;
	}
	mes "[鑽孔師]";
	mes "你也應該知道製作卡片插槽是非常危險的。^ff0000成功的機率大約是 50%。^000000 你想要繼續嗎？";
	next;
	if(select("取消:算了吧！") == 1) {
		mes "[鑽孔師]";
		mes "下次見囉。";
		close;
	}
	if (.@equip_id == 2966) {
		set .@slotted, 2967; //RWC_2012_Ring_
		set .@name$,"RWC 2012 Memorial Ring";
		set .@str$,"ring";
	} else if (.@equip_id == 2968) {
		set .@slotted, 2969; //RWC_2012_Pendant_
		set .@name$,"RWC 2012 Memorial Pendant";
		set .@str$,"pendant";
	} else {
		mes "[鑽孔師]";
		mes "我無法識別你戴在右手邊的飾品。我無法對它進行作業。";
		close;
	}
	if (getequipcardid(.@part,3) > 0) {
		mes "[鑽孔師]";
		mes "這個物品已經被附魔了。我不能對它進行作業，因為這違反了規定。";
		close;
	}

	// anti-hack
	if (callfunc("F_IsEquipIDHack", .@part, .@equip_id))
		close;

	delequip .@part;
	if (rand(1,10) > 5) {
		getitem .@slotted,1;
		specialeffect2 EF_REPAIRWEAPON;
		mes "[鑽孔師]";
		mes "耶！成功了！你的 "+.@name$+" 現在有一個卡片插槽了。檢查一下吧！";
		close;
	} else {
		specialeffect2 EF_LORD;
		mes "[鑽孔師]";
		mes "唉... 該死的弱 "+.@str$+"... 在過程中壞掉了。我很抱歉。";
		close;
	}
}

prontera,147,59,3	script	高柏格#pron	878,{
	disable_items;
	if (checkweight(1201,1) == 0) {
		mes "你攜帶太多物品了。整理好你的背包後再來吧。";
		close;
	}
	if (MaxWeight - Weight < 10000) {
		mes "你因為超重而無法繼續。";
		close;
	}
	mes "[高柏格]";
	mes "哈囉！我負責用一些神秘力量來附魔 RWC 紀念飾品。";
	next;
	set .@part, EQI_ACC_L;
	if (!getequipisequiped(.@part)) {
		mes "[高柏格]";
		mes "很抱歉，但你沒有在右邊飾品欄位裝備任何物品。";
		close;
	}
	set .@equip_id, getequipid(.@part);
	if (.@equip_id < 2966 || .@equip_id > 2969) {
		mes "[高柏格]";
		mes "然而，我發現你現在戴的飾品不是我可以處理的東西。請裝備一個 RWC 紀念飾品。";
		close;
	}
	set .@select, select("抱歉，沒興趣。:請賦予我的飾品力量。:移除附魔。")-1;
	if (.@select == 0) {
		mes "[高柏格]";
		mes "好吧，下次見了...";
		close;
	}
	setarray .@equip_card[0], getequipcardid(.@part,0),getequipcardid(.@part,1),getequipcardid(.@part,2),getequipcardid(.@part,3);
	if (.@select == 1) {
		switch(getequipid(.@part)) {
		case 2966:
			setarray .@option[0],2,2,4,4;
			break;
		case 2967:
			setarray .@option[0],0,2,4,4;
			break;
		case 2968:
			setarray .@option[0],1,1,3,3;
			break;
		case 2969:
			setarray .@option[0],0,1,3,3;
			break;
		default:
			mes "[高柏格]";
			mes "很抱歉，但我無法對你目前配戴的飾品進行作業。";
			mes "如果你將 RWC 紀念飾品裝備在左側，請嘗試將其交換到右側。";
			close;
		}
		for(set .@i,3; .@i>=0; set .@i,.@i-1) {
			if (.@equip_card[.@i] == 0) {
				set .@slot, .@i;
				set .@op_type, .@option[.@i];
				break;
			}
		}
		switch(.@op_type) {
		case 4:
			mes "[高柏格]";
			mes "你想要注入哪種附魔？";
			next;
			setarray .@enchant_select[0],1,2,3,4;
			set .@i, select("取消:鬥志:ATK (%):Max HP:HP")-2;
			break;
		case 3:
			setarray .@enchant_select[0],5,6,7;
			set .@i, select("取消:法術:MATK (%):SP")-2;
			break;
		case 2:
			mes "[高柏格]";
			mes "^ff0000小心！附魔失敗的機率約為 25%。如果發生這種情況，物品將會被摧毀。^000000 你想要注入哪種附魔？";
			next;
			setarray .@enchant_select[0],8,9,10,11,12,13,14;
			set .@i, select("取消:STR:AGI:VIT:INT:DEX:LUK:SP")-2;
			break;
		case 1:
			mes "[高柏格]";
			mes "^ff0000附魔失敗的機率約為 25%。如果發生這種情況，物品將會被摧毀。^000000 你想要注入哪種附魔？";
			next;
			setarray .@enchant_select[0],8,9,10,11,12,13,15,16;
			set .@i, select("取消:STR:AGI:VIT:INT:DEX:LUK:MHP:HP")-2;
			break;
		case 0:
			mes "[高柏格]";
			mes "你的飾品已經被附魔太多次了，我幾乎無法再對它進行作業了。";
			close;
		}
		if (.@i == -1) {
			mes "[高柏格]";
			mes "好吧，下次見了。";
			close;
		}
		mes "[高柏格]";
		mes "附魔的力量將會隨機選擇。^ff0000一旦注入，附魔將無法移除。^000000 我們要繼續嗎？";
		next;
		if(select("不，請停止。:是的，請繼續。") == 1) {
			mes "[高柏格]";
			mes "好吧，下次見了...";
			close;
		}
		set .@enchant_type, .@enchant_select[.@i];
		if (!getequipisequiped(.@part)) {
			mes "[高柏格]";
			mes "我在工作時不要脫掉你的裝備，好嗎？";
			close;
		}
		switch(.@enchant_type) {
		case 1:
			setarray .@enc[0],4811,4810,4809; //Fighting_Spirit1,Fighting_Spirit2,Fighting_Spirit3
			break;
		case 2:
			setarray .@enc[0],4819,4766,4767; //Atk1,Atk2,Atk3
			break;
		case 3:
			setarray .@enc[0],4861,4862,4867; //MHP1,MHP2,MHP3
			break;
		case 4:
			setarray .@enc[0],4795,4796,4797; //HP100,HP200,HP300
			break;
		case 5:
			setarray .@enc[0],4760,4761,4806; //Matk1,Matk2,Matk3
			break;
		case 6:
			setarray .@enc[0],4815,4814,4813; //Spell1,Spell2,Spell3
			break;
		case 7:
			setarray .@enc[0],4870,4800,4871; //SP25,SP50,SP75
			break;
		case 8:
			setarray .@enc[0],4700,4701,4702; //Strength1,Strength2,Strength3
			break;
		case 9:
			setarray .@enc[0],4730,4731,4732; //Agility1,Agility2,Agility3
			break;
		case 10:
			setarray .@enc[0],4740,4741,4742; //Vitality1,Vitality2,Vitality3
			break;
		case 11:
			setarray .@enc[0],4710,4711,4712; //Inteligence1,Inteligence2,Inteligence3
			break;
		case 12:
			setarray .@enc[0],4720,4721,4722; //Dexterity1,Dexterity2,Dexterity3
			break;
		case 13:
			setarray .@enc[0],4750,4751,4752; //Luck1,Luck2,Luck3
			break;
		case 14:
			setarray .@enc[0],4870,4800,4871; //SP25,SP50,SP75
			break;
		case 15:
			setarray .@enc[0],4861,4862,4867; //MHP1,MHP2,MHP3
			break;
		case 16:
			setarray .@enc[0],4795,4796,4797; //HP100,HP200,HP300
			break;
		default:
			mes "[高柏格]";
			mes "我們遇到問題了，讓我檢查一下。";
			close;
		}

		if (.@enchant_type < 8)
			set .@i, rand(1,300); // 0% break chance.
		else
			set .@i, rand(1,400); // 25% break chance.

		     if (.@i < 151) set .@enchant, .@enc[0];
		else if (.@i < 251) set .@enchant, .@enc[1];
		else if (.@i < 301) set .@enchant, .@enc[2];
		else set .@enchant,9;

		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@part, .@equip_id) ||
		    callfunc("F_IsEquipCardHack", .@part, .@equip_card[0], .@equip_card[1], .@equip_card[2], .@equip_card[3]))
			close;

		set .@equip_card[.@slot], .@enchant;
		if (.@slot == 2 && .@enchant == 0) {
			set .@equip_card[3],0;
		} else if (.@slot == 1 && .@enchant == 0) {
			set .@equip_card[2],0;
			set .@equip_card[3],0;
		} else if (.@slot == 0 && .@enchant == 0) {
			set .@equip_card[1],0;
			set .@equip_card[2],0;
			set .@equip_card[3],0;
		}

		set .@equip_refine, getequiprefinerycnt(.@part);
		delequip .@part;
		if (.@enchant == 9) {
			mes "[高柏格]";
			mes "噢，天啊！";
 			mes "這個物品不夠堅固，無法承受附魔，因此被摧毀了。我很抱歉。";
			specialeffect2 EF_LORD;
			close;
		}
		if (.@enchant == 0) { // Should never happen.
			mes "[高柏格]";
			mes "喔... 看起來所有注入的力量之間存在某種不穩定性。這導致所有附魔都消失了。很可惜，但請再試一次！";
		} else {
			mes "[高柏格]";
			mes "太棒了！";
			mes "附魔成功了！它將被應用在插槽編號 ^990000"+(.@slot+1)+"^000000。";
			specialeffect2 EF_REPAIRWEAPON;
		}

//		GetNonSlotItemSock2 .@equip_refine .@equip_id .@equip_card[0] .@equip_card[1] .@equip_card[2] .@equip_card[3]
		getitem2 .@equip_id,1,1,.@equip_refine,0,.@equip_card[0],.@equip_card[1],.@equip_card[2],.@equip_card[3];

		close;
	} else if (.@select == 2) {
		mes "[高柏格]";
		mes "我將僅初始化附魔選項，而不會對已插入的卡片做任何事情。你想繼續嗎？";
		next;
		if(select("我會停止。:是的，確定，繼續。") == 1) {
			mes "[高柏格]";
			mes "如果你改變主意，就回來吧。";
			close;
		}
		if (countitem(6665) == 0) {
			mes "[高柏格]";
			mes "我很抱歉。但是你沒有 RWC 初始化優惠券。你可以檢查一下你的背包嗎？";
			close;
		}
		if (getiteminfo(.@equip_card[3], ITEMINFO_SUBTYPE) != CARD_ENCHANT) {
			mes "[高柏格]";
			mes "嗯... 這個裝備是乾淨的。如果沒有東西，我就無法初始化它！再檢查一次。";
			close;
		}
		specialeffect2 EF_REPAIRWEAPON;
		mes "[高柏格]";
		mes "你物品上的附魔選項將被初始化。";
		delitem 6665,1; //RWC_Inicializer
		delequip .@part;
	
//		GetNonSlotItemSock2 .@equip_refine .@equip_id .@equip_card[0] .@equip_card[1] .@equip_card[2] .@equip_card[3]
		for ( .@i = getiteminfo(.@equip_id, ITEMINFO_SLOT); .@i < MAX_SLOTS; .@i++ ) {
			if (getiteminfo(.@equip_card[.@i], ITEMINFO_SUBTYPE) == CARD_ENCHANT)
				.@equip_card[.@i] = 0;// Armor Enchant System
		}
		getitem2 .@equip_id,1,1,.@equip_refine,0,.@equip_card[0],.@equip_card[1],.@equip_card[2],.@equip_card[3];
		
		close;
	}
}
