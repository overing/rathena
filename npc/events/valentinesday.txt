// Stephen - Valentine Event Chocolate seller
//============================================================
alberta,26,243,4	script	Stephen	58,{
	mes "[史蒂芬]";
	mes "猜猜我得到了什麼？";
	mes "一種在米德加爾特不容易找到的美味點心....";
	next;
	mes "[史蒂芬]";
	mes "巧克力！";
	mes "沒錯，你不喜歡巧克力嗎....我很喜歡。";
	mes "而且你很幸運，因為我現在只賣 5,000 zeny 一個！";
	next;
	if(select("我想要一些巧克力！:不用了，謝謝。")==2) {
		mes "[史蒂芬]";
		mes "你不想來點巧克力？";
		mes "我跟你說！你會後悔的！";
		mes "你最好現在買一些...你不會再遇到像這樣的巧克力了！";
		mes "好好考慮一下，有空再來找我。";
		close; }
	mes "[史蒂芬]";
	mes "哈！";
	mes "我就知道！";
	mes "但我一次最多只能賣你 5 個...不過，如果你真的很需要更多....";
	mes "你可以再來找我。";
	mes "那麼你想要幾個？";
	next;
	input .@i;
	if (.@i <= 0) close;
	mes "[史蒂芬]";
	if (.@i > 5)  {
		mes "抱歉，我不能給你這麼多。";
		close; }
	if (Zeny < .@i*5000) {
		mes "抱歉，但看來你買不起我的東西。";
		close; }
	set Zeny, Zeny - .@i*5000;
	getitem 558,.@i;
	mes "給你！";
	mes "你可以把它當作禮物送給別人，或者自己享用！";
	mes "嗯... 香甜的巧克力...";
	mes "隨時再來找我...！";
	close;
}
