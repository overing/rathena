payon,162,176,4	script	萬聖節魔術師#iRO08	704,{
	if (Hallow08 < 1) {
		mes "[萬聖節魔術師]";
		mes "Kkkkkkkkk!";
		mes "今年萬聖節我有一個特別的活動，測試你的運氣和敏捷。";
		mes "你有興趣嗎？";
		next;
		mes "[萬聖節魔術師]";
		mes "拜託！別當個孬種！";
		mes "如果你收集到足夠的票券，就可以獲得很棒的獎品！";
		mes "所以你怎麼說？";
		next;
		goto MainMenu;
	}
	if (Hallow08Kill == 1) {
		mes "[萬聖節魔術師]";
		mes "你是膽小鬼之類的嗎？！";
		mes "你知道你想要再試一次...";
		mes "你知道規則嗎？";
		next;
		set Hallow08Kill,0;
		if(select("是的，我知道。:不，我不知道。") == 2) {
			goto Rules;
		}
		goto Participate;
	}
	if (Hallow08Kill == 2) {
		mes "[萬聖節魔術師]";
		mes "喔，做得好！你還不錯嘛！";
		mes "和殭屍在一起是不是很有趣？";
		mes "你知道嗎，殭屍也是人啊！";
		mes "哈！";
		mes "Kkkkkkk.";
		next;
		getitem 7941,1;
		set Hallow08Kill,0;
		mes "[萬聖節魔術師]";
		mes "正如我承諾的";
		mes "你可以用萬聖節票券換取酷炫的物品。";
		close;
	}
	if (Hallow08 > 0) {
		mes "[萬聖節魔術師]";
		mes "嗯，你想要再次聽聽規則，還是直接開始...";
		next;
		goto MainMenu2;
	}

		Rules:
			mes "[萬聖節魔術師]";
			mes "這個村莊就像一個虛擬的沛雍。";
			mes "有殭屍和食屍鬼在四處遊蕩，以及三個南邊的出口，但只有一個有效。";
			mes "這取決於你來找出它。";
			next;
			mes "[萬聖節魔術師]";
			mes "你不能使用任何技能來殺死食屍鬼或殭屍。";
			mes "還有一件事...";
			mes "你不應該忘記...";
			next;
			mes "[萬聖節魔術師]";
			mes "所有參與者都應該什麼都不穿。";
			mes "將所有物品放在你的倉庫中，當你的重量為 '0' 時回到這裡。";
			next;
			mes "[萬聖節魔術師]";
			mes "喔，還有一件事！";
			mes "你不能騎著波利或擁有手推車。";
			mes "如果你有，我會在您進入之前將它們移除。";
			mes "明白了嗎？";
			next;
			mes "[萬聖節魔術師]";
			mes "記住，有三個出口但只有一個隨機有效，在那裡四處遊蕩的殭屍和食屍鬼無法被殺死，而且你不能穿任何東西。";
			next;
			if (Hallow08 > 0) {
				mes "[萬聖節魔術師]";
				mes "嘿...";
				mes "準備好後再回來。";
				close;
			} else {
				goto MainMenu;
			}

		Participate:
			mes "[萬聖節魔術師]";
			mes "好的，你準備好了。";
			mes "讓我檢查你的重量。";
			next;
			if (Weight > 0) {
				mes "[萬聖節魔術師]";
				mes "哎呀！";
				mes "任何地方總會有一隻害群之馬。";
				next;
				mes "[萬聖節魔術師]";
				mes "你認為我不會注意到你的重量高於 '0' 嗎？";
				mes "你超重了...";
				close;
			} else {
				mes "[萬聖節魔術師]";
				mes "你似乎準備好了，你的重量剛剛好。";
				next;
				mes "[萬聖節魔術師]";
				mes "我工作太忙了，所以我很累。";
				next;
				mes "[萬聖節魔術師]";
				mes "我偶爾會忘記送你過去...";
				next;
				mes "[萬聖節魔術師]";
				mes "我希望你能平安歸來。";
				close2;
				if (Hallow08 < 1) {
					set Hallow08,1;
				}
				set Hallow08Kill,1;
				set @Hallow08Warp, rand(1,3);
				percentheal -98,0;
				setriding 0;
				setcart 0;
				warp "evt_zombie",155,246;
				end;
			}

		TicketExchange:
			mes "[萬聖節魔術師]";
			mes "你想用票券兌換獎品嗎？";
			mes "做得好！Kkkkkk!";
			next;
			mes "[萬聖節魔術師]";
			mes "讓我告訴你可以兌換哪些物品。";
			next;
			mes "[萬聖節魔術師]";
			mes "5 張票券換南瓜派。";
			mes "20 張票券換南瓜頭。";
			mes "50 張票券換舊藍色箱子。";
			mes "70 張票券換舊紫色箱子。";
			mes "200 張票券換舊卡冊。";
			next;
			mes "[萬聖節魔術師]";
			mes "你想要兌換什麼？";
			next;
			switch(select("南瓜派:南瓜頭:舊藍色箱子:舊紫色箱子:舊卡冊")) {
			case 1:
				if (countitem(7941) < 5) {
					goto NotEnough;
				} else {
					delitem 7941,5;
					getitem 12192,1;
					goto Enough;
				}
				break;
			case 2:
				if (countitem(7941) < 20) {
					goto NotEnough;
				} else {
					delitem 7941,20;
					getitem 5134,1;
					goto Enough;
				}
				break;
			case 3:
				if (countitem(7941) < 50) {
					goto NotEnough;
				} else {
					delitem 7941,50;
					getitem 603,1;
					goto Enough;
				}
				break;
			case 4:
				if (countitem(7941) < 70) {
					goto NotEnough;
				} else {
					delitem 7941,70;
					getitem 617,1;
					goto Enough;
				}
				break;
			case 5:
				if (countitem(7941) < 200) {
					goto NotEnough;
				} else {
					delitem 7941,200;
					getitem 616,1;
					goto Enough;
				}
			}

		NextTime:
			mes "[萬聖節魔術師]";
			mes "好的，再見。";
			mes "Kkkkkkkk.";
			close;

	MainMenu:
		menu "Explain it to me.",Rules,"I want to participate.",Participate,"I want to exchange tickets for prizes.",TicketExchange,"I'll come back next time.",NextTime;

	MainMenu2:
		menu "Get me back there now!",Participate,"Please, tell me the rules",Rules,"I want to exchange tickets for prizes.",TicketExchange,"I'll come back next time.",NextTime;

	NotEnough:
		mes "[萬聖節魔術師]";
		mes "你的票券不足！";
		mes "你甚至不會數數嗎？";
		mes "請帶著正確數量的票券來這裡。";
		close;

	Enough:
		mes "[萬聖節魔術師]";
		mes "給你。";
		mes "你還需要...";
		mes "別的東西嗎？";
		next;
		if (Hallow08 == 1) {
			goto MainMenu2;
		} else {
			goto MainMenu;
		}
}
