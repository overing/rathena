payon_in01,193,30,4	script	米店老奶奶#rat	78,{
	mes "[米店老奶奶]";
	if (!lunar_rat) {
		mes "那些骯髒的小";
		mes "東西！偷偷摸摸的";
		mes "到處亂竄，偷我的";
		mes "東西！如果牠們沒";
		mes "那麼快就好了...";
		next;
		select("怎麼了？");
		mes "[米店老奶奶]";
		mes "喔，我只是在做";
		mes "一些麻糬和";
		mes "糕點來慶祝";
		mes "新年，但是這些";
		mes "動物一直在偷";
		mes "我準備好的米袋。";
		next;
		select("米袋？");
		mes "[米店老奶奶]";
		mes "是的，我兒子需要那些";
		mes "米袋來將米";
		mes "在石臼中搗碎，但如果我";
		mes "連米都沒有，就沒辦法";
		mes "做糕點了。你";
		mes "覺得你可以幫我嗎？";
		next;
		if (select("當然可以。:我甚至不認識妳。") == 1) {
			set lunar_rat,1;
			mes "[" + strcharinfo(0)+"]";
			mes "當然可以。";
			mes "我能做什麼？";
			next;
			mes "[米店老奶奶]";
			mes "嗯，你覺得你";
			mes "可以抓住那些";
			mes "偷走我米袋的動物嗎？";
			mes "牠們是藍色和";
			mes "白色的老鼠和這些";
			mes "討厭的小鼴鼠。";
			next;
			select("沒問題。");
			mes "[" + strcharinfo(0)+"]";
			mes "沒問題。";
			mes "我會抓住那些動物，";
			mes "並帶回任何我能找到的";
			mes "米袋。";
			next;
			mes "[米店老奶奶]";
			mes "喔，謝謝你！";
			mes "祝你好運，幫我";
			mes "抓住那些害蟲～";
			close;
		}
		mes "[米店老奶奶]";
		mes "我... 嗯...";
		mes "那是真的，但我";
		mes "打算給你一些好";
		mes "東西來回報你的";
		mes "幫助。我知道你不是";
		mes "壞人，所以...";
		close;
	}
	else if ((lunar_rat >= 1 && lunar_rat <= 2) || lunar_rat == 4) {
		if (countitem(7770) >= 1 && lunar_rat == 2) {
			mes "喔，那是";
			mes "我兒子做的甜米嗎？";
			mes "你能讓我用一下嗎？";
			mes "請在這裡稍等一下，";
			mes "我會給你做一些糕點～";
			delitem 7770,1; //Sweet_rice
			set lunar_rat,3;
			close;
		}
		else if (lunar_rat == 4) {
			if (countitem(7770) >= 1) {
				mes "喔，那是";
				mes "我兒子做的甜米嗎？";
				mes "你能讓我用一下嗎？";
				mes "請在這裡稍等一下，";
				mes "我會給你做一些糕點～";
				delitem 7770,1; //Sweet_rice
				set lunar_rat,5;
				close;
			}
			mes "喔，所有的";
			mes "米袋在哪裡？我們需要";
			mes "牠們來做更多的麻糬...";
			close;
		}
		else if (countitem(7869) < 1 && lunar_rat <= 2) {
			mes "喔，那些白色和";
			mes "藍色的老鼠可能";
			mes "偷了我的米袋";
			mes "那些鼴鼠可能也拿走了";
			mes "牠們。真是討厭的小";
			mes "東西，不是嗎？";
			close;
		}
		mes "喔，那是我的";
		mes "米袋嗎？我很抱歉，";
		mes "但我屁股很痛...";
		mes "你能幫我一個忙，";
		mes "把它交給我";
		mes "兒子嗎？";
		set lunar_rat,2;
		close;
	}
	else if (lunar_rat == 3 || lunar_rat == 5) {
		set .@reward,rand(1,100);
		if (.@reward <= 5) {
			if (lunar_rat == 3) {
				getitem 9038,1; // New_Year_Doll_Egg
				set lunar_rat,4;
			}
			else if (lunar_rat == 5) {
				getitem 668,1; // Red_Envelope
			}
		}
		else if (.@reward >= 6 && .@reward <= 10) getitem 12198,2; // Lucky_Rice_Cake
		else if (.@reward >= 11 && .@reward <= 30) getitem 12195,3; // Plain_Rice_Cake
		else if (.@reward >= 31 && .@reward <= 40) getitem 12196,2; // Hearty_Rice_Cake
		else if (.@reward >= 41 && .@reward <= 55) getitem 12123,2; // Honey_Pastry
		else if (.@reward >= 56 && .@reward <= 70) getitem 12122,2; // Sesame_Pastry
		else if (.@reward >= 71 && .@reward <= 80) getitem 12124,2; // Rainbow_Cake
		else if (.@reward >= 81 && .@reward <= 90) getitem 12198,2; // Korean_Rice_Cake
		else if (.@reward >= 91 && .@reward <= 100) getitem 12197,3; // Salty_Rice_Cake
		// After getting the egg there is a chance to get an envelope with each turn in
		if (rand(1,20) <= 5 && lunar_rat == 4) getitem 668,1; // Red_Envelope
		// Set the quest accordingly for repeat turn-ins.
		if (lunar_rat == 3) set lunar_rat,2;
		else if (lunar_rat == 5) set lunar_rat,4;
		mes "這遠遠不足以";
		mes "回報你為我所做的";
		mes "一切，但我希望";
		mes "你能擁有我";
		mes "剛做的這個糕點。請享用～";
		next;
		mes "[米店老奶奶]";
		mes "謝謝你的";
		mes "甜米！我一定會";
		mes "為你做一些";
		mes "美味的東西，如果你";
		mes "再給我帶來更多，好嗎？";
		close;
	}
}
