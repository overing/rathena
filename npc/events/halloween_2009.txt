prontera,152,192,5	script	南瓜帽研究員	712,{

		mes "[南瓜帽研究員]";

		mes "[南瓜帽研究員]";

	}

	mes "你喜歡南瓜派嗎？";
	next;
	mes "[南瓜帽研究員]";
	mes "我是南瓜帽研究員，你為什麼不聽聽我的故事？";
	next;
	while(1) {
		switch(select("聽故事.:詢問南瓜帽.:取得南瓜帽.:停止對話.")) {
			case 1:
				mes "[南瓜帽研究員]";
				mes "我一直在研究升級版的南瓜帽。";
				mes "我發現這是一個非常簡單的過程。";
				next;
				mes "[南瓜帽研究員]";
				mes "這個過程相當簡單。";
				mes "如果你給我 ^4a4aff20 傑克南瓜^000000，我可以展示給你。";
				mes "這不是很吸引人的提議嗎？";
				next;
				break;
			case 2:
				mes "[南瓜帽研究員]";
				mes "這個升級版的南瓜帽非常強大！";
				next;
				mes "[南瓜帽研究員]";
				mes "它可以製作南瓜派，使用濃縮能量來恢復配戴者大量的 HP 和 SP。";
				next;
				mes "[南瓜帽研究員]";
				mes "你只需要 ^4a4aff20 傑克南瓜^000000s。";
				next;
				break;
			case 3:
				mes "[南瓜帽研究員]";
				mes "你想要取得南瓜派嗎？好的，讓我來數數你帶來的傑克南瓜。";
				next;
				if(countitem(1062) < 20) {
					mes "[南瓜帽研究員]";
					mes "我需要 ^4a4aff20 傑克南瓜^000000。";
					mes "我又不是煉金術士或巫師，無法憑空創造東西。";
					next;
					mes "[南瓜帽研究員]";
					mes "好的，去狩獵怪物然後回來。";
					mes "我會在這裡待一陣子，所以慢慢來。";
					next;
					break;
				}else{
					mes "[南瓜帽研究員]";
					mes "我希望這對你有用。別忘了在戰鬥時配戴它來取得你的南瓜派。";
					delitem 1062,20;	// Jack o' Pumpkin
					getitem 5668,1;		// Weird Pumpkin Hat
					next;
					mes "[南瓜帽研究員]";
					mes "我想我應該回去研究了。";
					close;
				}
			case 4:
				mes "[南瓜帽研究員]";
				mes "再見，直到我們再次相見。";
				mes "祝你好運...";
				close;
		}
	}
}

-	script	萬聖節巫師#iRO09::09HallowWiz	737,{
	mes "[萬聖節巫師]";
	mes "...";
	mes "你想對某人惡作劇嗎？";
	next;
	while(1) {
		switch(select("什麼惡作劇？:當然:不要。")) {
			case 1:
				mes "[萬聖節巫師]";
				mes "我只需要一些材料就可以在世界其他地方召喚怪物。";
				next;
				mes "[萬聖節巫師]";
				mes "聽起來很有趣，對吧？";
				next;
				mes "[萬聖節巫師]";
				mes "如果你給我布料、傑克南瓜、破損的布料或壓碎的南瓜，我就可以召喚怪物。";
				next;
				break;
			case 2:
				mes "[萬聖節巫師]";
				mes "你想在哪個城鎮惡作劇？";
				next;
				getmapxy(.@mapname$,.@mapx,.@mapy,BL_PC,""+strcharinfo(0)+"");
				if (.@mapname$ == "prontera") {
					switch(select("吉芬:斐揚:艾爾貝塔:艾爾德巴朗")) {
						case 1:
							set .@HallowTown,3;
							break;
						case 2:
							set .@HallowTown,2;
							break;
						case 3:
							set .@HallowTown,4;
							break;
						case 4:
							set .@HallowTown,5;
							break;
					}
				}
				else if (.@mapname$ == "payon") {
					switch(select("普隆德拉:吉芬:艾爾貝塔:艾爾德巴朗")) {
						case 1:
							set .@HallowTown,1;
							break;
						case 2:
							set .@HallowTown,3;
							break;
						case 3:
							set .@HallowTown,4;
							break;
						case 4:
							set .@HallowTown,5;
							break;
					}
				}
				else if (.@mapname$ == "geffen") {
					switch(select("普隆德拉:斐揚:艾爾貝塔:艾爾德巴朗")) {
						case 1:
							set .@HallowTown,1;
							break;
						case 2:
							set .@HallowTown,2;
							break;
						case 3:
							set .@HallowTown,4;
							break;
						case 4:
							set .@HallowTown,5;
							break;
					}
				}
				else if (.@mapname$ == "alberta") {
					switch(select("普隆德拉:吉芬:斐揚:艾爾德巴朗")) {
						case 1:
							set .@HallowTown,1;
							break;
						case 2:
							set .@HallowTown,3;
							break;
						case 3:
							set .@HallowTown,2;
							break;
						case 4:
							set .@HallowTown,5;
							break;
					}
				}
				else if (.@mapname$ == "aldebaran") {
					switch(select("普隆德拉:吉芬:斐揚:艾爾貝塔")) {
						case 1:
							set .@HallowTown,1;
							break;
						case 2:
							set .@HallowTown,3;
							break;
						case 3:
							set .@HallowTown,2;
							break;
						case 4:
							set .@HallowTown,4;
							break;
					}
				}
				setarray .@HallowTowns$[1],"prontera","payon","geffen","alberta","aldebaran";
				mes "[萬聖節巫師]";
				mes "好吧，那麼讓我們進入下一步。";
				next;
				mes "[萬聖節巫師]";
				mes "你想使用多少布料或傑克南瓜？不要超過 100，因為這是我可以使用的最大數量。";
				next;
				input .@input;
				if (.@input == 0) {
					mes "[萬聖節巫師]";
					mes "你沒有明確的想法。";
					mes "這沒什麼大不了的。";
					mes "告訴我吧。";
					next;
					break;
				}
				else if (.@input > 100) {
					mes "[萬聖節巫師]";
					mes "我告訴過你，它必須在 1 到 100 之間！";
					mes "你沒有注意聽！";
					next;
					break;
				}
				else {
					set .@fabric,countitem(1059);
					set .@jack,countitem(1062);
					set .@worn,countitem(6299);
					set .@crushed,countitem(6298);
					set .@whispers,0;
					set .@darklords,0;

					set .@total,.@fabric + .@jack + .@worn + .@crushed;

					if(.@total < .@input) {
						mes "[萬聖節巫師]";
						mes "重新計算你擁有的物品數量，並告訴我總數。";
						mes "呵呵呵呵呵呵...";
						next;
						break;
					}

					if(.@fabric > 0) {
						if(.@fabric >= .@input) {
							delitem 1059,.@input;
							set .@whispers,.@whispers+.@input;
							set .@input,0;
						}
						else{
							delitem 1059,.@fabric;
							set .@input,.@input - .@fabric;
							set .@whispers,.@whispers+.@fabric;
						}
					}
					if(.@worn > 0 && .@input != 0) {
						if(.@worn >= .@input) {
							delitem 6299,.@input;
							set .@whispers,.@whispers+.@input;
							set .@input,0;
						}
						else{
							delitem 6299,.@worn;
							set .@input,.@input - .@worn;
							set .@whispers,.@whispers+.@worn;
						}
					}
					if(.@jack > 0 && .@input != 0) {
						if(.@jack >= .@input) {
							delitem 1062,.@input;
							set .@darklords,.@darklords+.@input;
							set .@input,0;
						}
						else{
							delitem 1062,.@jack;
							set .@input,.@input - .@jack;
							set .@darklords,.@darklords+.@jack;
						}
					}
					if(.@crushed > 0 && .@input != 0) {
						if(.@crushed >= .@input) {
							delitem 6298,.@input;
							set .@darklords,.@darklords+.@input;
							set .@input,0;
						}
						else{
							delitem 6298,.@crushed;
							set .@input,.@input - .@crushed;
							set .@darklords,.@darklords+.@crushed;
						}
					}
					if (.@input > 0) {
						mes "Theres a problem.";
						close;
					}
					monster ""+.@HallowTowns$[.@HallowTown]+"",0,0,"Halloween Whisper",3014,.@whispers;
					monster ""+.@HallowTowns$[.@HallowTown]+"",0,0,"Halloween Dark Lord",3015,.@darklords;
					mes "[萬聖節巫師]";
					mes "這就是你想要的。";
					mes "想像一下，其他村莊的人們一定在想什麼？";
					mes "Muahahaha";
					close;
				}
			case 3:
				mes "[萬聖節巫師]";
				mes "如果你改變主意，就回到這裡...";
				mes "我會在這裡待一陣子...";
				mes "Kkkk...";
				close;
		}
	}
}
