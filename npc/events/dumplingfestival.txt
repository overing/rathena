payon,93,81,4	script	驅魔師法海	834,{

	mes "[驅魔師法海]";
	mes "年輕的戰士，你好，我是驅魔師法海。";
	mes "我追蹤難以捉摸的巴風特已經有一段時間了。";
	next;
	mes "[驅魔師法海]";
	mes "我覺得是時候讓另一位戰士嘗試完成";
	mes "我的任務了。我有一些物品可以協助你完成我的任務。";
	next;
	menu "I wish to buy",-, "No thanks",M_FIN;
  
	mes "[驅魔師法海]";
	mes "我有 2 件物品可能對你有用。";
	next;
	menu "Realgar Wine",-, "Exorcize Herb",M_HERB;
  
	mes "[驅魔師法海]";
	mes "你需要多少個？";
	mes "你一次只能購買 5 個。";
	mes "每個價值 20000z。";
	mes "（輸入 0 取消）";
	next;
	input @input;
	if(@input==0) close;
	if(@input>5) goto L_SORRY;
	if(Zeny< @input*20000) goto L_NOZENY;
	set Zeny,Zeny-20000*@input;
	getitem 682,@input;
	mes "[驅魔師法海]";
	mes "給你，我希望你能成功完成我的任務。";
	close;
  
L_SORRY:
	mes "[驅魔師法海]";
	mes "你一定不能吝嗇，誠實是神的道路。";
	close;
  
M_FIN:
	mes "[驅魔師法海]";
	mes "完全消滅邪惡的道路還很長，請在神的道路上幫助我。";
	close;
  
M_HERB:
	mes "[驅魔師法海]";
	mes "你需要多少個？";
	mes "你一次只能購買 5 個。";
	mes "每個價值 10000z。";
	mes "（輸入 0 取消）";
	next;
	input @input;
	if(@input==0) close;
	if(@input>5) goto L_SORRY;
	if(Zeny< @input*10000) goto L_NOZENY;
	set Zeny,Zeny-10000*@input;
	getitem 683,@input;
	mes "[驅魔師法海]";
	mes "給你，我希望你能成功完成我的任務。";
	close;

L_NOZENY:
	mes "[驅魔師法海]";
	mes "金錢不會給每個人帶來快樂，但我們需要它來支持寺廟和我自己。請嘗試殺死一些怪物並拿走它們的掉落物。";
	close;
}
