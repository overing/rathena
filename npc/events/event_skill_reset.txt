//= Skill Reset with payment, minimum BaseLv.60 to reset skill.
//= Each 1 BaseLv need 20.000 Zeny, so be careful to use skill.
//============================================================
yuno,138,187,4	script	Hypnotist Teacher	124,{
	set @npcname$,"[^D5A500Hypnotist^000000]";
	mes @npcname$;

	if(MISC_QUEST & 1024){
		mes "我已經告訴過你，你只能完成這個事件一次。";
		close;
	}
	if(BaseLevel < 60) {
		mes "請在你達到 BaseLv 60 或更高時再來。";
		close;
	}
	if(SkillPoint != 0) {
		mes "如果你希望我繼續，你需要用完你所有的技能點數。";
		mes "請盡快再來！";
		close;
	}
	if(Weight || checkfalcon() || checkcart() || checkriding()) {
		mes "你想要重置技能嗎？";
		mes "我很抱歉，但是...";
		next;
	mes @npcname$;
		if(Weight) {
			mes "當你持有";
			mes "任何物品時";
			mes "無法重置技能。";
		} else if(checkcart())
			mes "請放下你的手推車，我們才能繼續。";
		else if(checkfalcon())
			mes "請放生你的獵鷹，我們才能繼續。";
		else if(checkriding())
			mes "請放生你的波利波利雞，我們才能繼續。";
		next;
		mes @npcname$;
		mes "歡迎再次光臨！";
		close;
	}
	mes "哈囉，我是技能重置者。";
	mes "你的名字是 ^0080FF"+strcharinfo(0)+"^000000。";
	mes "有什麼我可以幫忙的嗎？";
	next;
	menu "^009500Information about Reset skills.^000000",L_Info,"^00B6FFReset skills.^000000",L_Reset,"^000088Nevermind^000000",-;

	mes @npcname$;
	mes "如果你想要重置技能，";
	mes "你知道在哪裡可以找到我！！";
	close;

L_Info:
	mes @npcname$;
	mes "這個技能重置並不是免費的！！";
	mes "重置技能的費用是 ^D5A50020000 Zeny x BaseLv^000000。";
	mes "沒錯 ... 每一個 BaseLv 都需要花費 20000 Zeny 來重置技能。";
	next;
	mes @npcname$;
	mes "喔對，還有一件事！";
	mes "如果你重置技能，任何你裝備的手推車、獵鷹或波利波利雞";
	mes "都會被移除。";
	next;
	mes @npcname$;
	mes "只有一次機會可以再次重置技能點數";
	mes "從現在開始小心你的技能。";
	close;
	
L_Reset:
	mes @npcname$;
	mes "在開始技能重置之前。";
	mes "你必須先告訴我你的 Base 等級。";
	next;
	mes "^D5A500[" + strcharinfo(0) + "]^000000";
	mes "我的 Base 等級是 ^AA00AALevel " + BaseLevel + "BaseLv.^000000";
	next;
	set @zeny,BaseLevel*20000;
	mes @npcname$;
	mes "技能重置服務總共需要 ^529DFF" + @zeny + "Zeny^000000 。";
	next;
	if(Zeny < @zeny) {
		mes @npcname$;
		mes "看來你的錢不夠。";
		mes "我們也等待機會。";
		emotion ET_SCRATCH;
		close;
	}
	set Zeny, Zeny-@zeny;
	set MISC_QUEST,MISC_QUEST | 1024;
	resetskill;
	mes @npcname$;
	mes "謝謝你。";
	emotion ET_THANKS;
//	logmes "SKILL RESET EVENT";
	close;
}
