//callfunc "F_CoolEventCorp","<menu>","<location>","<save map>",<x>,<y>,"<warp map>";
function	script	F_CoolEventCorp	{
	cutin "zonda_01",2;
	mes "[酷炫活動公司工作人員]";
	mes "歡迎來到酷炫活動公司。";
	mes "我們的職員總是致力於";
	mes "超越您對於高品質服務的";
	mes "期望。那麼今天有什麼";
	mes "可以為您服務的嗎？";
	next;
	switch(select(getarg(0))) {
	case 1:
		mes "[酷炫活動公司工作人員]";
		mes "您的復活點";
		mes "已經儲存於此";
		mes getarg(1)+".";
		mes "感謝您使用";
		mes "酷炫活動公司的服務～";
		savepoint getarg(2),getarg(3),getarg(4),1,1;
		close2;
		break;
	case 2:
		mes "[酷炫活動公司工作人員]";
		if(!callfunc("F_CanOpenStorage")){
			mes "很抱歉，您需要";
			mes "新手基本技能等級 6";
			mes "才能使用";
			mes "倉庫服務。";
			close2;
			break;
		}
		if(Zeny < 40){
			mes "很抱歉，您沒有";
			mes "足夠的 Zeny 來使用";
			mes "倉庫服務。我們的";
			mes "倉庫使用費為 40 Zeny。";
			close2;
			break;
		}
		set Zeny, Zeny - 40;
		mes "讓我立即為您開啟";
		mes "您的個人倉庫。";
		mes "感謝您使用我們的";
		mes "服務來支持酷";
		mes "活動公司。祝您有個美好的一天～";
		callfunc("F_CheckKafCode");	//check your storage password, if set
		close2;
		openstorage;
		break;
	case 3:
		mes "請選擇";
		mes "您的目的地。";
		next;
		if (VIP_SCRIPT && !vip_status(VIP_STATUS_ACTIVE))
			set .@cost,4400;
		else
			set .@cost,2200;
		.@destination$ = getarg(5);
		if (select(.@destination$ +" -> "+ .@cost +" z:取消") == 1) {
			if ((countitem(7060) != 0) || (Zeny >= .@cost)) {
				if (countitem(7060) != 0) {
					delitem 7060,1; //Warp_Free_Ticket
				}
				else {
					set Zeny,Zeny-.@cost;
				}
				if (.@destination$ == "Veins") warp "veins",205,101;
				else if (.@destination$ == "Rachel") warp "rachel",115,125;
				cutin "",255;
				end;
			}
			else {
				mes "[酷炫活動公司工作人員]";
				mes "很抱歉，您沒有足夠的";
				mes "zeny 來使用傳送";
				mes "服務。傳送到";
				mes .@destination$ +" 的費用是 "+.@cost+" zeny。";
			}
		}
		close2;
		break;
	case 4:
		mes "[酷炫活動公司工作人員]";
		if(BaseClass != Job_Merchant){
			mes "很抱歉，";
			mes "手推車租借服務";
			if (checkre(0))
				mes "僅提供給商人職業。";
			else {
				mes "僅提供給商人、";
				mes "鐵匠、神工匠、";
				mes "鍊金術師和創造者。";
			}
			close2;
			break;
		}
		if(getskilllv("MC_PUSHCART") == 0){
			mes "您必須學會";
			mes "手推車技能後才能租借手推車。";
			close2;
			break;
		}
		if(checkcart() == 1){
			mes "您已經裝備了";
			mes "手推車。";
			mes "很遺憾，我們一次";
			mes "無法為每位顧客";
			mes "租借超過一台手推車。";
			close2;
			break;
		}
		if(countitem(7061)){
			delitem 7061,1; //Cart_Free_Ticket
			setcart;
			close2;
			break;
		}
		mes "手推車租借";
		mes "費用為 800 Zeny。您";
		mes "想要租借手推車嗎？";
		next;
		if(select("租借手推車:取消") == 1){
			if(Zeny < 800){
				mes "[酷炫活動公司工作人員]";
				mes "很抱歉，您";
				mes "沒有足夠的";
				mes "Zeny 來支付手推車";
				mes "的租借費用 800 Zeny。";
			} else {
				set Zeny, Zeny - 800;
				setcart;
			}
		}
		close2;
		break;
	case 5:
		callfunc("F_SetKafCode","[酷炫活動公司工作人員]","酷炫活動公司.");
		//the function closes msg and removes cutins
	case 6:
	default:
		mes "[酷炫活動公司工作人員]";
		mes "酷炫活動公司總是";
		mes "努力為我們的客戶";
		mes "提供最好的服務。";
		mes "請提供您的意見";
		mes "和誠實的回饋來幫助我們";
		mes "成為最好的公司。謝謝您。";
		close2;
	}
	cutin "", 255;
	end;
}
