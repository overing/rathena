lou_fild01,240,354,4	script	Mantis Researcher	878,{

	if (checkquest(60179) != -1) {
		if (checkquest(60179,HUNTING) != 2 ) {
			mes "[螳螂研究員]";
			mes "你完成狩獵 50 隻螳螂了嗎？";
			next;
			switch(select("是:否:我想放棄")) {
			case 1:
				mes "[螳螂研究員]";
				mes "嗯，我不認為你已經獵殺了 50 隻。";
				close;
			case 2:
				mes "[螳螂研究員]";
				mes "記住，我需要你幫忙狩獵螳螂，所以去獵殺 50 隻吧。";
				close;
			case 3:
				mes "[螳螂研究員]";
				mes "你確定你要停止狩獵嗎？";
				mes "你所做的任何進度都將被清除";
				next;
				switch(select("是:否")) {
				case 1:
					mes "[螳螂研究員]";
					mes "好吧，如果你改變主意，就回來這裡。";
					erasequest 60179;
					close;
				case 2:
					mes "[螳螂研究員]";
					mes "請殺死 50 隻螳螂，讓這個地方成為人們可以去並獲得經驗的地方。";
					close;
				}
			}
		}
		mes "[螳螂研究員]";
		mes "太棒了，你以驚人的速度完成了。";
		getexp 18818,7527;
		erasequest 60179;
		close;
	}
	if ((BaseLevel > 34) && (BaseLevel < 71)) {
		mes "[螳螂研究員]";
		mes "喔！太好了！";
		mes "你是來幫助我的，對吧？";
		mes "如果不是，我不知道該如何處理這個地方的怪物！";
		next;
		switch(select("當然:不")) {
		case 1:
			mes "[螳螂研究員]";
			mes "太好了！我一看到你就知道我可以依靠你！";
			mes "我需要你減少目前在這個地圖上肆虐的怪物數量！";
			mes "周圍的螳螂太多了，這使得低等級的玩家無法在這裡獲得良好的經驗。";
			next;
			mes "[螳螂研究員]";
			mes "那麼，你仍然願意幫助我嗎？";
			next;
			switch(select("當然:取消")) {
			case 1:
				if (BaseLevel > 70) {
					mes "[螳螂研究員]";
					mes "感謝你對螳螂的巨大幫助！";
					mes "目前情況穩定，我不再需要你的幫助來殺死牠們。";
					close;
				}
				mes "[螳螂研究員]";
				mes "殺死 50 隻螳螂，完成後告訴我。我會獎勵你的努力。";
				setquest 60179;
				close;
			case 2:
				mes "[螳螂研究員]";
				mes "如果你改變主意，請回來。";
				close;
			}
		case 2:
			mes "[螳螂研究員]";
			mes "如果你改變主意，請回來。";
			close;
		}
	}
	mes "[螳螂研究員]";
	mes "噢，天啊！";
	next;
	mes "[螳螂研究員]";
	mes "抱歉，我現在無法說話，我正處於痛苦之中，這些該死的螳螂要殺了我了。";
	close;
}
