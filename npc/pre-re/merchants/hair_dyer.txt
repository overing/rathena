prt_in,243,168,4	script	Jovovich	91,{
	mes "[喬維琪美髮師]";
	mes "歡迎光臨～！";
	mes "有什麼我可以幫忙的嗎？";
	next;
	mes "[喬維琪美髮師]";
	if (Sex == SEX_MALE) mes "喔，不！你的頭髮受損了。看來你可能需要專業護理。請過來這邊坐下。來。";
	else {
		if (rand(20) > 11) {
			mes "咦！？我的天啊！喔不不不不！你的頭髮實在是太受損了！如果你讓頭髮維持這樣是不好的。";
			next;
			mes "[喬維琪美髮師]";
			mes "你願意讓我護理你的頭髮嗎？拜託？";
		}
		else mes "哇！一旦染髮後你的頭髮會變得完美～要不要換個髮色？";
	}
	next;
	while(1) {
		switch(select("染髮:秘訣與資訊:取消")) {
		case 1:
			mes "[喬維琪美髮師]";
			mes "是的是的，好選擇～";
			mes "那麼，這就是你改頭換面的機會，你綻放美麗的機會！選擇你想要的顏色。";
			next;
			while(1) {
				if (.@choose_success == 1) {
					mes "[喬維琪美髮師]";
					mes "你覺得怎麼樣？你想要不同的顏色嗎？";
					next;
					if (select("是:否") == 1) {
						mes "[喬維琪美髮師]";
						mes "好的！選擇你想要的顏色。";
						next;
					}
					else  {
						mes "[喬維琪美髮師]";
						mes "嗯，我有點失望。我想要做得更好。但我保證下次會做得更好。請再來～";
						close;
					}
				}

				while(1) {
					switch(select("紅色，拜託。:黃色，拜託。:紫色，拜託。:橘色，拜託。:綠色，拜託。:藍色，拜託。:白色，拜託。:黑色，拜託。:其實，我喜歡我現在的髮色。")) {
					case 1: set .@headpalette,8; break;
					case 2: set .@headpalette,1; break;
					case 3: set .@headpalette,2; break;
					case 4: set .@headpalette,3; break;
					case 5: set .@headpalette,4; break;
					case 6: set .@headpalette,5; break;
					case 7: set .@headpalette,6; break;
					case 8: set .@headpalette,7; break;
					case 9:
						if (.@choose_success != 0) {
							mes "[喬維琪美髮師]";
							mes "你一定很喜歡你的髮色～";
							close;
						}
						mes "[喬維琪美髮師]";
						mes "欸？你不打算染髮嗎？我有點難過...";
						close;
					}

					if (.@headpalette == getlook(VAR_HEADPALETTE)) {
						mes "[喬維琪美髮師]";
						mes "欸？但那是你已經有的髮色。請選擇不同的顏色。";
						next;
					}
					else {
						switch(.@headpalette) {
						case 1: callsub S_NoDye,976,"yellow"; break;
						case 2: callsub S_NoDye,978,"violet"; break;
						case 3: callsub S_NoDye,980,"orange"; break;
						case 4: callsub S_NoDye,979,"green"; break;
						case 5: callsub S_NoDye,981,"blue"; break;
						case 6: callsub S_NoDye,982,"white"; break;
						case 7: callsub S_NoDye,983,"black"; break;
						case 8: callsub S_NoDye,975,"red"; break;
						}
						if (Zeny < 1000) {
							mes "[喬維琪美髮師]";
							mes "費用是 1000 Ｚeny。你錢不夠嗎...？";
							close;
						}

						switch(.@headpalette) {
						case 1: delitem 976,1; break; //Lemon_Dyestuffs
						case 2: delitem 978,1; break; //Violet_Dyestuffs
						case 3: delitem 980,1; break; //Orange_Dyestuffs
						case 4: delitem 979,1; break; //Darkgreen_Dyestuffs
						case 5: delitem 981,1; break; //Cobaltblue_Dyestuffs
						case 6: delitem 982,1; break; //White_Dyestuffs
						case 7: delitem 983,1; break; //Black_Dyestuffs
						case 8: delitem 975,1; break; //Scarlet_Dyestuffs
						}
						set Zeny, Zeny-1000;
						setlook VAR_HEADPALETTE,.@headpalette;
						set .@choose_success,1;
						break;
					}

				}

			}

		case 2:
			mes "[喬維琪美髮師]";
			mes "當你感到沮喪時，當你被甩時，當你想給某人留下深刻印象時，甚至當你只是想脫穎而出時...";
			next;
			mes "[喬維琪美髮師]";
			mes "對於那個特別的地點和時間，你難道不想要一個你專屬的髮型嗎？只要你有適當的染髮劑，我就會讓你的頭髮看起來很棒。";
			next;
			mes "[喬維琪美髮師]";
			mes "並且盡量不要太擔心費用。獲得美麗就等於獲得一切。呵呵～";
			next;
			mes "[喬維琪美髮師]";
			mes "* 費用與資訊 *";
			mes "- 1 個你想要的髮色的染髮劑道具。";
			mes "- 1000 Ｚeny 費用。";
			close;

		case 3:
			mes "[喬維琪美髮師]";
			mes "男人或女人...";
			mes "每個人都有變美的權利和義務。";
			close;
		}

	}

S_NoDye:
	if (countitem(getarg(0)) == 0) {
		mes "[喬維琪美髮師]";
		mes "欸？！但你需要道具 '"+getitemname(getarg(0))+"' 才能將你的頭髮染成 "+getarg(1)+" ...";
		close;
	}
	return;
}
