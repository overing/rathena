que_ng,187,149,3	script	Magazine Dealer Kenny::mdk	4_M_01,{
	mes "[肯尼]";
	if (BaseClass == Job_Gunslinger) {
		mes "[肯尼]";
		mes "歡迎來到我的彈匣商店。";
		mes "您可能知道，大量的";
		mes "子彈裝在彈匣裡更容易";
		mes "攜帶。現在，您對";
		mes "任何東西特別感興趣嗎？";
		next;
		switch(select("風靈彈匣包:暗靈彈匣包:毒靈彈匣包:水靈彈匣包:火靈彈匣包:彈夾:血腥彈夾:銀彈夾:取消")) {
		case 1: callsub S_Casing,13204,12144;// <bullet id>, <pack id>
		case 2: callsub S_Casing,13206,12145;
		case 3: callsub S_Casing,13205,12146;
		case 4: callsub S_Casing,13207,12147;
		case 5: callsub S_Casing,13203,12148;
		case 6: callsub S_Casing,13200,12149;
		case 7: callsub S_Casing,13202,12150;
		case 8: callsub S_Casing,13201,12151;
		case 9:
			mes "[肯尼]";
			mes "嗯，如果您發現";
			mes "您的子彈太多了";
			mes "就來找我。";
			mes "將子彈儲存在我的";
			mes "彈匣裡是個聰明的主意。";
			close;
		}
	}
	mes "歡迎來到我的商店。";
	mes "在這裡，我為神槍手";
	mes "提供彈匣和彈夾。";
	mes "抱歉，看來";
	mes "我的服務對您來說";
	mes "沒有任何用處，冒險家。";
	next;
	mes "[肯尼]";
	mes "呃，但是如果您剛好";
	mes "認識任何神槍手，請";
	mes "將他們送到我這裡來。子彈";
	mes "永遠不會嫌多。";
	close;

S_Casing:
	.@bullet_id = getarg(0);
	.@pack_id = getarg(1);

	mes "[肯尼]";
	mes "現在，您可以交易";
	mes callfunc("F_InsertPlural",500,getitemname(.@bullet_id));
	mes "和 500 zeny 來換取 1 個 "+getitemname(.@pack_id)+"，所以請確保";
	mes "您有足夠的子彈";
	mes "和 zenny 來進行此交換。";
	next;
	mes "[肯尼]";
	mes "您可以將 "+callfunc("F_InsertPlural",500,getitemname(.@bullet_id))+" 和 500 zeny";
	mes "換成 1 個 "+getitemname(.@pack_id)+"。";
	next;
	mes "[肯尼]";
	mes "請記住，我一次最多只能";
	mes "給您 50 個彈匣。";
	mes "現在請輸入您想要";
	mes "獲得的彈匣數量。如果要取消，";
	mes "只需輸入 ''0''。";
	next;
	input .@amount;
	mes "[肯尼]";
	if (.@amount > 50 || .@amount == 0) {
		mes "嘿，我不能給你";
		mes "那麼多彈匣。";
		mes "請重試，並且";
		mes "輸入一個不大於";
		mes "50 的數字。";
	}
	else if (countitem(.@bullet_id) >= (500*.@amount)) {
		if (Zeny >= (500*.@amount)) {
			if (checkweight(.@pack_id,.@amount) == 0) {
				mes "嘿，你的背包裡";
				mes "塞滿了很多垃圾。";
				mes "騰出一些";
				mes "空間，然後再來";
				mes "用你的子彈換";
				mes "一些彈匣，好嗎？";
			} else {
				mes "好的，這是";
				mes "你的彈匣。謝謝";
				mes "您光臨我的商店，並且";
				mes "我希望您明智地使用";
				mes "您的所有彈藥。";
				Zeny = Zeny -500*.@amount;
				delitem .@bullet_id,(500*.@amount);
				getitem .@pack_id,.@amount;
			}
		} else {
			mes "抱歉，您的";
			mes "zeny 不夠";
			mes "無法交換此彈匣。";
			mes "請在存夠";
			mes "更多錢後再來";
			mes "我的商店。";
		}
	} else {
		mes "抱歉，您的";
		mes "子彈不夠";
		mes "無法交換此彈匣。";
		mes "也許您應該仔細";
		mes "檢查您的背包，然後";
		mes "稍後再來找我。";
	}
	close;
}
