	cutin "prt_soldier",2;
	mes "[艾爾貝塔導遊]";
	mes "歡迎來到艾爾貝塔，";
	mes "這座港口城市。";
	mes "如果你在鎮上找不到";
	mes "任何東西，或是需要";
	mes "城市導覽，";
	mes "隨時都可以來問我。";
	set .@loop1,1;
	while (.@loop1) {
		next;
		switch(select("城市導覽:從迷你地圖移除標記:注意事項:取消")) {
		case 1:
			mes "[艾爾貝塔導遊]";
			mes "請從";
			mes "以下選單中";
			mes "選擇一個地點。";
			if (.@compass_check == 0) {
				mes "需要我";
				mes "在你的迷你地圖上";
				mes "標記位置嗎？";
				next;
				if (select("是:否") == 1) set .@compass_check,1;
			}
			set .@loop2,1;
			while (.@loop2) {
				if (.@wait_button_chk == 0) set .@wait_button_chk,1;
				else next;
				
				switch(select("^FF0000商人公會^000000:武器商店:工具商店:旅館:鍛造屋:取消")) {
				case 1:
					mes "[艾爾貝塔導遊]";
					mes "商人公會";
					mes "負責轉職成";
					mes "商人，";
					mes "位於";
					mes "艾爾貝塔的";
					mes "西南角。";
					if (.@compass_check)
						viewpoint 1,33,41,2,0xFF0000;
					break;
				case 2:
					mes "[艾爾貝塔導遊]";
					mes "武器商店";
					mes "位於";
					mes "艾爾貝塔的";
					mes "南邊。";
					if (.@compass_check)
						viewpoint 1,117,37,3,0xFF00FF;
					break;
				case 3:
					mes "[艾爾貝塔導遊]";
					mes "工具商店";
					mes "非常靠近";
					mes "艾爾貝塔的";
					mes "中心。應該";
					mes "不難找到。";
					if (.@compass_check)
						viewpoint 1,98,154,4,0xFF00FF;
					break;
				case 4:
					mes "[艾爾貝塔導遊]";
					mes "在艾爾貝塔";
					mes "的北邊";
					mes "有一間旅館";
					mes "可以讓你休息。";
					if (.@compass_check)
						viewpoint 1,65,233,5,0xFF00FF;
					break;
				case 5:
					mes "[艾爾貝塔導遊]";
					mes "艾爾貝塔的鍛造屋";
					mes "和商人公會";
					mes "在同一棟建築裡。";
					mes "在西南方。";
					if (.@compass_check)
						viewpoint 1,35,41,6,0xFF00FF;
					break;
				case 6:
					mes "[艾爾貝塔導遊]";
					mes "如果你不再想要";
					mes "在迷你地圖上顯示";
					mes "位置標記，請告訴我";
					mes "''從迷你地圖移除標記''。";
					set .@loop2,0;
				}
			}
			break;
		case 2:
			viewpoint 2,237,41,2,0xFF0000;
			viewpoint 2,237,41,3,0xFF00FF;
			viewpoint 2,46,345,4,0xFF00FF;
			viewpoint 2,175,220,5,0xFF00FF;
			viewpoint 2,175,220,6,0xFF00FF;
			set .@compass_check,0;
			break;
		case 3:
			mes "[艾爾貝塔導遊]";
			mes "由於魔法和";
			mes "科技的進步，";
			mes "我們更新了資訊";
			mes "系統，現在可以在你的";
			mes "迷你地圖上標記";
			mes "位置，以便更容易導航。";
			next;
			mes "[艾爾貝塔導遊]";
			mes "你的迷你地圖";
			mes "位於螢幕的";
			mes "右上角。如果你";
			mes "看不到，請按下 Ctrl + Tab";
			mes "鍵，或點擊基本資訊視窗中的 ''地圖'' 按鈕。";
			next;
			mes "[艾爾貝塔導遊]";
			mes "在你的迷你地圖上，";
			mes "點擊 ''+'' 和 ''-''";
			mes "符號可以放大和";
			mes "縮小你的迷你地圖。";
			mes "希望你在艾爾貝塔";
			mes "玩得開心。";
			break;
		case 4:
			mes "[艾爾貝塔導遊]";
			mes "旅途平安，";
			mes "如果你有任何";
			mes "關於艾爾貝塔的";
			mes "問題，隨時都可以問我。";
			close2;
			set .@loop1,0;
		}
	}
	cutin "prt_soldier",255;
	end;
}
