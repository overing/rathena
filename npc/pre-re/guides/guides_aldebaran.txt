aldebaran,139,63,4	script	Soldier#alde	105,{
	cutin "prt_soldier",2;
	mes "[艾爾貝塔守衛]";
	mes "我只是一個普通的守衛";
	mes "你可以在任何其他城市找到。 我想我甚至沒有名字...";
	next;
	mes "[艾爾貝塔守衛]";
	mes "我負責艾爾貝塔駐軍的服務指南。 讓我引導你";
	mes "遊覽我們的城鎮！";
	next;
	if (select("取得位置指南。:結束對話。") == 1) {
		viewpoint 1,61,229,0,0xFF6633;
		viewpoint 1,72,197,1,0x0000FF;
		viewpoint 1,223,222,2,0x00FFFF;
		viewpoint 1,233,105,3,0x515151;
		viewpoint 1,197,70,4,0x3355FF;
		viewpoint 1,60,60,5,0xFF5555;
		mes "^FF6633+^000000 -> 卡普拉總部 ";
		mes "^0000FF+^000000 -> 武器商店 ";
		mes "^00FFFF+^000000 -> 魔法師公會 (關閉)";
		mes "^515151+^000000 -> 酒館";
		mes "^3355FF+^000000 -> 道具商店";
		mes "^FF5555+^000000 -> 鍊金術師公會";
		close2;
		cutin "prt_soldier",255;
		end;
	}
	mes "[艾爾貝塔守衛]";
	mes "我們宣誓保護艾爾貝塔！願邪惡勢力永遠被";
	mes "正義的鐵拳粉碎！";
	close2;
	cutin "prt_soldier",255;
	end;
}

aldebaran,133,108,4	script	Soldier#2alde	105,{
	cutin "prt_soldier",2;
	mes "[艾爾貝塔守衛]";
	mes "我只是一個";
	mes "普通的守衛，";
	mes "你可以在";
	mes "任何其他城市找到的那種。";
	next;
	mes "[艾爾貝塔守衛]";
	mes "當我不忙於";
	mes "保護艾爾貝塔";
	mes "人民時，我在這裡為像你這樣的";
	mes "冒險家指路。";
	next;
	switch(select("卡普拉總部 :武器商店 :魔法師公會 :酒館 :道具商店 :鍊金術師公會 :結束對話 ")) {
	case 1:
		viewpoint 1,61,229,0,0xFF6633;
		mes "^FF6633+^000000 -> 卡普拉總部 ";
		break;
	case 2:
		viewpoint 1,72,197,1,0x0000FF;
		mes "^0000FF+^000000 -> 武器商店 ";
		break;
	case 3:
		viewpoint 1,223,222,2,0x00FFFF;
		mes "^00FFFF+^000000 -> 魔法師公會 (關閉)";
		break;
	case 4:
		viewpoint 1,233,105,3,0x515151;
		mes "^515151+^000000 -> 酒館 ";
		break;
	case 5:
		viewpoint 1,197,70,4,0x3355FF;
		mes "^3355FF+^000000 -> 道具商店 ";
		break;
	case 6:
		viewpoint 1,60,60,5,0xFF5555;
		mes "^FF5555+^000000 -> 鍊金術師公會";
		break;
	case 7:
	mes "[艾爾貝塔守衛]";
	mes "我們宣誓";
	mes "保護艾爾貝塔！";
	mes "願善良的力量";
	mes "永遠戰勝邪惡~";
		break;
	}
	close2;
	cutin "prt_soldier",255;
	end;
}
