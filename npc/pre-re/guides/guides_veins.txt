veins,210,345,5	script	Veins Guide#1::ve_guide	934,{
	mes "[靜脈引導]";
	mes "沙漠城市靜脈歡迎尋求庇護，躲避嚴酷沙塵暴的冒險者。";
	mes "如果這是您第一次使用引導服務，何不查看...";
	set .@loop1,1;
	while(.@loop1) {
		next;
		switch(select("村莊引導:從迷你地圖移除標記:注意:取消")) {
		case 1:
			mes "[靜脈引導]";
			mes "只要在靜脈，我就可以告訴你任何建築物的位置。";
			mes "你想去哪裡？";
			if (.@compass_check == 0) {
				mes "你想要我";
				mes "在地圖上標記位置";
				mes "在你的迷你地圖上嗎？";
				next;
				if (select("是:否") == 1) set .@compass_check,1;
			}
			set .@loop2,1;
			while(.@loop2) {
				if (.@wait_button_chk == 0) set .@wait_button_chk,1;
				else next;

				switch(select("神殿:旅館:武器店:道具店:飛船:酒館:地質研究所:取消")) {
				case 1:
					mes "[靜脈引導]";
					mes "我們的神殿位於北方，";
					mes "總是擠滿了芙蕾雅女神虔誠的信徒。";
					if (.@compass_check == 1) viewpoint 1,196,258,1,0xFF0000;
					break;
				case 2:
					mes "[靜脈引導]";
					mes "您可以在旅館休息，消除旅途的疲勞。";
					mes "我旁邊左邊的建築物是靜脈旅館。";
					if (.@compass_check == 1) viewpoint 1,128,266,2,0xFF00FF;
					break;
				case 3:
					mes "[靜脈引導]";
					mes "是的，您應該自己保護自己免受危險。";
					mes "以實惠的價格購買高品質的武器。";
					mes "靜脈武器店位於西方。";
					if (.@compass_check == 1) viewpoint 1,150,175,3,0x99FFFF;
					break;
				case 4:
					mes "[靜脈引導]";
					mes "您是否為您的冒險準備了足夠的必需品？";
					mes "如果沒有，我建議您查看位於市中心的靜脈所能為您提供的東西。";
					if (.@compass_check == 1) viewpoint 1,230,161,4,0x0000FF;
					break;
				case 5:
					mes "[靜脈引導]";
					mes "請注意，靜脈僅營運貨運飛船。";
					if (.@compass_check == 1) viewpoint 1,273,285,5,0x00FF00;
					break;
				case 6:
					mes "[靜脈引導]";
					mes "如果您想與";
					mes "鎮民交朋友，我建議您";
					mes "去西邊的酒館喝一杯。";
					if (.@compass_check == 1) viewpoint 1,150,217,6,0x00FF00;
					break;
				case 7:
					mes "[靜脈引導]";
					mes "您對研究地質學感興趣嗎？";
					mes "那麼你最好去看看";
					mes "武器店二樓的";
					mes "地質研究所。";
					if (.@compass_check == 1) viewpoint 1,150,175,7,0x00FF00;
					break;
				case 8:
					mes "[靜脈引導]";
					mes "如果您想擺脫迷你地圖上的所有位置標記，";
					mes "只要再問我一次，然後選擇“從迷你地圖移除標記”菜單。";
					set .@loop2,0;
					break;
				}
			}
			break;

		case 2:
			viewpoint 2,196,258,1,0xFF0000;
			viewpoint 2,128,266,2,0xFF00FF;
			viewpoint 2,150,175,3,0x99FFFF;
			viewpoint 2,230,161,4,0x0000FF;
			viewpoint 2,273,285,5,0x00FF00;
			viewpoint 2,150,217,6,0x00FF00;
			viewpoint 2,150,175,7,0x00FF00;
			mes "[靜脈引導]";
			mes "好的，它們現在消失了。如果您還有其他位置要問，請告訴我。";
			mes "祝您在靜脈玩得愉快。";
			break;
		case 3:
			mes "[靜脈引導]";
			mes "當您使用“村莊引導”菜單時，";
			mes "請確保建築物的位置將在螢幕右上角的迷你地圖上標記。";
			mes "如果您看不到迷你地圖，請使用快捷鍵“ctrl+tab”或按下基本信息視窗上的“地圖”按鈕，好嗎？";
			mes "如果您無法查看村莊的整個地圖，您也可以使用“-”按鈕縮小迷你地圖。";
			break;
		case 4:
			mes "[靜脈引導]";
			mes "祝您在靜脈玩得愉快。";
			close2;
			set .@loop1,0;
			break;
		}
	}
	close;
}

veins,189,101,5	duplicate(ve_guide)	Veins Guide#2	934
