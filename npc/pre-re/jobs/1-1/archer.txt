payon_in02,64,71,4	script	弓箭手公會人員#archer	85,{
	if (Upper == 1) {
		if (Class == Job_Novice_High && (ADVJOB == Job_Sniper || ADVJOB == Job_Clown || ADVJOB == Job_Gypsy)) {
			mes "[弓箭手公會人員]";
			mes "嘿，我知道你。";
			mes "你之前參加過這個測試";
			mes "不是嗎？";
			next;
			mes "[弓箭手公會人員]";
			mes "啊，你一定是";
			mes "去過瓦爾哈拉並且重生了。";
			mes "哇，真令人印象深刻！";
			next;
			if (!callfunc("F_CanChangeJob")) {
				mes "[弓箭手公會人員]";
				mes "呃...";
				mes "你最好先學會所有的基礎技能，才能成為弓箭手。";
				next;
				mes "[弓箭手公會人員]";
				mes "好的，待會見。";
				close;
			}
			mes "[弓箭手公會人員]";
			mes "那麼。我不需要";
			mes "再說什麼了。";
			mes "我知道你會成為一個偉大的弓箭手...";
			next;
			skill "NV_TRICKDEAD",0,SKILL_PERM;
			jobchange Job_Archer_high;
			skill "AC_MAKINGARROW",1,SKILL_PERM;
			skill "AC_CHARGEARROW",1,SKILL_PERM;
			mes "[弓箭手公會人員]";
			mes "雖然這次沒有特別的";
			mes "獎勵給你，但我希望你能理解。照顧好自己。";
			close;
		}
		else {
			mes "[弓箭手公會人員]";
			mes "喔...?";
			mes "嘿，你在";
			mes "這裡做什麼...?";
			next;
			mes "[弓箭手公會人員]";
			mes "我看得出來你不適合當弓箭手。感覺你好像注定要";
			mes "做別的事情...";
			close;
		}
	}
	mes "[弓箭手公會人員]";
	mes "您好。有什麼可以幫您的嗎？";
	next;
	switch(select("我想成為弓箭手。:我需要知道成為弓箭手的條件。:不用了，謝謝。")) {
	case 1:
		if (BaseJob == Job_Archer) {
			mes "[弓箭手公會人員]";
			mes "你已經是弓箭手了...";
			close;
		}
		else if (BaseJob != Job_Archer && BaseJob != Job_Novice) {
			mes "[弓箭手公會人員]";
			mes "嗯...";
			mes "你一點都不像初心者...";
			next;
			mes "[弓箭手公會人員]";
			mes "總之，不管你現在是什麼身份，因為你已經有職業了，所以你不能選擇弓箭手這個職業。";
			close;
		}
		if (job_archer_q == 0) {
			mes "[弓箭手公會人員]";
			mes "你想成為弓箭手嗎？";
			mes "如果是的話，你需要填寫這份申請表。";
			next;
			if (select("申請。:取消") == 1) {
				set job_archer_q,1;
				setquest 1004;
				mes "[弓箭手公會人員]";
				mes "好的，在這裡簽名。好的，嗯，一旦你符合資格，我就會晉升你。";
				next;
				mes "[弓箭手公會人員]";
				mes "如果你認為你已經符合資格了，我們可以現在檢查。";
				mes "你準備好了嗎？";
				next;
				if (select("是的，我準備好了。:還沒。") == 1) {
					mes "[弓箭手公會人員]";
					mes "好的，讓我檢查一下。";
					next;
				}
				else {
					mes "[弓箭手公會人員]";
					mes "我明白了。如果你想看看需求，請自便。";
					close;
				}
			}
			else {
				mes "[弓箭手公會人員]";
				mes "好吧，好的。";
				mes "下次見。";
				close;
			}
		}
		mes "[弓箭手公會人員]";
		mes "你是..." + strcharinfo(0) + "嗎？";
		next;
		if (!callfunc("F_CanChangeJob")) {
			mes "[弓箭手公會人員]";
			mes "嗯，你的職業等級不夠。請再次檢查要求。";
			next;
			mes "[弓箭手公會人員]";
			mes "你的職業等級必須至少達到 10 級，而且別忘了你應該學習所有的基礎技能。一旦你完成了，就回來吧。";
			close;
		}
		if (job_archer_q == 1) {
			set .@archer_item1,countitem(1066) * 5;
			set .@archer_item2,countitem(1067) * 3;
			set .@archer_item3,countitem(1068) * 2;
			set .@archer_item4,countitem(1019);
			set .@total_archer,.@archer_item1 + .@archer_item2 + .@archer_item3 + .@archer_item4;
			set .@total_archer2,(((.@archer_item2 + .@archer_item3) * 2) + .@archer_item4);
			mes "[弓箭手公會人員]";
			mes "太棒了！";
			mes "那麼，";
			mes "讓我看看...";
			next;
			mes "[弓箭手公會人員]";
			mes "我將鑑定你帶來的各種樹幹的價值，這些樹幹是用來製造弓的。";
			next;
			if (countitem(1066) == 0 && countitem(1067) == 0 && countitem(1068) == 0 && countitem(1019) == 0) {
				mes "[弓箭手公會人員]";
				mes "嗯...";
				mes "不幸的是，你沒有帶來任何要求的物品。我沒有什麼可以鑑定的。";
				close;
			}
			mes "[弓箭手公會人員]";
			if (countitem(1066) != 0) mes " S級 : " + countitem(1066) + " 個, 等級: " + .@archer_item1 + " . ";
			if (countitem(1067) != 0) mes " A級 : " + countitem(1067) + " 個, 等級 : " + .@archer_item2 + " . ";
			if (countitem(1068) != 0) mes " B級 : " + countitem(1068) + " 個, 等級 : " + .@archer_item3 + " . ";
			if (countitem(1019) != 0) mes " C級 : " + countitem(1019) + " 個, 等級 : " + .@archer_item4 + " . ";
			if (.@total_archer < 25) {
				mes "總等級: ^FF0000" + .@total_archer + "^000000 / 40";
				next;
				mes "[弓箭手公會人員]";
				mes "少於 25 分！？你必須至少獲得 25 分！加油，再努力一點！";
				close;
			}
			else {
				mes "總等級: ^0000FF" + .@total_archer + "^000000 / 40";
				next;
				mes "[弓箭手公會人員]";
				if (.@total_archer > 40) {
					mes "哇！超過 40 分！";
					mes "太棒了！恭喜你！";
				}
				else if (.@total_archer > 30) {
					mes "超過 30 分！幹得好！";
					mes "恭喜你！";
				}
				else {
					mes "*嘆氣* 嗯，你只是勉強通過了... 總之，做得很好。";
				}
			}
			next;
			mes "[弓箭手公會人員]";
			mes "我會把這些樹幹轉交給我們的弓生產部門。既然你已經符合要求了，讓我馬上晉升你！";
			if (countitem(1066) != 0) delitem 1066,countitem(1066); //Tree_Of_Archer_1
			if (countitem(1067) != 0) delitem 1067,countitem(1067); //Tree_Of_Archer_2
			if (countitem(1068) != 0) delitem 1068,countitem(1068); //Tree_Of_Archer_3
			if (countitem(1019) != 0) delitem 1019,countitem(1019); //Wooden_Block
		}
		next;
		callfunc "Job_Change",Job_Archer;
		callfunc "F_ClearJobVar";
		completequest 1004;
		mes "[弓箭手公會人員]";
		mes "恭喜你！";
		mes "你現在是一名弓箭手了！";
		next;
		mes "[弓箭手公會人員]";
		mes "當然，我們期望您能以您的努力為弓箭手公會的未來做出貢獻。";
		next;
		mes "[弓箭手公會人員]";
		mes "啊，你的弓已經從弓生產部門送來了。給你，拿去吧！這是你的～";
		getitem 1702,1; //Bow_
		getitem 1750,.@total_archer2; //Arrow
		next;
		mes "[弓箭手公會人員]";
		mes "現在，走吧。帶著身為最佳訓練者之一的驕傲去狩獵吧！";
		close;
	case 2:
		mes "[弓箭手公會人員]";
		mes "我將解釋成為弓箭手的要求。";
		if (BaseJob != Job_Novice) {
			if (BaseJob == Job_Archer) {
				next;
				mes "[弓箭手公會人員]";
				mes "但是...";
				mes "你已經是弓箭手了。你應該已經知道了...";
			}
			else {
				next;
				mes "[弓箭手公會人員]";
				mes "等一下。你已經選擇了另一個職業。你不需要知道這個～";
			}
			mes "所以...是的...沒有真正的原因告訴你這些要求...";
		}
		next;
		mes "[弓箭手公會人員]";
		mes "首先，你必須是職業等級 9 的初心者，並且知道所有的基礎技能。";
		next;
		mes "[弓箭手公會人員]";
		mes "弓箭手需要極高的集中力和反應能力，所以我們不接受那些缺乏耐心的人。";
		next;
		mes "[弓箭手公會人員]";
		mes "你還必須收集 ^FF0000樹幹^000000。有 4 種不同類型的樹幹，每種的品質都不同。根據它們的品質，你會得到不同的等級。";
		next;
		mes "[弓箭手公會人員]";
		mes "為了成為弓箭手，你必須從 40 分中獲得至少 ^0000FF25^000000 分的等級。你可以從樹「柳樹」那裡獲得樹幹。不過要小心，它們可能是很難對付的怪物。";
		close;
	case 3:
		close;
	}
}
