	}

		}
	}
	if (job_thief_q == 3) {

		mes "你還沒";

		mes "嗯...";
		mes "你是來參加測試的，對吧？我能從你的眼神中看出你知道些什麼。";
		next;
		mes "[無關緊要先生]";
		mes "你的名字是 " + strcharinfo(0) + " 嗎？啊，在名單上。好吧，我讓你進入 Mushroom Farm，但我不能保證你的安全...";
		close2;
		set job_thief_q,3;
		switch(rand(5)) {
		case 1: warp "job_thief1",228,106; end;

		close;
	}
	mes "[無關緊要先生]";
	mes "嘿，新手！為什麼不加入盜賊公會的行列呢？我們隨時歡迎你們這些新手加入我們和我們自私的事業。";
	next;
	mes "[無關緊要先生]";
	mes "你可以在金字塔 1 樓的地下室獲得更多信息。";
	close;
}
