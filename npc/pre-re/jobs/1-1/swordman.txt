function	script	F_JobSwdStaff	{
	mes "[測試大廳人員]";
	mes "你要投降嗎？";
	next;
	if (select("Yes.:No.") == 1) {
		mapannounce "job_sword1","Applicant " + strcharinfo(0) + " quit the test.",bc_map;

		end;
	}
	mes "[測試大廳人員]";
	mes "太棒了！再試一次！";
	close;
}
