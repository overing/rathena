-	script	::Crewman_bra2	-1,{
	mes "[船員]";
	mes "嘿，你有聽說過一個叫做巴西的地方嗎？";
	mes "那是一個熱帶城市，像沙漠一樣炎熱，但又多雨。是一個非常神秘的地方。";
	next;
	mes "[船員]";
	mes "我們最近找到了一條新的海上航線，可以輕鬆到達那裡。";
	if (VIP_SCRIPT) {
		set .@cost, vip_status(VIP_STATUS_ACTIVE)?1000:10000;
		mes "來回一趟只要 10,000 zeny，VIP 只要 1,000！所以你想去嗎？"; //custom translation (VIP)
	} else {
		set .@cost,10000;
		mes "來回一趟只要 10,000 zeny！所以你想去嗎？";
	}
	next;
	switch(select("帶我去巴西！:我待在這裡。")) {
	case 1:
		if (Zeny < .@cost) {
			mes "[船員]";
			mes "我說的是 "+callfunc("F_InsertComma",.@cost)+" zeny。";
			close;
		}
		else {
			mes "[船員]";
			mes "酷喔～！！我們走吧～！";
			set Zeny, Zeny-.@cost;
			close2;
			warp "brasilis",314,60;
			end;
		}
	case 2:
		mes "[船員]";
		mes "如果你有興趣，隨時告訴我，我可以帶你過去。";
		close;
	}
}

brasilis,137,77,5	script	冰淇淋商	85,{
	mes "[冰淇淋商]";
	mes "來～來～";
	mes "冰淇淋是炎熱天氣的完美點心～";
	mes "只要 ^3355FF100 Zeny^000000～";
	mes "冰淇淋～";
	mes "來買冰淇淋！";
	next;
	switch(select("給我一個！:冰淇淋？:取消。")) {
	case 1:
		mes "[冰淇淋商]";
		mes "由於有很多人想要買冰涼的冰淇淋，你一次只能訂購 5 個。";
		mes "所以你想要幾個？";
		next;
		while(1) {
			input .@input; //,1,5;
			if (.@input == 0) {
				mes "[冰淇淋商]";
				mes "沒有？";
				mes "好吧，讓開，我還要服務其他客人。";
				close;
			}
			else if ((.@input < 0) || (.@input > 5)) {
				mes "[冰淇淋商]";
				mes "哇。";
				mes "你點太多了。";
				mes "如果你吃超過 5 個，你可能需要在你的胃裡和怪物戰鬥。冷靜點，老兄。";
				next;
			}
			else
				break;
		}
		set .@icecream_hap,.@input*100;
		if (Zeny < .@icecream_hap) {
			mes "[冰淇淋商]";
			mes "老兄～！你沒有足夠的錢。";
			mes "只要 ^3355FF100 Zeny^000000～ 說真的！";
			close;
		}
		if (!checkweight(536,.@input)) {
			mes "[冰淇淋商]";
			mes "你好像帶太多東西了。";
			mes "在購買這個之前，先減輕你的負擔。";
			close;
		}
		set Zeny, Zeny-.@icecream_hap;
		getitem 536,.@input; //Ice_Cream
		close;
	case 2:
		mes "[冰淇淋商]";
		mes "「冰淇淋是…";
		mes "等等，你不知道";
		mes "什麼是冰淇淋？";
		mes "你一直住在";
		mes "哪個石頭下面？";
		next;
		mes "[冰淇淋商]";
		mes "我甚至不想開始說這聽起來有多奇怪。";
		mes "總之，趁它又好又冷的時候，在這裡買你的冰淇淋。";
		close;
	case 3:
		mes "[冰淇淋商]";
		mes "不要錯過這個機會，來吃這片土地上最棒的冰淇淋～！";
		close;
	}
}
