-	script	::Sailor_izlude	-1,{
	mes "[船員]";
	mes "各位旅客您好！";
	mes "注意了，注意了！";
	mes "快來搭乘令人著迷的";
	mes "船隻，乘風破浪！";
	mes "快點，快點！";
	next;
	switch(select("貝陽島 -> 150 Zeny.:艾爾貝塔碼頭 -> 500 Zeny.:取消。")) {
	case 1:
		if (Zeny < 150) {
			mes "[船員]";
			mes "150 Zeny！";
			mes "只要 150 Zeny就能搭乘！";
			close;
		}
		set Zeny, Zeny - 150;
		warp "izlu2dun",107,50;
		end;
	case 2:
		if (Zeny < 500) {
			mes "[船員]";
			mes "500 Zeny！";
			mes "只要 500 Zeny就能搭乘！";
			close;
		}
		set Zeny, Zeny - 500;
		warp "alberta",188,169;
		end;
	case 3:
		close;
	}
}
