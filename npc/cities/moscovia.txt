	mes "[莫斯科比亞公關人員]";

	mes "在無盡的海洋上擴展的天堂...";
	mes "歡迎來到莫斯科比亞";
	mes "這裡充滿了冒險和神秘。";
	next;
	switch(select("關於莫斯科比亞...:前往莫斯科比亞.:取消")) {
		case 1:
			mes "[莫斯科比亞公關人員]";
			mes "莫斯科比亞是一個美麗的王國";
			mes "位於盧恩-";
			mes "米德加茲北部的島嶼上。";
			next;
			mes "[莫斯科比亞公關人員]";
			mes "我相信你一定會被";
			mes "莫斯科比亞美麗的風景";
			mes "和華麗的宮殿所";
			mes "深深吸引。";
			next;
			mes "[莫斯科比亞公關人員]";
			mes "既然我們漫長的冬天已經";
			mes "過去了，";
			mes "我很高興現在可以向你展示";
			mes "我們華麗的家鄉。";
			next;
			mes "[莫斯科比亞公關人員]";
			mes "如果你想參觀莫斯科比亞";
			mes "現在就抓住機會！";
			mes "我會幫助你有一個美好的";
			mes "莫斯科比亞之旅！";
			close;

		case 2:
			mes "[莫斯科比亞公關人員]";
			mes "好的，那我們現在開始吧。";
			mes "你需要付給我 10,000 zeny";
			mes "才能前往莫斯科比亞。";
			mes "但是當你回來的時候，";
			mes "你就不需要付錢了。";
			next;
			mes "[莫斯科比亞公關人員]";
			mes "我們現在可以出發了嗎？";
			next;
			if (select("出發吧！:取消") == 2) {
				mes "[莫斯科比亞公關人員]";
				mes "如果你現在太忙了，";
				mes "請隨時告訴我你想去。";
				mes "我隨時準備引導任何人前往莫斯科比亞。";
				close;
			}
			if (Zeny < 10000) {
				mes "[莫斯科比亞公關人員]";
				mes "我很抱歉，但你現在沒有";
				mes "足夠的zeny";
				mes "你需要 10,000 zeny";
				mes "才能前往莫斯科比亞";
				mes "謝謝。";
				close;
			} else {
				mes "[莫斯科比亞公關人員]";
				mes "好的，那我們現在出發了。";
				close2;
				set Zeny,Zeny - 10000;
				warp "moscovia",163,55;
				end;
			}
		case 3:
			mes "[莫斯科比亞公關人員]";
			mes "如果你現在太忙了，";
			mes "請隨時告訴我你想去。";
			mes "我隨時準備引導任何人前往莫斯科比亞。";
			close;
	}
}
