lou_in01,25,23,5	script	看起來很親切的小姐#lo	817,{
	mes "[洪淼]";
	mes "歡迎。";
	next;
	mes "[洪淼]";
	mes "這是一個通往";
	mes "觀景塔的電梯。我們以實惠的價格為您提供安全快捷的轉乘服務。您想使用這項服務嗎？";
	next;
	switch(select("資訊.:是的.:下次再說。")) {
	case 1:
		mes "[洪淼]";
		mes "在許多建議和提案被發送到洛陽旅遊局後，建造了觀景塔，以便遊客可以欣賞美景。";
		next;
		mes "[洪淼]";
		mes "由於地理";
		mes "洛陽的特色，很難享受到我們土地所能提供的令人嘆為觀止的景色。";
		next;
		mes "[洪淼]";
		mes "您可以乘坐這裡的電梯上塔。我們以每人 500 zeny 的價格提供這種快速安全的轉乘服務。";
		close;
	case 2:
		if (Zeny < 500) {
			mes "[洪淼]";
			mes "抱歉，您的 zeny 不夠。我希望您以後再來享受觀景塔。祝您有美好的一天。";
			close;
		}
		mes "[洪淼]";
		mes "感謝您的惠顧。";
		mes "我們正在努力為您提供最好的服務。請";
		mes "再次光臨。";
		next;
		set Zeny, Zeny-500;
		warp "lou_in01",17,19;
		end;
	case 3:
		mes "[洪淼]";
		mes "請";
		mes "稍後再來。";
		mes "祝您有美好的一天。";
		close;
	}
