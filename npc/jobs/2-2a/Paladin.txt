valkyrie,53,39,3	script	Paladin#Valkyrie	752,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[聖騎士]";
			mes "恭喜你。";
			mes "向戰士致敬！";
			close;
		}
		mes "[聖騎士]";
		mes "別忘了";
		mes "聖戰";
		mes "即將到來！";
		mes "我們必須做好準備！";
		mes "願正義之光";
		mes "永遠照亮我們的道路！";
		close;
	}
	else if (ADVJOB == Job_Paladin && Class == Job_Swordman_High && JobLevel > 39) {
		mes "[聖騎士]";
		mes "聖戰將";
		mes "在我們意識到之前降臨。";
		mes "我們比以往任何時候都更";
		mes "需要堅強的男人";
		mes "和女人為";
		mes "美好和正確的事物而戰。";
		next;
		mes "[聖騎士]";
		mes "你願意以";
		mes "聖騎士的身分";
		mes "為正義而戰嗎？";
		next;
		if (select("不要:是的。") == 1) {
			mes "[聖騎士]";
			mes "當你準備好的時候，";
			mes "隨時回來。";
			mes "向戰士致敬！";
			close;
		}
		if (SkillPoint) {
			mes "[聖騎士]";
			mes "你仍然可以學習更多技能。請使用";
			mes "所有剩餘的技能點數再回來找我。";
			close;
		}
		jobchange Job_Paladin;
		set ADVJOB,0;
		mes "[聖騎士]";
		mes "恭喜你。";
		mes "身為聖騎士，我希望";
		mes "你能保護那些";
		mes "比你弱小的人，並為";
		mes "我們在即將到來的";
		mes "善與惡之戰中帶來勝利。";
		close;
	}
	mes "[聖騎士]";
	mes "歡迎來到";
	mes "瓦爾哈拉，";
	mes "榮耀殿堂。";
	next;
	mes "[聖騎士]";
	mes "請";
	mes "在這裡";
	mes "放鬆身心。";
	mes "向戰士致敬！";
	close;
}
