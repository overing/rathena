valkyrie,53,50,3	script	Biochemist#Valkyrie	122,{
	if (Class == Job_Creator && bioeth == 13 && getskilllv("AM_BIOETHICS") == 0) {
		mes "[生化學者]";
		mes "啊，你是來";
		mes "取回你失去的";
		mes "記憶的嗎？是的，你";
		mes "一定是為了";
		mes "曾經屬於你的";
		mes "生命奧秘而來的...";
		next;
		if (select("是:否") == 1) {
			mes "[生化學者]";
			mes "閉上你的眼睛並";
			mes "讓你的思緒平靜下來。";
			mes "我們將回到你的";
			mes "過去，去收集你失去的";
			mes "記憶碎片。";
			next;
			mes "[生化學者]";
			mes "當你睜開眼睛時，";
			mes "你將清楚地記得";
			mes "生命的奧秘。你也會";
			mes "記得利用這些";
			mes "秘密來達到正確目的的";
			mes "責任之重...";
			next;
			skill "AM_BIOETHICS",1,SKILL_PERM;
			mes "[生化學者]";
			mes "睜開你的眼睛...";
			mes "既然你已經";
			mes "記起如何";
			mes "創造人造生命，我只";
			mes "要求你尊重你所有的";
			mes "造物。";
			close;
		}
		mes "[生化學者]";
		mes "如果你想";
		mes "取回你失去的";
		mes "記憶，請";
		mes "回到我身邊。";
		mes "創造";
		mes "生命的秘密絕非小事...";
		close;
	}
	else if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[生化學者]";
			mes "恭喜你。";
			mes "向戰士們致敬！";
			close;
		}
		mes "[生化學者]";
		mes "像我這樣的人";
		mes "在這裡真奇怪。";
		mes "但即使是擅長";
		mes "科學方法的人";
		mes "也能成為英雄。";
		next;
		mes "[生化學者]";
		mes "在這種情況下，";
		mes "不一定是我使用的";
		mes "手段，而是";
		mes "我為之奮鬥的";
		mes "目標，為我贏得了";
		mes "在瓦爾哈拉的一席之地...";
		close;
	}
	else if (ADVJOB == Job_Creator && Class == Job_Merchant_High && JobLevel > 39) {
		mes "[生化學者]";
		mes "是的...";
		mes "是時候了。";
		mes "我們在米德加爾特需要更多";
		mes "像你這樣的天才。";
		next;
		mes "[生化學者]";
		mes "你想成為";
		mes "生化學者嗎？";
		next;
		if (select("否.:是。") == 1) {
			mes "[生化學者]";
			mes "當你準備好的時候，";
			mes "隨時回來找我。";
			mes "向戰士們致敬！";
			close;
		}
		if (SkillPoint) {
			mes "[生化學者]";
			mes "你仍然有可能學習更多技能。請在使用";
			mes "完所有剩餘的技能點數後再回來找我。";
			close;
		}
		jobchange Job_Creator;
		if (bioeth == 13)
			skill "AM_BIOETHICS",1,SKILL_PERM;
		set ADVJOB,0;
		mes "[生化學者]";
		mes "恭喜你！";
		mes "身為生化學者，";
		mes "我希望你為了";
		mes "正確的目的使用你";
		mes "廣博的知識。";
		close;
	}
	else {
		mes "[生化學者]";
		mes "歡迎";
		mes "來到瓦爾哈拉，";
		mes "榮耀大廳。";
		next;
		mes "[生化學者]";
		mes "請";
		mes "在這裡";
		mes "放輕鬆。";
		mes "向戰士們致敬！";
		close;
	}
}
