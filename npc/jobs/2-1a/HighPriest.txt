valkyrie,44,42,5	script	High Priest#Valkyrie	60,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[高階祭司]";
			mes "恭喜。";
			mes "向戰士們致敬！";
			close;
		}
		mes "[高階祭司]";
		mes "藉由神聖";
		mes "的力量，願我們";
		mes "找到平靜、力量";
		mes "和保護。將我們";
		mes "從邪惡勢力中拯救...";
		close;
	}
	else if (ADVJOB == Job_High_Priest && Class == Job_Acolyte_High && JobLevel > 39) {
		mes "[高階祭司]";
		mes "我們的世界需要";
		mes "有才能和信念的人。";
		mes "請繼續您的";
		mes "善行，成為一位更";
		mes "偉大的神聖英雄...";
		next;
		mes "[高階祭司]";
		mes "您想要成為";
		mes "高階祭司嗎？";
		next;
		if (select("不要。:是的。") == 1) {
			mes "[高階祭司]";
			mes "當您準備好時，";
			mes "隨時回來。";
			mes "向戰士們致敬！";
			close;
		}
		if (SkillPoint) {
			mes "[高階祭司]";
			mes "您仍然可以學習更多技能。請在使用完剩餘的所有技能點數後再回來找我。";
			close;
		}
		jobchange Job_High_Priest;
		set ADVJOB,0;
		mes "[高階祭司]";
		mes "恭喜。";
		mes "身為高階祭司，";
		mes "我希望您能引導";
		mes "其他人走上";
		mes "神聖之路...";
		close;
	}
	mes "[高階祭司]";
	mes "歡迎來到";
	mes "瓦爾哈拉，";
	mes "榮譽殿堂。";
	next;
	mes "[高階祭司]";
	mes "請在這裡";
	mes "放鬆身心。";
	mes "向戰士們致敬！";
	close;
}
