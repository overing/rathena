valkyrie,44,47,5	script	High Wizard#Valkyrie	735,{
	if (ADVJOB == 0 ||  Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[高等巫師]";
			mes "恭喜。";
			mes "向戰士們致敬！";
			close;
		}
		mes "[高等巫師]";
		mes "我們高等巫師有";
		mes "責任去";
		mes "為了正確的目的";
		mes "使用我們具有破壞性的魔法。";
		next;
		mes "[高等巫師]";
		mes "要成為";
		mes "高等巫師，需要";
		mes "一生的訓練。你能想像";
		mes "如果我們的力量";
		mes "被放在錯誤的人手中會發生什麼事嗎？！";
		close;
	}
	else if (ADVJOB == Job_High_Wizard && Class == Job_Mage_High && JobLevel > 39) {
		mes "[高等巫師]";
		mes "時機到了。";
		mes "而米德加爾特";
		mes "需要那些能";
		mes "運用最強大魔法的人...";
		next;
		mes "[高等巫師]";
		mes "你想成為";
		mes "高等巫師嗎？";
		next;
		if (select("不要。:是的。") == 1) {
			mes "[高等巫師]";
			mes "當你準備好時，";
			mes "隨時可以回來。";
			mes "向戰士們致敬！";
			close;
		}
		if (SkillPoint) {
			mes "[高等巫師]";
			mes "你仍然有可能學習更多技能。請在使用";
			mes "完所有剩餘的技能點數後再回來找我。";
			close;
		}
		jobchange Job_high_Wizard;
		set ADVJOB,0;
		mes "[高等巫師]";
		mes "恭喜。";
		mes "身為一位高等巫師，";
		mes "我希望你使用";
		mes "你的力量來為";
		mes "受壓迫者帶來和平。";
		close;
	}
	mes "[高等巫師]";
	mes "歡迎來到";
	mes "瓦爾哈拉，";
	mes "榮譽殿堂。";
	next;
	mes "[高等巫師]";
	mes "請";
	mes "在這裡";
	mes "好好放鬆。";
	mes "向戰士們致敬！";
	close;
}
