valkyrie,44,39,5	script	Lord Knight#Valkyrie	56,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[騎士領主]";
			mes "恭喜。";
			mes "向戰士致敬！";
			close;
		}
		mes "[騎士領主]";
		mes "我們騎士有著";
		mes "重大的責任...";
		mes "去服務與保護。";
		next;
		mes "[騎士領主]";
		mes "即使要付出";
		mes "我們自己的生命，";
		mes "我們也必須保護";
		mes "我們同伴的福祉。";
		close;
	}
	else if (ADVJOB == Job_Lord_Knight && Class == Job_Swordman_High && JobLevel > 39) {
		mes "[騎士領主]";
		mes "你的時刻到了！";
		mes "世界仍然需要你。";
		mes "請以新的外觀，";
		mes "繼續你的英雄生涯。";
		next;
		mes "[騎士領主]";
		mes "你想要";
		mes "成為騎士領主嗎？";
		next;
		if (select("不要。:是的。") == 1) {
			mes "[騎士領主]";
			mes "當你準備好的時候，";
			mes "隨時歡迎回來。";
			mes "向戰士致敬！";
			close;
		}
		if (SkillPoint) {
			mes "[騎士領主]";
			mes "你仍然有可能學習更多技能。請在使用";
			mes "完你所有剩餘的技能點數後再回來找我。";
			close;
		}
		jobchange Job_Lord_Knight;
		set ADVJOB,0;
		mes "[騎士領主]";
		mes "恭喜！";
		mes "身為騎士領主，";
		mes "我希望你能夠";
		mes "站在戰鬥的最前線，";
		mes "並帶領你的盟友走向勝利！";
		close;
	}
	mes "[騎士領主]";
	mes "歡迎";
	mes "來到英靈殿，";
	mes "榮譽殿堂。";
	next;
	mes "[騎士領主]";
	mes "請在這裡";
	mes "盡情放鬆，";
	mes "享受舒適的時光。";
	mes "向戰士致敬！";
	close;
}
