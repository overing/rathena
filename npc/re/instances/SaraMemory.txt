dali,139,118,4	script	次元移動機#sara	10007,{
	if (BaseLevel < 99) {
		mes "[探險家雷文]";
		mes "嘿！ 你太年輕了，不能玩這個！ 至少達到 ^0000FF等級 99^000000 後再來！";
		close;
	}
	if ((saramemory < 1) || (saramemory == 2) || (saramemory == 3) || (saramemory == 5)) {
		mes "[探險家雷文]";
		mes "在你玩弄它之前； 你為什麼不過來和我談談呢？";
		close;
	}

	// 20 hour cooldown
	set .@ins_sara_time,checkquest(15002,PLAYTIME);
	if (.@ins_sara_time == -1) {

		set .@party_id,getcharid(1);
		set .@p_name$,getpartyname(.@party_id);
		set .@md_name$,"莎拉的記憶";

		if (!.@party_id) {
			mes "[探險家雷文]";
			mes "在你進入之前，你需要組織一個隊伍！ 你不知道怎麼做嗎？ 輸入 /organize 隊伍名稱在此。";
			close;
		}

		if (is_party_leader() == true)
			set .@menu$, "啟動維度設備。:使用維度設備。:取消。";
		else
			set .@menu$, ":使用維度設備。:取消。";

		switch(select(.@menu$)) {
		case 1:
			if (instance_create(.@md_name$) < 0) {
				mes "隊伍名稱: "+.@p_name$;
				mes "隊伍隊長: "+strcharinfo(0);
				mes "^0000ff"+.@md_name$+" ^000000- 預約失敗!";
				close;
			}
			mes "^FF0000次元裝置已順利啟動。使用該裝置進入時空的裂縫。^000000";
			close;
		case 2:
			switch(instance_enter(.@md_name$)) {
			case IE_OTHER:
				mes "發生不明錯誤。";
				close;
			case IE_NOINSTANCE:
				mes "紀念迷宮 "+.@md_name$+" 不存在。";
				mes "隊伍隊長尚未產生該迷宮。";
				close;
			case IE_NOMEMBER:
				mes "只有已註冊的成員才能進入 "+.@md_name$+" 副本。";
				close;
			case IE_OK:
				mapannounce "dali",strcharinfo(0)+", 隊伍 "+.@p_name$+" 的成員進入了 "+.@md_name$+" 副本。",bc_map,"0x00ff99";
				setquest 15002;
				end;
			}
		case 3:
			mes "^FF0000隊伍隊長尚未啟動次元裝置。^000000";
			close;
		}
	} else if (.@ins_sara_time == 0 || .@ins_sara_time == 1) {
		mes "[冒險者雷文]";
		mes "我想在你上次進入裂縫後大約 20 小時，它會變得足夠穩定，讓你再次穿越它。";
		close;
	}
	else {
		erasequest 15002;
		mes "^0000FF[ 莎拉的記憶 ] 冷卻時間已清除。現在可以再次使用次元裝置。^000000";
		close;
	}
}
