		close;
	}

	case -1:
		break;
	case 0:
	case 1:
		mes "- 次元裝置已關閉。";
		mes "似乎無法進入。 -";
		close;
	case 2:
		break;
	}
	if (isbegin_quest(9335) > 0) {
		mes "- 次元裝置已關閉。";
		mes "似乎無法進入。 -";
		close;
	}
	if (select( "不要進入裝置。", "進入時間裝置" ) == 1) {
		if (getcharid(1) > 0) {
			mes "- 看來次元裝置 ";
			mes "會扭曲時空 -";
			close;
		}
		mes "- 這裡有一股潮濕的氣息。 -";
		close;
	}
	.@md_name$ = "Fenrir and Sarah";
	switch( instance_enter(.@md_name$) ) {
	case IE_OTHER:
		mes "發生未知的錯誤。";
		close;
	case IE_NOINSTANCE:
		mes "副本 " + .@md_name$ + " 不存在。";
		mes "隊長尚未建立副本。";
		close;
	case IE_NOMEMBER:
		mes "只有隊伍成員才能進入副本。";
		close;
	case IE_OK:
		mapannounce "dali02", "隊伍 " + getpartyname( getcharid(1) ) + " 的成員 " + strcharinfo(0) + " 進入房間 " + .@md_name$ + "", bc_map,0xFF99,FW_NORMAL,12;
		setquest 9335;// Search for shards of Gigantes
		if (isbegin_quest(9336) > 0)
			erasequest 9336;
		// warp "1@glast",367,304;
		end;
	}
}
