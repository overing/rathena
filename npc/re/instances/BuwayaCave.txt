ma_fild02,312,317,5	script	Guard#buwaya_cave	570,{
	if (BaseLevel < 130) {
		mes "[守衛]";
		mes "等級 ^ff0000130^000000 以下的人";
		mes "無法進入這個地方。";
		mes "這個地方很危險。請回去吧。";
		close;
	}

	set .@party_id,getcharid(1);
	set .@md_name$,"Buwaya Cave";

	if (!.@party_id) {
		mes "[守衛]";
		mes "布瓦亞很危險。";
		mes "請與其他人組隊後再來。";
		close;
	}
	if (is_party_leader() == true) {
		mes "[守衛]";
		mes "這個地方很危險。請回去吧。";
		mes "如果你沒有任何事情，請回去吧。";
		next;
		while(1) {
			switch(select("為什麼這麼危險？:我是來獵殺布瓦亞的。:結束對話。")) {
			case 1:
				mes "[守衛]";
				mes "最近，發生了一起";
				mes "村民失?的案件。";
				mes "在調查蹤跡後，";
				mes "我們發現是巨型怪物布瓦亞";
				mes "造成的。";
				next;
				mes "[守衛]";
				mes "士兵和傭兵已被派遣去";
				mes "獵殺布瓦亞，但從未";
				mes "完全成功。";
				next;
				mes "[守衛]";
				mes "直到今天，布瓦亞仍在綁架";
				mes "村民。";
				mes "請召集一些人來除掉布瓦亞。";
				next;
				break;
			case 2:
				set .@playtime, checkquest(4229,PLAYTIME);
				if (.@playtime == -1) {
					// fall through
				} else if (.@playtime == 0 || .@playtime == 1) {
					mes "[守衛]";
					mes "布瓦亞仍然躲藏著。";
					mes "即使你現在進入，布瓦亞也不會出來。";
					mes "請稍後再來。";
					close;
				} else {
					erasequest 4229;
					// fall through
				}
				if (instance_create(.@md_name$) < 0) {
					mes "[守衛]";
					mes "隊伍名稱是... "+getpartyname(.@party_id)+".";
					mes "隊伍領袖是... "+strcharinfo(0);
					mes "^0000ff"+.@md_name$+"^000000 現在無法開啟。";
					mes "請稍後再試。";
					close;
				}
				mes "[守衛]";
				mes "我將打開通往 ^0000ff"+.@md_name$+"^000000 的隧道。";
				mes "祝你好運。";
				close;
			case 3:
				mes "[守衛]";
				mes "這個地方很危險。你不能進入。";
				close;
			}
		}
	} else {
		mes "[守衛]";
		mes "讓我跟隊伍領袖談談。";
		mes "請叫隊伍領袖出來。";
		close;
	}
}

ma_fild02,315,323,0	script	Cave Entrance#buwaya	45,2,2,{
	end;
OnTouch:
	if (BaseLevel < 130) {
		mes "[守衛]";
		mes "等級 ^ff0000130^000000 以下的人";
		mes "無法進入這個地方。";
		mes "這個地方很危險。請回去吧。";
		close;
	}
	set .@playtime, checkquest(4229,PLAYTIME);
	if (.@playtime == -1) {
		// fall through
	} else if (.@playtime == 0 || .@playtime == 1) {
		mes "[守衛]";
		mes "布瓦亞仍然躲藏著。";
		mes "即使你現在進入，布瓦亞也不會出來。";
		mes "請稍後再來。";
		close;
	} else {
		erasequest 4229;
		// fall through
	}
	switch(select("進入。:返回。")) {
	case 1:
		switch(instance_enter("Buwaya Cave")) {
		case IE_OTHER:
			mes "[守衛]";
			mes "哦，現在不是好時機。";
			mes "請稍後再試。";
			close;
		case IE_NOINSTANCE:
		case IE_NOMEMBER:
			mes "[守衛]";
			mes "這個地方很危險。請不要進入。";
			close;
		case IE_OK:
			mapannounce "ma_fild02",getpartyname(getcharid(1))+" 隊伍的 "+strcharinfo(0)+" 成員開始在布瓦亞洞穴中獵殺布瓦亞。",bc_map,"0x00ff99"; //FW_NORMAL 12 0 0
			setquest 4229;
			//warp "1@ma_c",35,57;
			end;
		default:
			mes "- 由於不明力量，無法進入。 -";
			close;
		}
	case 2:
		mes "[守衛]";
		mes "你必須珍惜你的生命。";
		mes "最好在行動之前仔細考慮。";
		close;
	}
}
