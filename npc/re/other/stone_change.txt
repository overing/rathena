	mes "[礦石測試員 索提]";
	mes "你的物品欄裡有 ^3131FF任何佔用大量空間的同類型礦石^000000嗎？";
	mes "我可以立即區分你的礦石並整理它們～！";
	next;
	switch(select("提煉我的礦石。:取消。")) {
	case 1:
		mes "[礦石測試員 索提]";
		mes "目前我只能區分 ^FF0000鋼鐵^000000、^FF0000鐵^000000、^FF0000鐵礦石^000000 和 ^FF0000粗糙的奧里吉金^000000 並整理它們。";
		mes "請注意，^3131FF其他物品無法整理^000000！";
		next;
		mes "[礦石測試員 索提]";
		mes "那麼，你想要我整理你的礦石嗎？";
		next;
		.@i = select("請整理鋼鐵。:請整理鐵。:請整理鐵礦石。:請整理粗糙的奧里吉金。:不，謝謝。");
		if (.@i == 5) {
			mes "[礦石測試員 索提]";
			mes "呼，我真的很討厭邋遢的人！";
			close;
		}
		setarray .@ores[0],999,998,1002,756;
		.@item = .@ores[.@i-1];
		if (countitem(.@item) == 0) {
			mes "[礦石測試員 索提]";
			mes "你沒有任何"+getitemname(.@item)+"...";
			close;
		}
		// while(1) {
			// if (countitem(.@item)) {
				// set .@count, countitem(.@item);
				// set .@total, .@total + .@count;
				// delitem .@item, .@count;
			// } else {
				// getitem .@item, .@total;
				// break;
			// }
		// }
		.@count = countitem(.@item);
		delitem .@item, .@count;
		getitem .@item, .@count;
		mes "[礦石測試員 索提]";
		mes "完成了。看起來很棒！你不覺得嗎？";
		close;
	case 2:
		mes "[礦石測試員 索提]";
		mes "看來你的包包很乾淨？呼呼！你可能是清潔大師！";
		close;
	}
}
