//===== rAthena Script =======================================
//= Dynamic NPC: Kachua's Secret Box
//===== Description: =========================================
//- [Official conversion]
//= Kachua's Secret Box is a Gachapon NPC.
//= It lets the player exchange a Kachua's Secret Key for a random item.
//= It also gives a Kachua's Mileage Coupon for each pull.
//===== Changelogs: ==========================================
//= 1.0 First version. [secretdataz]
//============================================================

// !!! 伺服器程式碼寫死 NPC 名稱改動會無法觸發道具功能 !!!
sec_in02,126,178,3	script	Kachua's Secret Box#bm	4_TREASURE_BOX,{
	.@key$ = "K_Secret_Key";
	.@keyname$ = getitemname( .@key$ );
	.@box$ = "Main_Lucky_Box";
	.@boxname$ = getitemname( .@box$ );
	mes "一個沒人知道裡面有什麼的神秘盒子。";
	mes "^4d4dff你可以消耗 " + mesitemlink( .@key$ ) + " 來打開這個盒子。^000000";
	next;
	switch(select("^4d4dff打開盒子 1 次 (1 個 " + .@keyname$ + ")^000000","^4d4dff打開盒子 10 次 (10 個 " + .@keyname$ + "s)^000000")) {
		case 1:
			mes "^FF0000[注意]^000000";
			mes "^FF0000使用 1 個 " + .@keyname$ + " 隨機交換一個物品。^000000";
			mes "^FF0000交換的物品不能取回，也不能換回 " + .@keyname$ + "。^000000";
			next;
			if(select("繼續。","停止對話。") == 2) {
				mes "你決定不打開盒子。";
				close;
			} else if (countitem(.@key$) < 1) {
				mes "沒有足夠的 " + mesitemlink( .@key$ ) + "。";
				close;
			} else {
				if (checkweight(1201,1) == 0 || ((MaxWeight - Weight) * 100 / MaxWeight) < 10) {
					mes "^4d4dff請確認你的背包有足夠的空間。^000000";
					close;
				}
				delitem(.@key$, 1);
				consumeitem(.@box$);
				mes .@boxname$ + " 被打開了！";
				mes "你幸運嗎？";
				specialeffect2 EF_VALLENTINE;
				close;
			}
		case 2:
			mes "^FF0000[注意]^000000";
			mes "^FF0000使用 10 個 " + .@keyname$ + "s 隨機交換 10 個物品。^000000";
			mes "^FF0000交換的物品不能取回，也不能換回 " + .@keyname$ + "。^000000";
			next;
			if(select("繼續。","停止對話。") == 2) {
				mes "你決定不打開盒子。";
				close;
			} else if (countitem(.@key$) < 10) {
				mes "你沒有足夠的 " + mesitemlink( .@key$ ) + "。";
				close;
			} else {
				for (.@i = 1; .@i <= 10; ++.@i) {
					progressbar "4d4dff",2;
					if (checkweight(1201,1) == 0 || ((MaxWeight - Weight) * 100 / MaxWeight) < 10) {
						mes "^4d4dff請確認你的背包有足夠的空間。^000000";
						close;
					}
					if (countitem(.@key$) < 1) { // Custom check, just in case
						close;
					}
					delitem(.@key$, 1);
					consumeitem(.@box$);
					dispbottom .@boxname$ + " was opened " + .@i + " times. Another one is being opened.",0xFFFFFF;
					specialeffect2 EF_VALLENTINE;
				}
				mes .@boxname$ + " 被打開了 10 次！";
				mes "你幸運嗎？";
				close;
			}
	}
}
