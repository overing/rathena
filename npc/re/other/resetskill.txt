izlude,127,175,3	script	催眠師#novice	124,{
	mes "[催眠師]";
	mes "您好，冒險者。";
	mes "我是催眠師學院的一員";
	mes "被派到艾音布羅克";
	mes "以非常優惠的價格：免費！為特定的初心者角色提供技能重置服務。";
	next;
	mes "[催眠師]";
	mes "雖然我現在提供無限的";
	mes "屬性/技能重置，但我有";
	mes "兩個條件必須";
	mes "滿足。首先，你必須";
	mes "低於 ^FF0000Base 等級 50^000000。";
	next;
	mes "[催眠師]";
	mes "第二，你必須是";
	mes "^FF0000劍士、服事、法師、";
	mes "盜賊、弓箭手、商人、";
	mes "跆拳道男孩或跆拳道女孩、";
	mes "槍手和忍者^000000";
	mes "才能符合資格。";
	mes "現在，您有任何問題嗎？";
	next;
	switch(select("^FF0000屬性/技能重置？:重置屬性:重置技能^000000:取消")) {
	case 1:
		mes "[催眠師]";
		mes "屬性/技能重置允許冒險者";
		mes "如果他們對目前的技能";
		mes "不滿意，可以重新分配他們的技能";
		mes "點數。";
		next;
		mes "[催眠師]";
		mes "在進行";
		mes "屬性/技能重置之前，您必須";
		mes "減少所有";
		mes "角色所攜帶物品的重量";
		mes "到 0。您可以將";
		mes "多餘的物品放入卡普拉倉庫。";
		next;
		mes "[催眠師]";
		mes "啊，還有重要的是";
		mes "移除您的 ^FF0000手推車^000000";
		mes "如果你裝備了手推車的話。";
		mes "否則，催眠術將無法";
		mes "工作，或會適得其反...";
		close;
	case 2:
		set .@str$,"Stat";
	case 3:
		if (.@str$ == "")
			set .@str$,"Skill";
		mes "[催眠師]";
		mes "您確定您";
		mes "想要繼續進行";
		mes "我的 ^FF0000"+.@str$+" 重置^000000 服務嗎？";
		next;
		if(select("是:取消") == 1)
			break;
	case 4:
		mes "[催眠師]";
		mes "謝謝您，祝您";
		mes "在冒險中好運。";
		mes "請安全旅行～";
		close;
	}
	if ((Class >= Job_Swordman && Class <= Job_Thief) || Class == Job_Taekwon || Class == Job_Gunslinger || Class == Job_Ninja) {
		if (checkcart()) {
			mes "[催眠師]";
			mes "噢！請在繼續進行";
			mes ""+.@str$+"重置服務前移除您的手推車。";
			mes "感謝您的合作～";
			close;
		}
		if (Weight != 0) {
			mes "[催眠師]";
			mes "如果您來這裡是為了我的 "+.@str$;
			mes "重置服務，請";
			mes "記住您無法";
			mes "重置您的 "+.@str$+"s，直到";
			mes "^FF0000您的物品欄中攜帶的物品重量減少到 0^000000。";
			next;
			mes "[催眠師]";
			mes "您為什麼不把您的";
			mes "東西放到卡普拉倉庫";
			mes "暫時？這樣，您就可以";
			mes "安全地保存您的所有物品。";
			close;
		}
		if (BaseLevel >= 50 && (Class == Job_Taekwon || Class == Job_Gunslinger || Class == Job_Ninja)) {
			mes "[催眠師]";
			mes "很抱歉，但是基礎等級";
			mes "高於 50 的角色不符合我提供的";
			mes .@str$+" 重置服務的資格。";
			close;
		}
		mes "[催眠師]";
		mes "感謝您使用";
		mes "我的 "+.@str$+" 重新分配";
		mes "服務。噢，祝您";
		mes "旅途愉快，";
		mes "冒險者。";
		if (.@str$ == "Stat")
			resetstatus;
		else
			resetskill;
		close;
	} else {
		mes "[催眠師]";
		mes "很抱歉，但是您的";
		mes "職業不符合";
		mes "我提供的 "+.@str$+" 重置服務";
		mes "的資格。我只能";
		mes "提供 "+.@str$+" 重置給";
		mes "以下職業...";
		next;
		mes "[催眠師]";
		mes "^FF0000劍士、服事、";
		mes "法師、盜賊、弓箭手、";
		mes "商人、跆拳道";
		mes "男孩、跆拳道女孩、";
		mes "槍手和忍者^000000。";
		close;
	}
}
izlude_a,127,175,3	duplicate(催眠師#novice)	催眠師#novice_a	124
