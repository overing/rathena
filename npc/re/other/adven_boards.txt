function	script	F_AvenBoard	{
	.@mo_board = select( "依樓層查詢怪物",
					"Monster Information Retrieval System",
					( countitem(909) == 30000 ? "Request" : "" ),
					"Cancel" );
	if (.@mo_board == 1)
		return 1;
	else if (.@mo_board == 2) {
		while(1) {
			mes "[怪物資訊檢索系統]";
			mes "這裡是怪物資訊檢索系統。可以搜尋出現在這個地牢中的怪物資訊。搜尋時請輸入怪物名稱，包含空格。";
			next;
			input .@inputstr$;
			setarray .@element$[0],"Neutral","Water","Earth","Fire","Wind","Poison","Holy","Shadow","Ghost","Undead";
			setarray .@size$[0],"Medium","Small","Big";
			setarray .@race$[0],"Formless","Undead","Animal","Plant","Insect","Fish","Demon","Demihuman","Angel","Dragon";
			.@size = getargcount();
			.@i = 0;
			while( .@i < .@size && getarg(.@i+1,"") != .@inputstr$ ) .@i += 3;
			mes "[怪物資訊檢索系統]";
			mes "現在搜尋中";
			next;
			mes "[怪物資訊檢索系統]";
			mes "搜尋已完成。";
			mes "現在顯示搜尋結果。";
			next;
			mes "[怪物資訊檢索系統]";
			if (.@i >= .@size) {
				mes "沒有該怪物的可用資訊。";
				mes "請檢查您的怪物名稱。";
			}
			else {
				.@mob_id = getarg(.@i);
				mes "- 名稱 : "+ getarg(.@i+1);
				mes "- 屬性 : "+ .@element$[ getmonsterinfo(.@mob_id,MOB_ELEMENT) ];
				mes "- 尺寸 : "+ .@size$[ getmonsterinfo(.@mob_id,MOB_SIZE) ];
				mes "- 種族 : "+ .@race$[ getmonsterinfo(.@mob_id,MOB_RACE) ];
				mes "--------------------------------^FD8D02";
				mes getarg(.@i+2);
			}
			next;
			mes "[怪物資訊檢索系統]";
			mes "您要搜尋其他怪物資訊嗎？";
			next;
			if (select( "搜尋","取消" ) == 1)
				continue;
			mes "[怪物資訊檢索系統]";
			mes "怪物搜尋系統現在將關閉。";
			next;
			return 2;
		}
	}
	else if (.@mo_board == 3) {
		mes "請求的資訊將會顯示 - 如果有顯示就沒有問題";
		next;
		return 3;
	}
	else if (.@mo_board == 4)
		close;
	else {
		mes "操作錯誤。";
		close;
	}
}

function	script	F_InfoBoard	{
	next;
	mes "[冒險者公會]";
	mes "如需更多關於怪物的資訊";
	mes "必須使用怪物資訊檢索系統。";
	next;
	return;
}
