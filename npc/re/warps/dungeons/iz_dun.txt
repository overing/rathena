iz_dun04,130,234,5	script	Gatekeeper#iz_dun	413,{
	mes "[守門人]";
	mes "止步！";
	mes "冒險者，是什麼風把你吹到這裡？";
	mes "我們在這個水底城市發現了一個新的樓層。";
	next;
	mes "[守門人]";
	mes "你認為你準備好迎接挑戰了嗎？";
	mes "這裡的怪物等級介於 110 到 124 級之間。";
	mes "而且你必須小心克拉肯！";
	next;
	switch(select("釋放克拉肯！:我會待在這裡。")) {
	case 1:
		mes "[守門人]";
		mes "小心點，祝你好運。";
		mes "如果你看到克拉肯，務必確保你的朋友能幫上忙，或者像我們一樣逃跑。";
		close2;
		warp "iz_dun05",141,187;
		end;
	case 2:
		mes "[守門人]";
		mes "在探索新區域之前做好準備總是最好的。";
		mes "準備好再去吧。";
		close;
	}
}
