prontera,246,62,3	script	Training Instructor#sol1	8W_SOLDIER,{
	mes "[訓練教官]";
	mes "如果你想在更廣闊的區域訓練，我會讓你進去。";
	next;
	mes "[訓練教官]";
	mes "訓練場地分為兩個主要區域。";
	next;
	mes "[訓練教官]";
	mes "上方區域，你可以在那裡根據魔物的大小檢查施加的力量。";
	next;
	mes "[訓練教官]";
	mes "下方區域，你可以在那裡根據魔物的屬性和形態檢查施加的力量。";
	next;
	mes "[訓練教官]";
	mes "你想進入哪個區域？";
	next;
	switch( select( "上方區域", "下方區域", "不進入" ) ) {
	case 1:
		mes "[訓練教官]";
		mes "那麼我會引導你。";
		close2;
		warp "tra_fild",112,167;
		end;
	case 2:
		mes "[訓練教官]";
		mes "那麼我會引導你。";
		close2;
		warp "tra_fild",111,74;
		end;
	case 3:
		mes "[訓練教官]";
		mes "如果你需要引導，來找我。";
		close;
	}
}
