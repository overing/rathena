-	script	GuildWarehouse	-1,{
	mes "[倉庫管理員]";
	mes "你好嗎？我們專門經營公會倉庫。這是我們的 ^0000cd故事^000000：";
	mes "為什麼公會成員不能共享一個倉庫？我們從這個簡單的問題開始。";
	next;
	mes "[倉庫管理員]";
	mes "你可以花費 1000 Zeny 開啟倉庫，或查看公會的使用紀錄。";
	mes "我可以幫你什麼忙？";
	next;
	if( select( "開啟公會倉庫:查看倉庫使用紀錄" ) == 1 ){
		.@guildid = getcharid( 2 );

		if( .@guildid == 0 ){
			mes "[倉庫管理員]";
			mes "公會倉庫僅供公會成員使用。";
			close;
		}

		if( getgdskilllv( .@guildid, "GD_GUILD_STORAGE" ) == 0 || !guild_has_permission( GUILD_PERM_STORAGE ) ){
			mes "[倉庫管理員]";
			mes "您的公會似乎還不能使用公會倉庫，或者您沒有存取倉庫的權限。";
			mes "請在確認公會技能和存取倉庫的權限後再來。";
			close;
		}

		if( Zeny < 1000 ){
			mes "[倉庫管理員]";
			mes "恐怕你沒有足夠的錢來支付費用。";
			mes "費用是 1000 Zeny。";
			close;
		}

		mes "[倉庫管理員]";
		mes "那我為您開啟公會倉庫。祝您有個難忘的時光！";
		close2;

		if( Zeny < 1000 ){
			// Cheat prevention
			end;
		}

		if( guildopenstorage() == GSTORAGE_OPEN ){
			Zeny -= 1000;
			end;
		}else{
			mes "[倉庫管理員]";
			mes "恐怕現在有人正在使用倉庫。";
			mes "請稍後再試。";
			close;
		}
	}else{
		.@guildid = getcharid( 2 );

		if( .@guildid == 0 ){
			mes "[倉庫管理員]";
			mes "公會倉庫僅供公會成員使用。";
			close;
		}

		if( getgdskilllv( .@guildid, "GD_GUILD_STORAGE" ) == 0 || !guild_has_permission( GUILD_PERM_STORAGE ) ){
			mes "[倉庫管理員]";
			mes "您的公會似乎還不能使用公會倉庫，或者您沒有存取倉庫的權限。";
			mes "請在確認公會技能和存取倉庫的權限後再來。";
			close;
		}

		mes "[倉庫管理員]";
		mes "我將向您展示倉庫的使用歷史。使用歷史將保留最多 3 個月。";
		mes "祝您有個難忘的時光！";
		close2;
		guildopenstorage_log();
		end;
	}
}
