// Main NPC :: mal_jerun
//============================================================
malangdo,221,174,6	script	Holink#mal_cash	559,{
	disable_items;
	mes "[霍林克]";
	mes "我是喵～鐵匠霍林克～";
	mes "精鍊大師，霍林克～";
	mes "我是向莫洛克學習的特別貓咪霍林克～";
	mes "我的女兒以我為榮，霍林克～";
	mes "今天霍林克～要精鍊什麼呢？";
	next;

	setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
	for (set .@i,1; .@i<=10; set .@i,.@i+1)
		set .@menu$, .@menu$ + ( getequipisequiped(.@indices[.@i]) ? getequipname(.@indices[.@i]) : F_getpositionname(.@indices[.@i]) +"-[Empty]" ) +":";
	set .@part, .@indices[ select(@menu$) ];

	if (!getequipisequiped(.@part)) {
		mes "[霍林克]";
		switch(.@part) {
		case EQI_HEAD_TOP:
			mes "我的老師阿拉岡說喵～";
			mes "愚蠢是沒有藥醫的...";
			break;
		case EQI_ARMOR:
			mes "這裡沒什麼好看的，喵！！";
			break;
		case EQI_HAND_L:
			mes "喵？你想要我用這隻左手做什麼...？";
			break;
		case EQI_HAND_R:
			mes "喵？你想要我用這隻右手做什麼...？";
			break;
		case EQI_GARMENT:
			mes "喵？你身上什麼都沒穿。";
			break;
		case EQI_SHOES:
			mes "Kyang~! 不要弄我的敏感嗅覺，喵～。";
			break;
		case EQI_ACC_L:
		case EQI_ACC_R:
			mes "喵？飾品在哪裡？";
			break;
		case EQI_HEAD_MID:
		case EQI_HEAD_LOW:
			mes "喵？你說的是頭部其他部位，喵？～";
			break;
		}
		close;
	}
	if (!getequipisenableref(.@part)) {
		mes "[霍林克]";
		mes "連阿拉岡都無法精鍊這種東西，喵。";
		close;
	}
	if (getequiprefinerycnt(.@part) >= 10) {
		mes "[霍林克]";
		mes "喵～完美的精鍊。阿拉岡做的嗎，喵？～";
		close;
	}
	mes "[霍林克]";
	.@refineitemid = getequipid(.@part); // save id of the item
	.@itemtype = getiteminfo( .@refineitemid, ITEMINFO_TYPE );
	.@refinerycnt = getequiprefinerycnt(.@part); //save refinery count
	setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);
// Holink has different hardcoded prices
//	.@price = getequiprefinecost(.@part, REFINE_COST_NORMAL, REFINE_ZENY_COST);
	.@material = getequiprefinecost(.@part, REFINE_COST_NORMAL, REFINE_MATERIAL_ID);

	if( .@itemtype == IT_ARMOR ){
		.@equip_lv = getequiparmorlv( .@part );
		.@type$ = "armor";
		mes "你選擇了一件盔甲，喵～";

		switch(.@equip_lv) {
			case 1:
				.@price = 15000;
				break;
			default:
				// TODO:
				close;
		}
	}else if( .@itemtype == IT_WEAPON ){
		.@equip_lv = getequipweaponlv( .@part );
		.@type$ = "weapon";

		switch( .@equip_lv ){
			case 1: // Level 1 Weapon
				.@price = 500;
				mes "1級武器...？";
				break;
			case 2: // Level 2 Weapon
				.@price = 2000;
				mes "喵，2級武器...？";
				break;
			case 3: // Level 3 Weapon
				.@price = 20000;
				mes "喵喵~~ 3級武器~~";
				break;
			case 4: // Level 4 Weapon
				.@price = 50000;
				mes "喵-喵！... 4級武器...！";
				mes "我在向阿拉岡學習的時候只看過兩次... 喵-喵！！";
				break;
			default:
				// TODO:
				close;
		}
	}else{
		// TODO:
		close;
	}

	mes "你需要 ^ff9999"+getitemname(.@material)+"^000000 和 ^ff9999"+.@price+"^000000 Zeny 才能精鍊，喵～";
	mes "想要繼續嗎，喵？～";
	next;
	if(select("是的！！:不要！！") == 2) {
		mes "[霍林克]";
		mes "Kyack！！";
		mes "你不相信精鍊大師霍林克嗎，喵？～";
		close;
	}
	if (getequippercentrefinery(.@part, true) < 100) {
		mes "[霍林克]";
		mes "喵！！";
		if (.@type$ == "armor")
			mes "這件盔甲已經被精鍊很多次了，喵。";
		else {
			mes "危險。危險～";
			mes "這件武器已經被精鍊很多次了，喵～";
			next;
			mes "[霍林克]";
		}
		mes "如果你繼續嘗試精鍊這件物品，你可能會弄壞它，喵。";
		next;
		mes "[霍林克]";
		mes "一旦 "+.@type$+" 損壞，你";
		mes "就再也無法使用它了，喵。更不用說...目前所有的";
		mes "^ff0000卡片和附魔肯定會消失^000000。";
		mes "你還是想試試看嗎，喵？～";
		next;
		if(select("是的，我願意！！:算了吧！！") == 2) {
			mes "[霍林克]";
			mes "喵！明智的選擇，喵。";
			mes "但是！！";
			mes "看到你懷疑精鍊大師霍林克，我不高興，喵～";
			close;
		}
	}
	if (countitem(.@material) == 0 || Zeny < .@price) {
		mes "[霍林克]";
		mes "你沒有材料。";
		mes "你需要 ^ff9999"+getitemname(.@material)+"^000000 和 ^ff9999"+.@price+"^000000 Zeny，喵～";
		mes "去拿吧，喵～";
		close;
	}
	delitem .@material,1;
	set Zeny, Zeny-.@price;

	// anti-hack
	if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) ||
		callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt)) {
		mes "[霍林克]";
		emotion ET_FRET;
		mes "等一下...";
		mes "你覺得我笨嗎？！";
		mes "我不注意的時候，你偷偷換了物品！滾開！";
		close;
	}

	if (getequippercentrefinery(.@part, true) > rand(100)) {
		successrefitem .@part;
		mes "[霍林克]";
		mes "喵～喵～喵！有趣的精鍊～";
		next;
		emotion ET_CHUP;
		mes "[霍林克]";
		mes "完美！！完美，喵！！";
		mes "我是精鍊巫師阿拉岡的弟子～";
		mes "霍林克！！";
		mes "又是成功精鍊的一天，喵！！";
		close;
	}
	failedrefitem .@part;
	mes "[霍林克]";
	mes "喵～喵～Kyaaak！！";
	next;
	switch(rand(1,5)) {
		case 1: emotion ET_CRY; break;
		case 2: emotion ET_PROFUSELY_SWEAT; break;
		case 3: emotion ET_KEK; break;
		case 4: emotion ET_SCRATCH; break;
		case 5: emotion ET_BIGTHROB; break;
	}
	mes "[霍林克]";
	mes "喵！！Aaaaakk~~!!!!";
	mes "Kyaaak！！我失敗了，喵！！";
	next;
	mes "[霍林克]";
	mes "......";
	mes "......";
	mes "全部～所有～都壞掉了，喵...";
	next;
	mes "[霍林克]";
	mes "喵.... 阿拉岡大師曾經說過，";
	mes "從你的失敗中學習...";
	mes "人類，這一次失敗將會是你未來成功的開始。";
	close;
}
