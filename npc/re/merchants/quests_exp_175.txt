einbech,149,257,4	script	Site manager Elmen#extdq0	4_TOWER_16,{
	if (checkweight(25814,100) == 0 || checkweight(1201,1) == 0 || (MaxWeight - Weight) < 10000) {
		mes "- 你不能繼續，因為你攜帶的物品過多。 -";
		close;
	}
	mes "[艾爾門]";
	mes "你好。冒險者。";
	mes "如果你來這裡是要購買戴納特礦物採集設備或將礦石提煉成戴納特，你來對地方了。";
	next;
	switch( select( "戴納特提煉", "戴納特商店", "戴納特裝備兌換" ) ) {
	case 1:
		npctalk "Elmen: 你一次最多可以提煉 100 個戴納特。這台機器很小。", "", bc_self;
		mes "[艾爾門]";
		mes "" + mesitemlink(25814) + " 精煉礦石。";
		mes "" + mesitemlink(25809) + " ^777777(目前 " + countitem(25809) + ")^000000";
		mes "" + mesitemlink(25810) + " ^777777(目前 " + countitem(25810) + ")^000000";
		mes "" + mesitemlink(25811) + " ^777777(目前 " + countitem(25811) + ")^000000";
		mes "" + mesitemlink(25812) + " ^777777(目前 " + countitem(25812) + ")^000000";
		mes "" + mesitemlink(25813) + " ^777777(目前 " + countitem(25813) + ")^000000";
		mes "你需要五種礦石各一個。";
		next;
		input .@val;
		if (.@val > 100) {
			mes "[艾爾門]";
			mes "我很抱歉。冒險者。";
			mes "戴納特提煉一次最多只能提煉 100 個？ 這台機器很小，所以沒辦法。請理解。";
			close;
		}
		if (.@val == 0) {
			mes "[艾爾門]";
			mes "取消？";
			mes "如果你需要提煉戴納特，請隨時來找我。沒有人可以做到，除了我。";
			close;
		}
		if (countitem(25809) < .@val || countitem(25810) < .@val || countitem(25811) < .@val || countitem(25812) < .@val || countitem(25813) < .@val) {
			mes "[艾爾門]";
			mes "嘿！這樣會讓彼此都很為難，你難道不會用完提煉所需的材料嗎？這些材料！";
			close;
		}
		specialeffect EF_REPAIRWEAPON;
		progressbar_npc "FFFF00",1;
		delitem 25809,.@val;	// Ein_REDIUM
		delitem 25810,.@val;	// Ein_RINDIUM
		delitem 25811,.@val;	// Ein_ODIUM
		delitem 25812,.@val;	// Ein_PURDIUM
		delitem 25813,.@val;	// Ein_WHIDIUM
		getitem 25814,.@val;	// Ein_DYNITE
		mes "[艾爾門]";
		mes "" + .@val + " 戴納特已經被提煉了。 今天非常完美！";
		close;
	case 2:
		mes "[艾爾門]";
		mes "這裡有各種飾品和改造許可證。在購買之前仔細看看你想要什麼。";
		close2;
		callshop "barter_ein_dun03";
		end;
	case 3:
		mes "[艾爾門]";
		mes "我可以交換裝備與戴納特，但它已經在一個密封的盒子裡了。";
		mes "給我 50 個 ^4d4dff戴納特^000000，我就為你打開一個 ^4d4dff武器箱^000000。";
		next;
		mes "[艾爾門]";
		mes "我不知道會出現什麼，但品質很好。你想要打開它嗎？";
		next;
		if (select( "閉上眼睛交換。", "不要交換。" ) == 2) {
			mes "[艾爾門]";
			mes "嘖，這是一種浪費，這是一個很好的交易。";
			close;
		}
		if (countitem(25814) < 50) {
			mes "[艾爾門]";
			mes "你的戴納特朋友不夠多。";
			mes "我不知道會出現什麼，所以在這個價格下很難到任何地方獲得這種裝備。";
			close;
		}
		delitem 25814,50;
		// unknown text
		getgroupitem(IG_EIN_1HITEMBOX);
		end;
	}
}
