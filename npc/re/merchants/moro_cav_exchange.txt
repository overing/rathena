moro_cav,32,73,5	script	Hero Token Exchanger#1	4_F_SHADOWCHASER,{
	disable_items;
	mes "[英雄代幣兌換員]";
	mes "如果你與莫洛克戰鬥過";
	mes "並且收集到";
	mes "英雄代幣，";
	mes "我可以為你用它們兌換一個特殊的戒指。";
	mes "你想看看我的戒指嗎？";
	while(true) {
		next;
		if (checkweight(2163,2) == 0) {
			mes "[英雄代幣兌換員]";
			mes "你的背包裡有很多物品，";
			mes "請整理好後再來！";
			close;
		}
		switch( select( "戒指說明", "用戒指兌換。", "結束對話。" ) ) {
		case 1:
			mes "[英雄代幣兌換員]";
			mes "這個英雄戒指";
			mes "可能";
			mes "看起來";
			mes "並不";
			mes "起眼，";
			next;
			mes "[英雄代幣兌換員]";
			mes "但把它帶給";
			mes "專家附魔師拜恩斯，";
			mes "他可以為";
			mes "它附魔全部 4 個插槽";
			mes "並帶出";
			mes "它潛在的能力。";
			next;
			mes "[英雄代幣兌換員]";
			mes "英雄戒指";
			mes "可以幫助你";
			mes "征服";
			mes "莫洛克。";
			mes "還有其他問題嗎？";
			continue;
		case 2:
			mes "- 英雄戒指 -";
			mes "^006400為擊敗莫洛克的戰士們製作的戒指^000000";
			mes "^006400在任務中..^000000";
			mes "^006400當附魔時，^000000";
			mes "^006400它賦予其擁有者^000000";
			mes "^006400對抗莫洛克的力量。^000000";
			mes "^006400分類：飾品/DEF：0^000000";
			mes "^006400重量：20/需求等級：160^000000";
			mes "^006400插槽：0/裝備於：所有職業^000000";
			next;
			mes "[英雄代幣兌換員]";
			mes "英雄戒指需要";
			mes "30 個英雄代幣";
			mes "才能兌換。";
			mes "你要兌換嗎？";
			next;
			if (select( "上一個選單。", "兌換 30 個英雄代幣。" ) == 1) {
				mes "[英雄代幣兌換員]";
				mes "再想";
				mes "想。";
				continue;
			}
			if (countitem(6684) < 30) {
				mes "[英雄代幣兌換員]";
				mes "^006400英雄代幣^000000";
				mes "是給予";
				mes "那些勇敢地與莫洛克戰鬥的人。";
				mes "你沒有";
				mes "30 個 ^006400英雄代幣^000000，";
				mes "因此我無法幫助你。";
				continue;
			}
			mes "[英雄代幣兌換員]";
			mes "我將用";
			mes "30 個 ^006400英雄代幣^000000";
			mes "兌換 1 個 ^006400英雄戒指^000000。";
			mes "還有其他事嗎？";
			delitem 6684,30;	// TokenOfHero
			getitem 2981,1;		// RingOfHero
			continue;
		case 3:
			mes "[英雄代幣兌換員]";
			mes "總有一天";
			mes "你會發現自己";
			mes "需要英雄戒指。";
			close;
		}
	}
}

moro_cav,34,65,5	script	Expert Enchanter Byrnes	1_M_03,{
	disable_items;
	mes "[拜恩斯]";
	mes "你有聽說過英雄戒指嗎？只有真正的英雄才配擁有那枚華麗的戒指，而我一直在等待著。";
	mes "有什麼我可以幫你的嗎？";
	next;
	switch( select( "附魔英雄戒指。", "重置英雄戒指上的附魔。", "取消。" ) ) {
	case 1:
		mes "[拜恩斯]";
		mes "我只能附魔已裝備的戒指。 你想附魔哪個已裝備的戒指？";
		next;
		setarray .@eqi_slot[0], EQI_ACC_L, EQI_ACC_R;
		.@s = select( "左", "右" ) - 1;
		switch( getequipid(.@eqi_slot[.@s]) ) {
		case -1:
			mes "[拜恩斯]";
			mes "請裝備你想附魔的英雄戒指。";
			close;
		case 2981:
			break;
		default:
			mes "[拜恩斯]";
			mes "請裝備你的英雄戒指。 我無法附魔其他東西。";
			close;
		}
		// custom at this point
		setarray .@card[0],
			getequipcardid(.@eqi_slot[.@s],0),
			getequipcardid(.@eqi_slot[.@s],1),
			getequipcardid(.@eqi_slot[.@s],2),
			getequipcardid(.@eqi_slot[.@s],3);

		if (.@card[0] > 0) {
			mes "[拜恩斯]";
			mes "一個完全附魔的英雄戒指不能再被附魔了。";
			close;
		}
		if (.@card[3] == 0) {
			setarray .@enchant_list$[0], "DEF", "MDEF";
			.@slot = 4;
		}
		else if (.@card[2] == 0) {
			setarray .@enchant_list$[0], "Max HP", "Max SP";
			.@slot = 3;
		}
		else if (.@card[1] == 0) {
			setarray .@enchant_list$[0], "Atk", "MAtk";
			.@slot = 2;
		}
		else {
			setarray .@enchant_list$[0], "STR", "INT", "AGI", "VIT", "DEX", "LUK";
			.@slot = 1;
		}
		mes "[拜恩斯]";
		mes "你想要哪個能力作為插槽 " + .@slot + " 的附魔？";
		next;
		.@enchant_list_size = getarraysize(.@enchant_list$);
		for ( .@i = 0; .@i < .@enchant_list_size; ++.@i )
			.@menu$ += .@enchant_list$[.@i] + ":";
		.@type = select(.@menu$) - 1;

		switch(.@slot) {
		case 4:
			switch(.@type) {// enchant list index
			case 0:	// DEF
				setarray .@bonus[0], 4791, 4792, 4793, 4794, 4893;
				break;
			default:	// MDEF
				setarray .@bonus[0], 4890, 4786, 4891, 4787, 4892;
				break;
			}
			break;
		case 3:
			switch(.@type) {
			case 0:	// Max HP
				setarray .@bonus[0], 4861, 4862, 4867, 4868, 4900;
				break;
			default:	// Max SP
				setarray .@bonus[0], 4800, 4870, 4871, 4801, 4802;
				break;
			}
			break;
		case 2:
			switch(.@type) {
			case 0:	// Atk
				setarray .@bonus[0], 4882, 4766, 4767, 4894, 4895;
				break;
			default:	// Matk
				setarray .@bonus[0], 4883, 4896, 4897, 4898, 4899;
				break;
			}
			break;
		case 1:
			switch(.@type) {
			case 0:	// STR
				setarray .@bonus[0], 4700, 4701, 4702, 4703, 4704;
				break;
			case 1:	// INT
				setarray .@bonus[0], 4710, 4711, 4712, 4713, 4714; 
				break;
			case 2:	// AGI
				setarray .@bonus[0], 4730, 4731, 4732, 4733, 4734;
				break;
			case 3:	// VIT
				setarray .@bonus[0], 4740, 4741, 4742, 4743, 4744;
				break;
			case 4:	// DEX
				setarray .@bonus[0], 4720, 4721, 4722, 4723, 4724;
				break;
			default:	// LUK
				setarray .@bonus[0], 4750, 4751, 4752, 4753, 4754;
				break;
			}
			break;
		}

		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@eqi_slot[.@s], 2981) || callfunc("F_IsEquipCardHack", .@eqi_slot[.@s], .@card[0], .@card[1], .@card[2], .@card[3]))
			close;

		.@bonus_size = getarraysize(.@bonus);
		.@card[.@slot-1] = .@bonus[ rand(.@bonus_size) ];	// unknown rates
		specialeffect2 EF_ANGEL2;
		mes "[拜恩斯]";
		mes "附魔完成！";
		delequip .@eqi_slot[.@s];
		getitem2 2981,1,1,0,0,.@card[0],.@card[1],.@card[2],.@card[3];
		close;
	case 2:
		mes "[拜恩斯]";
		mes "要重置你英雄戒指上的附魔，我需要 3 個英雄代幣。";
		next;
		if (countitem(6684) < 3) {
			mes "[拜恩斯]";
			mes "你沒有足夠的英雄代幣。 我無法重置你的英雄戒指。";
			close;
		}
		mes "[拜恩斯]";
		mes "我只能重置已裝備的戒指。 你想重置哪個已裝備的戒指？";
		next;
		setarray .@eqi_slot[0], EQI_ACC_L, EQI_ACC_R;
		.@s = select( "左", "右" ) - 1;
		switch( getequipid(.@eqi_slot[.@s]) ) {
		case -1:
			mes "[拜恩斯]";
			mes "你必須先裝備物品才能重置其附魔狀態。";
			close;
		case 2981:
			break;
		default:
			mes "[拜恩斯]";
			mes "請裝備你的英雄戒指。 我無法重置其他東西。";
			close;
		}
		// custom at this point
		if (countitem(6684) < 3) {
			mes "[拜恩斯]";
			mes "你沒有足夠的英雄代幣。 我無法重置你的英雄戒指。";
			close;
		}
		setarray .@card[0],
			getequipcardid(.@eqi_slot[.@s],0),
			getequipcardid(.@eqi_slot[.@s],1),
			getequipcardid(.@eqi_slot[.@s],2),
			getequipcardid(.@eqi_slot[.@s],3);

		if (.@card[3] == 0) {
			mes "[拜恩斯]";
			mes "你的英雄戒指目前處於沒有附魔的狀態。";
			close;
		}

		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@eqi_slot[.@s], 2981) || callfunc("F_IsEquipCardHack", .@eqi_slot[.@s], .@card[0], .@card[1], .@card[2], .@card[3]))
			close;

		specialeffect2 EF_ANGEL2;
		mes "[拜恩斯]";
		mes "初始化完成。";
		delitem 6684,3;
		delequip .@eqi_slot[.@s];
		getitem 2981,1;
		close;
	case 3:
		close;
	}
}
