-	script	#GemstoneBagger	HIDDEN_NPC,{
	mes "[寶石裝袋者]";
	mes "我們以容易找到的方式將寶石裝在袋子裡。";
	mes "您想做什麼呢？";
	next;
	switch(select("兌換寶石:取消")) {
	case 1:
		mes "[寶石裝袋者]";
		mes "我會準備一個寶石袋。";
		close2;
		callshop "Gemstone_Bagger";
		break;
	case 2:
		mes "[寶石裝袋者]";
		mes "如果您需要幫助，請回來。";
		close;
	}
}
