//= Turns arrows into Arrow Quivers.
//= Breakdown of Subroutine "S_BuyQuiver"
//= arg(0): Type of Arrow to be packaged (item ID).
//= arg(1): How many of each 'getarg(0)' arrow per quiver.
//= arg(2): The cost of making a 'getarg(0)' quiver.
//= arg(3): The quiver given by the NPC (item ID).

// Mora :: ep14_1_tre
//============================================================
mora,106,117,3	script	Quiver Maker#mora	4_M_RAFLE_GR,{
	if (checkweight(1201,1) == 0) {
		mes "[箭矢製造商]";
		mes "你身上帶太多東西了。";
		mes "清出一些空間再回來。我會告訴你一些有趣的事情。";
		close;
	}
	if (MaxWeight - Weight < 2000) {
		mes "[箭矢製造商]";
		mes "你身上帶太多東西，看起來很疲憊。";
		mes "清出一些空間再回來。我會告訴你一些有趣的事情。";
		close;
	}
	mes "[箭矢製造商]";
	mes "莫拉村民問說箭矢筒有什麼用。他們根本不知道這個世界是怎麼運作的。";
	mes "難怪他們對箭矢筒一無所知 - 整天在樹葉上塗果醬。";
	next;
	if (getbattleflag("feature.barter")) {
		.@menu$ = "I want arrows";
	}
	switch( select("請幫我做一個箭矢筒。", .@menu$, "什麼是箭矢筒？") ) {
	case 1:
		if (getbattleflag("feature.barter")) {
			mes "[箭矢製造商]";
			mes "終於，出現了一個懂得箭矢筒真正價值的人。我會為那些懂得它價值的人製作。";
			close2;
			callshop "Quiver_Archer", 1;
			end;
		}
		mes "[箭矢製造商]";
		mes "終於有人懂得欣賞箭矢筒了！";
		mes "我可以製作精靈箭矢筒和獵人之箭矢筒。";
		mes "你需要哪一種？";
		next;
		switch(select("精靈箭矢筒:獵人之箭矢筒:我不需要箭矢筒。")) {
		case 1: callsub S_BuyQuiver,1773,500,500,12575; //Arrow_Of_Elf_Cntr
		case 2: callsub S_BuyQuiver,1774,500,500,12576; //Hunting_Arrow_Cntr
		case 3:
			mes "[箭矢製造商]";
			mes "你可以從我旁邊的工具商人那裡購買箭矢。";
			close;
		}
		end;
	case 2:
		if (getbattleflag("feature.barter")) {
			mes "[箭矢製造商]";
			mes "我不賣普通的箭矢。這些是只有在這裡才能買到的工匠箭矢。你真的會認出它的價值，對吧？";
			close2;
			callshop "Catch_Arrows", 1;
		}
		end;
	case 3:
		mes "[箭矢製造商]";
		mes "箭矢可能很細很輕，但攜帶成百上千的箭矢就像攜帶一整根原木。";
		next;
		mes "[箭矢製造商]";
		mes "但是當你有箭矢筒時，你可以把箭矢放進去，節省重量和空間。";
		mes "如果你對其中一個有興趣，我可以為你縫製一個。";
		close;
	}
	end;

S_BuyQuiver:
	.@item_req = getarg(0);
	.@req_amount = getarg(1);
	if (countitem(getarg(0)) < .@req_amount) {
		mes "[箭矢製造商]";
		mes "帶超過 "+ .@req_amount +" 個 "+ getitemname(.@item_req) +" 來，我就為你製作一個箭矢筒。";
		close;
	}
	mes "[箭矢製造商]";
	mes "喔，我看到你手上有 "+ getitemname(.@item_req) +"！";
	mes "你有興趣使用箭矢筒嗎？它真的很方便！";
	mes "如果你有興趣，我可以將 "+ .@req_amount +" 個箭矢換成一個箭矢筒，並收取 ^FF3131"+getarg(2)+" zeny^000000。";
	next;
	switch(select("交易你擁有的所有箭矢:只換一個箭矢筒:不交易")) {
	case 1:
		set .@arrows, countitem(.@item_req);
		set .@quiver, .@arrows / .@req_amount;
		set .@arrows_used, .@quiver * .@req_amount;
		set .@arrow_zeny01, .@quiver * getarg(2);
		mes "你擁有的箭矢數量：^3131FF"+.@arrows+"^000000";
		mes "可用的箭矢筒數量：^3131FF"+.@quiver+"^000000";
		mes "交易需要的Zeny：^3131FF"+.@arrow_zeny01+" zeny^000000";
		next;
		mes "交易？";
		next;
		if(select("交易:不交易") == 2) {
			mes "[箭矢製造商]";
			mes "嘿，你不是在懷疑我的技術吧？";
			close;
		}
		break;
	case 2:
		set .@quiver, 1;
		set .@arrows_used, .@req_amount;
		set .@arrow_zeny01, getarg(2);
		set .@zeny_mes,1;
		break;
	case 3:
		mes "[箭矢製造商]";
		mes "嘿，你不是在懷疑我的技術吧？";
		close;
	}
	if (Zeny < .@arrow_zeny01) {
		mes "[箭矢製造商]";
		if (.@zeny_mes == 1)
			mes "我說過我會為了你接受人類的錢幣，你還是不想花錢嗎？";
		else
			mes "你真的不期望不花任何代價就得到一件傑作，是嗎？";
		close;
	}
	mes "[箭矢製造商]";
	mes "嘿，給你。";
	mes "有 ^3131FF一些你需要知道的事情^000000 - 試著記住它。";
	mes "^FF0000當你的負重超過 70% 時，你不能使用箭矢筒。^000000";
	mes "你最好記住這一點，否則你以後可能會遇到麻煩。";
	set Zeny, Zeny-.@arrow_zeny01;
	delitem .@item_req,.@arrows_used;
	getitem getarg(3),.@quiver;
	close;
}
