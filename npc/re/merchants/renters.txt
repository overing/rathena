job3_rune01,88,62,5	script	Dragon Breeder	105,{
	mes "[龍族飼養員]";
	if ((eaclass()&EAJ_THIRDMASK)==EAJ_RUNE_KNIGHT) {
		mes "歡迎。您想要租借一隻龍嗎？";
		next;
		switch(select("是:否")) {
		case 1:
			//if (!getskilllv("RK_DRAGONTRAINING")) {
			if (!getskilllv("KN_RIDING")) {
				mes "[龍族飼養員]";
				mes "請先學習如何騎乘龍。";
				close;
			}
			else if (checkdragon()) {
				mes "[龍族飼養員]";
				mes "您已經有一隻龍了。";
				close;
			}
			else if (ismounting()) {
				mes "[龍族飼養員]";
				mes "請先移除您的商城坐騎。";
				close;
			}
			setdragon;
			close;
		case 2:
			mes "[龍族飼養員]";
			mes "我知道了。祝您有個美好的一天。";
			close;
		}
	}
	mes "你在這裡做什麼？";
	mes "只有盧恩騎士才能租借龍。";
	close;
}

// Mado Gear Renter :: madogear
//============================================================
-	script	::mgm	-1,{
	mes "[魔導機甲整備師]";
	if ((eaclass()&EAJ_THIRDMASK)==EAJ_MECHANIC) {
		mes "歡迎，機械工匠。";
		mes "您想要租借手推車或";
		mes "騎乘魔導機甲嗎？";
		next;
		switch(select("租借手推車:騎乘魔導機甲:購買緊急魔導機甲:升級冷卻裝置:取消")) {
		case 1:
			if (checkcart()) {
				mes "[魔導機甲整備師]";
				mes "抱歉，您已經";
				mes "擁有一台手推車了。";
				close;
			}
			setcart;
			mes "[魔導機甲整備師]";
			mes "給你！";
			close;
		case 2:
			if (checkmadogear()) {
				mes "[魔導機甲整備師]";
				mes "抱歉，您已經";
				mes "騎乘著魔導機甲了。";
				close;
			}
			else if (!getskilllv("NC_MADOLICENCE")) {
				mes "[魔導機甲整備師]";
				mes "請先學習技能以獲得魔導機甲執照。";
				close;
			}
			else if(ismounting()) {
				mes "[魔導機甲整備師]";
				mes "請先移除您的商城坐騎。";
				close;
			}
			setmadogear;
			mes "[魔導機甲整備師]";
			mes "玩得開心，請再來！";
			close;
		case 3:
			mes "[魔導機甲整備師]";
			mes "緊急魔導機甲對於緊急情況非常有用，售價為 1,000,000 Zeny。";
			next;
			if (select("購買:取消") == 2) {
				mes "[魔導機甲整備師]";
				mes "我知道了。如果您改變主意，";
				mes "請隨時來找我。";
				close;
			}
			if (countitem(23277) > 0) {
				mes "[魔導機甲整備師]";
				mes "抱歉，您已經擁有緊急魔導機甲了。";
				close;
			}
			if (Zeny < 1000000) {
				mes "[魔導機甲整備師]";
				mes "抱歉，您沒有足夠的 Zeny 來購買緊急魔導機甲。";
				close;
			}
			Zeny -= 1000000;
			getitem 23277,1; //Mado_Box
			mes "[魔導機甲整備師]";
			mes "給你！";
			close;
		case 4:
			mes "[魔導機甲整備師]";
			mes "您想要升級哪個裝置？";
			next;
			if (select("冷卻裝置:高品質冷卻器") == 1) {
				mes "[魔導機甲整備師]";
				mes "將冷卻裝置升級到高品質冷卻器需要 1 個冷卻裝置和 2,000,000 Zeny。";
				next;
				.@itemid = 2804;
				.@cost = 2000000;
			} else {
				mes "[魔導機甲整備師]";
				mes "將高品質冷卻器升級到特殊冷卻器需要 1 個高品質冷卻器和 4,000,000 Zeny。";
				next;
				.@itemid = 2809;
				.@cost = 4000000;
			}
			if (select("升級:取消") == 2) {
				mes "[魔導機甲整備師]";
				mes "我知道了。如果您改變主意，";
				mes "請隨時來找我。";
				close;
			}
			if (!countitem(.@itemid)) {
				mes "[魔導機甲整備師]";
				mes "抱歉，您沒有 " + getitemname(.@itemid) + "。";
				close;
			}
			if (Zeny < .@cost) {
				mes "[魔導機甲整備師]";
				mes "抱歉，您沒有足夠的 Zeny 來升級裝置。";
				close;
			}
			Zeny -= .@cost;
			delitem .@itemid,1;
			getitem (.@itemid == 2804 ? 2809 : 2810),1; //High_Quality_Cooler,Special_Cooler
			mes "[魔導機甲整備師]";
			mes "給你！您自己的 " + getitemname(.@itemid) + "。";
			close;
		case 5:
			close;
		}
	}
	mes "我能為您做些什麼？";
	mes "魔導機甲僅供機械工匠使用。";
	close;
}
