//===== rAthena Script =======================================
//= Insurance Exchanger
//===== Description: =========================================
//= Exchange items for New_Insurance item.
//===== Changelogs: ==========================================

//============================================================

prontera,82,108,5	script	Heart Merchant#life01	4_M_NFDEADMAN2,5,5,{
	.@old_item = 12209;
	.@new_item = 6413;
	.@old_item_name$ = getitemname(.@old_item);
	.@new_item_name$ = getitemname(.@new_item);

	if (!checkweight(1201,1) || (MaxWeight - Weight) < 500) {
		mes "由於您攜帶了大量的物品，無法繼續對話。";
		mes "請整理您的物品後再試一次。";
		close;
	}
	mes "[心之商人]";
	mes "您好！我看您是一位熱愛旅行、乘船橫跨大陸，並享受危險刺激的冒險家！";
	next;
	if(!countitem(.@old_item)){
		mes "[心之商人]";
		mes "您的倉庫裡有 " + mesitemlink( .@old_item ) + " 嗎？";
		next;
		mes "[心之商人]";
		mes "如果您給我一個 " + .@old_item_name$ + "，我會將它兌換成 " + mesitemlink( .@new_item ) + "。您覺得如何？您知道這是什麼物品嗎？";
	} else {
		mes "[心之商人]";
		mes "您有沒有想過將您的 " + mesitemlink( .@old_item ) + " 換成更好的東西？例如... " + mesitemlink( .@new_item ) + "。您覺得如何？";
	}
	next;
	switch(select("那是什麼？",(countitem(.@old_item)?"請兌換！":""),"我沒興趣")) {
	case 1:
		mes "[心之商人]";
		mes mesitemlink( .@old_item ) + " 是一種在您使用後的 30 分鐘內，防止死亡時經驗值損失的物品。";
		next;
		mes "[心之商人]";
		mes "但很多人都對時間限制感到難過，對吧？所以我研究了一下，準備了 " + mesitemlink( .@new_item ) + "。";
		next;
		mes "[心之商人]";
		mes mesitemlink( .@new_item ) + " 是一種只要擁有它，^EE0000無論持續時間^000000 都可以防止死亡時經驗值損失的物品。當然，死亡後，"+.@new_item_name$+" 會被消耗掉。";
		next;
		select("你為什麼要兌換它？");
		mes "[心之商人]";
		mes "您對此感到好奇嗎？我是一個商人。買賣東西是我的命運，無論它是否與生死有關。您認為我會錯過這個賺錢的機會嗎？我賺錢，而活著的人可以從死亡的痛苦中得到喘息。這不是一筆好交易嗎？";
		next;
		mes "[心之商人]";
		mes "匯率是 5 個 " + mesitemlink( .@old_item ) + " 換 1 個 " + mesitemlink( .@new_item ) + "。您有什麼問題嗎？";
		next;
		mes "[心之商人]";
		mes "嗯~ 您覺得如何？這不是一筆公平的交易嗎？您想一直擔心時間嗎？對於喜歡驚險冒險的冒險家來說，這是必備品。";
		close;
		
	case 2:	
		mes "[心之商人]";
		mes "是，是的。好決定。換成更好的，不是更好嗎？";
		next;
		mes "[心之商人]";
		mes "我可以將您的 ^0000cd5 "+.@old_item_name$+" 兌換成 1 "+.@new_item_name$+"。";
		next;
		mes "[心之商人]";
		mes "但是，^EE0000[非賣品] " + .@old_item_name$ + " 不符合兌換資格^000000，所以把它帶到這裡要求兌換是沒用的。";
		next;
		mes "[心之商人]";
		mes "我們現在兌換嗎？";
		next;
		if (select("請兌換！:我不想兌換。") == 2)
			break;
		mes "[心之商人]";
		mes "您要兌換多少個？ ^0000cd您一次最多可以兌換 10 個 "+.@new_item_name$+"^000000。";
		next;
		input .@amount;
		if (.@amount == 0) {
			mes "[心之商人]";
			mes "嗯？您不感興趣嗎？這是一個很棒的產品，而且對像您這樣的人來說真的很好。但如果您不想保護自己的生命，我也不會強迫您。";
			close;
		}
		if (.@amount > 10) {
			mes "[心之商人]";
			mes "就像我說的，^0000cd您一次最多只能兌換 10 個^000000。請再次確認數量。";
			close;
		}
		if (countitem(.@old_item) < (.@amount*5)) {
			mes "[心之商人]";
			mes "嗯，不好意思... 您的 " +.@old_item_name$+ " 數量不足，不是嗎？ 5 個 " + mesitemlink( .@old_item ) + " 換 1 個 " + mesitemlink( .@new_item ) + "。請檢查數量。";
			close;
		}
		mes "[心之商人]";
		mes "現在，有了這個，我已經將您的 ^0000cd" + .@old_item_name$ + "^000000 兌換成了 ^EE0000" + .@new_item_name$ + "^000000 ！！這是一筆好交易，不是嗎？下次您拿到 "+.@old_item_name$+" 時，再來找我！";
		delitem .@old_item,.@amount*5;
		getitem .@new_item,.@amount;
		close;
	case 3:
		break;
	}
	mes "[心之商人]";
	mes "這是一筆很棒的交易！冒險家，對於像您這樣在危險地方冒險的人來說，這是一個必備品！";
	next;
	mes "[心之商人]";
	mes "如果您對它產生興趣，請務必回來兌換。好嗎？";
	close;
	
OnTouch:
	if(countitem(12209))
		npctalk "You have something good. Would you like to replace your " + getitemname(12209) + " with a better one?", "你有好東西。你想把你的 " + getitemname(12209) + " 換成更好的嗎？", bc_self;
	end;
}
