//============================================================
// +11 and above Refiner Function
//============================================================
//= To allow auto safe refining/multiple refining set the
//= second argument to '1' in the function call.
//= If you enable this function, be sure to edit the value of
//= .@safe to the max safe refine in refine.yml as well.
//=
//= On official servers, if an item is unsuccessfully refined
//= it will break at a 20% rate and downgrade at an 80% rate.
//============================================================
function	script	refinenew	{
	.@npc_name$ = getarg(0);
	disable_items;
	mes "["+ .@npc_name$ +"]";
	mes "我是有史以來最好的護甲匠！";
	mes "我不精煉普通又無聊的物品。";
	mes "我只精煉等級 10 或更高的物品。";
	next;
	mes "["+ .@npc_name$ +"]";
	mes "總之，如果你的物品等級是 10 或更高，你可以使用我的服務。";
	mes "你想讓我精煉什麼？";
	next;

	setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
	for(.@i = 1; .@i<=10; ++.@i) {
		if (getequipisequiped(.@indices[.@i])) {
			.@menu$ = .@menu$ + F_getpositionname(.@indices[.@i]) + "-[" + getequipname(.@indices[.@i]) + "]";
			.@equipped = 1;
		}
		.@menu$ = .@menu$ + ":";
	}
	if (.@equipped == 0) {
		mes "["+ .@npc_name$ +"]";
		mes "我想我無法精煉你擁有的任何物品...";
		close;
	}
	.@part = .@indices[ select(@menu$) ];

	if (!getequipisequiped(.@part)) { //custom check
		mes "["+ .@npc_name$ +"]";
		mes "你沒有穿戴";
		mes "任何我可以";
		mes "精煉的東西。";
		emotion ET_FRET;
		close;
	}
	if (!getequipisenableref(.@part)) {
		mes "["+ .@npc_name$ +"]";
		mes "我想我完全無法";
		mes "精煉這個物品...";
		close;
	}
	.@refinerycnt = getequiprefinerycnt(.@part); //save refinery count
	if (.@refinerycnt < 10) {
		mes "["+ .@npc_name$ +"]";
		mes "我說過我不處理低於等級 10 的物品。";
		close;
	}
	if (.@refinerycnt >= 20) { //custom check
		mes "["+ .@npc_name$ +"]";
		mes "我不能再精煉";
		mes "這個了。這已經是";
		mes "精煉到極限了！";
		close;
	}
	.@refineitemid = getequipid(.@part); // save id of the item
	setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);
	.@price = getequiprefinecost(.@part, REFINE_COST_NORMAL, REFINE_ZENY_COST);
	.@material = getequiprefinecost(.@part, REFINE_COST_NORMAL, REFINE_MATERIAL_ID);
	.@safe = 10;
	.@itemtype = getiteminfo( .@refineitemid, ITEMINFO_TYPE );
	
	if( .@itemtype == IT_WEAPON ){
		.@article$ = "a";
		.@type$ = "weapon";
	}else if( .@itemtype == IT_ARMOR ){
		.@article$ = "an";
		.@type$ = "armor";
	}else{
		// TODO:
		close;
	}

	mes "["+ .@npc_name$ +"]";
	mes "嗯 " + .@article$ + " " + .@type$ + ", 這樣可以嗎？";
	mes "如果你想精煉這個 " + .@type$ + ",";
	mes "我需要 1 ^003366" + getitemname(.@material) + "^000000 和 " + callfunc("F_InsertComma",.@price) + " zeny。";
	mes "你確定要繼續嗎？";
	next;
	if (select("是的:不要") == 2) {
		mes "["+ .@npc_name$ +"]";
		mes "嗯...如果你介意...那就別在意...";
		close;
	}
	if (getarg(1) != 1) {
		if (getequippercentrefinery(.@part) < 100) {
			mes "["+ .@npc_name$ +"]";
			mes "這個 "+.@type$+" 已經被精煉過很多次了。";
			mes "如果我再次嘗試，它可能會被摧毀。";
			mes "它不會肯定會壞掉，但確實有很小的機率。";
			next;
			mes "["+ .@npc_name$ +"]";
			mes "你可能會^FF0000降低"+.@type$+"的升級等級^000000，";
			mes "或者如果它壞了，你將失去^FF0000任何卡片^000000或添加到它的特殊屬性。";
			next;
			mes "["+ .@npc_name$ +"]";
			mes "你仍然希望我精煉它嗎？";
			mes "我想我已經給了你足夠的警告。";
			next;
			if(select("是的。:不要。") == 2) {
				mes "["+ .@npc_name$ +"]";
				mes "好吧，沒有挑戰也是一種方式...";
				mes "沒有風險...這可能是明智的。";
				close;
			}
		}
		if (countitem(.@material) < 1 || Zeny < .@price) {
			mes "["+ .@npc_name$ +"]";
			mes "嗯。你似乎沒有足夠的錢或 "+getitemname(.@material)+"。";
			mes "請在你擁有它們時再來。";
			close;
		}
		Zeny = Zeny - .@price;
		delitem .@material,1;

		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) ||
		    callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt)) {
			mes "["+ .@npc_name$ +"]";
			emotion ET_FRET;
			mes "等一下...";
			mes "你覺得我很笨嗎？！";
			mes "趁我不注意，你換了物品！滾出去！";
			close;
		}

		if (getequippercentrefinery(.@part) > rand(100)) {
			mes "鏗！鏗！鏗！鏗！";
			successrefitem .@part;
			next;
			emotion ET_BEST;
			mes "["+ .@npc_name$ +"]";
			mes "好！成功！！！";
			mes "我是最好的護甲匠。";
			close;
		} else {
			if (rand(100) < 80) {
				mes "["+ .@npc_name$ +"]";
				mes "鏗！鏗！鏗！鏗！";
				downrefitem .@part,3; // Failed refine attempts decrease the item's refine level by 3
				next;
				emotion (!rand(5))?ET_MONEY:ET_HUK;
				mes "["+ .@npc_name$ +"]";
				mes "啊啊啊！！！";
				next;
				mes "["+ .@npc_name$ +"]";
				mes "我的天啊！";
				mes "升級等級下降了...";
			} else {
				mes "["+ .@npc_name$ +"]";
				mes "鏗！鏗！鏗！";
				failedrefitem .@part;
				next;
				emotion (!rand(5))?ET_MONEY:ET_HUK;
				mes "["+ .@npc_name$ +"]";
				mes "嗯嗯嗯！";
				next;
				mes "["+ .@npc_name$ +"]";
				mes "我的天啊！我未能精煉東西...";
				mes "我不是故意的！";
			}
			mes "即使我一直都是最好的護甲匠，我也可能會犯錯。";
			mes "這只是命中註定。";
			next;
			mes "["+ .@npc_name$ +"]";
			mes "下次我會做得更好！別擔心！";
			close;
		}
	}
// New +11 and above Refining Functions ========================
	if (.@refinerycnt < .@safe) {
		mes "["+ .@npc_name$ +"]";
		mes "我可以將其精煉到安全上限或想要的次數。由你選擇。";
		next;
		.@menu2 = select("請精煉到安全上限。","我來決定要精煉幾次。","我改變主意了...");
	} else
		.@menu2 = 2;
	switch(.@menu2){
	case 1: 
		.@refinecnt = .@safe - .@refinerycnt;
		break;
	case 2:
		mes "["+ .@npc_name$ +"]";
		mes "你想要我精煉你的物品多少次？";
		next;
		input .@refinecnt;
		.@refinecheck = .@refinecnt + .@refinerycnt;
		if (.@refinecnt < 1 || .@refinecheck > 20) {
			mes "["+ .@npc_name$ +"]";
			mes "我不能將這個物品精煉那麼多次。";
			close;
		}
		if (.@refinecheck > .@safe) {
			.@refinecheck = .@refinecheck - .@safe;
			mes "["+ .@npc_name$ +"]";
			mes "這將嘗試將裝備精煉超過安全上限 " + .@refinecheck + " 次。你的裝備可能會被摧毀...這樣可以嗎？";
			next;
			if(select("是的...","不要...") == 2){
				mes "["+ .@npc_name$ +"]";
				mes "你說的...就這樣吧。";
				close;
			}
		}
		break;
	case 3:
		mes "["+ .@npc_name$ +"]";
		mes "你說的...就這樣吧。";
		close;
	}
	.@fullprice = .@price * .@refinecnt;
	mes "["+ .@npc_name$ +"]";
	mes "那將花費你 " + .@refinecnt + " " + getitemname(.@material) + " 和 " + .@fullprice + " Zeny。這樣可以嗎？";
	next;
	if(select("是的:不要...") == 2){
		mes "["+ .@npc_name$ +"]";
		mes "你說的...就這樣吧。";
		close;
	}
	if (countitem(.@material) < .@refinecnt || Zeny < .@fullprice) {
		mes "["+ .@npc_name$ +"]";
		mes "嗯。你似乎沒有足夠的錢或 "+getitemname(.@material)+"。";
		mes "請在你擁有它們時再來。";
		close;
	}
	Zeny = Zeny - .@fullprice;
	delitem .@material,.@refinecnt;
	while(.@refinecnt){
		if (getequipisequiped(.@part) == 0) {
			mes "["+ .@npc_name$ +"]";
			mes "看看這裡...你沒有穿戴任何物品...";
			close;
		}
		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) ||
		    callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt))
			close;

		if (getequipid(.@part) != .@refineitemid || (.@menu2 == 1 && getequippercentrefinery(.@part) < 100)) {
			mes "["+ .@npc_name$ +"]";
			mes "鏗...不，但你認為我會那麼笨嗎？！";
			mes "你換了它...";
			mes "在我用我的錘子擊昏你之前滾出去！！";
			close;
		}
		if (getequippercentrefinery(.@part) > rand(100)) {
			mes "鏗！鏗！鏗！鏗！";
			successrefitem .@part;
			.@refinecnt = .@refinecnt - 1;
			next;
		} else {
			if (rand(100) < 80) {
				mes "["+ .@npc_name$ +"]";
				mes "鏗！鏗！鏗！鏗！";
				downrefitem .@part,3; // Failed refine attempts decrease the item's refine level by 3
				next;
				emotion (!rand(5))?ET_MONEY:ET_HUK;
				mes "["+ .@npc_name$ +"]";
				mes "啊啊啊！！！";
				next;
				mes "["+ .@npc_name$ +"]";
				mes "我的天啊！";
				mes "升級等級下降了...";
			} else {
				mes "["+ .@npc_name$ +"]";
				mes "鏗！鏗！鏗！";
				failedrefitem .@part;
				next;
				emotion (!rand(5))?ET_MONEY:ET_HUK;
				mes "["+ .@npc_name$ +"]";
				mes "嗯嗯嗯！";
				next;
				mes "["+ .@npc_name$ +"]";
				mes "我的天啊！我未能精煉東西...";
				mes "我不是故意的！";
			}
			mes "即使我一直都是最好的護甲匠，我也可能會犯錯。";
			mes "這只是命中註定。";
			next;
			mes "["+ .@npc_name$ +"]";
			mes "下次我會做得更好！別擔心！";
			close;
		}
		.@refinerycnt = getequiprefinerycnt(.@part);
	}
	emotion ET_BEST;
	mes "["+ .@npc_name$ +"]";
	mes "好棒！成功了！！！";
	mes "我是最棒的鐵匠。";
	close;
}

	next;
	if(select("是的，我有！！:算了吧！！") == 2) {
