grademk,34,184,4	script	Sratos#sratos	4_JP_GARM_H,{
	mes "[史特拉托斯]";
	mes "您好，親愛的顧客，走進來了，讓我們期待今天會有奇蹟發生。";
	mes "有什麼我可以幫你的嗎？";
	next;
	switch( select( "強化裝備的等級。", "兌換艾特爾道具", "什麼是等級強化？" ) ){
		case 1:
			mes "[史特拉托斯]";
			mes "我們的顧客。您想要強化裝備的等級。";
			mes "轉移寶石的魔力並不容易。";
			next;
			mes "[史特拉托斯]";
			mes "魔力可能會當場爆炸。";
			mes "那麼，將繼承魔力的武器也會爆炸！";
			next;
			switch( select( "我還是要試試看！", "我會考慮一下。" ) ){
				case 1:
					mes "[史特拉托斯]";
					mes "祝我們勇敢的顧客好運！";
					close2;
					enchantgradeui();
					end;
				case 2:
					mes "[史特拉托斯]";
					mes "無論何時您有挑戰的勇氣，請再來。";
					close;
			}
		case 2:
			mes "[史特拉托斯]";
			mes "這些是用於提升等級的寶石。";
			mes "艾特爾粉塵，將成為基礎的寶石，如果您給我一點費用，我會將其換成艾特爾寶石。";
			close2;
			callshop "EnchantGradeExchange";
			end;

		case 3:
			mes "[史特拉托斯]";
			mes "偶爾會發現具有大量純魔力的不穩定寶石。";
			mes "我們稱這些為以太寶石，對吧？";
			next;
			mes "[史特拉托斯]";
			mes "您可以將以太寶石的魔力轉移到其他武器上，但當然存在一定的風險。";
			mes "這很微妙，所以如果您不小心，砰！它會爆炸。";
			next;
			mes "[史特拉托斯]";
			mes "無論如何，如果您使用這種以太寶石，您可以強化您正在使用的武器。";
			mes "性能會更好。我們稱之為等級強化。";
			next;
			mes "[史特拉托斯]";
			mes "您也是，如果您喜歡的武器更上一層樓，您不會感到高興嗎？";
			mes "我們是協助它的人。雖然有點冒險。啊哈哈哈。";
			close;
	}

OnInit:
	setunittitle(getnpcid(0), "<Grade Enhancer>");
	end;
}
