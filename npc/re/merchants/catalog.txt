moc_para01,22,16,5	script	分類魔法師#catal01	704,{
	if (!checkweight(1301,3)) {
		mes "- 你的物品太多了。請清理出空間。 -";
		close;
	}
	set .@ticket_cost, 200;
	mes "[分類魔法師]";
	mes "看看... 吉芬的魔法學院";
	mes "現在直接販售";
	mes "全新的 ^007777萬用目錄銀幣^000000 了！";
	next;
	mes "他正在看著";
	mes "寫在他手上的字。";
	mes "他看起來太忙了";
	mes "沒空理你。";
	next;
	mes "[分類魔法師]";
	mes "全新的目錄";
	mes "剛剛推出了！";
	mes "你可以查看";
	mes "你能交易哪些物品";
	mes "以及商人那裡正在販售哪些物品";
	mes "。";
	next;
	mes "[分類魔法師]";
	mes "這個 ^007777萬用目錄銀幣^000000";
	mes "只需 200z！";
	mes "你一次最多可以購買 50 個！";
	next;
	switch(select("購買 ^007777萬用目錄銀幣^000000:我不需要！:你看起來不像商人！")) {
	case 1:
		mes "[分類魔法師]";
		mes "你... 想要全部 50 個？";
		mes "或者你需要多少個？";
		next;
		while (1) {
			input .@input;
			mes "[分類魔法師]";
			if (.@input == 0) {
				mes "交易已停止！";
				mes "我不知道接下來該做什麼...";
				mes "我應該怎麼辦？";
				emotion ET_PROFUSELY_SWEAT, playerattached();
				close;
			} else if (.@input < 1 || .@input > 50) {
				mes "應該少於 50 個。";
				next;
			} else
				break;
		}
		set .@sell, .@ticket_cost * .@input;
		mes "The total number of catalog(s) that you're trying to purchase is " + .@input + " pieces.";
		mes "It costs " + .@sell + " z.";
		if (Zeny < .@sell) {
			mes "看來你沒有足夠的錢。";
			mes "你可以賣掉你的裝備";
			mes "來賺更多錢... 如果你願意的話。";
			close;
		}
		mes "好的，給你。";
		set Zeny, Zeny - .@sell;
		getitem 12580,.@input; //Vending_Search_Scroll
		close;
	case 2:
		mes "[分類魔法師]";
		mes "啊... 你真的很需要這個";
		mes "物品... 你要我";
		mes "再解釋一遍嗎？我不知道";
		mes "接下來該做什麼...";
		mes "我應該怎麼辦？";
		emotion ET_PROFUSELY_SWEAT, playerattached();
		close;
	case 3:
		mes "[分類魔法師]";
		mes "我真正的職業是魔法師。";
		mes "我本打算請商人";
		mes "公會來賣它們，但他們";
		mes "試圖將它們作為掃帚銷售的";
		mes "附錄。這就是為什麼";
		mes "我獨自販售它們的原因。";
		next;
		mes "[分類魔法師]";
		mes "這裡有很多冒險者";
		mes "來來往往。";
		mes "我很確定我很快就能賣給他們所有人。";
		mes "你不覺得嗎？";
		mes "所以... 讓我們開始吧！！多買一些，拜託！拜託...！";
		close;
	}
}
