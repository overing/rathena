cmd_fild07,375,167,1	script	Ruins Black Trader#pa0829_01	4_F_JOB_HUNTER,{
	mes "[遺跡黑市商人]";
	mes "嘿，每次都走進走出這裡是不是很辛苦？ 只要花費 ^0000ff20,000^000000 Zeny，你就可以獲得一個捲軸，它會直接帶你到遺跡的入口，你覺得怎麼樣？";
Purchase:
	next;
	switch( select( "給我一個", "我認為不用了" )) {
		case 1:
			if (!checkweight(22980,1) || (MaxWeight - Weight) < 1000) {
				mes "由於您擁有大量的物品，因此無法繼續對話。";
				mes "請整理您的物品，然後再試一次。";
				close;
			}
			else if (Zeny < 20000) {
				mes "[遺跡黑市商人]";
				mes "你似乎沒有錢了。一個捲軸要 20,000 Zeny。";
				close;
			}
			Zeny -= 20000;
			getitem 22980,1;
			mes "[遺跡黑市商人]";
			mes "這是一筆好交易。還需要更多嗎？";
			goto Purchase;
		case 2:
			mes "[遺跡黑市商人]";
			mes "隨時再來";
			close;
	}
}

// Equipment Shop
cmd_fild07,57,275,5	script	Artifact Appraiser#pa0829_01	1_F_02,{
	if (!checkweight(1201,1) || (MaxWeight - Weight) < 1000) {
		mes "由於您擁有大量的物品，因此無法繼續對話。";
		mes "請整理您的物品，然後再試一次。";
		close;
	}
	.@stone_id = 6905;
	mes "[神器鑑定師]";
	mes "選擇您要購買的裝備類型。 只要您有 " + getitemname(.@stone_id) + "，就可以購買任意數量。";
	next;
	switch (select("取消:武器:防具")) {
		case 1:
			mes "[神器鑑定師]";
			mes "歡迎隨時再來～";
			close;

		case 2:
			setarray .@equip_id,1994,1938,13323,13126,28703,2024,16038,21014,28105,18128;
			.@price = 50;
			break;

		case 3:
			setarray .@equip_id,15141,22075,20779,19033;
			.@price = 50;
			break;
	}
	.@menu$ = "Cancel:";
	for (.@i = 0; .@i < getarraysize(.@equip_id); .@i++)
		.@menu$ += getitemname(.@equip_id[.@i]) + ":";
	.@s = select(.@menu$) - 1;
	switch (.@s) {
		case 0:
			mes "[神器鑑定師]";
			mes "歡迎隨時再來～";
			close;

		default:
			.@s--;
			mes "[神器鑑定師]";
			mes "您需要 ^0000FF" + .@price + "^000000 " + getitemname(.@stone_id) + " 才能購買 " + getitemname(.@equip_id[.@s]) + "～";
			next;
			if (select("取消:購買") == 1) {
				mes "[神器鑑定師]";
				mes "歡迎隨時再來～";
				close;
			}
			if (countitem(.@stone_id) < .@price) {
				mes "[神器鑑定師]";
				mes "您沒有足夠的 " + getitemname(.@stone_id) + " 來購買此物品。";
				close;
			}
			mes "[神器鑑定師]";
			mes "感謝您的信任。下次再來。";
			delitem .@stone_id,.@price;
			getitem .@equip_id[.@s],1;
			close;
	}
}

// Equipment Enchanter
cmd_fild07,60,275,3	script	Artifact Enhancer#pa0829_01	4_F_JOB_BLACKSMITH,{
	if (!checkweight(1201,1) || (MaxWeight - Weight) < 1000) {
		mes "由於您擁有大量的物品，因此無法繼續對話。";
		mes "請整理您的物品，然後再試一次。";
		close;
	}
	disable_items;
	.@stone_id = 6905;
	function equip_check;
	mes "[神器強化師]";
	mes "如果您正在尋找強化您在無限空間中獲得的裝備，那麼您來對地方了。";
	mes "您有任何 " + getitemname(.@stone_id) + " 嗎？";
	next;
	switch (select("如何強化我的裝備？:強化裝備:初始化裝備的附魔")) {
		case 1:
			mes "[神器強化師]";
			mes "如果您探索帕洛斯燈塔下的空間，就能夠獲得 " + getitemname(.@stone_id) + "。";
			next;
			mes "[神器強化師]";
			mes "有了這種材料，我可以強化您從無限空間獲得的裝備。";
			next;
			mes "[神器強化師]";
			mes "我只能附魔第 3 和第 4 個插槽。";
			next;
			mes "[神器強化師]";
			mes "在附魔過程中，裝備沒有被摧毀的機會，但是初始化附魔時，裝備有很高的機率被摧毀。";
			next;
			mes "[神器強化師]";
			mes "那麼，當您要附魔裝備時，再來找我。";
			break;

		case 2:
			.@stone_id = 6905;
			.@fee = 20;
			mes "[神器強化師]";
			mes "請選擇您要附魔的裝備。";
			next;
			switch (select("取消:武器:盔甲:鞋子:披肩:頭盔")) {
				case 1:
					mes "[神器強化師]";
					mes "那麼，當您要附魔裝備時，再來找我。";
					close;

				case 2:
					.@part = EQI_HAND_R;
					break;

				case 3:
					.@part = EQI_ARMOR;
					break;

				case 4:
					.@part = EQI_SHOES;
					break;

				case 5:
					.@part = EQI_GARMENT;
					break;

				case 6:
					.@part = EQI_HEAD_TOP;
					break;
			}
			.@equip_id = getequipid(.@part);
			.@refine = getequiprefinerycnt(.@part);
			equip_check(.@equip_id);
			for (.@i = 0;.@i < 4;.@i++) {
				.@card[.@i] = getequipcardid(.@part,.@i);
				.@check[.@i] = getequipcardid(.@part,.@i);
			}
			if (.@card[2] > 0) {
				mes "[神器強化師]";
				mes "我只能附魔到第三個插槽。您的裝備無法再被附魔。";
				close;
			}
			switch (.@part) {
				case EQI_HAND_R:
					setarray .@enchant$,
					"4700,4701:4710,4711:4720,4721",
					"4811,4810,4809,4808,4820,4821,4822,4823:4815,4814,4813,4812,4826,4827,4828,4829:4832,4833,4834,4835,4836,4837,4838,4839";
					break;

				case EQI_ARMOR:
				case EQI_SHOES:
					setarray .@enchant$,
					"4700,4701,4702,4703:4710,4711,4712,4713:4720,4721,4722,4723",
					"4795,4796,4797:4870,4871,4800:4870,4871,4800";
					break;

				case EQI_GARMENT:
				case EQI_HEAD_TOP:
					setarray .@enchant$,
					"4700,4701,4702,4703:4710,4711,4712,4713:4720,4721,4722,4723",
					"4861,4862,4867,4868,4900:4861,4862,4867,4868,4900:4861,4862,4867,4868,4900";
					break;
			}
			.@slot = (.@card[3] > 0 ? 2 : 3);
			.@index = (.@slot == 3 ? 0 : 1);
			mes "[神器強化師]";
			mes "您可以選擇 3 種附魔類型，附魔費用是 " + .@fee + " " + getitemname(.@stone_id) + "。";
			mes "我會確保它順利附魔。";
			next;
			.@type = select("離開:物理:魔法:遠程") - 2;
			mes "[神器強化師]";
			if (.@slot == 3)
				mes "好的，讓我們開始第一次附魔。";
			else
				mes "好的，讓我們開始第二次附魔。";
			next;
			if (select("我稍後回來。:請繼續。") == 1) {
				mes "[神器強化師]";
				mes "如果您改變主意，請回來找我。";
				close;
			}
			if (countitem(.@stone_id) < .@fee) {
				mes "[神器強化師]";
				mes "嗯。 順便說一句，我想你沒在聽，你沒有足夠的 " + getitemname(.@stone_id) + "。";
				close;
			}
			explode(.@T$,.@enchant$[.@index],":");
			explode(.@TT$,.@T$[.@type],",");
			.@enchant = atoi(.@TT$[rand(getarraysize(.@TT$))]);
			.@card[.@slot] = .@enchant;
			delitem .@stone_id,.@fee;
			if (callfunc("F_IsEquipIDHack", .@part, .@equip_id) || callfunc("F_IsEquipRefineHack", .@part, .@refine) || callfunc("F_IsEquipCardHack", .@part, .@check[0], .@check[1], .@check[2], .@check[3]))
				close;
			delequip .@part;
			getitem2 .@equip_id,1,1,.@refine,0,.@card[0],.@card[1],.@card[2],.@card[3];
			specialeffect2 EF_REPAIRWEAPON;
			mes "[神器強化師]";
			mes "嗯。 弄得很好。 立即檢查您的裝備。";
			break;
		
		case 3:
			.@stone_id = 6905;
			.@fee = 30;
			.@break_chance = 30;
			mes "[神器強化師]";
			mes "請選擇您要附魔的裝備。";
			next;
			switch (select("取消:武器:盔甲:鞋子:披肩:頭盔")) {
				case 1:
					mes "[神器強化師]";
					mes "那麼，當您要附魔裝備時，再來找我。";
					close;

				case 2:
					.@part = EQI_HAND_R;
					break;

				case 3:
					.@part = EQI_ARMOR;
					break;

				case 4:
					.@part = EQI_SHOES;
					break;

				case 5:
					.@part = EQI_GARMENT;
					break;

				case 6:
					.@part = EQI_HEAD_TOP;
					break;
			}
			.@equip_id = getequipid(.@part);
			.@refine = getequiprefinerycnt(.@part);
			equip_check(.@equip_id);
			for (.@i = 0; .@i < 4; .@i++) {
				.@card[.@i] = getequipcardid(.@part,.@i);
				.@check[.@i] = .@card[.@i];
			}
			if (.@card[3] == 0) {
				mes "[神器強化師]";
				mes "您的裝備上沒有任何附魔。";
				close;
			}
			mes "[神器強化師]";
			mes "您的無限空間裝備有可能在初始化過程中被摧毀。 您仍然願意繼續嗎？";
			next;
			if (select("離開:繼續") == 1) {
				mes "[神器強化師]";
				mes "如果您改變主意，請回來找我。";
				close;
			}
			if (countitem(.@stone_id) < .@fee) {
				mes "[神器強化師]";
				mes "嗯。 順便說一句，我想你沒在聽，你沒有足夠的 " + getitemname(.@stone_id) + "。";
				close;
			}
			delitem .@stone_id,.@fee;
			if (callfunc("F_IsEquipIDHack", .@part, .@equip_id) || callfunc("F_IsEquipRefineHack", .@part, .@refine) || callfunc("F_IsEquipCardHack", .@part, .@check[0], .@check[1], .@check[2], .@check[3]))
				close;
			delequip .@part;
			if (rand(1,100) > .@chance) {
				getitem2 .@equip_id,1,1,.@refine,0,.@card[0],.@card[1],0,0;
				specialeffect2 EF_REPAIRWEAPON;
				mes "[神器強化師]";
				mes "嗯。 弄得很好。 立即檢查您的裝備。";
			} else {
				specialeffect2 EF_REFINEFAIL;
				mes "[神器強化師]";
				mes "好吧，我確實警告過您。 您不走運，是嗎？";
			}
			break;
	}
	close;

	function	equip_check	{
		setarray .@equip_id,1994,1938,13323,13126,28703,2024,16038,21014,28105,18128,15141,22075,20779,19033;
		if (!getarg(0)) {
			mes "[神器強化師]";
			mes "您有脫下您的裝備嗎？";
			close;
		}
		if (inarray(.@equip_id,getarg(0)) == -1) {
			mes "[神器強化師]";
			mes "此裝備不適合附魔。 別忘了只有無限裝備系列可以附魔。";
			close;
		}
		return;
	}
