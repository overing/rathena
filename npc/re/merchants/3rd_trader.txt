job3_guil01,91,93,3	script	Rare Herb Collector	1_M_03,{
	if (checkweight(1201,1) == 0 || MaxWeight - Weight < 2000) {
		mes "- 等一下 !! -";
		mes "- 目前你身上攜帶 -";
		mes "- 太多物品了。 -";
		mes "- 請再試一次 -";
		mes "- 在你減輕一些重量之後。 -";
		close;
	}
	if (Class == Job_Guillotine_Cross || Class == Job_Guillotine_Cross_T || Class == Job_Baby_Guillotine_Cross) {
		mes "[稀有毒草蒐集家]";
		mes "我在世界各地遊蕩並蒐集稀有毒草。最近，我開始販賣一種叫做伊茲多（Izidor）的草藥。如果你有興趣，你可以購買它們。";
		next;
		.@i = select("我該如何購買？:用動物的血交換:用苦草交換:用致命的毒草交換:用冰凍的玫瑰交換:用薄荷交換:用希納勒交換")-2;
		if (.@i == -1) {
			mes "[稀有毒草蒐集家]";
			mes "你可以用 10,000 Zeny和以下其中一種稀有物品購買：動物的血、苦草、致命的毒草、冰凍的玫瑰、薄荷或希納勒。";
			next;
			mes "[稀有毒草蒐集家]";
			mes "為什麼我要收錢？這只是為了對像我這樣的收藏家表示一點敬意。哈哈哈...";
			close;
		} else {
			setarray .@exchange[0],702,621,631,749,605,703;
			.@item = .@exchange[.@i];
			.@price = 10000;

			mes "[稀有毒草蒐集家]";
			mes "你想要多少？";
			mes "你最多只能購買 '2000' 個。";
			mes "輸入 '0' 如果你想取消。";
			next;
			input .@amount;
			if (.@amount == 0) {
				mes "[稀有毒草蒐集家]";
				mes "你已經取消了交易。";
				close;
			}
			if (.@amount > 2000) {
				mes "[稀有毒草蒐集家]";
				mes "數量必須少於 2,000！";
				close;
			}
			.@total = .@amount * .@price;
			if (countitem(.@item) < .@amount || Zeny < .@total) {
				mes "[稀有毒草蒐集家]";
				mes "嘿，等你具備所有交換條件再來吧。";
				close;
			}
			mes "[稀有毒草蒐集家]";
			mes "很好。我已經收到錢和特殊物品了。";
			delitem .@item, .@amount;
			Zeny = Zeny - .@total;
			getitem 709, .@amount; //Izidor
			close;
		}
	}
	mes "[稀有毒草蒐集家]";
	mes "我在世界各地遊蕩並蒐集稀有毒草。但我不想把我的草藥賣給像你這樣的人……哈哈哈……";
	close;
}
