//===== rAthena Script =======================================
//= HD Refiners
//===== Description: =========================================
//= [Official Conversion]
//= Refiners that use HD ores to refine equipment. Upon
//= failure, the equipment is not destroyed; rather, its
//= refine level decreases by 1. The success rate is identical
//= to that for Enriched ores.
//= - "Blacksmith Mighty Hammer" only refines from +7~9.
//= - "Basta" only refines from +10 and up.
//===== Changelog: ===========================================
//= 1.0 First version. [Euphy]
//= 1.1 Removed re-roll behavior. [Secret]
//============================================================

// Blacksmith Mighty Hammer (+7~9) :: cash_smelting79
//============================================================
-	script	::MightyHammer	-1,{
	disable_items;
	mes "[鐵匠 重鎚]";
	mes "與其他人不同，我是一個只精鍊非常有限數量的物品的鐵匠。";
	mes "我只精鍊 ^CC0000+7 到 +9^000000 的物品。";
	next;
	mes "[鐵匠 重鎚]";
	mes "我的專長是，即使我的精鍊失敗，精鍊等級也會減少 1，而不會失去裝備。是不是很棒？";
	next;
	mes "[鐵匠 重鎚]";
	mes "所以讓我們加速，你說呢？你想要精鍊什麼物品？";
	next;
	setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
	for ( .@i = 1; .@i <= 10; ++.@i )
		.@menu$ = .@menu$ + ( getequipisequiped(.@indices[.@i]) ? getequipname(.@indices[.@i]) : F_getpositionname(.@indices[.@i]) + "-[Not equipped]" ) + ":";
	.@part = .@indices[ select(.@menu$) ];
	if (!getequipisequiped(.@part)) {
		mes "[鐵匠 重鎚]";
		switch(.@part) {
		case EQI_HEAD_TOP:
			mes "我是鐵匠，不是髮型師。";
			break;
		case EQI_ARMOR:
			mes "用我的錘子，我會讓你成為天空中的星星。";
			break;
		case EQI_HAND_L:
		case EQI_HAND_R:
			mes "製造人工手不是我的專長。";
			break;
		case EQI_GARMENT:
			mes "把物品拿出來，我才能精鍊它！";
			break;
		case EQI_SHOES:
			mes "這股腳臭味是從哪裡來的？";
			break;
		case EQI_ACC_L:
		case EQI_ACC_R:
			mes "飾品在哪裡？";
			break;
		case EQI_HEAD_MID:
			mes "你想要我精鍊什麼？";
			break;
		case EQI_HEAD_LOW:
			mes "嗯？你想要我做什麼？";
			break;
		}
		close;
	}
	if (!getequipisenableref(.@part)) {
		mes "[鐵匠 重鎚]";
		mes "此物品無法精鍊。";
		close;
	}
	switch( getequiprefinerycnt(.@part) ) {
	case 7:
		.@blacksmith_blessing_count = 1;
		break;
	case 8:
		.@blacksmith_blessing_count = 2;
		break;
	case 9:
		.@blacksmith_blessing_count = 4;
		break;
	default:
		mes "[鐵匠 重鎚]";
		mes "我只處理精鍊等級從 +7 到 +9 的物品。";
		close;
	}

	.@refineitemid = getequipid(.@part); // save id of the item
	.@itemtype = getiteminfo( .@refineitemid, ITEMINFO_TYPE );
	.@refinerycnt = getequiprefinerycnt(.@part); //save refinery count
	setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);
	.@price = getequiprefinecost(.@part, REFINE_COST_HD, REFINE_ZENY_COST);
	.@material = getequiprefinecost(.@part, REFINE_COST_HD, REFINE_MATERIAL_ID);

	mes "[鐵匠 重鎚]";
	mes "為了精鍊你選擇的裝備，你需要 ^ff9999" + getitemname(.@material) + "^000000 和 20,000 zeny 作為費用。";
	mes "你準備好了嗎？";
	next;
	if (select("是:否") == 2) {
		mes "[鐵匠 重鎚]";
		mes "我會等你準備好。";
		close;
	}
	if (getequippercentrefinery(.@part) < 100) {
		mes "[鐵匠 重鎚]";
		mes "看起來這個物品很可能會精鍊失敗。";
		mes "好吧，即使失敗了，也只會減少 1 個精鍊等級。";
		mes "你想要繼續精鍊嗎？";
		next;
		if (countitem(6635) < .@blacksmith_blessing_count)
			setarray .@menu$[0], "", "Yes", "Not yet";
		else {
			mes "[鐵匠 重鎚]";
			mes "啊！是 ^0000ff鐵匠的祝福^000000 嗎？";
			mes "有了鐵匠的祝福，如果精鍊失敗，裝備就不會消失！";
			next;
			mes "[鐵匠 重鎚]";
			if ( .@itemtype != IT_WEAPON )
				mes "對於 +" + getequiprefinerycnt(.@part) + " 裝備，使用^316AC5 " + .@blacksmith_blessing_count + " 個 鐵匠的祝福^000000 可以防止裝備消失。你想要使用它來精鍊嗎？";
			else
				mes "對於 +" + getequiprefinerycnt(.@part) + " 武器，使用^316AC5 " + .@blacksmith_blessing_count + " 個 鐵匠的祝福^000000 可以防止裝備消失。你想要使用它來精鍊嗎？";
			next;
			setarray .@menu$[0], "Use it to refine", "Refine directly without it", "Don't refine yet";
		}
		switch( select(.@menu$[0], .@menu$[1], .@menu$[2]) ) {
		case 1:
			.@bless_who = 1;
			break;
		case 2:
			break;
		case 3:
			mes "[鐵匠 重鎚]";
			mes "只有克服對失敗的恐懼的人才能獲得傑作。";
			close;
		}
	}
	if ((.@bless_who && countitem(6635) < .@blacksmith_blessing_count) || countitem(.@material) == 0 || Zeny < .@price) {
		mes "[鐵匠 重鎚]";
		mes "你不是才剛說你已經準備好一切了嗎？";
		close;
	}
	if (.@bless_who)
		delitem 6635, .@blacksmith_blessing_count;
	delitem .@material,1;
	Zeny = Zeny - .@price;

	// anti-hack
	if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) ||
		callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt)) {
		mes "[鐵匠 重鎚]";
		emotion ET_FRET;
		mes "等一下...";
		mes "你以為我是笨蛋嗎？！";
		mes "你趁我不注意的時候換了物品！給我滾出去！";
		close;
	}

	mes "[鐵匠 重鎚]";
	mes "鏘！鏘！鏘！";
	if (getequippercentrefinery(.@part, true) > rand(100)) {
		successrefitem .@part;
		next;
		emotion ET_BEST;
		mes "[鐵匠 重鎚]";
		mes "每次聽到這個聲音，我的頭腦都會煥然一新。";
		mes "給你，拿去吧。精鍊完美成功！";
		close;
	}
	if (.@bless_who == 1) {
		specialeffect EF_HOLYHIT;
		next;
		emotion ET_HUK;
		mes "[鐵匠 重鎚]";
		mes "什麼？！！";
		next;
		mes "[鐵匠 重鎚]";
		mes "哎呀！我現在變成無臉男了。哼哼。";
		close;
	}
	downrefitem .@part;
	next;
	emotion ET_HUK;
	mes "[鐵匠 重鎚]";
	mes "哎呀！！";
	next;
	mes "[鐵匠 重鎚]";
	mes "我相信像你這樣的人永遠不會因為精鍊等級降低 1 而責怪我。嗯。";
	close;
}

// Basta (+10 and up) :: cash_smelting
//============================================================
-	script	::巴斯塔	-1,{
	disable_items;
	mes "[巴斯塔]";
	mes "我是全世界最好的鐵匠，巴斯塔。";
	mes "但我沒有提供普通的精鍊服務。";
	mes "我只精鍊 ^CC0000+10 以上^000000 的裝備。";
	next;
	mes "[巴斯塔]";
	mes "您想要精煉哪個裝備呢？";
	next;
	setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
	for ( .@i = 1; .@i <= 10; ++.@i )
		.@menu$ = .@menu$ + ( getequipisequiped(.@indices[.@i]) ? getequipname(.@indices[.@i]) : F_getpositionname(.@indices[.@i]) + "-[Unequipped]" ) + ":";
	.@part = .@indices[ select(.@menu$) ];
	if (!getequipisequiped(.@part)) {
		mes "[巴斯塔]";
		switch(.@part) {
		case EQI_HEAD_TOP:
			mes "你的頭是裝備嗎？";
			break;
		case EQI_ARMOR:
			mes "您想要我做什麼？";
			break;
		case EQI_HAND_L:
		case EQI_HAND_R:
			mes "製作義肢不是我的專長。";
			break;
		case EQI_GARMENT:
			mes "您知道長袍是什麼嗎？";
			break;
		case EQI_SHOES:
			mes "如果您想要精煉您的腳，別來找我，試著跑個馬拉松。";
			break;
		case EQI_ACC_L:
		case EQI_ACC_R:
			mes "飾品在哪裡？";
			break;
		case EQI_HEAD_MID:
			mes "嗯...我沒有看到任何值得精煉的裝備。";
			break;
		case EQI_HEAD_LOW:
			mes "我不能讓您變聰明。去找學校老師吧。";
			break;
		}
		close;
	}
	if (!getequipisenableref(.@part)) {
		mes "[巴斯塔]";
		mes "即使是我也無法精煉這個物品。不可能的。";
		close;
	}
	.@refine_count = getequiprefinerycnt(.@part);
	if (.@refine_count < 10) {
		mes "[巴斯塔]";
		mes "我不是跟您說過了嗎？我只精煉 +10 或以上的裝備。";
		close;
	}
	else if (.@refine_count == 10)
		.@blacksmith_blessing_count = 7;
	else if (.@refine_count == 11)
		.@blacksmith_blessing_count = 11;
	else if (.@refine_count == 20) {
		mes "[巴斯塔]";
		mes "這把武器很完美，不需要再精煉了～";
		close;
	}
	.@refineitemid = getequipid(.@part); // save id of the item
	.@itemtype = getiteminfo( .@refineitemid, ITEMINFO_TYPE );
	.@refinerycnt = getequiprefinerycnt(.@part); //save refinery count
	setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);
	.@price = getequiprefinecost(.@part, REFINE_COST_HD, REFINE_ZENY_COST);
	.@material = getequiprefinecost(.@part, REFINE_COST_HD, REFINE_MATERIAL_ID);

	if ( .@itemtype != IT_WEAPON )
		.@type$ = "armor";
	else
		.@type$ = "weapon";
	mes "[巴斯塔]";
	mes "嗯...這是您想要精煉的嗎？";
	mes "要精煉這個裝備，我需要 1 個 ^ff9999" + getitemname(.@material) + "^000000 以及 " + callfunc("F_InsertComma",.@price) + " Zeny作為費用。";
	mes "您真的想要精煉這個嗎？";
	next;
	if (select("是:否") == 2) {
		mes "[巴斯塔]";
		mes "好的。如果這是您想要的...";
		close;
	}
	if (getequippercentrefinery(.@part, true) < 100) {
		mes "[巴斯塔]";
		mes "這個 " + .@type$ + " 已經被精煉到相當高的程度了。";
		mes "如果您嘗試繼續精煉它，精煉等級可能會下降。";
		next;
		mes "[巴斯塔]";
		mes "我和其他地方的鐵匠不一樣。";
		mes "精煉等級不可能下降，比如說，3 或 4 級...聽起來很可怕。";
		mes "在這裡只會下降 1 級。";
		next;
		if (.@blacksmith_blessing_count == 0 || countitem(6635) < .@blacksmith_blessing_count) {
			mes "[巴斯塔]";
			mes "與其他鐵匠相比，風險較小。";
			mes "我已經採取了所有預防措施。您想要試試看嗎？";
			next;
			setarray .@menu$[0], "", "Yes", "No";
		}
		else {
			mes "[巴斯塔]";
			mes "哇～ 是 ^316AC5鐵匠的祝福^000000 嗎？很難獲得～ 但是，您有！";
			next;
			mes "[巴斯塔]";
			mes "對於 +" + .@refine_count + " " + .@type$ + " 需要 ^316AC5" + .@blacksmith_blessing_count + " 個 鐵匠的祝福^000000 才能防止裝備消失。您想要使用鐵匠的祝福進行精煉嗎？";
			next;
			setarray .@menu$[0], "Refine with Blacksmith Blessing", "Refine without Blacksmith Blessing", "Don't refine yet";
		}
		switch( select(.@menu$[0], .@menu$[1], .@menu$[2]) ) {
		case 1:
			.@bless_who = 1;
			break;
		case 2:
			break;
		case 3:
			mes "[巴斯塔]";
			mes "嗯～";
			mes "完全沒有挑戰性也可能是一種人生的智慧。";
			close;
		}
	}
	if ((.@bless_who && countitem(6635) < .@blacksmith_blessing_count) || countitem(.@material) == 0 || Zeny < .@price) {
		mes "[巴斯塔]";
		mes "嗯...您沒有帶齊所有需要的材料。";
		mes "當您帶齊時再來吧。";
		close;
	}
	if (.@bless_who)
		delitem 6635, .@blacksmith_blessing_count;
	delitem .@material,1;
	Zeny = Zeny - .@price;

	// anti-hack
	if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) ||
		callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt)) {
		mes "[巴斯塔]";
		emotion ET_FRET;
		mes "等一下...";
		mes "您覺得我很笨嗎？！";
		mes "我在不注意的時候，您偷偷換了物品！給我滾出去！";
		close;
	}

	mes "Pow! Pow! Pow! Pow!";
	if (getequippercentrefinery(.@part, true) > rand(100)) {
		successrefitem .@part;
		next;
		emotion ET_BEST;
		mes "[巴斯塔]";
		mes "太棒了！做得好！！";
		mes "我真的是全世界最棒的鐵匠！";
		close;
	}
	if (.@bless_who == 1) {
		specialeffect EF_HOLYHIT;
		next;
		emotion (!rand(5))?ET_MONEY:ET_HUK;
		mes "[巴斯塔]";
		mes "啊啊啊啊啊啊啊啊！！！";
		next;
		mes "[巴斯塔]";
		mes "精煉失敗了！";
		mes "像我這樣全世界最棒的鐵匠...";
		mes "也不能保證 100% 成功～";
	}
	else {
		downrefitem .@part;
		next;
		emotion (!rand(5))?ET_MONEY:ET_HUK;
		mes "[巴斯塔]";
		mes "啊啊啊啊啊啊啊啊！！！";
		next;
		mes "[巴斯塔]";
		mes "可惡！";
		mes "精煉失敗，而且精煉等級下降了！";
		mes "即使是全世界最棒的鐵匠也不能保證 100% 成功！";
	}
	mes "太可惜了。";
	next;
	mes "[巴斯塔]";
	mes "我下次會做得更好！別擔心！";
	close;
}
