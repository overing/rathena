-	script	RareDiamondMerchant	58,{

	set .@npc$, "[稀有鑽石商人]";

	mes .@npc$;
	mes "各位先生女士！ ！！";
	mes "你們聽過傳聞，但從未親眼見過";
	mes "世界上最稀有的鑽石！";
	mes "這顆 ^FF82FF'17 克拉鑽石'^000000 目前正以低價出售！！";
	mes "如果您對珍貴的鑽石感興趣，請仔細聽！";
	next;
	
	mes .@npc$;
	mes "您可以將鑽石作為禮物送給您的愛人。";
	mes "收到 ^FF82FF'17 克拉鑽石'^000000 的人會珍惜它";
	mes "並在心中珍藏你。";
	emotion ET_BLABLA;
	next;
	
	mes .@npc$;
	mes "什麼？你沒有愛人？";
	mes "您仍然可以自己保留它。";
	mes "^FF82FF'17 克拉鑽石'^000000 的迷人光環很可能";
	mes "為你的生活帶來愛情。";
	next;
	
	mes .@npc$;
	mes "被譽為鑽石中最迷人的鑽石...";
	mes "它就是 ^FF82FF'17 克拉鑽石'^000000！";
	next;

	set .@menu$, "^FF82FF'17 Carat Diamond'^000000?";
	if (countitem(6024))
		set .@menu$, .@menu$ + ":Exchange my 17 Carat Diamond for zeny...";

	switch(select(.@menu$)) {
	case 1:
		mes .@npc$;
		mes "是的！";
		mes "呵呵，你現在看到的 ^FF82FF'17 克拉鑽石'^000000";
		mes "是品質最高的 17 克拉鑽石之一。";
		mes "將它與你可能從摩洛哥挖出的任何豌豆大小的無色石頭相比";
		mes "都是無法形容的！";
		emotion ET_HUK;
		next;
		
		mes .@npc$;
		mes "有些人聲稱這就是卡丘阿在克魔島看到的";
		mes "那顆鑽石！";
		next;
		
		if (select("結束對話:多少錢？") == 1) {
			mes .@npc$;
			mes "它相當便宜！";
			mes "如果您有興趣，隨時與我交談！";
			close;
		}
		
		mes .@npc$;
		mes "當然！最重要的是價格！";
		mes "正如我之前所說，^FF82FF'17 克拉鑽石'^000000 已被";
		mes "鑑定為 17 克拉鑽石。";
		mes "世界上像它這樣的並不多。";
		next;
		
		mes .@npc$;
		mes "價格僅為 5 億 Zeny！";
		mes "還有 1,000,000Z 的服務費。";
		mes "與鑽石的品質相比";
		mes "這是一個很小的代價。";
		next;
		
		if (select("^828282太貴了。^000000:我要買。") == 1) {
			mes .@npc$;
			mes "要說像這樣的鑽石";
			mes "^FF82FF'17 克拉鑽石'^000000 太貴了，";
			mes "我無言以對。";
			close;
		}
		
		mes .@npc$;
		mes "哦！從我見到你的那一刻起，我就知道你是這顆美麗鑽石的";
		mes "合法主人。";
		mes "我很高興我沒有錯！";
		mes "我相信這顆鑽石會非常適合你。";
		mes "我只需要你完成這份合約。";
		next;
		
		while(1) {
			if (select("^B9062F我將簽署合約。^000000:^828282算了，我改變主意了。^000000") == 2) {
				mes .@npc$;
				mes "我知道了，太可惜了..";
				close;
			}
			
			mes .@npc$;
			mes "你的名字是？";
			next;
			
			mes "[" + strcharinfo(0) + "]";
			mes "我的名字是 " + strcharinfo(0) + "。";
			next;

			mes .@npc$;
			mes "好的，請在這裡簽名。";
			next;
			
			input .@charname$;
			if (.@charname$ == strcharinfo(0)) {
				mes .@npc$;
				mes "好的，很好。";
				mes "現在剩下要做的就是付款了。";
				next;
				
				if (Zeny >= 501000000) {
					mes .@npc$;
					mes ". . . . . .";
					next;
					
					mes .@npc$;
					mes "正在計算價格和服務費..";
					mes "501,000,000 Zeny。";
					mes "金額已確認。";
					emotion ET_BLABLA;
					set Zeny, Zeny - 501000000;
					getitem 6024, 1;
					next;

					mes .@npc$;
					mes "您可以隨時將鑽石換回 Zeny。";
					mes "但是，將收取 1,000,000Z 的服務費。";
					mes "感謝您的惠顧。";
					emotion ET_THANKS;
					close;
				}
				else {
					mes .@npc$;
					mes "嗯？";
					mes "抱歉，您的資金不足。";
					mes "包括服務費在內，總共需要 501,000,000Z。";
					mes "請檢查您的 Zeny 餘額，然後重試。";
					emotion ET_SCRATCH;
					close;
				}
			}
			else {
				mes .@npc$;
				mes "這真的是你的簽名嗎？";
				mes "要再次簽名以確認嗎？";
				next;
			}
		}
	case 2:
		mes .@npc$;
		mes "啊。您想將您的 ^FF82FF'17 克拉鑽石'^000000 換成 Zeny 嗎？";
		emotion ET_SURPRISE;
		next;
		if (select("是:否") == 2) {
			mes .@npc$;
			mes "那麼，你想要什麼呢？..";
			mes ". . . . . ....";
			emotion ET_THINK;
			close;
		}

		mes .@npc$;
		mes "我知道了。在我收到你的簽名後，";
		mes "您必須提供您的 ^FF82FF'17 克拉鑽石'^000000。";
		mes "也不要忘記還有 1,000,000Z 的服務費！";
		next;
		
		mes .@npc$;
		mes "你的名字是？";
		next;
		
		mes "[" + strcharinfo(0) + "]";
		mes "我的名字是 " + strcharinfo(0) + "。";
		next;
		
		mes .@npc$;
		mes "好的，請在這裡簽名。";
		next;
		
		while (1) {
			if (select("簽名:不簽名") == 2) {
				mes .@npc$;
				mes "沒有您的簽名，交易無法完成。";
				close;
			}
			
			input .@charname$;
			if (.@charname$ == strcharinfo(0)) {
				mes .@npc$;
				mes "您的簽名已收到。";
				mes "我現在將拿走你的 ^FF82FF'17 克拉鑽石'^000000..";
				mes "您將在之後收到 ^0000FF499,000,000z^000000。";
				next;
				
				mes .@npc$;
				mes "感謝您的惠顧。";
				mes "無論何時您需要 ^FF82FF'17 克拉鑽石'^000000，";
				mes "隨時帶著 5 億 Zeny 回來。";
				emotion ET_THANKS;
				delitem 6024, 1;
				set Zeny, Zeny + 499000000;
				close;
			}
			else {
				mes .@npc$;
				mes "這真的是你的簽名嗎？";
				mes "要再次簽名以確認嗎？";
				next;
			}
		}
	}
}
