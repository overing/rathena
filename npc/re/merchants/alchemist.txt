alde_alche,31,186,3	script	Craft Book Merchant#alde	883,{
	mes "[製作書商人]";
	mes "歡迎光臨。";
	mes "我在這裡販售";
	mes "^0000FF基因學者專用的特殊製作書^000000。";
	next;
	mes "[製作書商人]";
	mes "基因學者有些技能";
	mes "需要製作書才能啟用。";
	next;
	mes "[製作書商人]";
	mes "你擁有的製作書越多，";
	mes "你可以使用的技能就越多。";
	next;
	mes "[製作書商人]";
	mes "請理解";
	mes "我專門販售這些特殊的書籍，並且";
	mes "我^FF0000不賣其他物品^000000。";
	next;
	mes "[製作書商人]";
	mes "您想看看我的製作書嗎？";
	next;
	if(select("當然。:不用了。") == 2) {
		mes "[製作書商人]";
		mes "感謝您光臨我的商店。";
		mes "祝您旅途平安。";
		close;
	}
	if (checkweight(1201,1) == 0) {
		mes "[製作書商人]";
		mes "很抱歉，您的背包幾乎滿了。";
		mes "請先清空您的背包。";
		close;
	}
	if (MaxWeight - Weight < 2500) {
		mes "[製作書商人]";
		mes "請先清空您的背包。";
		close;
	}
	mes "[製作書商人]";
	mes "好的，這是可用的製作書列表。";
	next;
	switch(select("[蘋果炸彈製作書] 100,000 zeny:[鳳梨炸彈製作書] 100,000 zeny:[椰子炸彈製作書] 100,000 zeny:[甜瓜炸彈製作書] 100,000 zeny:[香蕉炸彈製作書] 100,000 zeny:[植物基因培養法] 100,000 zeny:[高等藥水製作手冊] 100,000 zeny:[混合烹飪書] 100,000 zeny:[健康改善研究書] 100,000 zeny:[活力飲料配方] 100,000 zeny:關閉")){
	case 1: set .@item,6279; break;
	case 2: set .@item,6280; break;
	case 3: set .@item,6281; break;
	case 4: set .@item,6282; break;
	case 5: set .@item,6283; break;
	case 6: set .@item,6284; break;
	case 7: set .@item,6285; break;
	case 8: set .@item,11022; break;
	case 9: set .@item,11023; break;
	case 10: set .@item,11024; break;
	case 11:
		mes "[製作書商人]";
		mes "感謝您的惠顧。";
		mes "歡迎再來。";
		close;
	}
	mes "[製作書商人]";
	mes "^0000FF["+getitemname(.@item)+"] 售價為";
	mes "100,000 zeny。";
	mes "您想購買多少？";
	next;
	input .@amount;
	if (.@amount == 0) {
		mes "[製作書商人]";
		mes "您想看看其他不同的書嗎？";
		close;
	}
	if (.@amount < 1 || .@amount > 99 ){
		mes "[製作書商人]";
		mes "您一次最多只能購買 100 個。";
		close;
	}
	mes "[製作書商人]";
	mes "您輸入了 "+.@amount+" 個 ^0000FF["+getitemname(.@item)+"]^000000 要購買。";
	mes "您要繼續嗎？";
	next;
	if (select("是:否") == 1) {
		set .@total, .@amount * 100000;
		if (Zeny < .@total) {
			mes "[製作書商人]";
			mes "很抱歉，您的錢不夠。";
			close;
		}
		if (checkweight(.@item,.@amount) == 0){
			mes "[製作書商人]";
			mes "您似乎無法攜帶所有東西。";
			mes "請檢查您的背包空間。";
			close;
		}
		mes "[製作書商人]";
		mes "感謝您的惠顧。";
		set Zeny,Zeny - .@total;
		getitem .@item,.@amount;
		close;
	}
	mes "[製作書商人]";
	mes "請慢慢考慮";
	mes "在您做出決定之前。";
	close;
}
