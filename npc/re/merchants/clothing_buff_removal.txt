//===== rAthena Script =======================================
//= Clothing Effect Removal Service
//===== Description: =========================================
//= Service to remove clothing effects like Xmas, Wedding, ...
//===== Changelog: ===========================================
//= 1.0 Initial version. [Balferian, Everade, Lemongrass]
//============================================================

prontera,80,106,5	script	Clothing Effect Removal Service	4_EP16_COOK2,{
	mes "[德瑞克·肖特]";
	mes "你是否因為服裝的 Buff 而無法參與戰鬥？這一定很惱人。";
	next;
	mes "[德瑞克·肖特]";
	mes "雖然它在你身上看起來不錯，但它也可能變得非常煩人。我可以一個手勢就把它去除掉！！我會把它整理好。如果你準備好了，選擇你的選項！";
	next;
	switch( select( "保留它", "立即釋放它" ) ){
		case 1:
			mes "[德瑞克·肖特]";
			mes "如果你改變主意，就回來找我。";
			close;
		case 2:
			sleep2 500;
			specialeffect2 EF_SPHERE;
			sleep2 500;
			specialeffect2 EF_SPHERE;
			sleep2 500;
			specialeffect2 EF_SPHERE;
			mes "[德瑞克·肖特]";
			mes "移除已完成。請確認你現在可以自由移動。";
			atcommand "@changedress";
			close;
	}
}
