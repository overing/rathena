			close;
		}
		.@cost = 100000 * .@s;
		.@rate = 40 + 10 * .@s;

		if (Zeny < .@cost) {
			mes "[幻象附魔師]";
			mes "你沒有足夠的錢。";
			close;
		}

			close;

		Zeny -= .@cost;
		delequip .@loc;
		if (rand(100) < .@rate) {
			getitem2 .@eq_id,1,1,.@refine,0,.@card[0],.@card[1],0,0;
			mes "[幻象附魔師]";
			mes "完成了。請檢查。";
			specialeffect2 EF_REPAIRWEAPON;
			close;
		}
		mes "[幻象附魔師]";
		mes "啊，真是不幸。好吧，你不可能總是那麼幸運。";
		specialeffect2 EF_LORD;
		close;
	}
	end;
