//===== rAthena Script ======================================= 
//= Blessed Refiner
//===== Description: =========================================
//= [Official Conversion]
//= Refiners that use Blessed ores to refine equipment.
//= Upon failure, the equipment is not destroyed. The success
//= rate is identical to that for Enriched ores.
//= - "Blacksmith Dister" only refines from +6~12.
//= NOTE: This NPC is currently disabled on official servers.
//===== Changelog: ===========================================
//= 1.0 First version. [Euphy]
//= 1.1 Removed re-roll behavior. [Secret]
//============================================================

// Main NPC :: new_smelting612
//============================================================
-	script	::BlacksmithDister	-1,{
	disable_items;
	mes "[鐵匠迪斯特]";
	mes "在這競爭激烈的社會中，我們必須與眾不同才能生存！";
	mes "我只精煉 +6 ~ 12 級的裝備。";
	next;
	mes "[鐵匠迪斯特]";
	mes "武器我需要 ^ff9999神之金屬^000000，盔甲我需要 ^ff9999鋁^000000。";
	mes "精煉失敗^FF0000不會損壞或降低精煉等級^000000。";
	next;
	mes "[鐵匠迪斯特]";
	mes "怎麼樣？你想精煉嗎？";
	next;
	setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
	for(set .@i,1; .@i<=10; set .@i,.@i+1)
		set .@menu$, .@menu$ + ( getequipisequiped(.@indices[.@i]) ? getequipname(.@indices[.@i]) : F_getpositionname(.@indices[.@i]) +"-[Unequipped]" ) +":";
	set .@part, .@indices[ select(.@menu$) ];
	if (!getequipisequiped(.@part)) {
		mes "[鐵匠迪斯特]";
		switch(.@part) {
		case EQI_HEAD_TOP:
			mes "我是鐵匠，不是髮型師。";
			break;
		case EQI_ARMOR:
			mes "用我的錘子，我會讓你成為天空中的一顆星。";
			break;
		case EQI_HAND_L:
		case EQI_HAND_R:
			mes "製造人工手不是我的專長。";
			break;
		case EQI_GARMENT:
			mes "把物品拿出來，我才能精煉它！";
			break;
		case EQI_SHOES:
			mes "這奇怪的味道是從哪裡來的？";
			break;
		case EQI_ACC_L:
		case EQI_ACC_R:
			mes "飾品在哪裡？";
			break;
		case EQI_HEAD_MID:
			mes "你想讓我精煉什麼？";
			break;
		case EQI_HEAD_LOW:
			mes "嗯？你想讓我做什麼？";
			break;
		}
		close;
	}
	if (!getequipisenableref(.@part)) {
		mes "[鐵匠迪斯特]";
		mes "此物品無法精煉。";
		close;
	}
	.@equip_id = getequipid(.@part);
	.@itemtype = getiteminfo( .@equip_id, ITEMINFO_TYPE );
	.@equip_refine = getequiprefinerycnt(.@part);
	setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);

	if (.@equip_refine < 6 || .@equip_refine > 12) {
		mes "[鐵匠迪斯特]";
		mes "此裝備已精煉至 "+.@equip_refine+"。我只處理精煉等級從 +6 到 +12 的物品！";
		close;
	}
	if( .@itemtype == IT_ARMOR ){
		.@equip_lv = getequiparmorlv( .@part );
		.@type$ = "Armor";

		switch(.@equip_lv) {
			case 1:
				set .@price,20000;
				set .@material,6439; //Unbreakable_Def
				break;
			default:
				// TODO:
				close;
		}
	}else if( .@itemtype == IT_WEAPON ){
		.@equip_lv = getequipweaponlv( .@part );
		.@type$ = "Weapon";

		switch( .@equip_lv ){
			case 1:
				set .@price,1000;
				set .@material,6438; //Unbreakable_Weap
				break;
			case 2:
				set .@price,2000;
				set .@material,6438; //Unbreakable_Weap
				break;
			case 3:
				set .@price,20000;
				set .@material,6438; //Unbreakable_Weap
				break;
			case 4:
				set .@price,40000;
				set .@material,6438; //Unbreakable_Weap
				break;
			default:
				// TODO:
				close;
		}
	}else{
		// TODO:
		close;
	}
	set .@ore$,"^ff9999Blessed "+.@type$+" Ore^000000";
	mes "[鐵匠迪斯特]";
	mes "這件 "+.@type$+" 已經被精煉到 "+.@equip_refine+" 了。要精煉它，我需要 "+.@ore$+" 和 "+callfunc("F_InsertComma",.@price)+" zeny 的精煉費用。";
	mes "你想要繼續嗎？";
	next;
	if(select("是:否") == 2) {
		mes "[鐵匠迪斯特]";
		mes "我很忙，別跟我開玩笑！";
		close;
	}
	if (getequippercentrefinery(.@part) < 100) {
		mes "[鐵匠迪斯特]";
		mes "這個 "+.@type$+" 已經被精煉過很多次了。雖然它不會在失敗時消失，但 "+.@ore$+" 將會消失！";
		next;
		mes "[鐵匠迪斯特]";
		mes "當然，精煉費用是不會退還的！你想要繼續嗎？";
		next;
		if(select("是:否") == 2) {
			mes "[鐵匠迪斯特]";
			mes "很好。如果失敗了，我也會很難過的。";
			close;
		}
	}
	if (countitem(.@material) == 0 || Zeny < .@price) {
		mes "[鐵匠迪斯特]";
		mes "材料不足。";
		mes "要精煉 "+.@equip_lv+" 級的 "+.@type$+"，我需要 "+.@ore$+" 和 "+callfunc("F_InsertComma",.@price)+" zeny 的精煉費用。";
		close;
	}
	delitem .@material,1;
	set Zeny, Zeny-.@price;

	// anti-hack
	if (callfunc("F_IsEquipIDHack", .@part, .@equip_id) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) || callfunc("F_IsEquipRefineHack", .@part, .@equip_refine))
		close;

	mes "[鐵匠迪斯特]";
	mes "噹！噹！噹！";
	if (getequippercentrefinery(.@part, true) > rand(100)) {
		specialeffect EF_BLESSING;
		successrefitem .@part;
		next;
		mes "[鐵匠迪斯特]";
		mes "哈哈哈！！！";
		next;
		mes "[鐵匠迪斯特]";
		mes "精煉大成功～哈哈哈！";
		close;
	}
	specialeffect EF_CURSEATTACK;
	specialeffect2 EF_SUI_EXPLOSION;
	next;
	emotion (!rand(5))?ET_ANGER:ET_HUK;
	mes "[鐵匠迪斯特]";
	mes "我的天啊！！！！";
	next;
	mes "[鐵匠迪斯特]";
	mes "哦！雖然它受到 "+.@ore$+" 的祝福，得以保留裝備...";
	next;
	mes "[鐵匠迪斯特]";
	mes "我失敗時心情更糟！！";
	close;
}
