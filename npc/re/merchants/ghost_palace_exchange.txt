dali02,51,130,4	script	次元旅人	4_F_SHABBY,{
	if (checkweight(28100,1) == 0) {// Thanos Axe
		mes "- 你無法繼續進行，因為你身上攜帶太多物品了。 -";
		close;
	}
	.@item_name_req$ = getitemname(6672);// Gray_Shard
	mes "[次元旅人]";
	mes "冒險者朋友，我需要" + mesitemlink( 6672 ) + "。要不要用我有的東西跟你交換？";
	next;
	.@type = select( "給我看看你的武器。", "給我看看你的盔甲。", "有什麼特別的東西嗎？", "我不需要。" ) - 1;
	mes "[次元旅人]";
	switch(.@type) {
	case 0:
		mes "喔！你對武器有興趣嗎？我們有所有能從幽靈皇宮取得的武器。要不要選一個？";
		break;
	case 1:
		mes "盔甲？我們有所有能從幽靈皇宮取得的盔甲。要不要選一個？";
		break;
	case 2:
		mes "呼！看來很多人已經聽說我這裡有好東西了。隨意看看我的收藏吧。";
		break;
	case 3:
		mes "再見了朋友～歡迎再來！";
		close;
	}
	next;
	mes "[次元旅人]";
	mes "啊！需要寫在" + mesitemlink( 6672 ) + "旁邊的數字。";
	mes "請記住這一點。";

	switch(.@type) {
	case 0:
		setarray .@items[0],
			13441, 200,		// Thanos Sword
			21009, 200, 	// Thanos Great Sword
			1438,  200,		// Thanos Spear
			1496,  200,		// Thanos Long Spear
			1669,  200,		// Thanos Staff
			2023,  200,		// Thanos Rod
			18119, 200,		// Thanos Bow
			13093, 200,		// Thanos Dagger
			28000, 200,		// Thanos Katar
			1836,  200,		// Thanos Knuckle
			16028, 200,		// Thanos Hammer
			28100, 200,		// Thanos Axe
			1933,  200,		// Thanos Violin
			1988,  200;		// Thanos Whip
		break;
	case 1:
		setarray .@items[0],
			2187,  100,		// Shield Of Gray
			15090, 100, 	// Armor Of Gray
			15091, 100,		// Gray Robe
			20721, 100,		// Cloak Of Gray
			22033, 100,		// Boots Of Gray
			18820, 100;		// Gray Helmet
		break;
	case 2:
		setarray .@items[0],
			27184, 10000,		// Knight Sakray Card
			23676, 50, 			// Gray Charcoal Melee (Melee)
			23677, 50,			// Gray Charcoal Magic (Magic)
			23678, 50;			// Gray Charcoal Range (Range)
		break;
	}
	.@size = getarraysize(.@items);
	for ( .@i = 0; .@i < .@size; .@i += 2 )
		.@menu$ += getitemname(.@items[.@i]) + " - " + .@items[.@i+1] + ":";

	while(true) {
		next;
		.@s = (select(.@menu$) - 1) * 2;
		mes "[次元旅人]";
		if (.@type != 1)	// armor type doesn't display item info
			mes "你是說" + mesitemlink( .@items[.@s] ) + "嗎？";
		mes "需要" + .@items[.@s+1] + " ^006400" + .@item_name_req$ + "^000000 來交換。";
		next;
		switch( select( "讓我交換。", "讓我看看其他東西。", "我晚點再來。" ) ) {
		case 1:
			break;
		case 2:
			mes "[次元旅人]";
			switch(.@type) {
			case 0:
				mes "我有足夠的武器了。";
				break;
			case 1:
				mes "我們有很多盔甲給你。要不要選一個？";
				break;
			case 2:
				mes "如果你帶" + .@item_name_req$ + "給我，我就會用任何你想要的東西跟你交換。";
				break;
			}
			continue;
		case 3:
			mes "[次元旅人]";
			mes "保重朋友～下次見！";
			close;
		}
		if (countitem(6672) < .@items[.@s+1]) {
			mes "[次元旅人]";
			mes "你讓我感到為難了，朋友。";
			mes "我需要" + .@item_name_req$ + "。我會用 ^006400" + .@items[.@s+1] + " " + .@item_name_req$ + "^000000 來跟你交換。";
			close;
		}
		mes "[次元旅人]";
		mes "謝謝你。如果你需要其他東西，歡迎再來。";
		delitem 6672, .@items[.@s+1];
		getitem .@items[.@s], 1;
		close;
	}
}
