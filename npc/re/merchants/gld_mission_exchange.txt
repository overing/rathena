S_PurchaseGift:

	.@price = getarg(2);

	mes "您確定嗎？";
	next;
	switch(select("是的！","不要！")) {

			mes "[點點喵喵-GX]";
			if (countitem(6615) >= .@price) {
				mes "感謝您使用我們的服務，喵！";
				delitem 6615,.@price; // Siege_Guild_Coin
				getitem .@itemid,1;
			} else {
				mes "您需要 " + .@price + " 個紀念幣才能購買所選物品。";
				mes "您決定查看其他物品。";
			}
			break;

		case 2:
			mes "[點點喵喵-GX]";
			mes "您決定查看其他物品。";
			break;
	}
	return;

S_Kvasir:
	.@fromid = getarg(1);
	.@fromname$ = getarg(2);
	.@toid = getarg(3);
	.@toname$ = getarg(4);
	
	mes "[點點喵喵-GX]";

	if( countitem( .@fromid ) > 0 && countitem( 6615 ) > 9 ){
		mes "您有 " + .@fromname$ + "。";
		mes "您想把它換成 " + .@toname$ + " 嗎？";
		next;
		mes "[點點喵喵-GX]";
		switch( select( "是的！:離開" ) ){
			case 1:
				// Recheck the items once more
				if( countitem( .@fromid ) > 0 && countitem( 6615 ) > 9 ){
					mes "您已將 " + .@fromname$ + " 換成 " + .@toname$ + "。";
					delitem .@fromid, 1;
					delitem 6615, 10; // Siege_Guild_Coin
					getitem .@toid, 1;
					close;
				}
				break;
			case 2:
				mes "您決定查看其他物品。";
				return;
		}
	}
	mes "您需要 1 個 " + .@fromname$ + " 和 10 個紀念幣才能完成購買。";
