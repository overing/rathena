	if (checkweight(1201,1) == 0 || MaxWeight - Weight < 1000) {
		mes "[高莫]";
		mes "你攜帶太多東西了。放下一些行李再回來找我。";
		close;
	}
	if (countitem(1549)) {
		mes "[高莫]";
		mes "妳的重力加農炮用起來如何？";
		close;
	}
	if (Class != Job_Mechanic && Class != Job_Mechanic_T && Class != Job_Baby_Mechanic) {
		mes "[高莫]";
		mes "我已經退休了，而且是個機匠，我想我和你沒有任何關係。";
		close;
	}
	mes "[高莫]";
	mes "所以呢？搭乘瘋狂兔是什麼感覺？";
	mes "這是不允許那些未能尋找自己道路的人擁有的。";
	next;
	mes "[高莫]";
	mes "我？我很久以前就退休了。因為我出了意外...受了重傷，無法再次騎乘。我對此感到沮喪。";
	next;
	mes "[高莫]";
	mes "所以我決定留在這裡，為瘋狂兔鍛造齒輪。";
	mes "嗯？我不是說我對這份工作有任何抱怨。";
	next;
	mes "[高莫]";
	mes "被迫做你不喜歡的事情，沒有比這更悲慘的了。";
	next;
	mes "[高莫]";
	mes "有些人只是當作道具也很開心，你知道，就像成為別人可以依靠的牆。";
	next;
	mes "[高莫]";
	mes "這讓我感覺很好，我正要製作一個新的齒輪。我希望你幫助我。你覺得怎麼樣？";
	next;
	switch(select("我沒那麼感興趣。:你要做什麼？:你從哪裡獲得材料？:我帶了材料。")) {
	case 1:
		mes "[高莫]";
		mes "那我無能為力。";
		mes "下次見！";
		close;
	case 2:
		mes "[高莫]";
		mes "嗯，一根巨大的金屬棒，可以一擊粉碎對手！";
		next;
		mes "[高莫]";
		mes "它看起來並不花哨，但我設計了一個彈簧裝置，可以用巨大的力量進行攻擊。";
		next;
		mes "[高莫]";
		mes "它是隱藏的，所以當你找到機會時，一擊就能讓對手陷入危急狀態！";
		mes "我迫不及待想看到它在行動了！";
		next;
		mes "[高莫]";
		mes "如果你能想像這種武器有多好，我就會把它變成現實。";
		next;
		mes "[高莫]";
		mes "給我 200 個鋼鐵、30 個柔軟的管子和 1 個 布羅卡。 我可以用這些材料製造我剛才描述的東西。";
		close;
	case 3:
		mes "[高莫]";
		mes "您應該可以從許多怪物（如寇伯特）那裡獲得鋼鐵。";
		next;
		mes "[高莫]";
		mes "柔軟的管子比較稀有，由艾音布羅克和里希塔樂鎮附近的 金屬波利 掉落。";
		next;
		mes "[高莫]";
		mes "最後，找到 布羅卡 非常困難，但您應該能夠做些什麼。哈哈，去拿吧！";
		close;
	case 4:
		if (countitem(999) < 200 || countitem(7325) < 30 || countitem(1415) < 1) {
			mes "[高莫]";
			mes "你還沒有帶來我要求的所有材料。";
			next;
			mes "[高莫]";
			mes "我有一個習慣，就是放棄我一直在做的東西，當靈感來的時候，試圖製作新的東西。";
			mes "你最好快點！";
			close;
		} else {
			mes "[高莫]";
			mes "哦，我一直在等待。你沒有準時來，所以我想知道你是不是忘了。請在那裡稍等一下。";
			specialeffect EF_REPAIRWEAPON;
			next;
			mes "[高莫]";
			mes "完成了。你覺得怎麼樣？還不錯吧？雖然是我自己做的，但我不得不說它做得相當不錯。";
			next;
			mes "[高莫]";
			mes "重量、體積和看看它的顏色和光澤的組合。它觸動了我的心！";
			next;
			mes "[高莫]";
			mes "...........";
			next;
			mes "[高莫]";
			mes "呃，稍作改進，就可以大量生產和交付。我相信我可以製造更多這樣的東西，所以我會把這個給你。";
			next;
			mes "[高莫]";
			mes "哦，結果我根本不需要那麼多鋼鐵。所以，給你，把大部分拿回去。";
			delitem 999,50; //Steel
			delitem 7325,30; //Tube
			delitem 1415,1; //Skewer
			getitem 1549,1; //Pilebuncker
			getitem 1360,1; //Two_Handed_Axe
			close;
		}
	}
}
