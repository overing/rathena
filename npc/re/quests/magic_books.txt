function	script	Magic_Book_Essence	{
	if (MaxWeight - Weight < 1000) {
		mes "請減輕你的背包。";
		close;
	}
	.@item_req = getarg(1);
	.@req_amount = getarg(2);
	if (checkquest(getarg(0)) == 1 && countitem(.@item_req) >= .@req_amount) {
		specialeffect2 EF_BLIND;
		specialeffect2 EF_BEGINSPELL;
		progressbar "ffff00",4;
		delitem .@item_req,.@req_amount;
		if (rand(2)) {
			mes "你已經使用 "+ callfunc("F_InsertPlural",.@req_amount,getitemname(.@item_req)) +" 創造了 ^999900"+ getarg(3) +" 精髓^000000。";
			next;
			mes "^008800"+getarg(3)+" 精髓將永久存在，除非出現問題。^000000";
			completequest getarg(0);
			specialeffect2 EF_LORD;
			close;
		}
		mes "^008888你使用 "+ callfunc("F_InsertPlural",.@req_amount,getitemname(.@item_req)) +" 提煉精髓失敗了。^000000";
		close;
	}
	if (checkquest(getarg(0)) == 2) {
		mes "你已經在這個區域創造了 "+getarg(3)+" 精髓。";
		close;
	}
	mes "已經有人使用這個區域了。";
	close;
}
