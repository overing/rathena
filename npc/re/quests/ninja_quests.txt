que_ng,21,72,6	script	Master Armor Craftsman	4_M_RASWORD,{
	mes "[Gyo]";
	mes "你會如何定義勝利？";
	mes "擊倒敵人...";
	mes "然而，最重要的是...";
	mes "永遠不要讓自己倒下！";
	next;
	mes "[Gyo]";
	mes "為此，你必須更好地保護自己。";
	mes "當你阻礙'自身'的保護時，你無法保護'他人'。";
	next;
	if(select("合成盔甲。:結束對話。") == 2) {
		mes "[Gyo]";
		mes "謹慎也是戰鬥中重要的一種美德。";
		close;
	}
	switch(select("狼臂甲[1]:新月臂甲[1]:忍者鱗甲[1]:闇影君王鎧甲[1]:離開。")) {
	case 1:
		mes "[Gyo]";
		mes "狼臂甲... 一件野性的盔甲。";
		next;
		callsub L_Select,2172;
		if (countitem(2171) > 0 && countitem(6252) > 9) {  //custom translation
			delitem 2171,1; //Fox_Armguard
			delitem 6252,10; //Wolf_Blood
			getitem 2172,1; //Wolf_Armguard
			mes "[Gyo]";
			mes "你感覺到狼的本性了嗎？";
			mes "既狂野又迷人...";
			close;
		}
		break;
	case 2:
		mes "[Gyo]";
		mes "新月臂甲... 美麗與力量的結合... 沒有絲毫遜色。";
		next;
		callsub L_Select,2173;
		if (countitem(2171) > 0 && countitem(7321) > 99) {  //custom translation
			delitem 2171,1; //Fox_Armguard
			delitem 7321,100; //Fragment_Of_Crystal
			getitem 2173,1; //Crescent_Armguard
			mes "[Gyo]";
			mes "多麼美麗啊...";
			mes "彷彿全身都充滿了魔法。";
			close;
		}
		break;
	case 3:
		mes "[Gyo]";
		mes "忍者鱗甲... 據說是一位傳奇忍者曾經穿過的。";
		next;
		callsub L_Select,15054;
		if (countitem(7451) > 29 && countitem(7562) > 29 && countitem(6091) > 29) {  //custom translation
			delitem 7451,30; //Scale_Of_Red_Dragon
			delitem 7562,30; //Ice_Scale
			delitem 6091,30; //Dark_Red_Scale
			getitem 15054,1; //Ninja_Scale_Armor
			mes "[Gyo]";
			mes "這真是一件傳奇的盔甲...";
			mes "即使我不相信傳奇...";
			close;
		}
		break;
	case 4:
		mes "[Gyo]";
		mes "闇影君王鎧甲... 它是為了保護他人而製造的。";
		next;
		callsub L_Select,15055;
		if (countitem(15056) > 0 && countitem(6089) > 9) {  //custom translation
			delitem 15056,1; //Special_Ninja_Suit_
			delitem 6089,10; //Piece_Of_Darkness
			getitem 15055,1; //Tenebris_Latitantes
			mes "[Gyo]";
			mes "覺得怎麼樣？";
			mes "這件盔甲看起來是不是有兩個？";
			close;
		}
		break;
	case 5:
		close;
	}
	mes "[Gyo]";
	mes "你的材料不夠。";
	mes "再找找...";
	close;

L_Select:
	.@string1$ = "Check Ingredients.";
	.@string2$ = "Check Stats.";
	while(1) {
		switch(select(.@string1$,.@string2$,"合成。","離開。")) {
		case 1:
			.@string1$ = "";
			callsub L_Ingredients, getarg(0);
			break;
		case 2:
			.@string2$ = "";
			callsub L_Stats, getarg(0);
			break;
		case 3:
			callsub L_Combine;
			return;
		case 4:
			mes "[Gyo]";
			mes "看來你需要更多時間來決定。";
			close;
		}
	}

L_Ingredients:
	mes "[Gyo]";
	switch (getarg(0)) {
	case 2172:	//Wolf_Armguard
		mes "要製作 ^ff0000狼臂甲，你需要 1 個狐狸臂甲和 10 個狼的血^000000。";
		mes "你能感覺到嗎？狼血的跳動...";
		break;
	case 2173:	//Crescent_Armguard
		mes "要製作 ^ff0000新月臂甲，你需要 1 個狐狸臂甲和 100 個水晶碎片^000000。";
		mes "像水晶一樣美麗而清澈的新月將被刻在上面。";
		break;
	case 15054:	//Ninja_Scale_Armor
		mes "要製作 ^ff0000忍者鱗甲，你需要 30 個火龍鱗片、30 個冰鱗和 30 個暗紅鱗片^000000。";
		mes "你能感受到它傳奇的力量。";
		break;
	case 15055:	//Tenebris_Latitantes
		mes "要製作 ^ff0000闇影君王鎧甲，你需要一件有洞的特殊忍者服和 10 個黑暗碎片^000000。";
		mes "有了黑暗，它就能抹去存在本身...";
		break;
	}
	next;
	return;

L_Stats:
	mes "[" + getitemname(getarg(0)) + "]";
	switch (getarg(0)) {
	case 2172:	//Wolf_Armguard
		mes "近距離攻擊時，有機會獲得 5 秒 ATK+100，FLEE -50 的效果。";
		mes "盾牌類型，防禦 45，插槽 1";
		mes "需要等級 100，僅限忍者類型";
		break;
	case 2173:	//Crescent_Armguard
		mes "每強化一次，技能後延遲減少 2%。";
		mes "盾牌類型，防禦 70，插槽 1";
		mes "需要等級 100，僅限忍者類型";
		break;
	case 15054:	//Ninja_Scale_Armor
		mes "MHP+15%，MSP-30%";
		mes "盔甲類型，防禦 90，插槽 1";
		mes "需要等級 100，僅限忍者類型";
		break;
	case 15055:	//Tenebris_Latitantes
		mes "受到近距離攻擊時，有機會啟動等級 1 幻影 - 暗影。";
		mes "盔甲類型，防禦 60，插槽 1";
		mes "需要等級 100，僅限忍者類型";
		break;
	}
	next;
	return;

L_Combine:
	mes "[Gyo]";
	mes "請確保 ^ff0000你手邊有正確的材料和裝備^000000。";
	mes "之後再後悔也沒用了。";
	next;
	switch(select("合成。:離開。")) {
	case 1:
		if (checkweight(1201,1) == 0 || MaxWeight - Weight < 2000) {
			mes "- 等一下！！ -";
			mes "- 你無法獲得物品 -";
			mes "- 因為你攜帶的東西太多了。 -";
			mes "- 請稍後再試 -";
			mes "- 在減輕你的負擔之後。 -";
			close;
		}
		return;
	case 2:
		mes "[Gyo]";
		mes "看來你需要更多時間來決定。";
		close;
	}
	end;
}
