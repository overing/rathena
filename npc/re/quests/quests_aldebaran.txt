c_tower1,232,222,5	script	Belljamin Button#ct	968,{
	mes "[貝爾傑明·巴頓]";
	mes "這裡的怪物能夠控制時間。";
	if( BaseLevel < 150 ){
		mes "達到 ^0000FF等級 150^000000 時再回來。";
		close;
	}
	mes "時間、記憶、環境...一切都可能在這裡扭曲。";
	next;
	mes "[貝爾傑明·巴頓]";
	mes "我求你別再問更多了。這可能會讓你的生命陷入危險。";
	next;
	.@menu = select("不要問:前往扭曲之塔的第二層:前往扭曲之塔的第三層");
	switch(.@menu){
		case 1:
			break;
		case 2:
		case 3:
			if( .@menu == 2 ){
				.@map$ = "c_tower2_";
				.@x = 268;
				.@y = 26;
			}else{
				.@map$ = "c_tower3_";
				.@x = 65;
				.@y = 147;
			}
			mes "[貝爾傑明·巴頓]";
			mes "小心你的選擇。";
			next;
			mes "[貝爾傑明·巴頓]";
			mes "我需要 ^0000FF10,000z^000000 才能讓你進去。那是一個非常危險的地方。";
			next;
			switch( select( "退出:支付 10,000z:你為什麼需要錢？" ) ){
				case 1:
					break;
				case 2:
					if( Zeny < 10000 ){
						mes "[貝爾傑明·巴頓]";
						mes "真的嗎？你沒有錢還叫我帶你過去？";
						close;
					}
					Zeny -= 10000;
					mes "[貝爾傑明·巴頓]";
					mes "我不確定你將如何在裡面追蹤時間，但盡你所能地小心。";
					close2;
					warp .@map$, .@x, .@y;
					end;
				case 3:
					mes "[貝爾傑明·巴頓]";
					mes "我整天都在這裡為了冒險者的安全。我總是警告他們關於危險。這就是為什麼我需要某種形式的付款。";
					emotion ET_SWEAT;
					close;
			}
			break;
	}
	mes "[貝爾傑明·巴頓]";
	mes "呼...你做了一個很好的選擇。這個地方很殘酷，所以你可能應該回到一個安全的區域。";
	close;
}
