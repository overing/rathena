function	script	VIP_iRO_Acolyte	{
	if (!vip_status(VIP_STATUS_ACTIVE)) {
		mes getarg(0);
		mes "但是，我的全部力量只提供給那些 VIP。";
		mes "不過，我仍然可以提供你傳送到 "+getarg(1)+" 的服務。";
		next;
		switch(select(getarg(0)+" -> 1800Zeny:取消")) {
		case 1:
			if (Zeny < 1800) {
				mes getarg(0);
				mes "很抱歉，你沒有足夠的";
				mes "Zeny 來使用傳送";
				mes "服務。傳送到 "+getarg(1)+" 的費用是 1,800 Zeny。";
				close;
			}
			specialeffect2 EF_READYPORTAL;
			specialeffect2 EF_TELEPORTATION;
			specialeffect2 EF_PORTAL;
			next;
			set Zeny, Zeny - 1800;
			if (getarg(1) == "Juno")
				warp "yuno",158,125;
			else
				warp "prontera",116,72;
			close;
		case 2:
			close;
		}
	}
	return;
}
