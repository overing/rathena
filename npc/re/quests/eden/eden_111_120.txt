S_Quest:
	if (isbegin_quest(getarg(0)))
		erasequest getarg(0);
	setquest getarg(1);
	if (getarg(2,0))
		getexp getarg(2),getarg(3);
	return;
S_Hunting:
	.@total = getargcount();
	for ( .@i = 0; .@i < .@total; .@i += 3 ) {
		.@quest_id = getarg(.@i);
		switch( checkquest(.@quest_id,HUNTING) ) {
		case -1:
			if (checkquest(.@quest_id+1) > -1)
				erasequest (.@quest_id+1);
			continue;
		case 2:
			mes "[明敏]";
			mes "你完成了關於 "+ getarg(.@i+2) +" 的研究。謝謝你。";
			getexp 700000,getarg(.@i+1);
			break;
		case 0:
		case 1:
			mes "[明敏]";
			mes "我將從請求列表中刪除 "+ getarg(.@i+2) +" 的研究。";
		}
		erasequest .@quest_id;
		next;
	}
	return;
S_Gathering:
	.@quest_id = getarg(0);
	if (isbegin_quest(.@quest_id) > 0) {
		if (getarg(5,0))
			.@count_item2 = countitem(getarg(5));
		mes "[明敏]";
		if (countitem(getarg(3)) >= getarg(4) && .@count_item2 >= getarg(6,0)) {
			if (.@count_item2)
				mes "你完成了關於 "+ getitemname(getarg(3)) +" 和 "+ getitemname(getarg(5)) +" 的研究。謝謝你。";
			else
				mes "你完成了關於 "+ getitemname(getarg(3)) +" 的研究。謝謝你。";
			delitem getarg(3),getarg(4);
			if (getarg(5,0))
				delitem getarg(5),getarg(6);
			getexp 700000,getarg(1);
		}
		else {
			if (getarg(5,0))
				mes "我將從請求列表中刪除 "+  getitemname(getarg(3)) +" 和 "+ getitemname(getarg(5)) +" 的研究。";
			else
				mes "我將從請求列表中刪除 "+  getitemname(getarg(3)) +" 的研究。";
		}
		erasequest .@quest_id;
		next;
	}
	else if (isbegin_quest(.@quest_id+1) > 0)
		erasequest (.@quest_id+1);
	return;
