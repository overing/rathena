-	script	儲存箱#00::pggc	-1,{
	mes "這是一個物品儲藏櫃。";
	mes "在一張紙上寫著訊息。";
	next;
	mes "+ 櫃子是 +";
	mes "+ 伊甸園專用 +";
	mes "如果您想使用這個";
	mes "櫃子，請檢查";
	mes "以下內容：";
	next;
	mes "1.您是伊甸園";
	mes "   的成員嗎？";
	mes "2.您是否已經學會";
	mes "   足夠的基本技能？";
	mes "3.櫃子費用是";
	mes "  ^4d4dff500 zeny^000000！";
	next;
	if (callfunc("F_HasEdenGroupMark")) {
		mes "您需要投入 Zeny 才能使用櫃子。";
		mes "費用：500 Zeny";
		mes "您想使用它嗎？";
		next;
		switch(select("使用櫃子。:取消。")) {
			case 1:
				if (Zeny > 499) {
					if(!callfunc("F_CanOpenStorage")){
						mes "這個櫃子對我不起作用。";
						mes "也許我還沒有資格使用櫃子。";
						close;
					}
					set Zeny, Zeny - 500;
					close2;
					openstorage;
					end;
				}
				mes "我沒有足夠的 Zeny。";
				mes "我至少需要 500 Zeny 才能使用櫃子。";
				close;
			case 2:
				mes "我下次再用。";
				close;
		}
	}
	mes "我需要一個伊甸園的標記才能使用這個櫃子。";
	close;
}
