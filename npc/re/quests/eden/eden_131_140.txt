S_Quest:
	if (isbegin_quest(getarg(0)))
		erasequest getarg(0);
	setquest getarg(1);
	if (getarg(2,0))
		getexp getarg(2),getarg(3);
	return;
S_Erasequest:
	.@total = getargcount();
	for ( .@i = 0; .@i < .@total; .@i += 3 ) {
		.@quest_id = getarg(.@i);
		switch( checkquest(.@quest_id,HUNTING) ) {
		case -1:
			if (checkquest(.@quest_id+1) > -1)
				erasequest (.@quest_id+1);
			continue;
		case 0:
		case 1:
			mes "[拉吉]";
			switch( getarg(.@i+1) ) {
			case 1:
				mes "關於 "+ getarg(.@i+2) +" 的其餘故事...好吧，我會問其他人。";
				break;
			case 2:
			case 3:
				mes "我會問其他冒險家關於 "+ getarg(.@i+2) +" 的故事。";
				break;
			}
			break;
		case 2:
			mes "[拉吉]";
			switch( getarg(.@i+1) ) {
			case 1:
				mes "你聽說過關於 "+ getarg(.@i+2) +" 的故事嗎？ 謝謝你。 我保證會製作一首好歌。";
				getexp 1100000,300000;
				break;
			case 2:
				mes "夢羅克的的故事夠好聽能寫成歌嗎? 你做得很好。";
				getexp 1100000,200000;
				break;
			case 3:
				mes "你的冒險怎麼樣？ 好玩嗎？ 請讓我有一天聽到你的冒險故事。";
				getexp 1100000,200000;
				break;
			}
		}
		erasequest .@quest_id;
		next;
	}
	return;
