L_Quest:
	set .@quest1, getarg(0);
	set .@quest2, .@quest1+31;
	.@cutin$ = getarg(3);
	set .@playtime, checkquest(.@quest2,PLAYTIME);
	if (.@playtime == 0 || .@playtime == 1) {
		mes "[86-90 任務佈告欄]";
		mes "您需要等待 3 小時才能再次接受此任務。";
		close;
	}
	set .@hunting, checkquest(.@quest1,HUNTING);
	if (.@hunting == 0 || .@hunting == 1) {
		mes "[86-90 任務佈告欄]";
		mes "您是否已經完成獵殺所有 30 隻 "+getarg(1)+getarg(5,"")+" 了？";
		close;
	} else if (.@hunting == 2) {
		mes "[86-90 任務佈告欄]";
		mes "您已完成狩獵。";
		mes "請接受這份獎勵作為補償。";
		erasequest .@quest1;
		if (.@playtime > -1) erasequest .@quest2;
		setquest .@quest2;
		getexp getarg(2),0;
		close;
	}/* else if (.@hunting == 3) {
//		recall_completequest .@quest1
		erasequest .@quest1;
		mes "[86-90 任務佈告欄]";

		close;
	}*/ else {

		mes getarg(4);
		next;
		if (.@cutin$ != "") cutin .@cutin$,3;
		mes "[86-90 任務佈告欄]";
		mes "去那裡獵殺 30 隻 "+getarg(1)+" 怪獸，並在這個佈告欄上發布您的成功。";
		next;
		if (.@cutin$ != "") cutin .@cutin$,255;
		if(select("我會去獵殺它們。:不用了。") == 1) {
			mes "[86-90 任務佈告欄]";
			mes "當您完成任務後，將其發布在此佈告欄上並領取您的獎勵。";
