glast_01,195,273,3	script	Hugin's follower#1	2_F_MAGICMASTER,{
	mes "[胡金的追隨者]";
	mes "我無法像胡金大人那樣在時間和維度之間移動，但我可以假裝一下。";
	next;
	mes "[胡金的追隨者]";
	mes "是什麼原因導致這個古城格拉斯頓海姆被注入如此強大的黑暗力量？";
	next;
	if( select( "我不在乎:那很有趣，請告訴我更多" ) == 2 ){
		mes "[胡金的追隨者]";
		mes "感謝胡金，我找到了進入這個被強大黑暗力量注入的古城格拉斯頓海姆的方法。";
		next;
		mes "[胡金的追隨者]";
		mes "你覺得如何？你想去那裡探險嗎？";
		next;
		if( select( "我不想探險:我真的很想探險！" ) == 2 ){
			mes "[胡金的追隨者]";
			mes "當然，但這不是免費的服務！我必須得到一些回報。呵呵...";
			next;
			mes "[胡金的追隨者]";
			mes "我可以送你去兩個地方。你希望我送你去哪裡？";
			next;
			switch( select( "取消:修道院墓地 (Lv100~130) - 5000z:第二層走廊 (Lv140以上) - 10000z" ) ){
				case 1:
					mes "[胡金的追隨者]";
					mes "好吧，等你改變主意隨時回來。我想你有很多事情想和我談談...";
					close;
				case 2:
					.@map$ = "gl_chyard_";
					.@x = 148;
					.@y = 144;
					.@zeny = 5000;
					break;
				case 3:
					.@map$ = "gl_cas02_";
					.@x = 104;
					.@y = 31;
					.@zeny = 10000;
					break;
			}
			if( Zeny < .@zeny ){
				mes "[胡金的追隨者]";
				mes "什麼？你以為這項服務是免費的嗎？";
				close;
			}
			Zeny -= .@zeny;
			warp .@map$, .@x, .@y;
			end;
		}
	}
	mes "[胡金的追隨者]";
	mes "好吧，如果你不感興趣，那就別介意。";
	close;
}
