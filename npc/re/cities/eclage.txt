			if (!countitem(6081)) {
				mes "[接待員]";
				mes "看來您的硬幣不足。您是否可以再次檢查您的帳戶呢？";
				close;
			}
			mes "[接待員]";
			mes "請隨意使用～";
			close2;
			delitem 6081,1; //6081
			percentheal 100,100;
			warp "ecl_in02",167,49;
			end;
		case 2:
			if (Zeny < 5000) {
				mes "[接待員]";
				mes "看來您沒有足夠的資源來進行。您是否介意再次檢查您的帳戶？";
				close;
			}
			mes "[接待員]";
			mes "請隨意使用～";
			close2;
			Zeny = Zeny - 5000;
			percentheal 100,100;
			warp "ecl_in02",167,49;
			end;

			close;
		}
