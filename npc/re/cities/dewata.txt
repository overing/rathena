alberta,212,202,4	script	Dewata Sailor#alberta	536,{
	mes "[德瓦塔水手]";
	mes "您想要前往 ^8B4513德瓦塔島^000000 嗎？";
	mes "這裡有耀眼的海浪和迷人的景色，是個放鬆的好地方。";
	if (VIP_SCRIPT) {
		set .@cost, vip_status(VIP_STATUS_ACTIVE)?1000:10000;
		mes "交通費用為 10,000 Zeny，VIP 為 1,000。";
	} else {
		set .@cost,10000;
		mes "交通費用為 10,000 Zeny。";
	}
	next;
	switch(select("是的！:不用了。")) {
	case 1:
		if (Zeny >= .@cost) {
			mes "[德瓦塔水手]";
			mes "^8B4513德瓦塔^000000 是一個美麗又祥和的島國。";
			mes "祝您旅途愉快～";
			close2;
			set Zeny, Zeny - .@cost;
			warp "dewata",232,53;
			end;

			mes "[德瓦塔水手]";
			mes "你這小鬼沒錢啊。";
			mes "等你錢包鼓一點再來吧。";
			close;
		}
	case 2:
		mes "[德瓦塔水手]";
		mes "讓我知道";
		mes "當你準備好要旅行的時候～";
		close;
	}
}
