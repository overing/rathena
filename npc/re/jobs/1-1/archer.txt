		if (Class == Job_Novice_High && (ADVJOB == Job_Sniper || ADVJOB == Job_Clown || ADVJOB == Job_Gypsy)) {
			mes "[弓箭手公會人員]";
			mes "嘿，我認識你。";
			mes "你之前參加過";
			mes "這個測驗，不是嗎？";
			next;
			mes "[弓箭手公會人員]";
			mes "啊，你一定是去過";
			mes "瓦爾哈拉並重生了。";
			mes "哇，真令人印象深刻！";
			next;
			if (!callfunc("F_CanChangeJob")) {
				mes "[弓箭手公會人員]";
				mes "呃...";
				mes "你最好先學會所有的基本技能，才能成為弓箭手。";
				next;
				mes "[弓箭手公會人員]";
				mes "好吧，再見。";
				close;
			}
			mes "[弓箭手公會人員]";
			mes "那麼。我不需要";
			mes "再說什麼了。";
			mes "我知道你會成為一個偉大的弓箭手...";
			next;
			skill "NV_TRICKDEAD",0,SKILL_PERM;
			jobchange Job_Archer_high;
			skill "AC_MAKINGARROW",1,SKILL_PERM;
			skill "AC_CHARGEARROW",1,SKILL_PERM;
			mes "[弓箭手公會人員]";
			mes "雖然這次沒有特別的";
			mes "獎勵給你，但我希望你能諒解。保重。";
			close;
		}else{
			mes "[弓箭手公會人員]";
			mes "喔...？";
			mes "嘿，你在";
			mes "這裡做什麼...？";
			next;
			mes "[弓箭手公會人員]";
			mes "我可以看得出來你並不適合當弓箭手。感覺你應該去做";
			mes "別的事情...";
			close;
		}
	}
	mes "[弓箭手公會人員]";
	mes "很高興認識你。有什麼我可以幫忙的嗎？";
	next;
	switch(select("我想成為弓箭手。:我需要知道成為弓箭手的條件，謝謝。:沒事，謝謝。")) {
	case 1:
		mes "[弓箭手公會人員]";
		if(BaseJob != Job_Novice) {
			if(BaseJob == Job_Archer)
				mes "哈哈，你是在開玩笑吧...";
			else {
				mes "我很抱歉，但是只有初心者可以轉職。";
				mes "你已經有自己的職業了，不是嗎？";
			}
			close;
		}else{
			mes "你是... "+strcharinfo(0)+"，對吧？";
			next;
			if (!callfunc("F_CanChangeJob")) {
				mes "[弓箭手公會人員]";
				mes "嗯，你的技能等級還不夠。";
				mes "你的職業等級必須至少 ^4A4AFF10^000000 並且你的基本技能等級必須達到 ^4A4AFF9級";

				mes "[弓箭手公會人員]";
				mes "因為弓箭手需要極高的專注力，所以我們不接受那些沒有耐心的人。";
				close;
			}
			mes "[弓箭手公會人員]";
			mes "你的基本技能現在足夠了...";
			mes "....嗯~~所以你現在準備好成為弓箭手了。我會馬上進行。";
			next;
			mes "[弓箭手公會人員]";
			mes "恭喜！你現在是一名弓箭手了！還有，我們希望你能積極參與弓箭手公會復興的各項活動。";
			mes "啊，生產部門的物品已經送到了。給你，這些都是你的！";
			callfunc "Job_Change",Job_Archer;
			getitem 1742,1;	// N_Composite_Bow
			getitem 12004,1;	// Arrow_Container
			getitem 12009,1;	// Silver_Arrow_Container
			getitem 12008,1;	// Fire_Arrow_Container
			next;
			mes "[弓箭手公會人員]";
			mes "有了弓和箭，你現在就是真正的弓箭手了。";
			mes "如果你打開箭矢容器，裡面就會有箭矢，然後你就可以裝備它們。";
			mes "嗯，我希望聽到你更好的消息。是時候說再見了。";
			mes "再見。";
			close;
		}
	case 2:
		mes "[弓箭手公會人員]";
		mes "弓箭手擁有使用弓的技能，並具有各種天賦。";
		mes "弓箭手最大的能力是從遠距離攻擊敵人。";
		next;
		mes "[弓箭手公會人員]";
		mes "雖然弓箭手的 HP 較弱，但他或她可以從遠距離射擊敵人，";
		mes "所以在實際戰鬥中弓箭手更安全。";
		next;
		mes "[弓箭手公會人員]";
		mes "雖然仙境傳說中的弓箭手 HP 較低，但他或她具有很高的準確性和攻擊速度，因此弓箭手可以在怪物靠近之前殺死它們。";
		next;
		mes "[弓箭手公會人員]";
		mes "^8C2121弓箭手可以轉職成獵人。^000000";
		mes "^8C2121除了獵人之外，如果你是男性，你可以轉職成吟遊詩人，如果你是女性，你可以轉職成舞孃。^000000";
		close;
	case 3:
		mes "[弓箭手公會人員]";
		mes "如果您有任何問題，請隨時來問我。";
		close;
	}
}
