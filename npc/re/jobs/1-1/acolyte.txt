			next;
			mes "[馬勒西斯神父]";
			mes "永遠記得感謝上帝，祂一直照顧著我們。 在混亂和困難的時刻，用堅定的信念面對你的苦難。";
			callfunc "Job_Change",Job_Acolyte;
			getitem 1545,1;	// N_Mace
			next;
			mes "[馬勒西斯神父]";
			mes "最後，我想真誠地祝賀你堅持完成了你的懺悔試煉。";
			close;
		}
	case 2:
		mes "[馬勒西斯神父]";
		mes "你希望成為一名初心者嗎？";
		mes "那麼，你必須滿足以下要求，將它們視為上帝給予的練習。";
		next;
		mes "[馬勒西斯神父]";
		mes "首先，你必須至少達到 Novice Job Level 9 並且學習所有基礎技能。";
		mes "這是最基本的事情，所以你需要將其視為訓練自己的方式。";
		next;
		mes "[馬勒西斯神父]";
		mes "當你認為你已經滿足了這個要求時，請再次回到我身邊。 然後你將會得到一個神聖的職業，你可以藉此傳播上帝的旨意。";
		close;
	case 3:
		close;
	}
}
