ayothaya,203,169,4	script	Ayothaya Guide Noi#01ay	839,{
	if (strnpcinfo(0) == "Ayothaya Guide Noi#01ay")
		set .@n$, "[Noi]";
	else
		set .@n$, "[Noa]";
	mes .@n$;
	mes "哈囉~";
	mes "歡迎來到 ^8B4513阿瑜陀耶^000000。";
	mes "這個城鎮透過野外的森林與河流相連。";
	mes "這是一個非常平靜的國家。";
	mes "慢慢地四處看看吧。";
	F_Navi(.@n$);
	while (1) {
		next;
		switch(select("[ 武器商店 ]:[ 工具商店 ]:[ 酒館 ]:[ 神壇 ]:[ 釣魚點 ]:[ 艾巴克辛 ]:從小地圖上移除標記:取消")) {
		case 1:
			mes .@n$;
			mes "阿瑜陀耶的 "+F_Navi("Weapon Shop","ayothaya,163,90","^0000FF");
			mes "有特殊的武器。";
			mes "你可以在那裡找到很棒的武器。";
			mes "還有什麼我可以為你做的嗎？";
			viewpoint 1,163,90,0,0x0A82FF;
			break;
		case 2:
			mes .@n$;
			mes "我們的市民在 "+F_Navi("Tool Shop","ayothaya,131,86","^006400")+" 購買許多產品。";
			mes "確保你在那裡得到你所需要的一切。";
			mes "你需要別的什麼嗎？";
			viewpoint 1,131,86,1,0xAAFF00;
			break;
		case 3:
			mes .@n$;
			mes "冒險的一部分是收集資訊。";
			mes ""+F_Navi("酒館","ayothaya,229,72","^008080")+" 擠滿了許多旅行者，你可以在那裡獲得有用的資訊。";
			mes "還有什麼我可以為你做的嗎？";
			viewpoint 1,229,72,2,0x008080;
			break;
		case 4:
			mes .@n$;
			mes "為了平靜你的心，去 "+F_Navi("Shrine","ayothaya,208,280","^FF1493")+" 祈禱吧。";
			mes "還有什麼我可以為你做的嗎？";
			viewpoint 1,208,280,3,0xFF1493;
			break;
		case 5:
			mes .@n$;
			mes "阿瑜陀耶靠近海灘，所以大多數人都喜歡釣魚。";
			mes "他們真的很喜歡釣魚。";
			mes "去 "+F_Navi("fishing","ayothaya,255,99","^8B4513")+" 釣魚，試著抓一條大魚。";
			mes "還有什麼我可以為你做的嗎？";
			viewpoint 1,255,99,4,0x8B4513;
			break;
		case 6:
			mes .@n$;
			mes "如果你想回家";
			mes "找到 "+F_Navi("Aibakthing","ayothaya,151,68","^00BFFF");
			mes "並請他送你回去。";
			mes "還有什麼我可以為你做的嗎？";
			viewpoint 1,151,68,5,0x00BFFF;
			break;
		case 7:
			mes .@n$;
			mes "我會從小地圖上移除所有標記。";
			mes "你需要別的什麼嗎？？";
			viewpoint 2,1,1,0,0xFFFFFF;
			viewpoint 2,1,1,1,0xFFFFFF;
			viewpoint 2,1,1,2,0xFFFFFF;
			viewpoint 2,1,1,3,0xFFFFFF;
			viewpoint 2,1,1,4,0xFFFFFF;
			viewpoint 2,1,1,5,0xFFFFFF;
			break;
		case 8:
			mes .@n$;
			mes "旅途愉快~!";
			close;
		}
	}
}
