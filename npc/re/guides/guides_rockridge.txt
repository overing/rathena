harboro1,356,211,5	script	Guide#rockridge01	4_F_ANYA,{
	mes "[警長助理]";
	mes "歡迎來到岩石嶺，西部炎熱乾燥的沙漠，冒險正在等著您！";
	mes "我是警長的助理，我來這裡引導遊客。您想去哪裡嗎？";
	next;
	mes "[警長助理]";
	mes "告訴我想去的地方，我會告訴你它在哪裡。";
	mes "此外，您可以通過^B9062F點擊我向您展示的地點名稱^000000來使用^B9062F導航^000000模式。";
	while(true) {
		next;
		.@s = select( "[卡普拉職員]", "[旅館]", "[武器/盔甲商店]", "[工具商店]", "[警長辦公室]", "清除小地圖上的標記。", "取消。" );
		switch(.@s) {
		case 1:
			mes "[警長助理]";
			mes "好的，是^006400卡普拉職員^000000。請點擊您小地圖上的^0000FF方塊圖示^000000。";
			viewpoint 1,291,207,1,0xFFFF33;
			break;
		case 2:
			mes "[警長助理]";
			mes "我已經在地圖上標記了<NAVI>旅館<INFO>harboro1,156,215,0,000,0</INFO></NAVI>的位置。";
			mes "點擊名稱以啟動導航模式。";
			viewpoint 1,156,215,2,0xAAFF00;
			break;
		case 3:
			mes "[警長助理]";
			mes "我已經在地圖上標記了<NAVI>武器/盔甲商店<INFO>harboro1,312,193,0,000,0</INFO></NAVI>的位置。";
			mes "點擊名稱以啟動導航模式。";
			viewpoint 1,312,193,3,0x008080;
			break;
		case 4:
			mes "[警長助理]";
			mes "我已經在地圖上標記了<NAVI>工具商店<INFO>harboro1,312,193,0,000,0</INFO></NAVI>的位置。";
			mes "點擊名稱以啟動導航模式。";
			viewpoint 1,312,193,4,0xFF1493;
			break;
		case 5:
			mes "[警長助理]";
			mes "我已經在地圖上標記了<NAVI>警長辦公室<INFO>harboro1,241,218,0,000,0</INFO></NAVI>的位置。";
			mes "點擊名稱以啟動導航模式。";
			viewpoint 1,241,218,5,0xFFCCFF;
			break;
		case 6:
			mes "[警長助理]";
			mes "讓我移除您小地圖上的所有標記。";
			viewpoint 2,291,207,1,0xFFFF33;
			viewpoint 2,156,215,2,0xAAFF00;
			viewpoint 2,312,193,3,0x008080;
			viewpoint 2,312,193,4,0xFF1493;
			viewpoint 2,241,218,5,0xFFCCFF;
			break;
		case 7:
			mes "[警長助理]";
			mes "祝您在岩石嶺玩得開心！";
			close;
		}
		if (.@s == 7)
			mes "您還需要什麼嗎？";
		else
			mes "您還想去其他地方看看嗎？";
	}
}
