hugel,178,142,0	script	toairplane#hugel	45,1,1,{
	end;
OnTouch:
	mes "要使用飛空艇，您需要支付 1,200 zeny 或一張免費飛空艇票。";
	mes "您想要使用這項服務嗎？";
	next;
	if (select("是:否") == 1) {
		if (countitem(7311) > 0) {
			delitem 7311,1; //Free_Flying_Ship_Ticket
			warp "airplane",244,58;
			end;
		}
		if (Zeny >= 1200) {
			set Zeny, Zeny - 1200;
			warp "airplane",244,58;
			end;
		}
		mes "很抱歉，您沒有足夠的錢。";
		mes "請記住，您需要支付 1,200 zeny 才能使用這項服務。";
		close;
	}
	mes "謝謝，歡迎再次光臨。";
	close;
}
