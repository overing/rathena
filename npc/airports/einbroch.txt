airport,143,43,5	script	機場人員#airport1a::airport1	90,{
	mes "[機場人員]";
	mes "歡迎來到";
	mes "音博洛克機場，";
	mes "我們提供直達";
	mes "朱諾、里希塔樂鎮和毀葛";
	mes "等城市的航班。";
	next;
	if (select("搭乘飛空艇:取消") == 1) {
		mes "[機場人員]";
		mes "飛空艇的搭乘費用";
		mes "是 1,200 Zeny，但如果您有";
		mes "飛空艇免費搭乘券，";
		mes "則可免除費用。您要";
		mes "搭乘飛空艇嗎？";
		next;
		if (select("是:否") == 1) {
			if (countitem(7311) > 0) {
				delitem 7311,1; //Free_Flying_Ship_Ticket
				warp "airport",148,51;
				end;
			}
			if (Zeny >= 1200) {
				set Zeny, Zeny - 1200;
				warp "airport",148,51;
				end;
			}
			mes "[機場人員]";
			mes "很抱歉，您沒有";
			mes "飛空艇免費搭乘券";
			mes "也沒有足夠的Zeny";
			mes "搭乘飛空艇。請記住，";
			mes "搭乘費用是 1,200 Zeny。";
			close;
		}
	}
	mes "[機場人員]";
	mes "謝謝您，";
	mes "祝您有個愉快的一天。";
	close;
}
