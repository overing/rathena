y_airport,143,43,5	script	Airport Staff#y_air1a::y_airport1	90,{
	mes "[機場人員]";
	mes "歡迎來到朱諾機場，我們提供國內航班前往艾音布羅克、里希塔樂鎮和毀葛，";
	mes "以及國際航班前往艾音布洛克和拉赫爾。";
	mes "請問有什麼能為您服務的嗎？";
	next;
	if (select("搭乘飛空艇。:取消。") == 1) {
		mes "[機場人員]";
		mes "所有航班的登機費用";
		mes "都是 1,200 zeny。如果您";
		mes "使用飛空艇免費搭乘券，";
		mes "則免除登機費用。那麼您想出發嗎？";
		next;
		if (select("是:否") == 1) {
			if (countitem(7311) > 0) {
				delitem 7311,1; //Free_Flying_Ship_Ticket
				warp "y_airport",148,51;
				end;
			}
			if (Zeny >= 1200) {
				set Zeny, Zeny - 1200;
				warp "y_airport",148,51;
				end;
			}
			mes "[機場人員]";
			mes "很抱歉，您沒有";
			mes "1,200 zeny 可以支付";
			mes "登機費用。";
			close;
		}
	}
	mes "[機場人員]";
	mes "謝謝您，";
	mes "祝您有美好的一天。";
	close;
}
