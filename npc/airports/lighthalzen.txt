lhz_airport,143,43,5	script	Airport Staff#lhz_air1a::lhz_airport1	90,{
	mes "[機場人員]";
	mes "歡迎來到";
	mes "里希塔樂鎮機場，";
	mes "我們提供直飛";
	mes "艾音布羅克、朱諾和毀葛的航班。";
	next;
	if (select("搭乘飛空艇.:取消。") == 1) {
		mes "[機場人員]";
		mes "搭乘費用是";
		mes "1,200 Zeny，但你可以";
		mes "兌換一張飛空艇免費搭乘券來";
		mes "免除費用。";
		next;
		if (select("是:否") == 1) {
			if (countitem(7311) > 0) {
				delitem 7311,1; //Free_Flying_Ship_Ticket
				warp "lhz_airport",148,51;
				end;
			}
			if (Zeny >= 1200) {
				set Zeny, Zeny - 1200;
				warp "lhz_airport",148,51;
				end;
			}
			mes "[飛空艇人員]";
			mes "很抱歉，你沒有";
			mes "1,200 Zeny來支付";
			mes "搭乘費用。";
			close;
		}
	}
	mes "[機場人員]";
	mes "謝謝光臨";
	mes "歡迎再次光臨。";
	mes "祝您有個美好的一天~";
	close;
}
