		SF_AcceptBride();
		break;

		close;

function SF_RetrieveRingM {
	mes "["+@name$+"]";
	if ($@wed_ring && countitem($@wed_ring) < 1) {
		mes "你的 "+getitemname($@wed_ring)+" 怎麼了？ 你沒弄丟吧？ 我們需要它才能繼續進行儀式！";
		close;
	}
	if ($wed_bride_sex)
		set @item, 2634;	//Groom's wedding ring
	else
		set @item, 2635;	//Bride's wedding ring
	if (getnameditem(@item,$wed_groom$) == 0) {
		mes "你似乎沒有足夠的空間來攜帶戒指... 去清出一些空間，然後回來重新領取你伴侶的戒指。";
		close;
	}
	mes "這是給妳新娘的結婚戒指。";
	if ($@wed_ring) delitem $@wed_ring,1;
	set $wed_groom_progress,4;
	
	if ($wed_bride_progress == 4)
		SF_StartCeremony();
	else {
		mes "一旦妳的新娘取回戒指，典禮就會開始。";
		close;
	}
}

function SF_StartCeremony {
	mes "我現在將開始結婚典禮，你們將被宣告為夫妻。";
	set $wed_bride_progress,5;
	set $wed_groom_progress,5;
	set $@msg$,$wed_groom$;
	if (strcharinfo(0) == $wed_groom$)
		set $@msg$,$wed_bride$;
	if (marriage($@msg$) == 0) {
		next;
		mes "["+@name$+"]";
		mes ""+$@msg$+" 在哪裡?? 如果少了一位，我就無法為你們證婚...";
		close;
	}
	set $wed_bride_progress,6;
	set $wed_groom_progress,6;
	initnpctimer;
	close;
}

//Subfunction: Checks that the groom/bride is still wearing their stuff.
function SF_equip_check {
	if (Sex == SEX_MALE && getequipid(EQI_ARMOR) != 7170) {
		mes "["+@name$+"]";
		mes "孩子，你把你的 "+getitemname(7170)+" 拿去做什麼了？";
		emotion ET_THINK;
		return 0;
	}
	if (Sex == SEX_FEMALE && getequipid(EQI_ARMOR) != 2338) {
		mes "["+@name$+"]";
		mes "孩子，你在典禮期間應該一直戴著 "+getitemname(2338)+" ...";
		emotion ET_THINK;
		return 0;
	}
	if (Sex == SEX_FEMALE && $@wed_veil && getequipid(EQI_HEAD_TOP) != 2206) {
		mes "["+@name$+"]";
		mes "孩子，你還不能脫下你的 "+getitemname(2206)+" ....";
		emotion ET_THINK;
		return 0;
	}
	return 1;
}

function SF_Register {
	if ($@wed_allow) { //Role select
		mes "["+@name$+"]";
		mes "很好，你想登記為？";
		next;
		set @submenu, select("新郎","新娘","取消");
	} else if (Sex == SEX_MALE) { //Groom
		mes "["+@name$+"]";
		mes "很好，你要登記為新郎嗎？";
		next;
		if (select("是","我改變主意了。")==1)
			set @submenu, 1;
		else
			set @submenu, 3;
	} else { //Bride
		mes "["+@name$+"]";
		mes "很好，你要登記為新娘嗎？";
		next;
		if (select("是","我改變主意了。")==1)
			set @submenu, 2;
		else
			set @submenu, 3;
	}
	switch (@submenu) {
	case 1: //Groom
		SF_TryRegister(0);
		set $wed_progress,1;
		mes "["+@name$+"]";
		mes "你現在登記為新郎。";
		mes "告訴你的新娘盡快登記。";
		emotion ET_SCRATCH;
		initnpctimer;
		close;
	case 2: //Bride
		SF_TryRegister(1);
		set $wed_progress,1;
		mes "["+@name$+"]";
		mes "你現在登記為新娘。";
		mes "告訴你的新郎盡快登記。";
		emotion ET_SCRATCH;
		initnpctimer;
		close;
	default: //Cancel
		mes "["+@name$+"]";
		mes "準備好後再來。";
		close;
	}
}

//Subfunction SF_TryRegister (int bride)
function SF_TryRegister {
	set @bride, getarg(0);
	set @type$, "groom";
	if (@bride)
		set @type$, "bride";
		
	mes "["+@name$+"]";
	mes "在登記為 "+@type$+" 之前，讓我檢查您是否符合所有要求...";
	next;
	if (Upper == 2) {
		mes "["+@name$+"]";
		mes "哎呀，你太年輕了，不應該考慮結婚！";
		emotion ET_SURPRISE;
		close;
	}
	if (Sex == SEX_MALE)
		set @item, 7170;
	else
		set @item, 2338;
	
	if (getequipid(EQI_ARMOR) != @item) {
		mes "["+@name$+"]";
		mes "如果你想結婚，你應該穿著 "+getitemname(@item)+"。";
		close;
	}
	if (Sex == SEX_FEMALE && $@wed_veil && getequipid(EQI_HEAD_TOP) != 2206) {
		mes "["+@name$+"]";
		mes "你的 "+getitemname(2206)+" 在哪裡？這是你禮服的必要配件。";
		close;
	}
	if ($@wed_ring && countitem($@wed_ring) < 1) {
		mes "["+@name$+"]";
		mes "戒指在哪裡？你需要一個 "+getitemname($@wed_ring)+" 來交換戒指，親愛的。";
		close;
	}
	if (@bride)
		set @cost, $@wed_bride_reg;
	else 
		set @cost, $@wed_groom_reg;

	if (Zeny < @cost) {
		mes "["+@name$+"]";
		mes "我很抱歉，但你沒有足夠的錢來支付登記費。";
		mes "請在收集到 "+@cost+"z 後再來。";
		close;
	}
	set Zeny,Zeny-@cost;
	sc_start SC_Wedding,3600000,1; //Start Wedding Effect (SC_WEDDING)
	if (@bride) {
		set $wed_bride_progress,1;
		set $wed_bride$,strcharinfo(0);
		set $wed_bride_sex, Sex;
		if ($@wedding_effect == 1) //Store account id for effect.
			set $wedding_effect_id, getcharid(3);
	} else {
		set $wed_groom_progress,1;
		set $wed_groom$,strcharinfo(0);
		set $wed_groom_sex, Sex;
		if ($@wedding_effect == 2) //Store account id for effect.
			set $wedding_effect_id, getcharid(3);
	}
}
