//===== Description: =========================================
//= Black Jack card game.  Gameplay based off standard casino
//= black jack rules.  Dealer must have at least 17 to stay and will
//= automatically stay at 17 and up.  Player must have at least
//= 13 to stay.  Aces counted as 11 or 1.  Option to "Double Down".
//= Currently does not allow for "insurance", or "splitting"
//= of pairs.
//===== Additional Comments: =================================
//= 1.0 First version.
//============================================================

cmd_in02,188,89,1	script	Black Jack Dealer	57,{
	mes "[莊家]";
	mes "哈囉！想玩一局黑傑克嗎？";
M_Menu:
	next;
	menu "What are the rules?",M_0, "I want to play.",L_Play, "No thanks.",M_End;

M_0:
	mes "[莊家]";
	mes "想知道些什麼呢？";
sM_Menu0:
	next;
	menu "The basics.",sM_0a, "Winning and loosing.",sM_0b, "'Double Down'.",sM_0c,
	"Ace value.",sM_0d, "Nothing.",M_Menu;

sM_0a:
	mes "[莊家]";
	mes "黑傑克是一種卡牌遊戲，目標是取得 21 點，或盡可能";
	mes "接近 21 點，而不要超過 21 點。";
	next;
	mes "[莊家]";
	mes "在遊戲開始之前，玩家必須下注。一旦下注";
	mes "之後，莊家和玩家都會拿到 2 張牌";
	mes "每個人。根據你擁有的牌，你可以選擇^5533FF'停牌'^000000，";
	mes "或者你可以選擇^5533FF'要牌'^000000。";
	next;
	mes "[莊家]";
	mes "當你選擇'停牌'時，你是在告訴莊家你不需要";
	mes "更多牌了。這允許莊家在他/她";
	mes "想要的時候要牌。為了停牌，你必須^FF3355至少有 13 點^000000。";
	mes "莊家只有在擁有^FF3355 17 點或以上^000000時才能停牌。";
	next;
	mes "[莊家]";
	mes "當你選擇'要牌'時，你是在告訴莊家你想要";
	mes "另一張牌。通過要更多牌，你可以增加你的點數";
	mes "總計。只要你^FF3355少於 21 點^000000，你就可以從";
	mes "牌堆中要一張牌。";
	goto sM_Menu0;
sM_0b:
	mes "[莊家]";
	mes "在黑傑克中有 3 種贏和輸的方式。";
	next;
	mes "[莊家]";
	mes "1.) 在黑傑克回合結束時，如果你的點數比";
	mes "莊家多，你將贏得該回合。相反，如果你比";
	mes "莊家少，你就會輸。";
	next;
	mes "[莊家]";
	mes "2.) 如果你拿到一張牌使你的點數總和超過 21，你";
	mes "將自動輸掉該回合。這被稱為^5533FF'爆牌'^000000。如果";
	mes "莊家爆牌，那麼你將贏得該回合。";
	next;
	mes "[莊家]";
	mes "3.) 如果你用前 2 張牌的點數總和為 21 點，你";
	mes "將自動贏得該回合。這被稱為^5533FF'黑傑克'^000000";
	mes "並且發生在你拿到一張'A'和一張'10 點牌'時。如果";
	mes "莊家拿到黑傑克，他/她將自動贏得該回合。";
	next;
	mes "[莊家]";
	mes "4.) 除了贏和輸之外，你還可以與莊家打平。如果";
	mes "你和莊家在回合結束時的點數總和相同，";
	mes "這將導致平局，沒有贏家，也沒有金錢的損失或獲得。";
	next;
	mes "[莊家]";
	mes "這被稱為與莊家^5533FF'平手'^000000。這也";
	mes "適用於你和莊家同時擁有黑傑克。";
	goto sM_Menu0;
sM_0c:
	mes "[莊家]";
	mes "'加倍下注'選項允許你將當前賭注加倍，";
	mes "但缺點是你只能再要一張";
	mes "額外的牌。此選項僅在";
	mes "每回合開始時可用。";
	next;
	mes "[莊家]";
	mes "加倍下注何時有用的示例是，當";
	mes "你的前 2 張牌給你總共 11 點。你有一個很好的";
	mes "機會用你抽到的下一張牌獲得 21 點或 20 點。這";
	mes "將是一手適合加倍下注的好牌。";
	goto sM_Menu0;
sM_0d:
	mes "[莊家]";
	mes "'A'牌是黑傑克遊戲中一張獨特的牌，因為";
	mes "它可以有 2 個值。A 可以計為 11 點，";
	mes "或者僅僅 1 點。例如，如果你有一張 A 和一張 4，那將";
	mes "給你 15 點或 5 點。";
	next;
	mes "[莊家]";
	mes "如果你決定停牌，A 將自動計為";
	mes "11 點，使你總共得到 15 點。";
	next;
	mes "[莊家]";
	mes "如果你決定要牌並收到一張 9，那麼 A 將";
	mes "自動計為 1 點，使你總共得到 14";
	mes "點。如果 A 被計為 11 點，那麼你的";
	mes "點數總和將超過 21 點，並且會輸掉。";
	next;
	mes "[莊家]";
	mes "正是因為你擁有'A'的靈活性，才使";
	mes "它成為遊戲中最?大的牌。";
	goto sM_Menu0;
M_End:
	mes "[莊家]";
	mes "隨時歡迎回來";
	close;

//================
L_Play:
	mes "[莊家]";
	mes "請下注...";
	next;
	menu "2z",M_1a, "10z",M_1b, "20z",M_1c, "100z",M_1d, "Too rich for my blood....",M_End;

M_1a:
	if(Zeny < 2) goto sL_NotEnuf;
	set @bet, 2;
	goto L_Cont0;
M_1b:
	if(Zeny < 10) goto sL_NotEnuf;
	set @bet, 10;
	goto L_Cont0;
M_1c:
	if(Zeny < 20) goto sL_NotEnuf;
	set @bet, 20;
	goto L_Cont0;
M_1d:
	if(Zeny < 100) goto sL_NotEnuf;
	set @bet, 100;
	goto L_Cont0;

sL_NotEnuf:
	mes "[莊家]";
	mes "抱歉，你沒有足夠的 Zeny 來下注。";
	close;

L_Cont0:
	mes "(牌正在被發出)";
	next;
	deletearray $@card[0],13;
	set @dealerTurn, 0;
	set @numP, 0;
	set @numD, 0;
	set @pAce, 0;
	set @dAce, 0;
	callsub sF_GetCards, @numP, @playCard[@numP], @playCard$[@numP], @pAce;
	callsub sF_GetCards, @numP, @playCard[@numP], @playCard$[@numP], @pAce;
	callsub sF_GetCards, @numD, @dealCard[@numD], @dealCard$[@numD], @dAce;
	callsub sF_GetCards, @numD, @dealCard[@numD], @dealCard$[@numD], @dAce;

//==============
L_Start:
	callsub sF_GetTot;

	mes "- 這是 ^FF5533莊家^000000 的牌：";
	if (@numD==2) callsub sF_D2cards;
	if (@numD==3) callsub sF_D3cards;
	if (@numD==4) callsub sF_D4cards;
	if (@numD==5) callsub sF_D5cards;
	mes "    莊家有：^FF5533"+@dealTot+"^000000";
	mes " ";
	mes "- 這是 ^5533FF你^000000 的牌：";
	if (@numP==2) callsub sF_P2cards;
	if (@numP==3) callsub sF_P3cards;
	if (@numP==4) callsub sF_P4cards;
	if (@numP==5) callsub sF_P5cards;
	if(@pAce != 1 || @playTot == 21) mes "    你有：^5533FF" +@playTot+ "^000000";
	if(@pAce == 1 && @playTot != 21) mes "    你有：^5533FF" +@playTot+ "^000000, 或 ^5533FF" +(@playTot-10)+ "^000000";
	next;
	if(@playTot==21 && @dealTot==21) goto sL_Push;
	if(@numP==2 && @playTot == 21) goto sL_Win;
	if(@numD==2 && @dealTot == 21) goto sL_Lose;
	if(@playTot > 21) goto sL_Lose;
	if(@dealTot > 21) goto sL_Win;
	if(@numP==2 && @dealerTurn==0) menu "Hit me(pull)",M_Hit, "Stay",M_Stay, "Double Down",M_Double;
	if(@dealerTurn == 0) menu "Hit me(pull)",M_Hit, "Stay",M_Stay;

M_Stay:
	mes "[莊家]";
	if(@playTot < 13) goto sL_PlayToLow;
	if(@dealTot > 16) mes "莊家停牌。";
	if(@dealTot > 16 || @numD == 5) goto L_Check;
	mes "莊家將要牌";
	next;
	callsub sF_GetCards, @numD, @dealCard[@numD], @dealCard$[@numD], @dAce;
	set @dealerTurn, 1;
	goto L_Start;

sL_PlayToLow:
	mes "抱歉，你的總點數不夠高，無法停牌。你必須要牌。";
	next;
	goto M_Hit;

M_Hit:
	if(@numP == 5) goto M_Stay;
	callsub sF_GetCards, @numP, @playCard[@numP], @playCard$[@numP], @pAce;
	goto L_Start;
M_Double:
	mes "[莊家]";
	mes "玩家選擇加倍下注。你目前的賭注將會";
	mes "加倍，並且你只能再要 1 張牌。";
	next;
	set @dealerTurn, 1;
	set @bet, @bet*2;
	callsub sF_GetCards, @numP, @playCard[@numP], @playCard$[@numP], @pAce;
	goto L_Start;

//=============
L_Check:
	next;
	if(@playTot < @dealTot) goto sL_Lose;
	if(@playTot == @dealTot) goto sL_Push;

sL_Win:
	mes "[莊家]";
	mes "恭喜，你贏了！";
	next;
	set Zeny, Zeny + @bet;
	goto L_Play;
sL_Lose:
	mes "[莊家]";
	mes "抱歉，你輸了。";
	set Zeny, Zeny - @bet;
	next;
	goto L_Play;
sL_Push:
	mes "[莊家]";
	mes "平手。你與莊家打成平手。";
	next;
	goto L_Play;


//==================================
	// Sub function for dealing/pulling the cards
sF_GetCards:
	set @rnd, rand(1,13);
	if($@card[@rnd] == 4) goto sF_GetCards;
	set $@card[@rnd], $@card[@rnd] + 1;
	set getarg(1), @rnd;
	if(getarg(1) > 10) set getarg(1), 10;
	if(getarg(1) == 1 && getarg(3) < 1) set getarg(1), 11;
	if(getarg(1) == 11) set getarg(3), 1;
	set getarg(2), " " + getarg(1) + " ";
	if(@rnd == 10) set getarg(2), getarg(1);
	if(@rnd == 1) set getarg(2), " A ";
	if(@rnd == 11) set getarg(2), " J ";
	if(@rnd == 12) set getarg(2), " Q ";
	if(@rnd == 13) set getarg(2), " K ";
	set getarg(0), getarg(0) + 1;
	return;

//==================================
	// Sub function for getting the total score for each hand
sF_GetTot:
	set @i, 0;
	set @dealTot, 0;
	set @playTot, 0;

GetDealTot:
	set @dealTot, @dealTot + @dealCard[@i];
	set @i, @i + 1;
	if(@i < @numD) goto GetDealTot;
	if(@dAce == 1 && @dealTot > 21) set @dealTot, @dealTot - 10;
	set @i, 0;
GetPlayTot:
	set @playTot, @playTot + @playCard[@i];
	set @i, @i + 1;
	if(@i < @numP) goto GetPlayTot;
	if(@pAce == 1 && @playTot > 21) set @pAce, 2;
	if(@pAce > 0 && @playTot > 21) set @playTot, @playTot - 10;
	return;

//=======================================
	// Sub function for displaying the Cards
sF_D2cards:
	mes "    .-----.   .-----. ";
	mes "    | "+@dealCard$[0]+" |   | "+@dealCard$[1]+" | ";
	mes "    '-----'   '-----' ";
	return;
sF_P2cards:
	mes "    .-----.   .-----. ";
	mes "    | "+@playCard$[0]+" |   | "+@playCard$[1]+" | ";
	mes "    '-----'   '-----' ";
	return;
sF_D3cards:
	mes "    .-----.   .-----.   .-----. ";
	mes "    | "+@dealCard$[0]+" |   | "+@dealCard$[1]+" |   | "+@dealCard$[2]+" | ";
	mes "    '-----'   '-----'    '-----' ";
	return;
sF_P3cards:
	mes "    .-----.   .-----.   .-----. ";
	mes "    | "+@playCard$[0]+" |   | "+@playCard$[1]+" |   | "+@playCard$[2]+" | ";
	mes "    '-----'   '-----'    '-----' ";
	return;
sF_D4cards:
	mes " .-----.  .-----.  .-----. ";
	mes " | "+@dealCard$[0]+" |  | "+@dealCard$[1]+" |  | "+@dealCard$[2]+" | ";
	mes " '-----'  '-----'   '-----' ";
	mes "    .-----.  ";
	mes "    | "+@dealCard$[3]+" |";
	mes "    '-----'   ";
	return;
sF_P4cards:
	mes " .-----.  .-----.  .-----. ";
	mes " | "+@playCard$[0]+" |  | "+@playCard$[1]+" |  | "+@playCard$[2]+" |";
	mes " '-----'  '-----'   '-----' ";
	mes "    .-----.  ";
	mes "    | "+@playCard$[3]+" |";
	mes "    '-----'   ";
	return;
sF_D5cards:
	mes " .-----.  .-----.  .-----. ";
	mes " | "+@dealCard$[0]+" |  | "+@dealCard$[1]+" |  | "+@dealCard$[2]+" | ";
	mes " '-----'  '-----'   '-----' ";
	mes "    .-----.   .-----. ";
	mes "    | "+@dealCard$[3]+" |   | "+@dealCard$[4]+" | ";
	mes "    '-----'   '-----' ";
	return;
sF_P5cards:
	mes " .-----.  .-----.  .-----. ";
	mes " | "+@playCard$[0]+" |  | "+@playCard$[1]+" |  | "+@playCard$[2]+" | ";
	mes " '-----'  '-----'   '-----' ";
	mes "    .-----.   .-----. ";
	mes "    | "+@playCard$[3]+" |   | "+@playCard$[4]+" | ";
	mes "    '-----'   '-----' ";
	return;
}
