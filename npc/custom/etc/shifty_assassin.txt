//= Players buy ninjas to assassinate other players
//===== Additional Comments: =================================

//= 1.1.2 Updated WoE Check. [Paradox924X]

morocc,148,86,5	script	Shifty Assassin	725,{
	set .@ninja_price,250000;

	// STARTS THE MENU //
M_Start:
	mes "[狡猾的刺客]";
	mes "你想要什麼？";
	next;
	menu "Buy Ninjas",M_Buy,"Assassinate somebody",M_Kill,"Check your Ninjas",M_Check, ( getgmlevel() > 90 ? "Add Ninjas" : "" ),-,"Cancel",M_Exit;

	// GM MENU TO ADD NINJAS //
	mes "[狡猾的刺客]";
	mes "你想要提供多少忍者？";
	next;
	input .@add;
	set $ninja_avail, $ninja_avail+.@add;
	mes .@add +" 位忍者已新增。";
	close;

	// BUY NINJAS //
M_Buy:
	mes "[狡猾的刺客]";
	mes "你想購買多少忍者？";
	mes "目前有 ^0000FF" + $ninja_avail + "^000000 位忍者可供購買。";
	mes "每位價格 ^0000FF" + .@ninja_price + " zeny^000000。";

	input .@buy;
	next;
	if ($ninja_avail < 1) goto NoNinjas;
	if ($ninja_avail < .@buy) goto NotEnoughNinjas;
	set .@price, .@buy*.@ninja_price;
	if (Zeny < .@price) goto NoZeny;

	mes "[狡猾的刺客]";
	mes "總共需要 ^0000FF"+ .@price +" zeny^000000。";
	next;
	menu "Continue",-,"Cancel",M_Exit;

	set Zeny, Zeny - .@price;
	set #ninjas, #ninjas + .@buy;
	set $ninja_avail, $ninja_avail - .@buy;

	mes "[狡猾的刺客]";
	mes "謝謝你。";
	close;

	// ASSASSINATE SOMEBODY //
M_Kill:
	if (agitcheck()) goto M_Busy;
	mes "[狡猾的刺客]";
	mes "輸入目標的名稱。";
	mes "^FF0000請完全依照名稱輸入，否則我無法找到受害者。^000000";
	next;
	menu "Continue",-,"Cancel",M_Exit;
	input .@name$;
	if ( !getcharid( 3,.@name$ ) ) {
		mes "[狡猾的刺客]";
		mes .@name$ +" 並未上線。";
		close;
	}
	next;
	mes "[狡猾的刺客]";
	mes "活躍的忍者："+#ninjas;
	mes "休息中的忍者："+#ninjasr;
	mes "你想要派遣多少位？";
	input .@number;
	if (.@number < 1) goto NoNinjasSent;
	if (.@number > #ninjas) goto NotEnoughNinjas1;
	if (.@number > 10) goto TooManyNinjas;
	set .@chance, rand(1,12);
	set #ninjas,#ninjas-.@number;
	set #ninjas,#ninjas+#ninjasr;
	set #ninjasr,0;
	if (.@number < .@chance) goto M_Failure;

	// SUCCESSFUL ATTACK //
	mes "現在派遣忍者。";
	next;
	mes "[狡猾的刺客]";
	set .@ninjasurvived, rand(1,.@number);
	set #ninjasr,.@number-.@ninjasurvived;
	mes "你的攻擊成功了，但只有 ^FF0000" + #ninjasr + "^000000 位忍者存活下來。";
	set .@acc_id, getcharid( 3,.@name$ );
	if ( .@acc_id )
		unitkill .@acc_id;
	announce .@name$ +" has been assassinated by " + strcharinfo(0) +"'s Ninjas.",bc_npc;
	close;

	// FAILED ATTACK //
M_Failure:
	mes "現在派遣忍者。";
	next;
	mes "[狡猾的刺客]";
	set .@ninjasurvived, rand(1,.@number);
	set #ninjasr, .@number-.@ninjasurvived;
	mes "你的攻擊失敗了，只有 ^FF0000" + #ninjasr + "^000000 位忍者存活下來。";

	announce .@name$+" has survived " + strcharinfo(0) +"'s Ninja attack.",8;
	close;

	// NINJAS BUSY FOR WOE //
M_Busy:
	mes "[狡猾的刺客]";
	mes "抱歉，我所有的忍者都在忙著進行攻城戰。";
	close;

	// CHECK YOUR NINJAS //
M_Check:
	mes "[狡猾的刺客]";
	mes "你擁有：";
	mes "^FF0000" + #ninjas + "^000000 位活躍的忍者。";
	mes "^0000FF" + #ninjasr + "^000000 位休息中的忍者。";
	next;
	goto M_Start;


	// LIMIT //
NoNinjasSent:
	mes "[狡猾的刺客]";
	mes "沒有忍者的話，你無法殺任何人。";
	next;
	goto M_Start;

TooManyNinjas:
	mes "[狡猾的刺客]";
	mes "你最多只能派遣 10 位忍者。";
	next;
	goto M_Start;

NoZeny:
	mes "[狡猾的刺客]";
	mes "你沒有足夠的 zeny。";
	close;

NotEnoughNinjas:
	mes "[狡猾的刺客]";
	mes "沒有那麼多忍者可以購買。";
	next;
	goto M_Start;

NoNinjas:
	mes "[狡猾的刺客]";
	mes "沒有忍者可以購買了。";
	close;

NotEnoughNinjas1:
	mes "[狡猾的刺客]";
	mes "你沒有那麼多忍者。";
	next;
	goto M_Start;

M_Exit:
	mes "[狡猾的刺客]";
	mes "再見。";
	close;

// TIMER DELAY NINJA ADDER //
OnClock0600:
OnClock1200:
OnClock1500:
OnClock1900:
OnClock2000:
OnClock0000:
	set $ninja_avail,$ninja_avail+2;
	end;
OnClock1800:
	set $ninja_avail,$ninja_avail+3;
	end;
OnInit:
	set $ninja_avail,$ninja_avail+1;
	end;
}
