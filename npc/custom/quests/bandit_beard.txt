umbala,126,129,4	script	Bearded Man#bandit	120,{
	if(BEARD_QUEST == 4 ) goto L_MAKE;
	if(BEARD_QUEST >= 5 ) goto L_DONE;
	if(BEARD_QUEST >= 1 ) goto L_THREAD;

	mes "[鬍子男]";
	mes "鬍子！來買你想要的超讚鬍子！";
	mes "想要一撮鬍子嗎？";
	next;
	menu "Yes, I'd like some facial hair!",-,"No thanks, thats gross!",M_NO;

	mes "[鬍子男]";
	mes "HoHo~ 所以你想要一些鬍鬚。";
	emotion ET_SMILE;
	mes "嗯...我是很想給你我的鬍子，但它們可不便宜。";
	mes "事實上我只剩下一撮了...";
	mes "但我並不打算賣掉它。";
	mes "但別生氣！我可以做給你一撮。";
	next;
	set BEARD_QUEST,1;
L_THREAD:
	mes "[鬍子男]";
	mes "我需要一些 ^8080FF線^000000 來製作鬍子。";
	mes "去跟我在伊茲魯德的朋友要一些 ^8080FF線^000000。";
	next;

	mes "[鬍子男]";
	mes "而且我還需要你幫我準備一些鬍子的材料。";
	mes "它們是：";
L_LIST:
	mes "^8080FF鬆緊帶^000000";
	mes "^8080FF5 張動物皮革^000000";
	mes "^8080FF黑色染料^000000";
	mes "^8080FF老爺爺的鬍子^000000";
	mes "^8080FF100 個黏稠液體^000000";
	mes "^8080FF萬能藥^000000";
	mes " 和一顆 ^8080FF2 克拉鑽石^000000";
	close;

M_NO:
	mes "[鬍子男]";
	mes "好吧，隨你便！";
	emotion ET_ANGER;
	close;

L_DONE:
	mes "[鬍子男]";
	mes "抱歉，我只能一個人做一撮。";
	mes "我們現在不想要太多鬍子。對吧？";
	emotion ET_KIK;
	close;

L_MAKE:
	mes "[鬍子男]";
	mes "讓我們為你做鬍子吧！";
	next;
	if(countitem(7200) < 1 || countitem(919) < 1 ||	countitem(983) < 1 || countitem(2241) < 1 ||
	countitem(938) < 100 || countitem(973) < 1 || countitem(731) < 1) goto L_NOITEMS;
	delitem 7200, 1;
	delitem 919, 1;
	delitem 983, 1;
	delitem 2241, 1;
	delitem 938, 100;
	delitem 973, 1;
	delitem 731, 1;
	mes "^8080FF~你看到他正在裁剪並縫製鬍子~^000000";
	next;
	mes "^8080FF~他將完成的鬍子交給你~^000000";
	next;
	getitem 2237, 1;
	set BEARD_QUEST, 5;
	mes "[鬍子男]";
	mes "祝你有美好的一天！";
	emotion ET_KIK;
	close;

L_NOITEMS:
	mes "[鬍子男]";
	mes "搞什麼鬼！沒有這些材料我沒辦法做鬍子！";
	mes "這是清單：";
	emotion ET_HUK;
	goto L_LIST;
}

alberta,65,123,6	script	Man#bandit	85,{
	if(BEARD_QUEST == 3 && $@beardMobD) goto L_KILLED;
	if(BEARD_QUEST > 3) goto L_DONE;

	mes "[線的商人]";
	mes "啊啊啊！！我永遠無法將我的東西送到艾爾貝塔！";
	mes "我不斷被一些蟲子襲擊。";
	mes "它們偷走了我所有的物資和 ^8080FF線^000000。";
	next;
	mes "[線的商人]";

	if(BEARD_QUEST != 3) mes "我要怎麼辦？";
	if(BEARD_QUEST != 3) close;

	if($@beardmob > 0) mes "殺！殺了它們！！！";
	if($@beardmob > 0) close;
	mes "你是來幫助我的嗎？";
	emotion ET_QUESTION;
	next;
	menu "Yes",-,"No",M_NO;

	mes "[線的商人]";
	mes "喔，感謝諸神！";
	mes "如果你能殺死那些蟲子，我就能完成我的交貨。";
	next;
	mes "[線的商人]";
	mes "喔不！它們來了！！";
	emotion ET_PROFUSELY_SWEAT;
	donpcevent "BRDQ_MOBS::OnStart";
	close;

L_KILLED:
	set $@beardMobD,0;
	mes "[線的商人]";
	mes "謝謝你殺死那些盜蟲。";
	mes "作為我的謝意，這是一箱滿滿的 ^8080FF線^000000。";
	set BEARD_QUEST,4;
	close;

L_DONE:
	mes "[線的商人]";
	mes "再次感謝您！";
	emotion ET_THANKS;
	close;

M_NO:
	mes "[線的商人]";
	mes "該怎麼辦...";
	close;
}
