	mes "[蕾娜斯]";
	mes "我從瓦爾哈拉而來幫助您。";
	mes "我可以為您轉化許多瓦爾哈拉的物品，";
	mes "雖然它們需要許多稀有元素才能製作。";
	next;
	switch(select("妙爾尼爾:斯萊布尼爾:布里辛加曼:格萊普尼爾:梅金吉爾特:阿斯普里卡:布倫希爾德:赫沃爾:約爾姆加德:穆斯貝爾海姆:盧恩精髓:關閉")) {
		case 1:
			mes "[蕾娜斯]";
			mes "^3355FF妙爾尼爾^000000 是索爾的強大之槌。";
			setarray .@Items[0],1530,1522,2,984,20,985,5,969,40,7074,2,7075,4,7078,5,7091,5,7089,5;
			break;
		case 2:
			mes "[蕾娜斯]";
			mes "^3355FF斯萊布尼爾^000000 是奧丁的戰馬製作的靴子。";
			setarray .@Items[0],2410,2406,2,969,20,985,10,7076,3,7079,5,7083,3,7086,4,7087,3;
			break;
		case 3:
			mes "[蕾娜斯]";
			mes "^3355FF布里辛加曼^000000 是芙蕾雅，美麗女神的魔法項鍊。";
			setarray .@Items[0],2630,2603,1,726,2,722,3,727,10,724,5,969,20,7073,4,7077,4,7088,3,7090,3,7092,3;
			break;
		case 4:
			mes "[蕾娜斯]";
			mes "^3355FF格萊普尼爾^000000 是一種輕巧但堅固的繩索，是製作 ^3355FF梅金吉爾特^000000 所必需的";
			setarray .@Items[0],7058,7080,4,7081,5,7082,4,7084,3,7085,3;
			break;
		case 5:
			mes "[蕾娜斯]";
			mes "^3355FF梅金吉爾特^000000 是索爾強大的腰帶。";
			setarray .@Items[0],2629,7058,1,2627,1,969,20,726,10,984,10;
			break;
		case 6:
			mes "[蕾娜斯]";
			mes "^3355FF阿斯普里卡^000000 是眾神的光芒。";
			setarray .@Items[0],2541,7835,1,7836,1,7837,1,7838,1,2513,1,7063,100,7291,10,7293,10,985,20;
			break;
		case 7:
			mes "[蕾娜斯]";
			mes "^3355FF布倫希爾德^000000 是戰士的盔甲。";
			setarray .@Items[0],2383,7830,1,7831,1,7832,1,7833,1,7834,1,7510,100,2357,1,969,10,985,20;
			break;
		case 8:
			mes "[蕾娜斯]";
			mes "^3355FF赫沃爾^000000 以女武神的名字命名，她確保了戰士的勝利。";
			setarray .@Items[0],2181,6595,2,6596,1,6594,4,6597,3,6602,1,6604,1,6605,1,2115,1;
			break;
		case 9:
			mes "[蕾娜斯]";
			mes "^3355FF約爾姆加德^000000 以其毒害敵人的能力而聞名。";
			setarray .@Items[0],2020,6603,4,6599,1,6598,1,6601,4,6600,1,6604,1,6605,1,1473,1;
			break;
		case 10:
			mes "[蕾娜斯]";
			mes "^3355FF穆斯貝爾海姆^000000 是製作 ^3355FF約爾姆加德^000000 和 ^3355FF赫沃爾^000000 所必需的。";
			setarray .@Items[0],6605,984,50,985,50;
			break;
		case 11:
			mes "[蕾娜斯]";
			mes "^3355FF盧恩精髓^000000 是製作 ^3355FF約爾姆加德^000000 和 ^3355FF赫沃爾^000000 所必需的。";
			setarray .@Items[0],6604,12735,200,12736,100;
			break;
		default:
			close;
	}
	
	mes "^FF0000我需要的物品如下：^000000";
	set .@itemmissing,0;
	for(set .@i,1; .@i<getarraysize(.@Items); set .@i,.@i+2) {
		set .@amount,.@Items[.@i+1];
		set .@id,.@Items[.@i];
		mes " ~ "+.@amount+"x "+((countitem(.@id)<.@amount)?("^dd3355"):("^33dd55"))+getitemname(.@id)+((getitemslots(.@id)>0)?" ["+getitemslots(.@id)+"]":"")+"^000000";
		if (countitem(.@id) < .@amount) 
			set .@itemmissing,1;
	}
	if (.@itemmissing) {
		next;
		mes "[蕾娜斯]";
		mes "去尋找所有這些物品吧。";
		mes "我會在這裡等你幫助你。";
		close;
	}
	next;
	mes "[蕾娜斯]";
	mes "你想要 ^3355FF"+getitemname(.@Items[0])+"^000000 嗎？";
	next;
	if(select("否","是")==1) {
		mes "[蕾娜斯]";
		mes "如果你改變主意，請回來。";
		close;
	}
	mes "[蕾娜斯]";
	mes "這是你的 ^3355FF"+getitemname(.@Items[0])+"^000000。";
	mes "願它能好好為你服務。";
	//second check if the player tries to bypass the first
	for(set .@i,1; .@i<getarraysize(.@Items); set .@i,.@i+2) {
		set .@amount,.@Items[.@i+1];
		set .@id,.@Items[.@i];
		if (countitem(.@id) < .@amount) 
			close;
	}
	// Delete the items _after_ the second check
	for(set .@i,1; .@i<getarraysize(.@Items); set .@i,.@i+2) {
		set .@amount,.@Items[.@i+1];
		set .@id,.@Items[.@i];
		delitem .@id, .@amount;
	}
	getitem .@Items[0],1;
	close;
}
