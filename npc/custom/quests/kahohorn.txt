	setarray .@Items[0],7005, 968, 923, 734, 724,2268,7036,7020,5053,7018,741,742,743,750,751,752,753,754;
	setarray .@Drops[0],1272,1087,1039,1260,1046,1115,1252,1147,1038,1059;
	set .@n$, "[Lord Kaho's Servant]";

	mes .@n$;
	mes "我為了滿足卡浩大人對於頭飾的荒謬期望而操勞致死！";
	mes "既然我終於找到了完美頭飾的公式，我很樂意分享我的時間和才華。";
	next;
	mes .@n$;
	mes "你需要以下物品來獲得卡浩之角：";
	mes "3x ^0055FFEmperium^000000";
	next;
	mes .@n$;
	mes "喔，對了...";
	mes "我還忘了提到這些...";
	next;
	mes .@n$;
	for(set .@i,0; .@i<10; set .@i,.@i+1)
		mes "1x ^0055FF"+getitemname(.@Items[.@i])+"^000000 (來自 "+getmonsterinfo(.@Drops[.@i], MOB_NAME)+")";
	next;
	mes .@n$;
	mes "我們還沒完成，甜心...";
	mes "我是娃娃的超級粉絲，所以你需要帶給我這些可愛的東西！";
	for(set .@i,10; .@i<18; set .@i,.@i+1)
		mes "1x ^0055FF"+getitemname(.@Items[.@i])+"^000000";
	next;
	mes .@n$;
	mes "最後，我努力用上面提到的所有物品為我的主人製作了這些角...";
	mes "請包含 ^0055FF5 million Zeny^000000 作為我的努力報酬。";
	next;
	mes .@n$;
	mes "你準備好讓我製作這個特殊物品了嗎？";
	next;
	if(select("當然準備好了！:這些要求太難以置信了！")==2) {
		mes .@n$;
		mes "我所經歷的事情更難以置信...";
		mes "如果你成功獲得這些物品，你將擁有不可思議的力量！";
		close; }
	mes .@n$;
	for(set .@i,0; .@i<18; set .@i,.@i+1)
		if (!countitem(.@Items[.@i])) { set .@nr,1; break; }
	if (.@nr || countitem(714) < 3 || Zeny < 5000000) {
		mes "如果你沒有帶齊所有需要的材料，我就無法負擔製作這個物品。請理解這是為了造福像你這樣的英雄！";
		close; }
	for(set .@i,0; .@i<18; set .@i,.@i+1)
		delitem .@Items[.@i],1;
	delitem 714,3;
	set Zeny, Zeny-5000000;
	mes "哇！你真是勇敢！";
	getitem 5013,1;
	mes "享受成為米德加爾特的上帝吧！";
	close;
}
