//===== Description: ========================================= 
// Simple item trade-in quest, a person can get a Dead Branch
// by simply providing the items: Log x3 (7201), Wooden Heart x1 (7189), 
// Trunk x5 (1019), and Wooden Gnarl x1 (7222).
//===== Additional Comments: ================================= 
//= 1.1 Optimized, added Bloody Branch with 0.01% chance [Lupus]
//============================================================

niflheim,204,179,3	script	Mister Mobry	121,{

	mes "[莫布里先生]";
	mes "樹木擁有靈魂，你知道的...";
	next;

	mes "[莫布里先生]";
	mes "事實上，我知道一個關於這些靈魂的秘密，可以在某種程度上... 或許可以幫助你...";
	next;

	mes "[莫布里先生]";
	mes "我可以創造一個充滿活力的樹枝... 也就是說，如果你能提供我正確的材料...";
	next;
	goto L_MENU;

L_MENU:
	menu "What Do I Need?",-,"Where Can I find this Crap?",M_FIND,"Make me a Dead Branch!",M_CREATE,"Forget it...",M_END;

	mes "[莫布里先生]";
	mes "Mwehehe... 我看你想要控制樹木的靈魂，對嗎？好的... 我需要：";
	mes "^1354453 Logs^000000";
	mes "^1354451 Wooden Heart^000000";
	mes "^1354455 Trunk^000000";
	mes "^1354451 Wooden Gnarl^000000";
	next;
	goto L_MENU;

M_FIND:
	mes "[莫布里先生]";
	mes "你可以從某些怪物身上找到以下物品 heheh...";
	mes "^135445Logs can be found off of Tree Golems.^000000";
	mes "^135445Wooden Hearts the essence of Tree Golems.^000000";
	mes "^135445Trunks? Seriously, think wood...^000000";
	mes "^135445Wooden Gnarl... Gibbet...^000000";
	next;
	goto L_MENU;

L_NOLOG:
	mes "[莫布里先生]";
	mes "^135445Logs^000000 是身體... 我需要更多它們。";
	mes "給我三根原木，我就給你製造靈魂...";
	close;

L_NOHEART:
	mes "[莫布里先生]";
	mes "這個 ^135445Wooden Heart^000000 是靈魂的本質。";
	mes "給我一顆木製的心，我就給你製造靈魂...";
	close;

L_NOTRUNK:
	mes "[莫布里先生]";
	mes "什麼？你甚至找不到 ^1354455 Trunks^000000？";
	mes "殺死古老的柳樹或其他東西... 天啊... 沒有樹幹 heh... 可悲。";
	close;

L_NOGNARL:
	mes "[莫布里先生]";
	mes "沒有 ^135445Wooden Gnarl^000000 我怎麼做？";
	mes "我說了 Gibbet... 它們在這個地方到處都是...";
	close;

M_CREATE:
	mes "[莫布里先生]";
	mes "好吧... 讓我們看看你給我帶來了什麼...";
	next;
	if(countitem(7201)<3) goto L_NOLOG;
	if(countitem(7189)<1) goto L_NOHEART;
	if(countitem(1019)<5) goto L_NOTRUNK;
	if(countitem(7222)<1) goto L_NOGNARL;
	delitem 7201,3;
	delitem 7189,1;
	delitem 1019,5;
	delitem 7222,1;
	mes "[莫布里先生]";
	mes "Heh... 希望你和樹木的靈魂玩得開心... 小心點...";
	mes "Heh... 小心點？ 我才不在乎你死不死..... 沒事。";
	if(rand(1000)==0) goto L_GIVE2;
	getitem 604,1; //Dead Branch
	close;
L_GIVE2:
	getitem 12103,1; //Bloody Branch
	close;

M_END:
	mes "[莫布里先生]";
	mes "沒有靈魂給你... 走吧！";
	close;
}
