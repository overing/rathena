lighthalzen,108,229,2	script	La Fenn	766,{
	mes "[拉 芬恩]";
	if(BaseLevel<99) {
		if(Q_99LVL) {
			mes "你... 你變年輕了！！！但你肯定是 你！";
			mes "我沒想到你能夠滿足我的條件...";
			emotion ET_SCRATCH;
			next;
			mes "[拉 芬恩]";
			mes "這是你的獎勵。它永遠屬於你了。";
			set Q_99LVL,0;
			getitem 12106,1; //Jewel Case
			emotion ET_CONGRATULATION;
		} else {
			mes callfunc("F_Hi");
			mes "...";
			emotion ET_SLEEPY;
			next;
			mes "[拉 芬恩]";
			mes "我有一個謎題要給 2 轉且等級 99 的強者。";
			mes "*呵欠* 如果你遇到一些，請告訴他們我的謎題。";
		}
		close;
	}
	if(BaseLevel==99 && Upper==0) {
		if(Q_99LVL) {
			mes "現在前進並變得更年輕吧！";
			mes "如果你解開我的謎題，那麼我會給你一個裝有稀有飾品的魔法珠寶盒。";
		} else {
			mes callfunc("F_Hi")+" 英雄！";
			if(BaseExp<1000000) {
				mes "現在你很強了。";
				mes "但是... 還不夠強到可以解我的謎題。";
				if(rand(2)) {
					mes "再殺 "+((1000000-BaseExp)/3765+1)+" 隻海豹並回來。";
				} else {
					mes "再殺 "+((1000000-BaseExp)/81+1)+" 隻波利波利並回來。";
				}
			} else {
				mes "現在你真的 非常 強了。";
				mes "如果你變得比現在更年輕，我就會給你我的寶藏。";
				set Q_99LVL,1;
				emotion ET_BEST;
			}
		}
		close;
	}
	emotion ET_SCRATCH;
	mes "...";
	close;
}
