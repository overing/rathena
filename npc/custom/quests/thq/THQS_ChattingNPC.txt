yuno_in01,22,162,6	script	Female Assasin	725,{
	mes "[夏蕾特]";
	if (Event_THQS > 0) goto N_Member;
	mes "嗯...抱歉，如果你連會員都不是，我才懶得浪費時間跟你說話！";
	close;
N_Member:
	set @TEMP,rand(1,2);
	if (@TEMP == 1) goto N_Chat1;
	if (@TEMP == 2) goto N_Chat2;
	mes "你怎麼讀到這個的???";
	mes "除非你正在讀程式碼 :)";
	close;
N_Chat1:
	mes "歡迎你，尋寶獵人 "+strcharinfo(0)+"。";
	mes "你有沒有聽說那個獨自闖入古城格拉斯頓海姆的勇敢的人？！他一定是瘋了。";
	close;
N_Chat2:
	mes ""+strcharinfo(0)+" 你沒有什麼更好的事情可以做嗎？例如任務。";
	close;
}
