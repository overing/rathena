yuno_in01,124,164,6	script	Guild Leader	108,{
	mes "[奇根]";
	mes "是什麼風把你吹來的？有什麼話想說嗎？";
	next;
	menu "I want to be a Treasure Hunter",-,"Take a New Quest.",N_NQuest,"Take me to the Official Shop.",N_OShop,"Never mind.",N_NVM;
	if(Event_THQS==1) goto N_AlreadyReg;
	mes "[奇根]";
	mes "嗯..."+strcharinfo(0)+" 如果你想成為寶藏獵人，首先你必須...";
	mes " ";
	mes "^FF00001.^000000 你將會被兌換 ^FF000025,000z^000000 用於基本訓練以及你身為會員的證明。";
	mes " ";
	mes "嗯... 嗯... 這就是你需要做的全部了。哈哈哈。";
	next;
	menu "Pay ^FF000025,000z^000000.",-,"Thats way to high!!",N_HighPrice;
	if (Zeny < 25000) goto N_NoZeny;
	set Zeny,Zeny-25000;
	//getitem 7950, 1;
	//getitem 7951, 1;
	set Event_THQS,1;
	mes "[奇根]";
	mes "恭喜你！";
	emotion ET_CONGRATULATION;
	next;
	mes "[奇根]";
	mes "歡迎來到寶藏獵人公會。";
	mes "你現在可以隨時接受寶藏獵取任務。";
	close;
N_NoZeny:
	mes "[奇根]";
	mes "嗯... 你似乎沒有 ^FF000025,000z^000000 "+strcharinfo(0)+"。請等你有的時候再來。";
	close;
N_HighPrice:
	mes "[奇根]";
	mes "嗯... 如果你連區區 ^FF000025,000z^000000 都負擔不起，那你還不適合當寶藏獵人。";
	close;
N_AlreadyReg:
	mes "[奇根]";
	mes "哈哈哈...你已經是會員了。";
	close;
N_NQuest:
	if (Event_THQS!=1) goto A_NeedReg;
	mes "[奇根]";
	mes "嗯..."+strcharinfo(0)+" 如果你想接受任務，去和隔壁房間的寶藏獵人夥伴談談吧。";
	close;
A_NeedReg:
	mes "[奇根]";
	mes "抱歉，如果你想接受任務，你必須是寶藏獵人公會的會員。";
	mes " ";
	mes "此外，既然你現在是會員了，隨時都可以使用我們的商店，它們都位於這一層。";
	mes " ";
	mes "我們還有另一家僅供會員使用的商店，我們用一種叫做寶藏獵人代幣的小金屬鑽石來交換稀有商品。";
	mes " ";
	mes "你甚至可以使用我們專屬的個人地牢。";
	close;
N_OShop:
	if (Event_THQS!=1) goto N_NeedMem;
	mes "[奇根]";
	mes "是的 "+strcharinfo(0)+" 我這就帶你去我們的商店！";
	next;
	warp "prt_in",166,171;
	close;
	end;
N_NeedMem:
	mes "[奇根]";
	mes "抱歉，只有會員才能參觀我們的私人商店。";
	close;
N_NVM:
	mes "[奇根]";
	mes "好吧，等你空閒的時候再來吧。";
	close;
}
