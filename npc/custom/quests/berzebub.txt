prontera,165,178,8	script	老婦人	103,{

	setarray .@Items[0],714,25,1061,100,1095,200,618,10,1551,2,1614,1,727,20;
	set .@n$,"[老婦人]";

	mes .@n$;
	if(BerzQuest == 1) {
		mes "嘿，我記得你！我已經告訴過你，你只能完成這個任務一次。";
		close; }
	mes "你好，年輕人。我在吉芬附近採蘑菇的時候，發現了一個非常棒的東西。";
	next;
	if(select("繼續說:離開")==2) goto M_Leave;

	mes .@n$;
	mes "我從未見過這樣的物品。我聽說它可以放置在任何有插槽的飾品中。";
	next;
	mes .@n$;
	mes "如果你對這個物品有興趣，我就告訴你需要什麼東西，我才能為你製作一個。我只能給每個玩家一個，所以一旦你完成這個任務，你就不能再做了。";
	next;
	if(select("告訴我更多:算了吧，我不在乎")==2) goto M_Leave;

	mes .@n$;
	mes "我相信這個物品叫做 ^FF0000貝思波卡片^000000。它會讓你施法速度變得非常快！";
	next;
	mes .@n$;
	mes "我需要以下所有物品：";
	for(set .@i,0; .@i<getarraysize(.@Items); set .@i,.@i+2) {
		mes " ~ "+.@Items[.@i+1]+"x "+getitemname(.@Items[.@i]);
		if (countitem(.@Items[.@i]) < .@Items[.@i+1]) set .@nr,1; }
	next;
	if(select("我全部都有了！:我會盡快拿來的")==2) goto M_Leave;

	mes .@n$;
	if (.@nr) {
		mes "你說謊！滾出去，把東西拿來，你這個可悲的搖滾巨星藉口。";
		close; }
	for(set .@i,0; .@i<getarraysize(.@Items); set .@i,.@i+2)
		delitem .@Items[.@i], .@Items[.@i+1];
	getitem 4145,1;
	set BerzQuest, 1;
	mes "恭喜！你已經擁有所有物品。這是你承諾的 ^FF0000貝思波卡片^000000。";
	emotion ET_CONGRATULATION;
	close;

M_Leave:
	mes .@n$;
	mes "我手邊有這麼棒的東西...";
	close;
}
