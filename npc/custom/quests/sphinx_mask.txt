morocc,208,90,6	script	Turban Thief	58,{
	mes "[頭巾盜賊]";
	if(event_umbala < 2) {
		mes "你想跟我做什麼？走開！";
		close; }
	if(sphmask_q) {
		mes "你跟我沒有任何事，走開！";
		close; }
	mes "你好我的朋友，有沒有興趣看看這個稀有的面具？我向你保證它的價值是真的，我的朋友。真是太棒了！";
	next;
	callsub L_Menu,1,"1,000,000",1000000;
	callsub L_Menu,2,"750,000",750000;
	callsub L_Menu,3,"500,000",500000;
	next;
	mes "[頭巾盜賊]";
	mes "呃！算了！我可以在其他地方做得更好！";
	set sphmask_q,1;
	close;

L_Menu:
	mes "[頭巾盜賊]";
	switch(getarg(0)) {
		case 1: mes "哇？喔呵呵呵，所以你知道我從那些野蠻人那裡偷了這個東西，對吧？我不會還回去的。但如果給我一個價錢...你說呢？"; break;
		case 2: mes "啊，你是個生意人嗎？很好，那...怎麼樣？"; break;
		case 3: mes "嗯...你殺價很兇，好吧...我的最後報價..."; break; }
	next;
	if(select("支付 "+getarg(1)+"z:免談")==2) return;
	mes "[頭巾盜賊]";
	if (Zeny < getarg(2)) {
		mes "你是在耍我嗎？你沒錢！";
		close; }
	set Zeny, Zeny-getarg(2);
	mes "喔呵呵呵，成交！";
	getitem 7114,1;
	set sphmask_q,1;
	close;
}
