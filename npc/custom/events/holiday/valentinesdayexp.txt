	mes "[史蒂芬]";
	mes "猜猜我得到了什麼？";
	mes "一種在米德加茲不容易找到的美味佳餚....";
	next;
	mes "[史蒂芬]";
	mes "巧克力！";
	mes "沒錯，你不喜歡巧克力嗎....我喜歡。";
	mes "你很幸運，因為我賣的巧克力每塊只要 5,000 Zeny！";
	next;
	menu "I want some chocolate!",M_CHOCO,"No thanks.",-;

	mes "[史蒂芬]";
	mes "你不要巧克力嗎？";
	mes "我告訴你！你會後悔的！";
	mes "你最好現在就買一些...你再也遇不到這樣的巧克力了！";
	mes "好好想想，有空再來找我。";
	close;

M_CHOCO:
	mes "[史蒂芬]";
	mes "哈！";
	mes "我就知道！";
	mes "但是我一次最多只能賣你 5 塊....但是，如果你真的需要更多....";
	mes "你可以再回來。";
	mes "你想要多少？";
	next;
	set @needmon,0;
	input @flag_num;
	if (@flag_num <= 0) goto L_NONE;
	if (@flag_num > 5) goto L_ERR;
	set @needmon,@flag_num*5000;
	if (Zeny < @needmon) goto L_NOTENO;
	set Zeny,Zeny - @needmon;
	getitem 558,@flag_num;
	mes "[史蒂芬]";
	mes "給你！";
	mes "你可以把它當作禮物送給別人，或者自己享用！";
	mes "嗯....香甜的巧克力...";
	mes "隨時再來找我...！";
	close;

L_ERR:
	mes "[史蒂芬]";
	mes "對不起，但我不能給你那麼多。";
L_NONE:
	close;

L_NOTENO:
	mes "[史蒂芬]";
	mes "很抱歉，你似乎買不起我的巧克力。";
	close;
}
