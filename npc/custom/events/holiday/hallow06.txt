prontera,151,190,5	script	萬聖節活動	807,{
	mes "[萬聖節活動]";
	if(Class == Job_Novice){
		mes "很抱歉，你必須先轉職才能完成活動。";
		emotion ET_SWEAT, playerattached();
		close;
	}
	if((EVENT_HALWN06&1) && (EVENT_HALWN06&2)){
		mes "很抱歉，你已經完成過這個活動了。";
		emotion ET_SORRY;
		close;
	}
	mes "您好，活潑的人類同胞。";
	next;
	mes "[萬聖節活動]";
	mes "萬聖節即將到來，冒險進入少有人敢去的地方的時候也快到了。";
	next;
	mes "[萬聖節活動]";
	mes "不給糖就搗蛋，你喜歡哪一個？";
	next;
	menu "Trick",-, "Treat",L_Treat;

	mes "[萬聖節活動]";
	if(EVENT_HALWN06&2){
		mes "很抱歉，你已經被惡作劇過了。何不去選個糖果呢。";
		emotion ET_SORRY;
		close;
	}
	mes "請稍等...";
	next;
	set BaseExp,BaseExp+BaseLevel*1000;
	set EVENT_HALWN06,EVENT_HALWN06|2;

	mes "[萬聖節活動]";
	mes "這是你的糖果。你的Base EXP稍微增加了。";
	next;
	mes "[萬聖節活動]";
	mes "希望你喜歡這個惡作劇，萬聖節快樂！";
	close;

L_Treat:
	mes "[萬聖節活動]";
	if(EVENT_HALWN06&1){
		mes "很抱歉，你已經拿過糖果了。何不去選個惡作劇呢。";
		emotion ET_SORRY;
		close;
	}
	if(!(EVENT_HALWN06&4)){
		mes "太好了！這一定會非常有趣。";
		next;
		mes "[萬聖節活動]";
		mes "我會派你去尋找一些物品。";
		next;
		mes "[萬聖節活動]";
		mes "把所有需要的物品帶回來，我會為你製作一個糖果。";
		next;
		mes "[萬聖節活動]";
		mes "不是那種你可以吃的普通糖果，而是你可以戴在頭上的東西。";
		next;
		mes "[萬聖節活動]";
		mes "但是要製作任何東西，首先需要一些物品。";
		mes "清單：";
		mes "1 頂帽子，";
		mes "30 個南瓜，以及";
		mes "20 塊黑色布料";
		next;
		mes "[萬聖節活動]";
		mes "趕快行動吧。";
		set EVENT_HALWN06,EVENT_HALWN06|4;
		close;
	}
	mes "你拿到我要求的物品了嗎？還是需要再看一次清單？";
	next;
	menu "Got'em",-, "Don't Have",L_Dont_Have, "See List",L_List;

	mes "[萬聖節活動]";
	if((countitem(2220) < 1) || (countitem(535) < 30) || (countitem(7205) < 20)){
		mes "你的物品不夠。收集好後再來吧。";
		emotion ET_SORRY;
		close;
	}
	delitem 2220,1;
	delitem 535,30;
	delitem 7205,20;
	getitem 5134,1;
	set EVENT_HALWN06,EVENT_HALWN06|1;
	mes "做得好！希望你喜歡你的物品。";
	emotion ET_CONGRATULATION;
	next;
	mes "[萬聖節活動]";
	mes "祝你剩下的萬聖節愉快。";
	emotion ET_SCISSOR, playerattached();
	close;

L_Dont_Have:
	mes "[萬聖節活動]";
	mes "回去找找看吧。你想查看清單嗎？";
	next;
	menu "No",-, "Yes",L_List;
	close;

L_List:
	mes "[萬聖節活動]";
	mes "我需要以下物品：";
	mes "1 頂帽子，";
	mes "30 個南瓜，以及";
	mes "20 塊黑色布料。";
	close;
}
