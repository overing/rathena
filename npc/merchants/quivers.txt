payon_in01,5,134,5	script	發明家 傑克斯	89,{
	if (checkweight(1201,1) == 0 || MaxWeight - Weight < 2000) {
		mes "[發明家 傑克斯]";
		mes "嘿，你攜帶了";
		mes "太多東西了。為什麼不把它們存放在 Kafra 倉庫裡呢？在你做完之後我們可以再談，對吧？";
		close;
	}
	mes "[發明家 傑克斯]";
	mes "我叫傑克斯。";
	mes "我可以毫無自負地";
	mes "說我或許是";
	mes "當今時代最^663300偉大的發明家^000000。";
	next;
	mes "[發明家 傑克斯]";
	mes "這次，我";
	mes "創造了一些";
	mes "真正非凡的東西。我稱它們為... ^663300魔法箭筒^000000 !! 這將在歷史上被銘記為一次";
	mes "箭矢的革命！";
	emotion ET_BEST;
	next;
	mes "[發明家 傑克斯]";
	mes "我研究魔法和箭筒多年，夜以繼日地工作，直到我終於弄清楚如何用魔法來壓縮箭矢！有了魔法箭筒，你將攜帶更多的箭矢，但重量更輕！";
	next;
	mes "[發明家 傑克斯]";
	mes "你想要試試用我的箭筒嗎？我毫不懷疑像你這樣的人可以欣賞我的天才！";
	next;
	switch(select("箭筒:鐵箭箭筒:鋼箭箭筒:奧里吉金箭箭筒:火焰箭箭筒:銀箭箭筒:風靈箭箭筒:石箭箭筒:水晶箭箭筒:闇靈箭箭筒:無形箭箭筒:生鏽箭箭筒")) {
	case 1: callsub S_BuyQuiver,1750,500,500,12004;
	case 2: callsub S_BuyQuiver,1770,500,500,12005;
	case 3: callsub S_BuyQuiver,1753,500,500,12006;
	case 4: callsub S_BuyQuiver,1765,500,500,12007;
	case 5: callsub S_BuyQuiver,1752,500,500,12008;
	case 6: callsub S_BuyQuiver,1751,500,500,12009;

	case 9: callsub S_BuyQuiver,1754,500,500,12012;
	case 10: callsub S_BuyQuiver,1767,500,500,12013;
	case 11: callsub S_BuyQuiver,1757,500,500,12014;

	//This doesn't seem to be accessable...
	//mes "[發明家 傑克斯]";

S_BuyQuiver:
	.@item_id = getarg(0);
	.@amount = getarg(1);
	.@zeny_req = getarg(2);

	if (countitem(.@item_id) >= .@amount) {
		mes "[發明家 傑克斯]";
		mes "太棒了！";
		mes "你身上有帶箭矢嗎？我會提供你一個箭筒，它可以裝 "+ .@amount +" 支你的 "+ getitemname(.@item_id) +"，只需要 ^FF3131"+ .@zeny_req +" Zeny^000000。";
		next;
		switch(select("盡可能將箭矢儲存在箭筒中:購買 1 個箭筒:取消")) {
		case 1:
			.@arrows = countitem(.@item_id);
			set .@quiver,.@arrows / .@amount;
			set .@arrows_used,.@quiver * .@amount;
			.@arrow_zeny01 = .@quiver * .@zeny_req;
			mes "箭矢數量：";
			mes "箭矢: ^3131FF"+.@arrows+" ^000000";
			mes "可購買的";
			mes "箭筒最大數量：";
			mes "箭筒: ^3131FF"+.@quiver+" ^000000";
			mes "所需 Zeny: ^3131FF"+.@arrow_zeny01+" Zeny^000000";
			next;
			mes "[發明家 傑克斯]";
			mes "你想要";
			mes "盡可能地購買";
			mes "箭筒來裝你";
			mes "目前攜帶的箭矢嗎？";
			next;
			if (select("是:取消") == 1) {
				if (.@arrow_zeny01 < Zeny) {
					mes "[發明家 傑克斯]";
					mes "給你！";
					mes "請記住，^FF0000當你的攜帶重量達到最大重量限制的 90% 時，你將無法使用箭筒^000000。";
					set Zeny, Zeny-.@arrow_zeny01;
					delitem .@item_id,.@arrows_used;// Arrow
					getitem getarg(3),.@quiver; //Quiver
					next;
					mes "[發明家 傑克斯]";
					mes "所以...";
					mes "請隨時";
					mes "注意你攜帶的";
					mes "重量，這樣你";
					mes "應該就沒事了。";
					close;
				}
				else {
					mes "[發明家 傑克斯]";
					mes "我很抱歉，但你沒有足夠的 Zeny。在花了多年時間";
					mes "開發這種革命性的技術之後，我不能就這樣把它們送出去！";
					close;
				}
			}
			mes "[發明家 傑克斯]";
			mes "你改變主意了？";
			mes "當擁有";
			mes "箭筒的榮耀如此接近時？";
			close;
		case 2:
			if (Zeny > .@zeny_req) {
				mes "[發明家 傑克斯]";
				mes "給你！";
				mes "請記住，^FF0000當你的攜帶重量達到最大重量限制的 90% 時，你將無法使用箭筒^000000。";
				Zeny = Zeny -.@zeny_req;
				delitem .@item_id,.@amount;// Arrow
				getitem getarg(3),1; //Quiver
				next;
				mes "[發明家 傑克斯]";
				mes "所以...";
				mes "請隨時";
				mes "注意你攜帶的";
				mes "重量，這樣你";
				mes "應該就沒事了。";
				close;
			}
			else {
				mes "[發明家 傑克斯]";
				mes "你甚至";
				mes "沒有 "+ .@zeny_req +" Zeny?";
				mes "我非常抱歉。我不知道你竟然如此...";
				mes "^333333貧困^000000。";
				close;
			}
		case 3:
			mes "[發明家 傑克斯]";
			mes "什麼...?";
			mes "難道你沒有看到這個發明可以永遠改變箭矢的攜帶方式嗎？！未來就在眼前！";
			close;
		}
	}
	else {
		mes "[發明家 傑克斯]";
		mes "你可以在這個箭筒中攜帶最多 "+ .@amount +" 支箭矢。它是用我的秘密方法製成的，";
		mes "因此箭矢和箭筒的總重量小於單獨攜帶箭矢的重量。";
		next;
		mes "[發明家 傑克斯]";
		mes "這是一個科學奇蹟！如果你給我至少 "+ .@amount +" 支箭矢和每個箭筒 "+ .@zeny_req +" Zeny，你就可以親身體驗到。";
		close;
	}
}
