					specialeffect EF_LORD;
					mes "[莉亞布雷姆]";
					mes "太好了，看起來很成功。";
					mes "看起來做得很好。恭喜你！";
					delitem .@item_id,1;
					delitem .@item_req[0],.@req_amount[0];
					if (.@item_req[1] != 0 && .@req_amount[1] != 0)
						delitem .@item_req[1],.@req_amount[1];
					Zeny = Zeny - .@zeny_req *1000;
					getitem getarg(1),1;
					next;
					mes "[莉亞布雷姆]";
					mes "再見了，夥伴！";
					close;
				}
				else
				{
					specialeffect EF_SUI_EXPLOSION;
					mes "[莉亞布雷姆]";
					mes "哇！ ...我很抱歉，失敗了。";
					mes "但是，我是完全無辜的。";
					mes "這是你的運氣，而且是神注定的，好嗎？";
					mes "別太失望，";
					mes "下次再試試。";
					delitem .@item_id,1;
					delitem .@item_req[0],.@req_amount[0];
					if (.@item_req[1] != 0 && .@req_amount[1] != 0)
						delitem .@item_req[1],.@req_amount[1];
					Zeny = Zeny - .@zeny_req *1000;
					next;
					mes "[莉亞布雷姆]";
					mes "再見了，夥伴！";
					close;
				}
			}
			else
			{
				mes "[莉亞布雷姆]";
				mes "你是笨蛋嗎？你沒有帶所有需要的物品！";
				mes "快去把它們帶來！";
				close;
			}
		case 2:
			mes "[莉亞布雷姆]";
			mes "下次再見。";
			close;
	}
