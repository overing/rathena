function	script	Func_Socket	{
	.@item_id = getarg(0);
	.@zeny_req = getarg(4);
	setarray .@item_req[0], getarg(5), getarg(7,0);
	setarray .@req_amount[0], getarg(6), getarg(8,0);

	disable_items;
	mes "[西雅布勒姆]";
	mes "你想為 "+ getitemname(.@item_id) +" 添加一個插槽？";
	mes "好吧，請給我帶 ^FF0000"+ .@req_amount[0] +" "+ getitemname(.@item_req[0]) + (.@req_amount[0] > 1 && .@item_req[0] != 999 ? "個":"") +"^000000, "+ (.@item_req[1] != 0 && .@req_amount[1] != 0 ? ("^FF0000"+ .@req_amount[1] +" "+ getitemname(.@item_req[1]) + (.@req_amount[1] > 1 && .@item_req[1] != 999 ? "個":"") + "^000000 ") : "") + "以及我的 ^FF0000" + (.@zeny_req >= 1000 ? (.@zeny_req /1000 +",000") : .@zeny_req) +",000 zeny^000000 服務費。";
	mes "啊，別忘了帶上那個 "+ getitemname(.@item_id) +"！";
	next;
	mes "[西雅布勒姆]";
	if (getiteminfo(.@item_id, ITEMINFO_LOCATIONS) & EQP_HAND_R) // weapon
	{
		mes "如果你有需要的物品和 zeny，我現在可以嘗試添加一個插槽。";
		mes "但是，你應該知道我可能會失敗。";
		mes "因此，我需要給你一個公平的警告...";
		next;
		mes "[西雅布勒姆]";
		mes "如果嘗試向你的武器添加插槽失敗，那麼^FF0000武器^000000，^FF0000以及任何複合到它的卡片都會被銷毀^000000。";
	}
	else // armor
	{
		mes "如果你擁有所有必需的材料、我的 zeny 服務費和盔甲，那麼我們可以繼續進行插槽添加嘗試。";
		mes "但在那之前，我必須警告你風險。";
		next;
		mes "[西雅布勒姆]";
		mes "如果嘗試向你的盔甲添加插槽失敗，那麼^FF0000盔甲^000000，^FF0000它的升級^000000 ^FF0000以及任何複合到它的卡片都會被銷毀^000000。";
	}
	mes "你仍然想嘗試添加插槽嗎？";
	next;
	switch(select("嘗試添加插槽:取消"))
	{
		case 1:
			if ((Zeny >= .@zeny_req*1000) && countitem(.@item_req[0]) >= .@req_amount[0] && countitem(getarg(7,512)) >= .@req_amount[1] && countitem(.@item_id) > 0)// if there's no getarg(7) and getarg(8) provided, check itemcount against 0 (we still need a valid item ID though)
			{
				mes "[西雅布勒姆]";
				mes "好吧，讓工作開始吧！";
				mes "你最好祈禱能成功。";
				next;
				set .@a, rand(1,100);
				if ((.@a > getarg(2)) && (.@a < getarg(3)))
				{
					specialeffect getarg(3) == 51 ? EF_LORD : EF_SANCTUARY; // only S Class items have Lord of Vermillion effect
					mes "[西雅布勒姆]";
					mes "太棒了，似乎成功了。";
					mes "看起來做得很好。恭喜你！";
					delitem .@item_id,1;
					delitem .@item_req[0],.@req_amount[0];
					if (.@item_req[1] != 0 && .@req_amount[1] != 0)
						delitem .@item_req[1],.@req_amount[1];
					Zeny = Zeny - .@zeny_req *1000;
					getitem getarg(1),1;
					next;
					mes "[西雅布勒姆]";
					mes "再見，夥計！";
					close;
				}
				else
				{
					specialeffect EF_SUI_EXPLOSION;
					mes "[西雅布勒姆]";
					mes "哇！ ...我很抱歉，它失敗了。";
					mes "但是，我是完全無辜的。";
					mes "這是你的運氣，這是上帝註定的，好嗎？";
					mes "不要太失望，下次再試。";
					delitem .@item_id,1;
					delitem .@item_req[0],.@req_amount[0];
					if (.@item_req[1] != 0 && .@req_amount[1] != 0)
						delitem .@item_req[1],.@req_amount[1];
					Zeny = Zeny - .@zeny_req *1000;
					next;
					mes "[西雅布勒姆]";
					mes "我祝你下次好運！";
					close;
				}
			}
			else
			{
				mes "[西雅布勒姆]";
				mes "我想繼續進行這個插槽添加嘗試，但是你缺少一些東西。";
				mes "你確定你有裝備、所需的材料和 zeny 嗎？";
				close;
			}
		case 2:
			mes "[西雅布勒姆]";
			mes "需要一些時間來考慮一下，是嗎？";
			mes "好吧，我能理解。";
			mes "只要記住，如果你總是玩得很安全，生活就沒有樂趣了~";
			close;
	}
}
