alde_alche,24,188,3	script	Guild Dealer	740,{
	if (checkweight(1201,1) == 0) {
		mes "- 等一下！ -";
		mes "- 目前你攜帶了 -";
		mes "- 太多物品在身上了。 -";
		mes "- 請你先將一些物品 -";
		mes "- 存放到卡普拉倉庫後再回來。 -";
		close;
	}
	mes "[基佛·阿爾·夏普]";
	mes "歡迎來到";
	mes "鍊金術師聯盟。";
	mes "今天有什麼可以為您服務的嗎？";
	next;
	switch(select("購買材料。:購買生產手冊。:離開。")) {
	case 1:
		mes "[基佛·阿爾·夏普]";
		mes "您想要什麼？";
		next;
		if (select("藥水碗 - 8 Zeny:取消。") == 1) {
			mes "[基佛·阿爾·夏普]";
			mes "您想要多少個？";
			mes "輸入 '0' 如果您想要取消。";
			next;
			while(1) {
				input .@input,0,2001;
				if (.@input == 0) {
					mes "[基佛·阿爾·夏普]";
					mes "交易已取消。";
					mes "下次再來。";
					close;
				}
				else if (.@input < 1 || .@input > 2000) {
					mes "[基佛·阿爾·夏普]";
					mes "數量必須";
					mes "小於 2000。";
					next;
				}
				else {
					break;
				}
			}
			set .@sell,.@input * 8;
			set .@item_weight,.@input * 10;
			if (Zeny < .@sell) {
				mes "[基佛·阿爾·夏普]";
				mes "您的錢";
				mes "不夠。";
				mes "先確認您有多少錢。";
				close;
			}
			if ((MaxWeight - Weight) < .@item_weight) {
				mes "[基佛·阿爾·夏普]";
				mes "看來";
				mes "您無法全部攜帶。";
				mes "請檢查";
				mes "您物品欄的空間。";
				close;
			}
			set Zeny, Zeny-.@sell;
			getitem 7134,.@input; //Medicine_Bowl
			mes "[基佛·阿爾·夏普]";
			mes "謝謝。";
			mes "歡迎再來。";
			close;
		}
		mes "[基佛·阿爾·夏普]";
		mes "那麼，";
		mes "當您需要購買";
		mes "與鍊金術相關的";
		mes "材料時再來吧？";
		close;
	case 2:
		mes "[基佛·阿爾·夏普]";
		mes "您需要什麼？";
		mes "手冊通常是 100,000 Zeny。但有幾本特殊的手冊會更貴。";
		next;
		switch(select("藥水製作指南:酒精製作指南:瓶罐炸彈製作指南:酸性瓶罐製作指南:植物瓶罐製作指南:海洋精靈瓶罐製作指南:閃耀外袍製作指南:濃縮藥水製作指南:取消交易。")) {
		case 1:
			callsub S_SellManual,7144,100000; //Normal_Potion_Book
			break;
		case 2:
			callsub S_SellManual,7127,100000; //Alcol_Create_Book
			break;
		case 3:
			callsub S_SellManual,7128,100000; //FireBottle_Create_Book
			break;
		case 4:
			callsub S_SellManual,7129,100000; //Acid_Create_Book
			break;
		case 5:
			callsub S_SellManual,7130,100000; //Plant_Create_Book
			break;
		case 6:
			callsub S_SellManual,7131,100000; //Mine_Create_Book
			break;
		case 7:
			callsub S_SellManual,7132,100000; //Coating_Create_Book
			break;
		case 8:
			callsub S_SellManual,7133,240000; //Slim_Potion_Create_Book
			break;
		case 9:
			mes "[基佛·阿爾·夏普]";
			mes "那麼...";
			mes "如果您";
			mes "需要購買";
			mes "生產手冊";
			mes "再來吧。";
			close;
		}
	case 3:
		mes "[基佛·阿爾·夏普]";
		mes "好的，";
		mes "祝您有個美好的一天。";
		close;
	}

S_SellManual:
	.@item_id = getarg(0);
	.@zeny_req = getarg(1);

	if (Zeny < .@zeny_req) {
		mes "[基佛·阿爾·夏普]";
		mes "您的錢";
		mes "不夠。";
		mes "先確認您有多少錢。";
		close;
	}
	mes "[基佛·阿爾·夏普]";
	mes getitemname(.@item_id) +"?";
	mes "那要 "+ .@zeny_req +" zeny。";
	next;
	if (select("購買。:離開。") == 1) {
		Zeny = Zeny - .@zeny_req;
		getitem .@item_id,1;
		mes "[基佛·阿爾·夏普]";
		mes "感謝";
		mes "您的惠顧。";
		close;
	}
	mes "[基佛·阿爾·夏普]";
	mes "下次";
	mes "再來。";
	close;
}
