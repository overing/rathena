alberta_in,16,28,4	script	Pharmacist	61,{
	if (checkweight(1201,1) == 0) {
		mes "- 等一下！ -";
		mes "- 目前你攜帶 -";
		mes "- 過多的物品了。 -";
		mes "- 請稍後再回來 -";
		mes "- 等你把一些物品放到卡普拉倉庫後。 -";
		close;
	}
	mes "[老藥劑師]";
	mes "嗯...";
	mes "是什麼風把你吹來的...？";
	next;
	switch(select("製作藥水:交談:混合資訊:取消")) {
	case 1:
		if (MaxWeight - Weight < 5000) {
			mes "[老藥劑師]";
			mes "你為什麼攜帶這麼多東西！";
			mes "不要太貪心，只攜帶你需要的量！";
			next;
			mes "[老藥劑師]";
			mes "你太重了，無法從我這裡拿到藥水...";
			mes "先去你的倉庫存放一些物品！";
			close;
		} else {
			mes "[老藥劑師]";
			mes "你已經準備好所有的東西了嗎？你想要哪一種？";
			next;
			switch(select("紅色藥水:橘色藥水:黃色藥水:白色藥水:藍色藥水:綠色藥水:其實，我什麼都不想要。")) {
			case 1:
				callsub L_Making,507,3,501;
			case 2:
				mes "[老藥劑師]";
				if (countitem(507) < countitem(713)) {
					set .@max,countitem(507);
				}
				else if (countitem(508) < countitem(713)) {
					set .@max,countitem(508);
				}
				else {
					set .@max,countitem(713);
				}
				if ((countitem(507) < 1) || (countitem(508) < 1) || (countitem(713) == 0)) {
					mes "你這傢伙！你期望什麼？！什麼都沒帶就來這裡。嘖！";
					mes "滾開！";
					close;
				}
				if (Zeny < 3) {
					mes "你這傢伙！你期望什麼？！什麼都沒帶就來這裡。嘖！";
					mes "滾開！";
					close;
				}
				mes "要多少？";
				next;
				switch(select("盡可能多做一點:我想選擇數量:其實，我什麼都不想要。")) {
				case 1:
					if ((countitem(507) < .@max) || (countitem(508) < .@max) || (countitem(713) < .@max) || (Zeny < .@max*3)) {
						mes "[老藥劑師]";
						mes "你這傢伙！你甚至沒有所有的材料，還想要我幫你製作藥水？！";
						close;
					}
					set Zeny,Zeny-(.@max*5);
					delitem 507,.@max; //Red_Herb
					delitem 508,.@max; //Yellow_Herb
					delitem 713,.@max; //Empty_Bottle
					getitem 502,.@max; //Orange_Potion
					break;
				case 2:
					mes "[老藥劑師]";
					mes "那就選擇一個低於 100 的數字。如果你不想要任何東西，只需輸入 '0'。以你擁有的材料，你可以製作大約 "+.@max+" 瓶藥水。";
					next;
					input .@amount,0,101;
					if (.@amount == 0) {
						mes "[老藥劑師]";
						mes "下定決心，好嗎？！";
						close;
					}
					if (.@amount > 100) {
						mes "[老藥劑師]";
						mes "你聾了嗎？我說小於 100！";
						close;
					}
					if ((countitem(507) < .@amount) || (countitem(508) < .@amount) || (countitem(713) < .@amount) || (Zeny < .@amount*3)) {
						mes "[老藥劑師]";
						mes "你這傢伙！你甚至沒有所有的材料，還想要我幫你製作藥水？！";
						close;
					}
					next;
					set Zeny,Zeny-(.@amount*5);
					delitem 507,.@amount; //Red_Herb
					delitem 508,.@amount; //Yellow_Herb
					delitem 713,.@amount; //Empty_Bottle
					getitem 502,.@amount; //Orange_Potion
					break;
				case 3:
					mes "[老藥劑師]";
					mes "什麼？！";
					mes "咕...";
					mes "呸！";
					mes "滾開！";
					close;
				}
				mes "[老藥劑師]";
				mes "給你。一切都完成了，所以你可以拿走它。但是請記住！濫用藥物是不好的。";
				close;
			case 3:
				callsub L_Making,508,10,503;
			case 4:
				callsub L_Making,509,20,504;
			case 5:
				callsub L_Making,510,30,505;
			case 6:
				callsub L_Making,511,3,506;
			case 7:
				mes "[老藥劑師]";
				mes "什麼？！";
				mes "咕...";
				mes "呸！";
				mes "滾開！";
				close;
			}
		}
	case 2:
		mes "[老藥劑師]";
		mes "使用藥物，你可以提高一個人的再生能力。但是，它們只能到某種程度。*嘆氣* 我開始想起我年輕時的日子。我一定變老了。";
		next;
		mes "[老藥劑師]";
		mes "總之，藥水只不過是藥水。不多也不少。";
		close;
	case 3:
		mes "[老藥劑師]";
		mes "嗯...";
		mes "你們這些年輕人可能很煩人。但是，既然你問了，我就解釋一下。";
		next;
		mes "[老藥劑師]";
		mes "草藥本身效果很好，但是如果你使用我的特殊技術和技能來製作藥水，效果會更加強大。";
		next;
		mes "[老藥劑師]";
		mes "如果你熱切且有禮貌地提出要求，我會為你製作。但是，不是免費的... 不用擔心，我只收取少量費用，所以不會太貴。";
		next;
		mes "[老藥劑師]";
		mes "紅色藥水 - ^0098E52 紅色藥草, 1 個空瓶, 2 zeny 費用。^000000";
		mes "橘色藥水 - ^0098E51 紅色藥草, 1 個黃色藥草, 1 個空瓶, 5 zeny 費用。^000000";
		mes "黃色藥水 - ^0098E52 黃色藥草, 1 個空瓶, 10 zeny。^000000";
		next;
		mes "[老藥劑師]";
		mes "白色藥水 - ^0098E52 白色藥草, 1 個空瓶, 20 zeny 費用。^000000";
		mes "藍色藥水 - ^0098E52 藍色藥草, 1 個空瓶, 30 zeny 費用。^000000";
		mes "綠色藥水 - ^0098E52 綠色藥草, 1 個空瓶, 3 zeny 費用。^000000";
		close;
	case 4:
		mes "[老藥劑師]";
		mes "真是個無聊的人。如果你有話要說，為什麼不說出來？！";
		close;
	}
	
L_Making:
	.@item_req = getarg(0);
	.@req_amount = getarg(1);

	mes "[老藥劑師]";
	if ((countitem(.@item_req)/2) < countitem(713)) {
		.@max = countitem(.@item_req) /2;
	} else {
		.@max = countitem(713);
	}
	if (countitem(.@item_req) < 2 || countitem(713) == 0) {
		mes "你這傢伙！你期望什麼？！什麼都沒帶就來這裡。嘖！";
		mes "滾開！";
		close;
	}
	if (Zeny < 3) {
		mes "你這傢伙！你期望什麼？！什麼都沒帶就來這裡。嘖！";
		mes "滾開！";
		close;
	}
	mes "要多少？";
	next;
	switch(select("盡可能多做一點:我想選擇數量:其實，我什麼都不想要。")) {
	case 1:
		if ((countitem(.@item_req) < .@max*2) || countitem(713) < .@max || (Zeny < .@max*.@req_amount)) {
			mes "[老藥劑師]";
			mes "你這傢伙！你甚至沒有所有的材料，還想要我幫你製作藥水？！";
			close;
		}
		Zeny = Zeny - (.@max*.@req_amount);
		delitem .@item_req,.@max*2;
		delitem 713,.@max; //Empty_Bottle
		getitem getarg(2),.@max; 
		break;
	case 2:
		mes "[老藥劑師]";
		mes "那就選擇一個低於 100 的數字。如果你不想要任何東西，只需輸入 '0'。以你擁有的材料，你可以製作大約 "+.@max+" 瓶藥水。";
		next;
		input .@amount;
		if (.@amount == 0) {
			mes "[老藥劑師]";
			mes "下定決心，好嗎？！";
			close;
		}
		if (.@amount > 100) {
			mes "[老藥劑師]";
			mes "你聾了嗎？我說小於 100！";
			close;
		}
		if ((countitem(.@item_req) < .@amount*2) || countitem(713) < .@amount || (Zeny < .@amount*.@req_amount)) {
			mes "[老藥劑師]";
			mes "你這傢伙！你甚至沒有所有的材料，還想要我幫你製作藥水？！";
			close;
		}
		Zeny = Zeny - (.@max*.@req_amount);
		delitem .@item_req,.@amount*2;
		delitem 713,.@amount; //Empty_Bottle
		getitem getarg(2),.@amount; 
		break;
	case 3:
		mes "[老藥劑師]";
		mes "什麼？！";
		mes "咕...";
		mes "呸！";
		mes "滾開！";
		close;
	}
	mes "[老藥劑師]";
	mes "給你。一切都完成了，所以你可以拿走它。但是請記住！濫用藥物是不好的。";
	close;
}
