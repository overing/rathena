//============================================================
//= Main Refiner Function
//============================================================
//= To allow auto safe refining/multiple refining set the
//= second argument to '1' in the function call.
//= If you enable this function, be sure to edit the value of
//= .@safe to the max safe refine in refine.yml as well.
//============================================================
function	script	refinemain	{
	disable_items;
	.@npc_name$ = getarg(0);
	.@features = getarg(1);
	mes "["+ .@npc_name$ +"]";
	mes "我是武器工匠。";
	mes "我可以精鍊各種武器、盔甲和裝備，所以讓我";
	mes "知道你想讓我精鍊什麼。";
	next;

	setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
	for(.@i = 1; .@i<getarraysize(.@indices); ++.@i) {
		if(getequipisequiped(.@indices[.@i])) {
			.@menu$ = .@menu$ + F_getpositionname(.@indices[.@i]) + "-[" + getequipname(.@indices[.@i]) + "]";
			.@equipped = 1;
		}
		.@menu$ = .@menu$ + ":";
	}
	if (.@equipped == 0) {
		mes "["+ .@npc_name$ +"]";
		mes "我不認為我可以精鍊你擁有的任何物品...";
		close;
	}
	.@part = .@indices[select(.@menu$)];

	if(!getequipisequiped(.@part)) { //custom check
		mes "["+ .@npc_name$ +"]";
		mes "你沒有穿戴";
		mes "任何我可以";
		mes "精鍊的東西。";
		emotion ET_FRET;
		close;
	}
	//Check if the item is refinable...
	if(!getequipisenableref(.@part)) {
		mes "["+ .@npc_name$ +"]";
		mes "我不認為我能";
		mes "精鍊這個物品...";
		close;
	}
	//Check to see if the items is already +10
	if(getequiprefinerycnt(.@part) >= 10) {
		mes "["+ .@npc_name$ +"]";
		mes "我不能再精鍊";
		mes "這個了。這已經是";
		mes "精鍊到極限了！";
		close;
	}
	.@refineitemid = getequipid(.@part); // save id of the item
	.@refinerycnt = getequiprefinerycnt(.@part); //save refinery count
	setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);
	.@price = getequiprefinecost(.@part, REFINE_COST_NORMAL, REFINE_ZENY_COST);
	.@material = getequiprefinecost(.@part, REFINE_COST_NORMAL, REFINE_MATERIAL_ID);
	.@itemtype = getiteminfo( .@refineitemid, ITEMINFO_TYPE );

	if( .@itemtype == IT_ARMOR ){
		.@equip_lv = getequiparmorlv( .@part );

		switch( .@equip_lv ){
			case 1:
				.@safe = 4;
				break;
			default:
				// TODO:
				close;
		}

		// If the VIP system is enabled, the prices for non-VIP players are considerably higher.
		if( VIP_SCRIPT && !vip_status(VIP_STATUS_ACTIVE) ){
			switch( .@equip_lv ){
				case 1:
					.@price = .@price * 10;
					break;
				default:
					// TODO:
					close;
			}
		}
	}else if( .@itemtype == IT_WEAPON ){
		.@equip_lv = getequipweaponlv( .@part );

		switch( .@equip_lv ){
			case 1:
				.@safe = 7;
				break;
			case 2:
				.@safe = 6;
				break;
			case 3:
				.@safe = 5;
				break;
			case 4:
				.@safe = 4;
				break;
			default:
				// TODO:
				close;
		}

		// If the VIP system is enabled, the prices for non-VIP players are considerably higher.
		if( VIP_SCRIPT && !vip_status(VIP_STATUS_ACTIVE) ){
			switch( .@equip_lv ){
				case 1:
					.@price = .@price * 40;
					break;
				case 2:
					.@price = .@price * 50;
					break;
				case 3:
					.@price = .@price * 2;
					break;
				case 4:
					.@price = .@price * 2;
					break;
				default:
					// TODO:
					close;
			}
		}
	}else{
		.@safe = 4;
	}

	// TODO: What was this? Maybe shadow?
	// case 5: .@price = .@price * 10; break;

	if(.@features != 1) {
		mes "["+ .@npc_name$ +"]";
		mes "要精鍊這個我需要";
		mes "一個 ^003366"+getitemname(.@material)+"^000000 還有";
		mes "服務費 " + .@price + " Zeny。";
		mes "你真的要繼續嗎？";
		next;
		if(select("是:否") == 2){
			mes "["+ .@npc_name$ +"]";
			mes "是啊...";
			mes "沒有必要";
			mes "趕時間。慢慢來。";
			close;
		}
		if(getequippercentrefinery(.@part) < 100) {
			mes "["+ .@npc_name$ +"]";
			mes "喔不！如果我繼續";
			mes "精鍊這個，有風險它可能會";
			switch(.@material) {
			case 985:
				mes "被摧毀！這意味著 ^FF0000這件裝備^000000，以及 ^FF0000任何卡片^000000 或添加到這件盔甲上的特殊屬性，^FF0000都會消失^000000。";
				break;
			default:
				mes "被摧毀，而且你會 ^FF0000失去武器^000000，任何 ^FF0000武器中的卡片^000000，";
				mes "或任何添加的特殊屬性。";
				break;
			}
			next;
			mes "["+getarg(0)+"]";
			mes "我不能說得更清楚了。";
			mes "一旦武器被摧毀，";
			mes "就無法找回來了。";
			mes "你真的有機會";
			mes "^FF0000永遠失去這件武器^000000。";
			mes "你仍然要精鍊嗎？";
			next;
			if(select("是:否") == 2){
				mes "["+ .@npc_name$ +"]";
				mes "我完全同意...";
				mes "我可能是一個偉大的精鍊師，但有時甚至我都會犯錯。";
				close;
			}
		}
		if((countitem(.@material) < 1) || (Zeny < .@price)) {
			mes "["+ .@npc_name$ +"]";
			mes "你似乎沒有足夠的";
			mes "Zeny 或 "+getitemname(.@material)+"...";
			mes "去拿一些來。如果你需要我，我會";
			mes "一整天都在這裡。";
			close;
		}
		Zeny = Zeny-.@price;
		delitem .@material,1;

		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) ||
			callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt)) {
			mes "["+ .@npc_name$ +"]";
			emotion ET_FRET;
			mes "等一下...";
			mes "你覺得我很笨嗎？！";
			mes "趁我不注意的時候換了物品！滾出去！";
			close;
		}

		if(getequippercentrefinery(.@part) <= rand(100)) {
			failedrefitem .@part;
			mes "["+ .@npc_name$ +"]";
			emotion (!rand(5))?ET_MONEY:ET_HUK;
			.@lose = rand(1,3);
			if (.@lose == 1) {
				mes "喔！我的天啊！";
				mes "該死的！又來了！";
				mes "我非常抱歉，但你知道熟能生巧。";
				mes "嗯，對吧？呵呵...";
			} else if(.@lose == 2) {
				mes "不～～～～！";
				mes "它壞了！";
				mes "我-我很抱歉！";
			} else {
				mes "糟了！";
				mes "它無法承受";
				mes "更多的回火了！";
				mes "對不起...";
			}
			close;
		}
		mes "["+getarg(0)+"]";
		successrefitem .@part;
		emotion ET_SMILE;
		.@win = rand(1,3);
		if (.@win == 1) {
			mes "完美！";
			mes "呵呵！";
			mes "再一次，";
			mes "完美無瑕的";
			mes "大師級作品～";
		} else if(.@win == 2) {
			mes "成功...！";
			mes "再次，我驚人的";
			mes "天賦今天確實令人眼花繚亂";
			mes "並且閃耀。";
		} else {
			mes "呵呵！";
			mes "我完成了。";
			mes "毫無疑問，我的工作";
			mes "讓您滿意。";
			mes "純粹，徹底的完美～";
		}
		close;
	}

// New Refining Functions ========================
	if (.@refinerycnt < .@safe) {
		mes "["+ .@npc_name$ +"]";
		mes "我可以將其精鍊到安全上限或您想要的次數。由您決定。";
		next;
		.@menu2 = select("請精鍊到安全上限。","我來決定次數。","我改變主意了...");
	} else
		.@menu2 = 2;
	switch(.@menu2){
	case 1: 
		.@refinecnt = .@safe - .@refinerycnt;
		break;
	case 2:
		next;
		mes "["+ .@npc_name$ +"]";
		mes "您想讓我精鍊您的物品多少次？";
		next;
		input .@refinecnt;
		.@refinecheck = .@refinecnt + .@refinerycnt;
		if (.@refinecnt < 1 || .@refinecheck > 10) {
			mes "["+ .@npc_name$ +"]";
			mes "我不能將這個物品精鍊那麼多次。";
			close;
		}
		if(.@refinecheck > .@safe) {
			.@refinecheck = .@refinecheck - .@safe;
			mes "["+ .@npc_name$ +"]";
			mes "這會嘗試將裝備精鍊超過安全上限 " + .@refinecheck + " 次。您的裝備可能會被摧毀... 可以嗎？";
			next;
			if(select("可以...","不要...") == 2){
				mes "["+ .@npc_name$ +"]";
				mes "您說的... 就這樣吧。";
				close;
			}
		}
		break;
	case 3:
		next;
		mes "["+ .@npc_name$ +"]";
		mes "您說的... 就這樣吧。";
		close;
	}
	.@fullprice = .@price * .@refinecnt;
	mes "["+ .@npc_name$ +"]";
	mes "這將花費您 " + .@refinecnt + " " + getitemname(.@material) + " 和 " + .@fullprice + " Zeny。可以嗎？";
	next;
	if(select("是","不要...") == 2){
		mes "["+ .@npc_name$ +"]";
		mes "你這麼說... 就這樣吧。";
		close;
	}
	if(countitem(.@material) < .@refinecnt || Zeny < .@fullprice) {
		mes "["+ .@npc_name$ +"]";
		mes "這就是你所有的嗎？ 很遺憾，我不能以更低的價格為你工作。 試著站在我的角度想想。";
		close;
	}
	Zeny = Zeny - .@fullprice;
	delitem .@material,.@refinecnt;
	while(.@refinecnt){
		if (getequipisequiped(.@part) == 0) {
			mes "["+ .@npc_name$ +"]";
			mes "看看這裡... 你身上沒有任何物品...";
			close;
		}
		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) ||
				callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt) || (.@menu2 == 1 && getequippercentrefinery(.@part) < 100)) {
			mes "["+ .@npc_name$ +"]";
			mes "鏗... 不，但你有沒有想過我可能這麼蠢？！";
			mes "你換了...";
			mes "在我用鎚子擊暈你之前滾出去！！";
			close;
		} 
		mes "鏗、鏗！！！";
		if(.@menu2 == 2 && getequippercentrefinery(.@part) <= rand(100)) {
			failedrefitem .@part;
			emotion ET_HUK;
			mes "["+ .@npc_name$ +"]";
			mes "哇啊啊！！！ 我很抱歉... 我警告過你可能會發生這種情況...";
			.@refinecnt = .@refinecnt - 1;
			if(.@refinecnt == 0) close;
			mes "這是未使用的 Zeny 和材料...";
			getitem .@material,.@refinecnt;
			.@fullprice = .@refinecnt * .@price;
			Zeny = Zeny + .@fullprice;
			close;
		}
		successrefitem .@part;
		emotion ET_BEST;
		.@refinecnt = .@refinecnt - 1;
		.@refinerycnt = getequiprefinerycnt(.@part);
		next;
	}
	mes "["+ .@npc_name$ +"]";
	mes "全部完成了... 歡迎再來。";
	close;
}

			close;
		}
		else if (.@input < 0 || .@input > 500) {
			mes "["+ .@npc_name$ +"]";
			mes "好的，你可以";
			mes "購買多達 500 個。";
			mes "不能再多了，";
			mes "知道了？ 好。";
			next;
		}
		else {
			break;
		}
	}
	.@sell = .@input * .@price;
	if (Zeny < .@sell) {
		mes "["+ .@npc_name$ +"]";
		mes "呃...";
		mes "你沒有足夠的";
		mes "Zeny 來購買";
		mes ""+ .@input +" 個。";
		close;
	}
	if (checkweight(.@material,.@input) == 0) {
		mes "["+ .@npc_name$ +"]";
		mes "嗯...";
		mes "如果你的物品欄沒有足夠的空間，我就不能給你任何東西。 為什麼不把多餘的東西放在卡普拉倉庫裡再試一次？";
		close;
	}
	getitem .@material,.@input;
	Zeny = Zeny-.@sell;
	mes "["+ .@npc_name$ +"]";
	mes "給你！";
	mes "謝謝你的";
	mes "惠顧。";
	close;
}

// Equipment Repair Function
//============================================================
function	script	repairmain	{
	.@repairprice = 5000;
	.@npc_name$ = getarg(0);
	mes "["+ .@npc_name$ +"]";
	mes "嘿！";
	mes "你需要我";
	mes "修理任何物品嗎？";
	mes "你可以指望我";
	mes "進行物品維修！";
	next;
	switch(select("實際上，我有一些物品...:目前沒有。")) {
	case 1:
		.@checkitem = 1;
		while (1) {
			if (getbrokenid(.@checkitem) == 0) {
				break;
			}
			.@checkitem = .@checkitem+1;
		}
		.@checkitem = .@checkitem-1;
		if (!.@checkitem) {
			mes "["+ .@npc_name$ +"]";
			mes "哇，這太不可思議了！";
			mes "你一定非常小心地照顧你的東西。 你的物品都沒有損壞！";
			next;
			mes "["+ .@npc_name$ +"]";
			mes "如果每個人都像你一樣，我就要失業了！！ 哈哈~！";
			close;
		}
		mes "["+ .@npc_name$ +"]";
		mes "嗯...";
		mes "讓我想想...";
		mes "在你的所有物品中，";
		mes "" + .@checkitem + " 個已損壞。";
		mes "你想修理嗎？";
		next;
		.@totalcost = .@repairprice*.@checkitem;
		mes "["+ .@npc_name$ +"]";
		mes "每次維修費用為 " + .@repairprice + " Zeny。 因此，修理所有損壞的物品將花費 " + .@totalcost + " Zeny！ 你想修理這些物品嗎？";
		next;
		switch(select("是:否")) {
		case 1:
			if (Zeny < .@totalcost) {
				mes "["+ .@npc_name$ +"]";
				mes "哇哇哇...";
				mes "在收到維修帳單之前檢查你的錢包！ 我無法修理任何東西，因為你沒有足夠的 Zeny。";
				close;
			}
			.@checkitem2 = 1;
			while (1) {
				if (getbrokenid(.@checkitem2) == 0) {
					break;
				}
				.@checkitem2 = .@checkitem2+1;
			}
			.@checkitem2 = .@checkitem2-1;
			if (.@checkitem == .@checkitem2) {
				Zeny = Zeny-.@totalcost;
				while (.@checkitem) {
					repair(.@checkitem);
					.@checkitem = .@checkitem-1;
				}
				mes "["+ .@npc_name$ +"]";
				mes "好的！ 全部完成。 現在，盡量小心一點。 物品也有生命的，你知道。";
				close;
			}
			else {
				mes "["+ .@npc_name$ +"]";
				mes "嗯？ 有些不對勁。 等等... 裝備好你需要修理的物品，然後再回來找我。";
				close;
			}
		case 2:
			mes "["+ .@npc_name$ +"]";
			mes "嗯，這與我無關，但是讓物品損壞是不好的。 你應該盡快修理它們！";
			close;
		}
	case 2:
		mes "["+ .@npc_name$ +"]";
		mes "呵呵呵...";
		mes "如果你沒有";
		mes "任何物品需要修理";
		mes "就沒有任何";
		mes "事情要找我。";
		close;
	}
}
