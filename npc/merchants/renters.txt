// PecoPeco Breeder (for Knights) :: breeder
//============================================================
prontera,55,350,5	script	大嘴鳥 飼養員#knt	105,{
	if(Upper==0) set .@price,2500; //Normal Peco - default price
	if(Upper==1) set .@price,2500; //Armored Peco
	if(Upper==2) set .@price,2500; //Baby Peco

	mes "[大嘴鳥 飼養員]";
	if (BaseJob == Job_Knight && Class < Job_Rune_Knight) {
		mes "歡迎。";
		mes "尊敬的騎士，";
		mes "您想要租借";
		mes "大嘴鳥 嗎？租借";
		mes "費用是 "+.@price+" zeny。";
		next;
		switch(select("租借 大嘴鳥:取消")) {
		case 1:
			if (Zeny < .@price) {
				mes "[大嘴鳥 飼養員]";
				mes "您沒有";
				mes "足夠的 zeny。";
				mes "您是...";
				mes "破產了嗎？";
				close;
			}
			else if(getskilllv("KN_RIDING") == 0) {
				mes "[大嘴鳥 飼養員]";
				mes "很抱歉，您不符合";
				mes "這項服務的資格。請先去學習";
				mes "騎乘 大嘴鳥 技能。";
				close;
			}
			else if(checkriding()) {
				mes "[大嘴鳥 飼養員]";
				mes "您已經";
				mes "騎在";
				mes "大嘴鳥 上了。";
				close;
			}
			else if(ismounting()) {
				mes "[大嘴鳥 飼養員]";
				mes "請移除您的現金坐騎。";
				close;
			}
			set Zeny, Zeny-.@price;
			setriding;
			close;
		case 2:
			mes "[大嘴鳥 飼養員]";
			mes "我知道了。";
			mes "那麼，";
			mes "祝您有個美好的一天。";
			close;
		}
	} else {
		mes "很抱歉，這些";
		mes "大嘴鳥 僅";
		mes "提供給騎士";
		mes "和領主騎士使用。";
		close;
	}
}

// Grand PecoPeco Breeder (for Crusaders)
//============================================================
prontera,232,318,3	script	大嘴鳥 飼養員#cru	105,{
	if(Upper==0) set .@price,3500; //Normal Peco - default price
	if(Upper==1) set .@price,3500; //Armored Peco
	if(Upper==2) set .@price,3500; //Baby Peco

	mes "[大嘴鳥 飼養員]";
	if (BaseJob == Job_Crusader && Class < Job_Rune_Knight) {
		if(Upper != 1 ) mes "歡迎，十字軍。";
		else mes "歡迎，聖殿騎士。";
		mes "我們為您準備了";
		mes "特別的";
		mes "大嘴鳥。租借一隻";
		mes "將花費 "+.@price+" zeny。";
		next;
		switch(select("租借 大嘴鳥:離開")) {
		case 1:
			if (Zeny < .@price) {
				mes "[大嘴鳥 飼養員]";
				mes "您沒有";
				mes "足夠的 zeny。";
				mes "如果您想要";
				mes "大嘴鳥，請";
				mes "帶 "+.@price+" zeny 來...";
				close;
			}
			else if(getskilllv("KN_RIDING") == 0) {
				mes "[大嘴鳥 飼養員]";
				mes "在";
				mes "我可以租給您之前，您必須先學會";
				mes "騎乘 大嘴鳥。";
				close;
			}
			else if(checkriding()) {
				mes "[大嘴鳥 飼養員]";
				mes "您已經";
				mes "騎在 大嘴鳥 上了。";
				close;
			}
			else if(ismounting()) {
				mes "[大嘴鳥 飼養員]";
				mes "請移除您的現金坐騎。";
				close;
			}
			set Zeny, Zeny-.@price;
			setriding;
			close;
		case 2:
			mes "[大嘴鳥 飼養員]";
			mes "再見。";
			close;
		}
	} else {
		mes "我可以為您做些什麼？";
		mes "請注意，";
		mes "此 大嘴鳥 租借";
		mes "服務僅限";
		mes "十字軍和聖殿騎士使用。";
		close;
	}
}

// Falcon Master :: breeder

hu_in01,381,304,5	script	Falcon Breeder#hnt	105,{
	if(Upper==0) set .@price,2500; //Normal Falcon - default price
	if(Upper==1) set .@price,2500; //Scarf Falcon
	if(Upper==2) set .@price,2500; //Baby Falcon

	mes "[獵鷹飼養員]";
	if (BaseJob == Job_Hunter) {
		if (checkwug()) {
			mes "嗯...";
			mes "在您解雇座狼之前，";
			mes "您無法租借獵鷹！";
			close;
		}
		mes "您需要獵鷹嗎？";
		mes "您可以租借您自己的";
		mes "忠實的猛禽，";
		mes "費用僅需 "+.@price+" zeny~";
		next;
		switch(select("租借獵鷹:取消")) {
		case 1:
			if (checkwug()) {
				mes "[獵鷹飼養員]";
				mes "嗯...";
				mes "在您解雇座狼之前，";
				mes "您無法租借獵鷹！";
				close;
			}
			if (Zeny < .@price) {
				mes "[獵鷹飼養員]";
				mes "這是什麼？";
				mes "您沒有";
				mes "足夠的 zeny？！";
				mes "您最好開始";
				mes "獵取金錢";
				mes "而不是怪物~";
				close;
			}
			else if(getskilllv("HT_FALCON") == 0) {
				mes "[獵鷹飼養員]";
				mes "哎呀~";
				mes "先去學習如何";
				mes "管理獵鷹";
				mes "吧！如果您無法";
				mes "駕馭它，我就不能租";
				mes "給您，您知道的。";
				close;
			}
			else if(checkfalcon()) {
				mes "[獵鷹飼養員]";
				mes "嗯...";
				mes "您已經有";
				mes "一隻獵鷹了。它就在";
				mes "那裡，您看不到嗎？";
				close;
			}
			set Zeny, Zeny-.@price;
			setfalcon;
			close;
		case 2:
			mes "[獵鷹飼養員]";
			mes "等、等等，您要去";
			mes "哪裡？這些";
			mes "獵鷹都是一流的，";
			mes "我保證！來吧，喲~";
			close;
		}
	} else {
		mes "年輕的傻瓜！";
		mes "獵鷹只能";
		mes "由獵人和";
		mes "神射手使用，明白嗎？";
		mes "...嘿嘿，嫉妒嗎？";
		close;
	}
}
