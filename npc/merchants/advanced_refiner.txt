payon,157,146,6	script	順飛#cash	85,{
	disable_items;
	mes "[順飛]";
	mes "我是武器匠";
	mes "我可以精鍊各式各樣的武器、";
	mes "盔甲和裝備，所以請";
	mes "讓我知道你想精鍊什麼。";
	next;

	setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
	for(set .@i,1; .@i<=10; set .@i,.@i+1) {
		if (getequipisequiped(.@indices[.@i])) {
			set .@menu$, .@menu$ + F_getpositionname(.@indices[.@i]) + "-[" + getequipname(.@indices[.@i]) + "]";
			set .@equipped,1;
		}
		set .@menu$, .@menu$ + ":";
	}
	if (.@equipped == 0) {
		mes "[順飛]";
		mes "我不認為我可以精鍊你擁有的任何物品...";
		close;
	}
	set .@part, .@indices[ select(.@menu$) ];

	if (!getequipisequiped(.@part)) //custom check
		close;
	if (!getequipisenableref(.@part)) {
		mes "[順飛]";
		mes "去找其他的鐵匠吧。你不能精鍊這個東西。";
		close;
	}
	if (getequiprefinerycnt(.@part) >= 10) {
		mes "[順飛]";
		mes "嗯... 已經有人將它完美化了。我不認為我可以再對它做更多處理。";
		close;
	}

	.@refineitemid = getequipid(.@part); // save id of the item
	.@refinerycnt = getequiprefinerycnt(.@part); //save refinery count
	.@price = getequiprefinecost(.@part, REFINE_COST_ENRICHED, REFINE_ZENY_COST);
	.@material = getequiprefinecost(.@part, REFINE_COST_ENRICHED, REFINE_MATERIAL_ID);
	.@itemtype = getiteminfo( .@refineitemid, ITEMINFO_TYPE );

	// Make sure you have the necessary items and Zeny to refine your items
	// Determines chance of failure and verifies that you want to continue.
	callsub S_RefineValidate,.@itemtype,.@material,.@price,.@part,.@refineitemid,.@refinerycnt;

	mes "[順飛]";
	mes "鏗！鏗！鏗！";
	if (getequippercentrefinery(.@part, true) > rand(100)) {
		successrefitem .@part;
		next;
		emotion ET_BEST;
		mes "[順飛]";
		mes "給你！完成了。";
		mes "我已經很久沒製作出這麼好的"+((.@itemtype == IT_WEAPON)?"武器":"盔甲")+"了。你一定很高興它變得更強了！";
		close;
	}
	failedrefitem .@part;
	next;
	emotion (!rand(5))?ET_MONEY:ET_HUK;
	mes "[順飛]";
	mes "嗚嗚嗚嗚嗚嗚嗚嗚!!!";
	next;
	mes "[順飛]";
	mes "...";
	mes ".....";
	mes ".......呼呼呼呼～";
	mes "........這是你的選擇和我的能力，不後悔。";
	close;

S_RefineValidate:
	.@itemtype = getarg(0);
	.@item_req = getarg(1);
	.@price = getarg(2);
	.@part = getarg(3);
	setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);

	if( .@itemtype == IT_ARMOR ){
		.@equip_lv = getequiparmorlv( .@part );

		// If the VIP system is enabled, the prices for non-VIP players are considerably higher.
		if (VIP_SCRIPT && !vip_status(VIP_STATUS_ACTIVE)) {
			switch(.@equip_lv) {
				case 1:
					set .@price, .@price * 10;
					break;
				default:
					// TODO:
					close;
			}
		}
	}else if( .@itemtype == IT_WEAPON ){
		.@equip_lv = getequipweaponlv( .@part );

		// If the VIP system is enabled, the prices for non-VIP players are considerably higher.
		if (VIP_SCRIPT && !vip_status(VIP_STATUS_ACTIVE)) {
			switch( .@equip_lv ){
				case 1:
					set .@price, .@price * 40;
					break;
				case 2:
					set .@price, .@price * 50;
					break;
				case 3:
					set .@price, .@price * 2;
					break;
				case 4:
					set .@price, .@price * 2;
					break;
				default:
					// TODO:
					close;
			}
		}
	}else{
		// TODO:
		close;
	}

	mes "[順飛]";
	if (.@itemtype == IT_WEAPON)
		mes "你想精鍊一個等級 "+ .@equip_lv +" 的武器嗎？";
	mes "要精鍊它，你需要有一個 ^ff9999"+ getitemname(.@item_req) +"^000000 和 "+ .@price +" zeny。";
	mes "你想要繼續嗎？";
	next;
	if(select("是:否") == 1) {
		if (getequippercentrefinery(.@part) < 100) {
			if (.@itemtype == IT_WEAPON) {
				mes "[順飛]";
				mes "哇！！";
				mes "這件武器看起來";
				mes "可能已經被精鍊過...";
				mes "很多次了...";
				mes "如果再次精鍊";
				mes "它可能會損壞。";
				next;
				mes "如果它損壞了，";
				mes "你就不能再使用它了！";
				mes "裡面的所有卡片和屬性^ff0000都會遺失^000000！";
				mes "^ff0000此外，裝備也會損壞！^000000";
				mes "你確定你還想繼續嗎？";
				next;
				if(select("是:否") == 2) {
					mes "[順飛]";
					mes "很好。";
					mes "因為如果武器因為不合理的精鍊而損壞，我的心情也會很糟糕。";
					close;
				}
			} else {
				mes "[順飛]";
				mes "咯咯。咯咯。喔，你真有膽量，敢精鍊這個。";
				mes "你知道這非常冒險，不是嗎？";
				next;
				mes "如果你的防禦裝備損壞，你將永遠無法再次使用它。";
				mes "即使你的卡片和你的改造也將會^ff0000完全消失^000000。";
				//mes "一切都會消失。就像... 消失得無影無蹤！";
				mes "你真的希望繼續嗎？";
				next;
				if(select("是:否") == 2) {
					mes "[順飛]";
					mes "胡說八道。你浪費我寶貴的時間。";
					mes "滾開，混蛋。";
					close;
				}
			}
		}
		if (countitem(.@item_req) > 0 && Zeny > .@price) {
			delitem .@item_req,1;
			set Zeny, Zeny - .@price;

			// anti-hack
			if (callfunc("F_IsEquipIDHack", .@part, getarg(4)) ||
				callfunc("F_IsEquipRefineHack", .@part, getarg(5)) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3])) {
				mes "[霍林克]";
				emotion ET_FRET;
				mes "等一下...";
				mes "你認為我很笨嗎？！";
				mes "當我不注意的時候，你換了物品！滾出去！";
				close;
			}

			return;
		}
		mes "[順飛]";
		mes "這些就是你全部擁有的嗎？";
		mes "我很抱歉，但是如果沒有所有材料，我什麼都做不了。此外，我應該得到一些我的工作報酬，不是嗎？";
		close;
	}
	mes "[順飛]";
	mes "即使你不高興，我也無能為力...";
	close;
}
