prontera,123,102,5	script	商人#pron	85,{
	if (MaxWeight - Weight < 6301 || checkweight(1201,1) == 0) {
		mes "[商人]";
		mes "哈哈！";
		mes "你是什麼，超人嗎？";
		mes "你攜帶了這麼多的東西！";
		mes "你最好把一些東西";
		mes "放到卡普拉倉庫裡~";
		close;
	}
	mes "[商人]";
	mes "日安！";
	mes "你有收集任何";
	mes "像是貝殼或絨毛的物品嗎？";
	mes "果凍呢？ 喔對，";
	mes "我需要那些東西做點事。";
	next;
	mes "[商人]";
	mes "當然，我不會要求你";
	mes "免費給我那些東西。如果我用紅色藥水換 5 個貝殼、10 個絨毛，";
	mes "或 10 個果凍呢？";
	next;
	switch(select("當然好啊！:我覺得這是敲竹槓！:不用了，謝謝。")) {
	case 1:
		mes "[商人]";
		mes "所以你想";
		mes "帶什麼物品給我？";
		mes "貝殼、絨毛，還是果凍？";
		next;
		switch(select("貝殼:果凍:絨毛:取消")) {
		case 1: callfunc "F_PotExchange",935,5;
		case 2: callfunc "F_PotExchange",909,10;
		case 3: callfunc "F_PotExchange",914,10;
		case 4:
			mes "[商人]";
			mes "好的~";
			mes "沒問題。";
			close;
		}
	case 2:
		mes "[商人]";
		mes "敲竹槓...？";
		mes "如果你查看";
		mes "交易物品的市場價值，";
		mes "實際上被敲竹槓的人是我。";
		close;
	case 3:
		mes "[商人]";
		mes "好吧，";
		mes "沒問題。";
		mes "但如果你改變主意";
		mes "就回來找我吧。";

	}
}

function	script	F_PotExchange	{
	.@item_req = getarg(0);
	.@req_amount = getarg(1);

	if (countitem(.@item_req) < .@req_amount) {
		mes "[商人]";
		mes "嘿，你答應要給我";
		mes "的 "+ getitemname(.@item_req) +"";
		mes "在哪裡啊？";
		mes "給我 "+ getitemname(.@item_req) +"！";
		close;
	}
	else {
		mes "[商人]";
		mes "好，讓我檢查一下";
		mes "你有多少 "+ getitemname(.@item_req) +"。";
		mes "嗯...";
		next;
		mes "[商人]";
		mes "你總共有";
		mes countitem(.@item_req) +" 個 "+ getitemname(.@item_req) +"。";
		mes "我可以給你總共";
		mes (countitem(.@item_req)/.@req_amount) +" 瓶紅色藥水。";
		next;
		mes "[商人]";
		mes "你覺得如何？";
		mes "成交嗎？";
		next;
		if (select("成交。:不成交。") == 1) {
			mes "[商人]";
			mes "你知道你要用";
			mes "這些 "+ getitemname(.@item_req) +"";
			mes "換多少紅色藥水，";
			mes "對吧？";
			next;
			mes "[商人]";
			mes "現在，我可以一次";
			mes "跟你換最少 1 瓶、";
			mes "最多 100 瓶紅色藥水。";
			mes "如果你改變";
			mes "主意，輸入 '0' 取消即可。";
			next;
			while(1) {
				input .@input,0,101;
				if (.@input == 0) {
					mes "[商人]";
					mes "什麼...？";
					mes "你到底為什麼";
					mes "要來找我啊？";
					mes "真優柔寡斷，不是嗎？";
					close;
				}
				else if (.@input > 100) {
					mes "[商人]";
					mes "我一次不能給你";
					mes "超過 100 瓶紅色藥水。";
					mes "我們再";
					mes "試一次。";
					next;
				}
				else {
					break;
				}
			}
			.@put_out_item = .@input * .@req_amount;
			if (countitem(.@item_req) < .@put_out_item) {
				mes "[商人]";
				mes "喔喔，你輸入的數字好像不對。你最好再確認一次你可以換多少紅色藥水。";
				close;
			}
			mes "[商人]";
			mes "給你!";
			mes "檢查一下我";
			mes "給你的紅色藥水數量，應該沒錯。謝謝，這是一筆好交易～";
			delitem .@item_req,.@put_out_item; //Shell
			getitem 501,.@input; //Red_Potion
			close;
		}
		mes "[商人]";
		mes "嗯～";
		mes "好吧。";
		mes "不過對你這種";
		mes "冒險者來說，";
		mes "紅色藥水不是更有用嗎？";
		close;
	}
}
