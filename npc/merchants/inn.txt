		mes "[旅館員工]";

//======================= Inn Function ==============================
function	script	F_InnMaid	{
	.@npc_name$ = getarg(0);

	mes .@npc_name$;
	mes "歡迎來到";
	mes getarg(1) + "。";
	mes "有什麼需要幫忙的嗎？";
	next;
	switch(select("儲存:休息 -> 5000 Zeny:取消")) {
	case 1:
		mes .@npc_name$;
		mes "您的重生點";
		mes "已經儲存。";
		mes "謝謝您，";
		mes "請再次光臨。";
		savepoint getarg(2),getarg(3),getarg(4),1,1;
		close;
	case 2:
		mes .@npc_name$;
		if(Zeny < 5000){
			mes "抱歉，服務費是 5,000 Zeny。請確保您下次入住時有足夠的錢，好嗎？";
			close;
		}
		mes "謝謝您。";
		mes "祝您";
		mes "休息愉快～";
		close2;
		set Zeny,Zeny - 5000;
		percentheal 100,100;
		return;
	case 3:
		close;
	}
}
